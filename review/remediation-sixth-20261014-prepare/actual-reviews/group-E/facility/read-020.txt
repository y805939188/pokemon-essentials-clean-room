345:  def saveWins(challenge)
346:    if challenge.decision == 0     # if undecided
347:      @currentWins  = 0
348:      @currentSwaps = 0
349:    else
350:      if challenge.decision == 1   # if won
351:        @currentWins  = challenge.wins
352:        @currentSwaps = challenge.swaps
353:      else                       # if lost
354:        @currentWins  = 0
355:        @currentSwaps = 0
356:      end
357:      @maxWins       = [@maxWins, challenge.wins].max
358:      @previousWins  = challenge.wins
359:      @maxSwaps      = [@maxSwaps, challenge.swaps].max
360:      @previousSwaps = challenge.swaps
