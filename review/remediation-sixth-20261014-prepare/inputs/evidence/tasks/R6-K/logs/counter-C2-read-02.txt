728	  def pbFailsAgainstTarget?(user, target, show_message)
729	    if target.effects[PBEffects::Disable] > 0 || !target.lastRegularMoveUsed
730	      @battle.pbDisplay(_INTL("But it failed!")) if show_message
731	      return true
732	    end
733	    return true if pbMoveFailedAromaVeil?(user, target, show_message)
734	    canDisable = false
735	    target.eachMove do |m|
736	      next if m.id != target.lastRegularMoveUsed
737	      next if m.pp == 0 && m.total_pp > 0
738	      canDisable = true
739	      break
740	    end
741	    if !canDisable
742	      @battle.pbDisplay(_INTL("But it failed!")) if show_message
743	      return true
744	    end
745	    return false
746	  end
