317	  def can_attack?
318	    return false if self.effects[PBEffects::HyperBeam] > 0
319	    return false if status == :SLEEP && statusCount > 1
320	    return false if status == :FROZEN   # Only 20% chance of unthawing; assune it won't
321	    return false if self.effects[PBEffects::Truant] && has_active_ability?(:TRUANT)
322	    return false if self.effects[PBEffects::Flinch]
323	    # NOTE: Confusion/infatuation/paralysis have higher chances of allowing the
324	    #       attack, so the battler is treated as able to attack in those cases.
325	    return true
326	  end

336	  # NOTE: This specifically means "is not currently trapped but can become
337	  #       trapped by an effect". Similar to def pbCanSwitchOut? but this returns
338	  #       false if any certain switching OR certain trapping applies.
339	  def can_become_trapped?
340	    return false if fainted?
341	    # Ability/item effects that allow switching no matter what
342	    if ability_active? && Battle::AbilityEffects.triggerCertainSwitching(ability, battler, @ai.battle)
343	      return false
344	    end
345	    if item_active? && Battle::ItemEffects.triggerCertainSwitching(item, battler, @ai.battle)
346	      return false
347	    end
348	    # Other certain switching effects
349	    return false if Settings::MORE_TYPE_EFFECTS && has_type?(:GHOST)
350	    # Other certain trapping effects
351	    return false if battler.trappedInBattle?
352	    # Trapping abilities/items
353	    @ai.each_foe_battler(side) do |b, i|
354	      if b.ability_active? &&
355	         Battle::AbilityEffects.triggerTrappingByTarget(b.ability, battler, b.battler, @ai.battle)
356	        return false
357	      end
358	      if b.item_active? &&
359	         Battle::ItemEffects.triggerTrappingByTarget(b.item, battler, b.battler, @ai.battle)
360	        return false
361	      end
362	    end
363	    return true
364	  end

368	  def wants_status_problem?(new_status)
369	    return true if new_status == :NONE
370	    if ability_active?
371	      case ability_id
372	      when :GUTS
373	        return true if ![:SLEEP, :FROZEN].include?(new_status) &&
374	                       @ai.stat_raise_worthwhile?(self, :ATTACK, true)
375	      when :MARVELSCALE
376	        return true if @ai.stat_raise_worthwhile?(self, :DEFENSE, true)
377	      when :QUICKFEET
378	        return true if ![:SLEEP, :FROZEN].include?(new_status) &&
379	                       @ai.stat_raise_worthwhile?(self, :SPEED, true)
380	      when :FLAREBOOST
381	        return true if new_status == :BURN && @ai.stat_raise_worthwhile?(self, :SPECIAL_ATTACK, true)
382	      when :TOXICBOOST
383	        return true if new_status == :POISON && @ai.stat_raise_worthwhile?(self, :ATTACK, true)
384	      when :POISONHEAL
385	        return true if new_status == :POISON
386	      when :MAGICGUARD   # Want a harmless status problem to prevent getting a harmful one
387	        return true if new_status == :POISON ||
388	                       (new_status == :BURN && !@ai.stat_raise_worthwhile?(self, :ATTACK, true))
389	      end
390	    end
391	    return true if new_status == :SLEEP && check_for_move { |m| m.usableWhenAsleep? }
392	    if has_move_with_function?("DoublePowerIfUserPoisonedBurnedParalyzed")
393	      return true if [:POISON, :BURN, :PARALYSIS].include?(new_status)
394	    end
395	    return false
396	  end

400	  # Returns a value indicating how beneficial the given ability will be to this
401	  # battler if it has it.
402	  # Return values are typically between -10 and +10. 0 is indifferent, positive
403	  # values mean this battler benefits, negative values mean this battler suffers.
404	  # NOTE: This method assumes the ability isn't being negated. The calculations
405	  #       that call this method separately check for it being negated, because
406	  #       they need to do something special in that case.
407	  def wants_ability?(ability = :NONE)
408	    ability = ability.id if !ability.is_a?(Symbol) && ability.respond_to?("id")
409	    # Get the base ability rating
410	    ret = 0
411	    Battle::AI::BASE_ABILITY_RATINGS.each_pair do |val, abilities|
412	      next if !abilities.include?(ability)
413	      ret = val
414	      break
415	    end
416	    # Modify the rating based on ability-specific contexts
417	    if @ai.trainer.medium_skill?
418	      ret = Battle::AI::Handlers.modify_ability_ranking(ability, ret, self, @ai)
419	    end
420	    return ret
