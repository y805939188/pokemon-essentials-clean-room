503	# Pokémon doesn't have an ability that makes it immune to a foe's move, but a

508	Battle::AI::Handlers::ShouldSwitch.add(:absorb_foe_move,
509	  proc { |battler, reserves, ai, battle|
510	    next false if !ai.trainer.medium_skill?
511	    # Not worth it if the battler is evasive enough
512	    next false if battler.stages[:EVASION] >= 3
513	    # Not worth it if abilities are being negated
514	    next false if battle.pbCheckGlobalAbility(:NEUTRALIZINGGAS)
515	    # Get the foe move with the highest power (or a random damaging move)
516	    foe_moves = []
517	    ai.each_foe_battler(battler.side) do |b, i|
518	      b.moves.each do |move|
519	        next if move.statusMove?
520	        m_power = move.power
521	        m_power = 100 if move.is_a?(Battle::Move::OHKO)
522	        m_type = move.pbCalcType(b.battler)
523	        foe_moves.push([m_power, m_type, move])
524	      end
525	    end
526	    next false if foe_moves.empty?
527	    if ai.trainer.high_skill?
528	      foe_moves.sort! { |a, b| a[0] <=> b[0] }   # Highest power move
529	      chosen_move = foe_moves.last
530	    else
531	      chosen_move = foe_moves[ai.pbAIRandom(foe_moves.length)]   # Random move
532	    end
533	    # Get the chosen move's information
534	    move_power = chosen_move[0]
535	    move_type = chosen_move[1]
536	    move = chosen_move[2]
537	    # Don't bother if the foe's best move isn't too strong
538	    next false if move_power < 70
539	    # Check battler for absorbing ability
540	    next false if ai.pokemon_can_absorb_move?(battler, move, move_type)
541	    # battler can't absorb move; find a party Pokémon that can
542	    if reserves.any? { |pkmn| ai.pokemon_can_absorb_move?(pkmn, move, move_type) }
543	      next false if ai.pbAIRandom(100) < 70
544	      PBDebug.log_ai("#{battler.name} wants to switch because it can't absorb a foe's move but a reserve can")
