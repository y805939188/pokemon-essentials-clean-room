3	  # Decide whether the trainer is allowed to tell the Pokémon to use the given
4	  # move. Called when choosing a command for the round.
5	  # Also called when processing the Pokémon's action, because these effects also
6	  # prevent Pokémon action. Relevant because these effects can become active
7	  # earlier in the same round (after choosing the command but before using the
8	  # move) or an unusable move may be called by another move such as Metronome.
9	  #=============================================================================
10	  def pbCanChooseMove?(move, commandPhase, showMessages = true, specialUsage = false)
11	    # Disable
12	    if @effects[PBEffects::DisableMove] == move.id && !specialUsage
13	      if showMessages
14	        msg = _INTL("{1}'s {2} is disabled!", pbThis, move.name)
15	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
16	      end
17	      return false
18	    end
19	    # Heal Block
20	    if @effects[PBEffects::HealBlock] > 0 && move.healingMove?
21	      if showMessages
22	        msg = _INTL("{1} can't use {2} because of Heal Block!", pbThis, move.name)
23	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
24	      end
25	      return false
26	    end
27	    # Gravity
28	    if @battle.field.effects[PBEffects::Gravity] > 0 && move.unusableInGravity?
29	      if showMessages
30	        msg = _INTL("{1} can't use {2} because of gravity!", pbThis, move.name)
31	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
32	      end
33	      return false
34	    end
35	    # Throat Chop
36	    if @effects[PBEffects::ThroatChop] > 0 && move.soundMove?
37	      if showMessages
38	        msg = _INTL("{1} can't use {2} because of Throat Chop!", pbThis, move.name)
39	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
40	      end
41	      return false
42	    end
43	    # Choice Band/Gorilla Tactics
44	    @effects[PBEffects::ChoiceBand] = nil if !pbHasMove?(@effects[PBEffects::ChoiceBand])
45	    if @effects[PBEffects::ChoiceBand] && move.id != @effects[PBEffects::ChoiceBand]
46	      choiced_move = GameData::Move.try_get(@effects[PBEffects::ChoiceBand])
47	      if choiced_move
48	        if hasActiveItem?([:CHOICEBAND, :CHOICESPECS, :CHOICESCARF])
49	          if showMessages
50	            msg = _INTL("The {1} only allows the use of {2}!", itemName, choiced_move.name)
51	            (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
52	          end
53	          return false
54	        elsif hasActiveAbility?(:GORILLATACTICS)
55	          if showMessages
56	            msg = _INTL("{1} can only use {2}!", pbThis, choiced_move.name)
57	            (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
58	          end
59	          return false
60	        end
61	      end
62	    end
63	    # Taunt
64	    if @effects[PBEffects::Taunt] > 0 && move.statusMove? && !specialUsage
65	      if showMessages
66	        msg = _INTL("{1} can't use {2} after the taunt!", pbThis, move.name)
67	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
68	      end
69	      return false
70	    end
71	    # Torment
72	    if @effects[PBEffects::Torment] && !@effects[PBEffects::Instructed] && !specialUsage &&
73	       @lastMoveUsed && move.id == @lastMoveUsed && move.id != @battle.struggle.id
74	      if showMessages
75	        msg = _INTL("{1} can't use the same move twice in a row due to the torment!", pbThis)
76	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
77	      end
78	      return false
79	    end
80	    # Imprison
81	    if @battle.allOtherSideBattlers(@index).any? { |b| b.effects[PBEffects::Imprison] && b.pbHasMove?(move.id) }
82	      if showMessages
83	        msg = _INTL("{1} can't use its sealed {2}!", pbThis, move.name)
84	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
85	      end
86	      return false
87	    end
88	    # Assault Vest (prevents choosing status moves but doesn't prevent
89	    # executing them)
90	    if hasActiveItem?(:ASSAULTVEST) && move.statusMove? && move.function_code != "UseMoveTargetIsAboutToUse" && commandPhase
91	      if showMessages
92	        msg = _INTL("The effects of the {1} prevent status moves from being used!", itemName)
93	        (commandPhase) ? @battle.pbDisplayPaused(msg) : @battle.pbDisplay(msg)
94	      end
95	      return false
96	    end
97	    # Belch
98	    return false if !move.pbCanChooseMove?(self, commandPhase, showMessages)
99	    return true
100	  end

105	  # Return true if Pokémon continues attacking (although it may have chosen to
106	  # use a different move in disobedience), or false if attack stops.
107	  def pbObedienceCheck?(choice)
108	    return true if usingMultiTurnAttack?
109	    return true if choice[0] != :UseMove
110	    return true if !@battle.internalBattle
111	    return true if !@battle.pbOwnedByPlayer?(@index)
112	    disobedient = false
113	    # Pokémon may be disobedient; calculate if it is
114	    badge_level = 10 * (@battle.pbPlayer.badge_count + 1)
115	    badge_level = GameData::GrowthRate.max_level if @battle.pbPlayer.badge_count >= 8
116	    if Settings::ANY_HIGH_LEVEL_POKEMON_CAN_DISOBEY ||
117	       (Settings::FOREIGN_HIGH_LEVEL_POKEMON_CAN_DISOBEY && @pokemon.foreign?(@battle.pbPlayer))
118	      if @level > badge_level
119	        a = ((@level + badge_level) * @battle.pbRandom(256) / 256).floor
120	        disobedient |= (a >= badge_level)
121	      end
122	    end
123	    disobedient |= !pbHyperModeObedience(choice[2])
124	    return true if !disobedient
125	    # Pokémon is disobedient; make it do something else
126	    return pbDisobey(choice, badge_level)
127	  end

129	  def pbDisobey(choice, badge_level)
130	    move = choice[2]
131	    PBDebug.log("[Disobedience] #{pbThis} disobeyed")
132	    @effects[PBEffects::Rage] = false
133	    # Do nothing if using Snore/Sleep Talk
134	    if @status == :SLEEP && move.usableWhenAsleep?
135	      @battle.pbDisplay(_INTL("{1} ignored orders and kept sleeping!", pbThis))
136	      return false
137	    end
138	    b = ((@level + badge_level) * @battle.pbRandom(256) / 256).floor
139	    # Use another move
140	    if b < badge_level
141	      @battle.pbDisplay(_INTL("{1} ignored orders!", pbThis))
142	      return false if !@battle.pbCanShowFightMenu?(@index)
143	      otherMoves = []
144	      eachMoveWithIndex do |_m, i|
145	        next if i == choice[1]
146	        otherMoves.push(i) if @battle.pbCanChooseMove?(@index, i, false)
147	      end
148	      return false if otherMoves.length == 0   # No other move to use; do nothing
149	      newChoice = otherMoves[@battle.pbRandom(otherMoves.length)]
150	      choice[1] = newChoice
151	      choice[2] = @moves[newChoice]
152	      choice[3] = -1
153	      return true
154	    end
155	    c = @level - badge_level
156	    r = @battle.pbRandom(256)
157	    # Fall asleep
158	    if r < c && pbCanSleep?(self, false)
159	      pbSleepSelf(_INTL("{1} began to nap!", pbThis))
160	      return false
161	    end
162	    # Hurt self in confusion
163	    r -= c
164	    if r < c && @status != :SLEEP
165	      pbConfusionDamage(_INTL("{1} won't obey! It hurt itself in its confusion!", pbThis))
166	      return false
167	    end
168	    # Show refusal message and do nothing
169	    case @battle.pbRandom(4)
170	    when 0 then @battle.pbDisplay(_INTL("{1} won't obey!", pbThis))
171	    when 1 then @battle.pbDisplay(_INTL("{1} turned away!", pbThis))
172	    when 2 then @battle.pbDisplay(_INTL("{1} is loafing around!", pbThis))
173	    when 3 then @battle.pbDisplay(_INTL("{1} pretended not to notice!", pbThis))
174	    end
175	    return false
176	  end

179	  # Check whether the user (self) is able to take action at all.
180	  # If this returns true, and if PP isn't a problem, the move will be considered
181	  # to have been used (even if it then fails for whatever reason).
182	  #=============================================================================
183	  def pbTryUseMove(choice, move, specialUsage, skipAccuracyCheck)
184	    # Check whether it's possible for self to use the given move
185	    # NOTE: Encore has already changed the move being used, no need to have a
186	    #       check for it here.
187	    if !pbCanChooseMove?(move, false, true, specialUsage)
188	      @lastMoveFailed = true
189	      return false
190	    end
191	    # Check whether it's possible for self to do anything at all
192	    if @effects[PBEffects::SkyDrop] >= 0   # Intentionally no message here
193	      PBDebug.log("[Move failed] #{pbThis} can't use #{move.name} because of being Sky Dropped")
194	      return false
195	    end
196	    if @effects[PBEffects::HyperBeam] > 0   # Intentionally before Truant
197	      PBDebug.log("[Move failed] #{pbThis} is recharging after using #{move.name}")
198	      @battle.pbDisplay(_INTL("{1} must recharge!", pbThis))
199	      @effects[PBEffects::Truant] = !@effects[PBEffects::Truant] if hasActiveAbility?(:TRUANT)
200	      return false
201	    end
202	    if choice[1] == -2   # Battle Palace
203	      PBDebug.log("[Move failed] #{pbThis} can't act in the Battle Palace somehow")
204	      @battle.pbDisplay(_INTL("{1} appears incapable of using its power!", pbThis))
205	      return false
206	    end
207	    # Skip checking all applied effects that could make self fail doing something
208	    return true if skipAccuracyCheck
209	    # Check status problems and continue their effects/cure them
210	    case @status
211	    when :SLEEP
212	      self.statusCount -= 1
213	      if @statusCount <= 0
214	        pbCureStatus
215	      else
216	        pbContinueStatus
217	        if !move.usableWhenAsleep?   # Snore/Sleep Talk
218	          PBDebug.log("[Move failed] #{pbThis} is asleep")
219	          @lastMoveFailed = true
220	          return false
221	        end
222	      end
223	    when :FROZEN
224	      if !move.thawsUser?
225	        if @battle.pbRandom(100) < 20
226	          pbCureStatus
227	        else
228	          pbContinueStatus
229	          PBDebug.log("[Move failed] #{pbThis} is frozen")
230	          @lastMoveFailed = true
231	          return false
232	        end
233	      end
234	    end
235	    # Obedience check
236	    return false if !pbObedienceCheck?(choice)
237	    # Truant
238	    if hasActiveAbility?(:TRUANT)
239	      @effects[PBEffects::Truant] = !@effects[PBEffects::Truant]
240	      if !@effects[PBEffects::Truant]   # True means loafing, but was just inverted
