105:   def pbJudge
106:     judgearray = []
107:     if @lastPokemon
108:       judgearray.push([-1, @lastPokemon.species, pbBugContestScore(@lastPokemon)])
109:     end
110:     maps_with_encounters = []
111:     @contestMaps.each do |map|
112:       enc_type = :BugContest
113:       enc_type = :Land if !$PokemonEncounters.map_has_encounter_type?(map, enc_type)
114:       if $PokemonEncounters.map_has_encounter_type?(map, enc_type)
115:         maps_with_encounters.push([map, enc_type])
116:       end
117:     end
118:     raise _INTL("There are no Bug Contest/Land encounters for any Bug Contest maps.") if maps_with_encounters.empty?
119:     @contestants.each do |cont|
120:       enc_data = maps_with_encounters.sample
121:       enc = $PokemonEncounters.choose_wild_pokemon_for_map(enc_data[0], enc_data[1])
122:       raise _INTL("No encounters for map {1} somehow, so can't judge contest.", enc_data[0]) if !enc
123:       pokemon = Pokemon.new(enc[0], enc[1])
124:       pokemon.hp = rand(1...pokemon.totalhp)
125:       score = pbBugContestScore(pokemon)
126:       judgearray.push([cont, pokemon.species, score])
127:     end
128:     raise _INTL("Too few bug-catching contestants") if judgearray.length < 3
129:     judgearray.sort! { |a, b| b[2] <=> a[2] }   # sort by score in descending order
130:     @places.push(judgearray[0])
131:     @places.push(judgearray[1])
132:     @places.push(judgearray[2])
133:   end
