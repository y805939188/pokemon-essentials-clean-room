361: def pbBugContestBattle(pkmn, level = 1)
362:   # Record information about party Pokémon to be used at the end of battle (e.g.
363:   # comparing levels for an evolution check)
364:   EventHandlers.trigger(:on_start_battle)
365:   # Generate a wild Pokémon based on the species and level
366:   pkmn = pbGenerateWildPokemon(pkmn, level) if !pkmn.is_a?(Pokemon)
367:   foeParty = [pkmn]
368:   # Calculate who the trainers and their party are
369:   playerTrainer     = [$player]
370:   playerParty       = $player.party
371:   playerPartyStarts = [0]
372:   # Create the battle scene (the visual side of it)
373:   scene = BattleCreationHelperMethods.create_battle_scene
374:   # Create the battle class (the mechanics side of it)
375:   battle = BugContestBattle.new(scene, playerParty, foeParty, playerTrainer, nil)
376:   battle.party1starts = playerPartyStarts
377:   battle.ballCount    = pbBugContestState.ballcount
378:   setBattleRule("single")
379:   BattleCreationHelperMethods.prepare_battle(battle)
380:   # Perform the battle itself
381:   decision = 0
382:   pbBattleAnimation(pbGetWildBattleBGM(foeParty), 0, foeParty) do
383:     decision = battle.pbStartBattle
384:     BattleCreationHelperMethods.after_battle(decision, true)
385:     if [2, 5].include?(decision)   # Lost or drew
386:       $game_system.bgm_unpause
387:       $game_system.bgs_unpause
388:       pbBugContestStartOver
389:     end
390:   end
391:   Input.update
392:   # Update Bug Contest game data based on result of battle
393:   pbBugContestState.ballcount = battle.ballCount
394:   if pbBugContestState.ballcount == 0
395:     pbMessage(_INTL("ANNOUNCER:  The Bug-Catching Contest is over!"))
396:     pbBugContestState.pbStartJudging
397:   end
398:   # Save the result of the battle in Game Variable 1
399:   BattleCreationHelperMethods.set_outcome(decision, 1)
400:   # Used by the Poké Radar to update/break the chain
401:   EventHandlers.trigger(:on_wild_battle_end, pkmn.species_data.id, pkmn.level, decision)
402:   # Return false if the player lost or drew the battle, and true if any other result
403:   return (decision != 2 && decision != 5)
404: end
