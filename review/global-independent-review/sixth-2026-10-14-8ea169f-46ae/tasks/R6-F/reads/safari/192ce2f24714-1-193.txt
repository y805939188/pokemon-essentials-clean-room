173:   def play_time
174:     if $game_temp&.last_uptime_refreshed_play_time
175:       now = System.uptime
176:       @play_time += now - $game_temp.last_uptime_refreshed_play_time
177:       $game_temp.last_uptime_refreshed_play_time = now
178:     end
179:     return @play_time
180:   end
