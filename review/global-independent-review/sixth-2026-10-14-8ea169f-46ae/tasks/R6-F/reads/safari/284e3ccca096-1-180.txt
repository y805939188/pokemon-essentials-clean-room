97:   def pbStartPokemonMenu
98:     if !$player
99:       if $DEBUG
100:         pbMessage(_INTL("The player trainer was not defined, so the pause menu can't be displayed."))
101:         pbMessage(_INTL("Please see the documentation to learn how to set up the trainer player."))
102:       end
103:       return
104:     end
105:     @scene.pbStartScene
106:     # Show extra info window if relevant
107:     pbShowInfo
108:     # Get all commands
109:     command_list = []
110:     commands = []
111:     MenuHandlers.each_available(:pause_menu) do |option, hash, name|
112:       command_list.push(name)
113:       commands.push(hash)
114:     end
115:     # Main loop
116:     end_scene = false
117:     loop do
118:       choice = @scene.pbShowCommands(command_list)
119:       if choice < 0
120:         pbPlayCloseMenuSE
121:         end_scene = true
122:         break
123:       end
124:       break if commands[choice]["effect"].call(@scene)
125:     end
126:     @scene.pbEndScene if end_scene
127:   end
128: end
