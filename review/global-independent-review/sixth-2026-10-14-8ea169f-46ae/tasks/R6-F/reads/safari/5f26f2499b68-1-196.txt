64:   if pbSafariState.inProgress?

66:     # map can be outdoors, with its own grassy patches.
67:     reception = pbSafariState.pbReceptionMap
68:     return true if $game_map.map_id == reception
69:     return true if $game_map.metadata&.safari_map
70:   end
71:   return false
72: end

107: EventHandlers.add(:on_calling_wild_battle, :safari_battle,
108:   proc { |pkmn, handled|
109:     # handled is an array: [nil]. If [true] or [false], the battle has already
110:     # been overridden (the boolean is its outcome), so don't do anything that
111:     # would override it again
112:     next if !handled[0].nil?
113:     next if !pbInSafari?
114:     handled[0] = pbSafariBattle(pkmn)
115:   }

118: def pbSafariBattle(pkmn, level = 1)
119:   # Generate a wild Pokémon based on the species and level
120:   pkmn = pbGenerateWildPokemon(pkmn, level) if !pkmn.is_a?(Pokemon)
121:   foeParty = [pkmn]
122:   # Calculate who the trainer is
123:   playerTrainer = $player
124:   # Create the battle scene (the visual side of it)
125:   scene = BattleCreationHelperMethods.create_battle_scene
126:   # Create the battle class (the mechanics side of it)
127:   battle = SafariBattle.new(scene, playerTrainer, foeParty)
128:   battle.ballCount = pbSafariState.ballcount
129:   BattleCreationHelperMethods.prepare_battle(battle)
130:   # Perform the battle itself
131:   decision = 0
132:   pbBattleAnimation(pbGetWildBattleBGM(foeParty), 0, foeParty) do
133:     pbSceneStandby { decision = battle.pbStartBattle }
134:   end
135:   Input.update
136:   # Update Safari game data based on result of battle
137:   pbSafariState.ballcount = battle.ballCount
138:   if pbSafariState.ballcount <= 0
139:     if decision != 2   # Last Safari Ball was used to catch the wild Pokémon
140:       pbMessage(_INTL("Announcer: You're out of Safari Balls! Game over!"))
141:     end
142:     pbSafariState.decision = 1
143:     pbSafariState.pbGoToStart
144:   end
145:   # Save the result of the battle in Game Variable 1
146:   #    0 - Undecided or aborted
147:   #    2 - Player ran out of Safari Balls
148:   #    3 - Player or wild Pokémon ran from battle, or player forfeited the match
149:   #    4 - Wild Pokémon was caught
150:   if decision == 4
151:     $stats.safari_pokemon_caught += 1
152:     pbSafariState.captures += 1
153:     $stats.most_captures_per_safari_game = [$stats.most_captures_per_safari_game, pbSafariState.captures].max
154:   end
155:   pbSet(1, decision)
156:   # Used by the Poké Radar to update/break the chain
157:   EventHandlers.trigger(:on_wild_battle_end, pkmn.species_data.id, pkmn.level, decision)
158:   # Return the outcome of the battle
159:   return decision
160: end
