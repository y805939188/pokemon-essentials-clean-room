101: SaveData.register_conversion(:v20_refactor_planted_berries_data) do
102:   essentials_version 20
103:   display_title "Updating berry plant data format"
104:   to_value :global_metadata do |global|
105:     if global.eventvars
106:       global.eventvars.each_pair do |key, value|
107:         next if !value || !value.is_a?(Array)
108:         case value.length
109:         when 6   # Old berry plant data
110:           data = BerryPlantData.new
111:           if value[1].is_a?(Symbol)
112:             plant_data = GameData::BerryPlant.get(value[1])
113:             data.new_mechanics      = false
114:             data.berry_id           = value[1]
115:             data.time_alive         = value[0] * plant_data.hours_per_stage * 3600
116:             data.time_last_updated  = value[3]
117:             data.growth_stage       = value[0]
118:             data.replant_count      = value[5]
119:             data.watered_this_stage = value[2]
120:             data.watering_count     = value[4]
121:           end
122:           global.eventvars[key] = data
123:         when 7, 8   # New berry plant data
124:           data = BerryPlantData.new
125:           if value[1].is_a?(Symbol)
126:             data.new_mechanics     = true
127:             data.berry_id          = value[1]
128:             data.mulch_id          = value[7] if value[7].is_a?(Symbol)
129:             data.time_alive        = value[2]
130:             data.time_last_updated = value[3]
131:             data.growth_stage      = value[0]
132:             data.replant_count     = value[5]
133:             data.moisture_level    = value[4]
134:             data.yield_penalty     = value[6]
135:           end
136:           global.eventvars[key] = data
137:         end
138:       end
139:     end
140:   end
141: end

145: SaveData.register_conversion(:v20_refactor_follower_data) do
146:   essentials_version 20
147:   display_title "Updating follower data format"
148:   to_value :global_metadata do |global|
149:     # NOTE: dependentEvents is still defined in class PokemonGlobalMetadata just
150:     #       for the sake of this conversion. It is deprecated and will be
151:     #       removed in v22.
152:     if global.dependentEvents && global.dependentEvents.length > 0
153:       global.followers = []
154:       global.dependentEvents.each do |follower|
155:         data = FollowerData.new(follower[0], follower[1], "reflection",
156:                                 follower[2], follower[3], follower[4],
157:                                 follower[5], follower[6], follower[7])
158:         data.name            = follower[8]
159:         data.common_event_id = follower[9]
160:         global.followers.push(data)
161:       end
162:     end
163:     global.dependentEvents = nil
164:   end
165: end
