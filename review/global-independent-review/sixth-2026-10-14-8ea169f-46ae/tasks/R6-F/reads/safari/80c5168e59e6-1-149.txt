52:   def can_run?
53:     return @move_speed > 3 if @move_route_forcing
54:     return false if @bumping
55:     return false if $game_temp.in_menu || $game_temp.in_battle ||
56:                     $game_temp.message_window_showing || pbMapInterpreterRunning?

64:   def set_movement_type(type)
65:     meta = GameData::PlayerMetadata.get($player&.character_ID || 1)
66:     new_charset = nil
67:     case type
68:     when :fishing
69:       new_charset = pbGetPlayerCharset(meta.fish_charset)
70:     when :surf_fishing
71:       new_charset = pbGetPlayerCharset(meta.surf_fish_charset)
72:     when :diving, :diving_fast, :diving_jumping, :diving_stopped
73:       self.move_speed = 3 if !@move_route_forcing
74:       new_charset = pbGetPlayerCharset(meta.dive_charset)
75:     when :surfing, :surfing_fast, :surfing_jumping, :surfing_stopped
76:       if !@move_route_forcing
77:         self.move_speed = (type == :surfing_jumping) ? 3 : 4
78:       end
79:       new_charset = pbGetPlayerCharset(meta.surf_charset)
80:     when :descending_waterfall, :ascending_waterfall
81:       self.move_speed = 2 if !@move_route_forcing
82:       new_charset = pbGetPlayerCharset(meta.surf_charset)
83:     when :cycling, :cycling_fast, :cycling_jumping, :cycling_stopped
84:       if !@move_route_forcing
85:         self.move_speed = (type == :cycling_jumping) ? 3 : 5
86:       end
87:       new_charset = pbGetPlayerCharset(meta.cycle_charset)
88:     when :running
89:       self.move_speed = 4 if !@move_route_forcing
90:       new_charset = pbGetPlayerCharset(meta.run_charset)
91:     when :ice_sliding
92:       self.move_speed = 4 if !@move_route_forcing
93:       new_charset = pbGetPlayerCharset(meta.walk_charset)
94:     else   # :walking, :jumping, :walking_stopped
95:       self.move_speed = 3 if !@move_route_forcing
96:       new_charset = pbGetPlayerCharset(meta.walk_charset)
97:     end
98:     self.move_speed = 3 if @bumping
99:     @character_name = new_charset if new_charset
100:   end

102:   # Called when the player's character or outfit changes. Assumes the player
103:   # isn't moving.
104:   def refresh_charset
105:     meta = GameData::PlayerMetadata.get($player&.character_ID || 1)
106:     new_charset = nil
107:     if $PokemonGlobal&.diving
108:       new_charset = pbGetPlayerCharset(meta.dive_charset)
109:     elsif $PokemonGlobal&.surfing
110:       new_charset = pbGetPlayerCharset(meta.surf_charset)
111:     elsif $PokemonGlobal&.bicycle
112:       new_charset = pbGetPlayerCharset(meta.cycle_charset)
113:     else
114:       new_charset = pbGetPlayerCharset(meta.walk_charset)
115:     end
116:     @character_name = new_charset if new_charset
117:   end
