204:   # Calculate how many shakes a thrown Poké Ball will make (4 = capture)
205:   #=============================================================================
206:   def pbCaptureCalc(pkmn, battler, catch_rate, ball)
207:     return 4 if $DEBUG && Input.press?(Input::CTRL)
208:     # Get a catch rate if one wasn't provided
209:     catch_rate = pkmn.species_data.catch_rate if !catch_rate
210:     # Modify catch_rate depending on the Poké Ball's effect
211:     if !pkmn.species_data.has_flag?("UltraBeast") || ball == :BEASTBALL
212:       catch_rate = Battle::PokeBallEffects.modifyCatchRate(ball, catch_rate, self, battler)
213:     else
214:       catch_rate /= 10
215:     end
216:     # First half of the shakes calculation
217:     a = battler.totalhp
218:     b = battler.hp
219:     x = (((3 * a) - (2 * b)) * catch_rate.to_f) / (3 * a)
220:     # Calculation modifiers
221:     if battler.status == :SLEEP || battler.status == :FROZEN
222:       x *= 2.5
223:     elsif battler.status != :NONE
224:       x *= 1.5
225:     end
226:     x = x.floor
227:     x = 1 if x < 1
228:     # Definite capture, no need to perform randomness checks
229:     return 4 if x >= 255 || Battle::PokeBallEffects.isUnconditional?(ball, self, battler)
230:     # Second half of the shakes calculation
231:     y = (65_536 / ((255.0 / x)**0.1875)).floor
232:     # Critical capture check
233:     if Settings::ENABLE_CRITICAL_CAPTURES
234:       dex_modifier = 0
235:       numOwned = $player.pokedex.owned_count
236:       if numOwned > 600
237:         dex_modifier = 5
238:       elsif numOwned > 450
239:         dex_modifier = 4
240:       elsif numOwned > 300
241:         dex_modifier = 3
242:       elsif numOwned > 150
243:         dex_modifier = 2
244:       elsif numOwned > 30
245:         dex_modifier = 1
246:       end
247:       dex_modifier *= 2 if $bag.has?(:CATCHINGCHARM)
248:       c = x * dex_modifier / 12
249:       # Calculate the number of shakes
250:       if c > 0 && pbRandom(256) < c
251:         @criticalCapture = true
252:         return 4 if pbRandom(65_536) < y
253:         return 0
254:       end
255:     end
256:     # Calculate the number of shakes
257:     numShakes = 0
258:     4.times do |i|
259:       break if numShakes < i
260:       numShakes += 1 if pbRandom(65_536) < y
261:     end
262:     return numShakes
263:   end
