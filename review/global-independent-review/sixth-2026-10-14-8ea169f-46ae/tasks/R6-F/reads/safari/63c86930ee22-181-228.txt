199: def pbRoamingPokemonBattle(pkmn, level = 1)
200:   # Get the roaming Pokémon to encounter; generate it based on the species and
201:   # level if it doesn't already exist
202:   idxRoamer = $game_temp.roamer_index_for_encounter
203:   if !$PokemonGlobal.roamPokemon[idxRoamer] ||
204:      !$PokemonGlobal.roamPokemon[idxRoamer].is_a?(Pokemon)
205:     if pkmn.is_a?(Pokemon)
206:       $PokemonGlobal.roamPokemon[idxRoamer] = pbGenerateWildPokemon(pkmn.species_data.id, pkmn.level, true)
207:     else
208:       $PokemonGlobal.roamPokemon[idxRoamer] = pbGenerateWildPokemon(pkmn, level, true)
209:     end
210:   end
211:   # Set some battle rules
212:   setBattleRule("single")
213:   setBattleRule("roamerFlees")
214:   # Perform the battle
215:   decision = WildBattle.start_core($PokemonGlobal.roamPokemon[idxRoamer])
216:   # Update Roaming Pokémon data based on result of battle
217:   if [1, 4].include?(decision)   # Defeated or caught
218:     $PokemonGlobal.roamPokemon[idxRoamer]       = true
219:     $PokemonGlobal.roamPokemonCaught[idxRoamer] = (decision == 4)
220:   end
221:   $PokemonGlobal.roamEncounter = nil
222:   $PokemonGlobal.roamedAlready = true
223:   $game_temp.roamer_index_for_encounter = nil
224:   # Used by the Poké Radar to update/break the chain
225:   EventHandlers.trigger(:on_wild_battle_end, pkmn.species_data.id, pkmn.level, decision)
