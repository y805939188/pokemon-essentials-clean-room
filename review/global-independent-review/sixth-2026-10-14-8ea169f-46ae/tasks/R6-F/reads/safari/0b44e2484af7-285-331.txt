292: class Spriteset_Map
293:   attr_reader :usersprites
294: 
295:   alias _animationSprite_initialize initialize unless private_method_defined?(:_animationSprite_initialize)
296:   alias _animationSprite_update update unless method_defined?(:_animationSprite_update)
297:   alias _animationSprite_dispose dispose unless method_defined?(:_animationSprite_dispose)
298: 
299:   def initialize(map = nil)
300:     @usersprites = []
301:     _animationSprite_initialize(map)
302:   end
303: 
304:   def addUserAnimation(animID, x, y, tinting = false, height = 3)
305:     sprite = AnimationContainerSprite.new(animID, self.map, x, y, @@viewport1, tinting, height)
306:     addUserSprite(sprite)
307:     return sprite
308:   end
309: 
310:   def addUserSprite(new_sprite)
311:     @usersprites.each_with_index do |sprite, i|
312:       next if sprite && !sprite.disposed?
313:       @usersprites[i] = new_sprite
314:       return
315:     end
316:     @usersprites.push(new_sprite)
317:   end
318: 
319:   def dispose
320:     _animationSprite_dispose
321:     @usersprites.each { |sprite| sprite.dispose }
322:     @usersprites.clear
323:   end
324: 
325:   def update
326:     @@viewport3.tone.set(0, 0, 0, 0)
327:     _animationSprite_update
328:     @usersprites.each { |sprite| sprite.update if !sprite.disposed? }
329:     @usersprites.delete_if { |sprite| sprite.disposed? }
330:   end
331: end
