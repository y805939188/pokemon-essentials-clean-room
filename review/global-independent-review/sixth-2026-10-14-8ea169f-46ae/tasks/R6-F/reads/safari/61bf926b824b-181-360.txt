227:   def getRelativePos(thisMapID, thisX, thisY, otherMapID, otherX, otherY)
228:     if thisMapID == otherMapID   # Both events share the same map
229:       return [otherX - thisX, otherY - thisY]
230:     end
231:     MapFactoryHelper.eachConnectionForMap(thisMapID) do |conn|
232:       if conn[0] == otherMapID
233:         posX = conn[4] - conn[1] + otherX - thisX
234:         posY = conn[5] - conn[2] + otherY - thisY
235:         return [posX, posY]
236:       elsif conn[3] == otherMapID
237:         posX =  conn[1] - conn[4] + otherX - thisX
238:         posY =  conn[2] - conn[5] + otherY - thisY
239:         return [posX, posY]

241:     end

307:   def getRealTilePos(mapID, x, y)
308:     id = mapID
309:     return [id, x, y] if getMapNoAdd(id).valid?(x, y)
310:     MapFactoryHelper.eachConnectionForMap(id) do |conn|
311:       if conn[0] == id
312:         newX = x + conn[4] - conn[1]
313:         newY = y + conn[5] - conn[2]
314:         next if newX < 0 || newY < 0
315:         dims = MapFactoryHelper.getMapDims(conn[3])
316:         next if newX >= dims[0] || newY >= dims[1]
317:         return [conn[3], newX, newY]
318:       else
319:         newX = x + conn[1] - conn[4]
320:         newY = y + conn[2] - conn[5]
321:         next if newX < 0 || newY < 0
322:         dims = MapFactoryHelper.getMapDims(conn[0])
323:         next if newX >= dims[0] || newY >= dims[1]
324:         return [conn[0], newX, newY]
325:       end
326:     end
327:     return nil
328:   end
