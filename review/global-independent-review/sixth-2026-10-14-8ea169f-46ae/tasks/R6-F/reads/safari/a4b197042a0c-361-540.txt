412:     ret = []
413:     species_id = nil
414:     args.each do |arg|
415:       case arg
416:       when Pokemon
417:         raise _INTL("Species {1} was given but not a level.", species_id) if species_id
418:         ret.push(arg)
419:       when Array
420:         raise _INTL("Species {1} was given but not a level.", species_id) if species_id
421:         species = GameData::Species.get(arg[0]).id
422:         pkmn = pbGenerateWildPokemon(species, arg[1])
423:         ret.push(pkmn)
424:       else
425:         if species_id   # Expecting level
426:           if !arg.is_a?(Integer) || !(1..GameData::GrowthRate.max_level).include?(arg)
427:             raise _INTL("Expected a level (1..{1}) but {2} is not a number or not a valid level.", GameData::GrowthRate.max_level, arg)
428:           end
429:           ret.push(pbGenerateWildPokemon(species_id, arg))
430:           species_id = nil
431:         else   # Expecting species ID
432:           if !GameData::Species.exists?(arg)
433:             raise _INTL("Species {1} does not exist.", arg)
434:           end
435:           species_id = arg
436:         end
437:       end
438:     end
439:     raise _INTL("Species {1} was given but not a level.", species_id) if species_id
440:     return ret

448:   # Used by most trainer events, which can be positioned in such a way that
449:   # multiple trainer events spot the player at once. The extra code in this
450:   # method deals with that case and can cause a double trainer battle instead.
451:   def self.start(*args)
452:     # If there is another NPC trainer who spotted the player at the same time,
453:     # and it is possible to have a double battle (the player has 2+ able Pokémon
454:     # or has a partner trainer), then record this first NPC trainer into
455:     # $game_temp.waiting_trainer and end this method. That second NPC event will
456:     # then trigger and cause the battle to happen against this first trainer and
457:     # themselves.
458:     if !$game_temp.waiting_trainer && pbMapInterpreterRunning? && pbCanDoubleBattle?
459:       thisEvent = pbMapInterpreter.get_self
460:       # Find all other triggered trainer events
461:       triggeredEvents = $game_player.pbTriggeredTrainerEvents([2], false, true)
462:       otherEvent = []
463:       triggeredEvents.each do |i|
464:         next if i.id == thisEvent.id
465:         next if $game_self_switches[[$game_map.map_id, i.id, "A"]]
466:         otherEvent.push(i)
467:       end
468:       # If there is exactly 1 other triggered trainer event, this trainer can be
469:       # stored up to battle with that one
470:       if otherEvent.length == 1
471:         trainers, _items, _end_speeches, _party, _party_starts = TrainerBattle.generate_foes(*args)
472:         # If this is just 1 trainer with 6 or fewer Pokémon, it can be stored up
473:         # to battle alongside the other trainer
474:         if trainers.length == 1 && trainers[0].party.length <= Settings::MAX_PARTY_SIZE
475:           $game_temp.waiting_trainer = [trainers[0], thisEvent.id]
476:           return false
477:         end
478:       end
479:     end
480:     # Perform the battle
481:     if $game_temp.waiting_trainer
482:       new_args = args + [$game_temp.waiting_trainer[0]]
483:       outcome = TrainerBattle.start_core(*new_args)
484:       pbMapInterpreter.pbSetSelfSwitch($game_temp.waiting_trainer[1], "A", true) if outcome == 1
485:       $game_temp.waiting_trainer = nil
486:     else
487:       outcome = TrainerBattle.start_core(*args)
488:     end
489:     # Return true if the player won the battle, and false if any other result
490:     return outcome == 1
491:   end
