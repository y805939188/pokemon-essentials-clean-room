5:   def pbCanUseItemOnPokemon?(item, pkmn, battler, scene, showMessages = true)
6:     if !pkmn || pkmn.egg?
7:       scene.pbDisplay(_INTL("It won't have any effect.")) if showMessages
8:       return false
9:     end
10:     # Embargo
11:     if battler && battler.effects[PBEffects::Embargo] > 0
12:       if showMessages
13:         scene.pbDisplay(_INTL("Embargo's effect prevents the item's use on {1}!",
14:                               battler.pbThis(true)))
15:       end
16:       return false
17:     end
18:     # Hyper Mode and non-Scents
19:     if pkmn.hyper_mode && !GameData::Item.get(item)&.is_scent?
20:       scene.pbDisplay(_INTL("It won't have any effect.")) if showMessages
21:       return false
22:     end
23:     return true
24:   end

34:   def pbRegisterItem(idxBattler, item, idxTarget = nil, idxMove = nil)
35:     # Register for use of item on a Pokémon in the party
36:     @choices[idxBattler][0] = :UseItem
37:     @choices[idxBattler][1] = item        # ID of item to be used
38:     @choices[idxBattler][2] = idxTarget   # Party index of Pokémon to use item on
39:     @choices[idxBattler][3] = idxMove     # Index of move to recharge (Ethers)
40:     # Delete the item from the Bag. If it turns out it will have no effect, it
41:     # will be re-added to the Bag later.
42:     pbConsumeItemInBag(item, idxBattler)
43:     return true
44:   end

48:   #=============================================================================
49:   def pbConsumeItemInBag(item, idxBattler)
50:     return if !item
51:     return if !GameData::Item.get(item).consumed_after_use?
52:     if pbOwnedByPlayer?(idxBattler)
53:       if !$bag.remove(item)
54:         raise _INTL("Tried to consume item that wasn't in the Bag somehow.")
55:       end
56:     else
57:       items = pbGetOwnerItems(idxBattler)
58:       items.delete_at(items.index(item)) if items
59:     end
60:   end
61: 
62:   def pbReturnUnusedItemToBag(item, idxBattler)
63:     return if !item
64:     return if !GameData::Item.get(item).consumed_after_use?
65:     if pbOwnedByPlayer?(idxBattler)
66:       if $bag&.can_add?(item)
67:         $bag.add(item)
68:       else
69:         raise _INTL("Couldn't return unused item to Bag somehow.")
70:       end
71:     else
72:       items = pbGetOwnerItems(idxBattler)
73:       items&.push(item)
74:     end
75:   end

86:   # Uses an item on a Pokémon in the trainer's party.
87:   def pbUseItemOnPokemon(item, idxParty, userBattler)
88:     trainerName = pbGetOwnerName(userBattler.index)
89:     pbUseItemMessage(item, trainerName)
90:     pkmn = pbParty(userBattler.index)[idxParty]
91:     battler = pbFindBattler(idxParty, userBattler.index)
92:     ch = @choices[userBattler.index]
93:     if ItemHandlers.triggerCanUseInBattle(item, pkmn, battler, ch[3], true, self, @scene, false)
94:       ItemHandlers.triggerBattleUseOnPokemon(item, pkmn, battler, ch, @scene)
95:       ch[1] = nil   # Delete item from choice
96:       return
97:     end
98:     pbDisplay(_INTL("But it had no effect!"))
99:     # Return unused item to Bag
100:     pbReturnUnusedItemToBag(item, userBattler.index)
101:   end

103:   # Uses an item on a Pokémon in battle that belongs to the trainer.
104:   def pbUseItemOnBattler(item, idxParty, userBattler)

106:     pbUseItemMessage(item, trainerName)
107:     battler = pbFindBattler(idxParty, userBattler.index)
108:     ch = @choices[userBattler.index]

110:       if ItemHandlers.triggerCanUseInBattle(item, battler.pokemon, battler, ch[3], true, self, @scene, false)
111:         ItemHandlers.triggerBattleUseOnBattler(item, battler, @scene)
112:         ch[1] = nil   # Delete item from choice
113:         battler.pbItemOnStatDropped

115:       else
116:         pbDisplay(_INTL("But it had no effect!"))
117:       end
118:     else
119:       pbDisplay(_INTL("But it's not where this item can be used!"))
120:     end
121:     # Return unused item to Bag
122:     pbReturnUnusedItemToBag(item, userBattler.index)
123:   end
