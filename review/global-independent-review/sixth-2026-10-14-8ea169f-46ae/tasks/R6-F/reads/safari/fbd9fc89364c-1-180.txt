100:   # Returns whether a wild encounter should happen, based on its encounter
101:   # chance. Called when taking a step and by Rock Smash.
102:   def encounter_triggered?(enc_type, repel_active = false, triggered_by_step = true)
103:     if !enc_type || !GameData::EncounterType.exists?(enc_type)
104:       raise ArgumentError.new(_INTL("Encounter type {1} does not exist", enc_type))
105:     end
106:     return false if $game_system.encounter_disabled
107:     return false if !$player
108:     return false if $DEBUG && Input.press?(Input::CTRL)
109:     # Check if enc_type has a defined step chance/encounter table
110:     return false if !@step_chances[enc_type] || @step_chances[enc_type] == 0
111:     return false if !has_encounter_type?(enc_type)
112:     # Poké Radar encounters always happen, ignoring the minimum step period and
113:     # trigger probabilities
114:     return true if pbPokeRadarOnShakingGrass
115:     # Get base encounter chance and minimum steps grace period
116:     encounter_chance = @step_chances[enc_type].to_f
117:     min_steps_needed = (8 - (encounter_chance / 10)).clamp(0, 8).to_f
118:     # Apply modifiers to the encounter chance and the minimum steps amount
119:     if triggered_by_step
120:       encounter_chance += @chance_accumulator / 200
121:       encounter_chance *= 0.8 if $PokemonGlobal.bicycle
122:     end
123:     if $PokemonMap.lower_encounter_rate
124:       encounter_chance /= 2
125:       min_steps_needed *= 2
126:     elsif $PokemonMap.higher_encounter_rate
127:       encounter_chance *= 1.5
128:       min_steps_needed /= 2
129:     end
130:     first_pkmn = $player.first_pokemon
131:     if first_pkmn
132:       case first_pkmn.item_id
133:       when :CLEANSETAG
134:         encounter_chance *= 2.0 / 3
135:         min_steps_needed *= 4 / 3.0
136:       when :PUREINCENSE
137:         encounter_chance *= 2.0 / 3
138:         min_steps_needed *= 4 / 3.0
139:       else   # Ignore ability effects if an item effect applies
140:         case first_pkmn.ability_id
141:         when :STENCH, :WHITESMOKE, :QUICKFEET
142:           encounter_chance /= 2
143:           min_steps_needed *= 2
144:         when :INFILTRATOR
145:           if Settings::MORE_ABILITIES_AFFECT_WILD_ENCOUNTERS
146:             encounter_chance /= 2
147:             min_steps_needed *= 2
148:           end
149:         when :SNOWCLOAK
150:           if GameData::Weather.get($game_screen.weather_type).category == :Hail
151:             encounter_chance /= 2
152:             min_steps_needed *= 2
153:           end
154:         when :SANDVEIL
155:           if GameData::Weather.get($game_screen.weather_type).category == :Sandstorm
156:             encounter_chance /= 2
157:             min_steps_needed *= 2
158:           end
159:         when :SWARM
160:           encounter_chance *= 1.5
161:           min_steps_needed /= 2
162:         when :ILLUMINATE, :ARENATRAP, :NOGUARD
163:           encounter_chance *= 2
164:           min_steps_needed /= 2
165:         end
166:       end
167:     end
168:     # Wild encounters are much less likely to happen for the first few steps
169:     # after a previous wild encounter
170:     if triggered_by_step && @step_count < min_steps_needed
171:       @step_count += 1
172:       return false if rand(100) >= encounter_chance * 5 / (@step_chances[enc_type] + (@chance_accumulator / 200))
173:     end
174:     # Decide whether the wild encounter should actually happen
175:     return true if rand(100) < encounter_chance
176:     # If encounter didn't happen, make the next step more likely to produce one
177:     if triggered_by_step
178:       @chance_accumulator += @step_chances[enc_type]
179:       @chance_accumulator = 0 if repel_active
180:     end
