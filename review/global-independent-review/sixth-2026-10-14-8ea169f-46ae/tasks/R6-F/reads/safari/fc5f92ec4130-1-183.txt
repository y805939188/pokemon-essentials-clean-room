127: # This class keeps track of erased and moved events so their position
128: # can remain after a game is saved and loaded.  This class also includes
129: # variables that should remain valid only for the current map.
130: #===============================================================================
131: class PokemonMapMetadata
132:   attr_reader   :erasedEvents
133:   attr_reader   :movedEvents
134:   attr_accessor :strengthUsed
135:   attr_accessor :lower_encounter_rate    # Black Flute's old effect
136:   attr_accessor :higher_encounter_rate   # White Flute's old effect
137:   attr_accessor :lower_level_wild_pokemon    # White Flute's new effect
138:   attr_accessor :higher_level_wild_pokemon   # Black Flute's new effect
139: 
140:   def initialize
141:     clear
142:   end
143: 
144:   def clear
145:     @erasedEvents              = {}
146:     @movedEvents               = {}
147:     @strengthUsed              = false
148:     @lower_encounter_rate      = false   # Takes priority over @higher_encounter_rate
149:     @higher_encounter_rate     = false
150:     @lower_level_wild_pokemon  = false   # Takes priority over @higher_level_wild_pokemon
151:     @higher_level_wild_pokemon = false
152:   end
153: 
154:   def addErasedEvent(eventID)
155:     key = [$game_map.map_id, eventID]
156:     @erasedEvents[key] = true
157:   end
158: 
159:   def addMovedEvent(eventID)
160:     key = [$game_map.map_id, eventID]
161:     event = $game_map.events[eventID] if eventID.is_a?(Integer)
162:     @movedEvents[key] = [event.x, event.y, event.direction, event.through] if event
163:   end
164: 
165:   def updateMap
166:     @erasedEvents.each do |i|
167:       $game_map.events[i[0][1]]&.erase if i[0][0] == $game_map.map_id && i[1]
168:     end
169:     @movedEvents.each do |i|
170:       if i[0][0] == $game_map.map_id && i[1]
171:         next if !$game_map.events[i[0][1]]
172:         $game_map.events[i[0][1]].moveto(i[1][0], i[1][1])
173:         case i[1][2]
174:         when 2 then $game_map.events[i[0][1]].turn_down
175:         when 4 then $game_map.events[i[0][1]].turn_left
176:         when 6 then $game_map.events[i[0][1]].turn_right
177:         when 8 then $game_map.events[i[0][1]].turn_up
178:         end
179:       end
180:       $game_map.events[i[0][1]].through = i[1][3] if i[1][3]
181:     end
182:   end
183: end
