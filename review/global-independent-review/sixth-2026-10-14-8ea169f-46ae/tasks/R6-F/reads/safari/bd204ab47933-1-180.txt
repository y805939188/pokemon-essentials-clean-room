92: #===============================================================================
93: class Battle::Scene::Animation::ThrowBait < Battle::Scene::Animation
94:   include Battle::Scene::Animation::BallAnimationMixin
95: 
96:   def initialize(sprites, viewport, battler)
97:     @battler = battler
98:     @trainer = battler.battle.pbGetOwnerFromBattlerIndex(battler.index)
99:     super(sprites, viewport)
100:   end
101: 
102:   def createProcesses
103:     # Calculate start and end coordinates for battler sprite movement
104:     batSprite = @sprites["pokemon_#{@battler.index}"]
105:     traSprite = @sprites["player_1"]
106:     ballPos = Battle::Scene.pbBattlerPosition(@battler.index, batSprite.sideSize)
107:     ballStartX = traSprite.x
108:     ballStartY = traSprite.y - (traSprite.bitmap.height / 2)
109:     ballMidX   = 0   # Unused in arc calculation
110:     ballMidY   = 122
111:     ballEndX   = ballPos[0] - 40
112:     ballEndY   = ballPos[1] - 4
113:     # Set up trainer sprite
114:     trainer = addSprite(traSprite, PictureOrigin::BOTTOM)
115:     # Set up bait sprite
116:     ball = addNewSprite(ballStartX, ballStartY,
117:                         "Graphics/Battle animations/safari_bait", PictureOrigin::CENTER)
118:     ball.setZ(0, batSprite.z + 1)
119:     # Trainer animation
120:     if traSprite.bitmap.width >= traSprite.bitmap.height * 2
121:       ballStartX, ballStartY = trainerThrowingFrames(ball, trainer, traSprite)
122:     end
123:     delay = ball.totalDuration   # 0 or 7
124:     # Bait arc animation
125:     ball.setSE(delay, "Battle throw")
126:     createBallTrajectory(ball, delay, 12,
127:                          ballStartX, ballStartY, ballMidX, ballMidY, ballEndX, ballEndY)
128:     ball.setZ(9, batSprite.z + 1)
129:     delay = ball.totalDuration
130:     ball.moveOpacity(delay + 8, 2, 0)
131:     ball.setVisible(delay + 10, false)
132:     # Set up battler sprite
133:     battler = addSprite(batSprite, PictureOrigin::BOTTOM)
134:     # Show Pokémon jumping before eating the bait
135:     delay = ball.totalDuration + 3
136:     2.times do
137:       battler.setSE(delay, "player jump")
138:       battler.moveDelta(delay, 3, 0, -16)
139:       battler.moveDelta(delay + 4, 3, 0, 16)
140:       delay = battler.totalDuration + 1
141:     end
142:     # Show Pokémon eating the bait
143:     delay = battler.totalDuration + 3
144:     2.times do
145:       battler.moveAngle(delay, 7, 5)
146:       battler.moveDelta(delay, 7, 0, 6)
147:       battler.moveAngle(delay + 7, 7, 0)
148:       battler.moveDelta(delay + 7, 7, 0, -6)
149:       delay = battler.totalDuration
150:     end
151:   end
152: end
