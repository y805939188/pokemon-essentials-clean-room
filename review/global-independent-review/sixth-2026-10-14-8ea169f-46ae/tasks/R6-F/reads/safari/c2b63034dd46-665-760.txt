685: def pbSurf
686:   return false if !$game_player.can_ride_vehicle_with_follower?
687:   move = :SURF
688:   movefinder = $player.get_pokemon_with_move(move)
689:   if !pbCheckHiddenMoveBadge(Settings::BADGE_FOR_SURF, false) || (!$DEBUG && !movefinder)
690:     return false
691:   end
692:   if pbConfirmMessage(_INTL("The water is a deep blue color... Would you like to use Surf on it?"))
693:     speciesname = (movefinder) ? movefinder.name : $player.name
694:     pbMessage(_INTL("{1} used {2}!", speciesname, GameData::Move.get(move).name))
695:     pbCancelVehicles
696:     pbHiddenMoveAnimation(movefinder)
697:     surfbgm = GameData::Metadata.get.surf_BGM
698:     pbCueBGM(surfbgm, 0.5) if surfbgm
699:     pbStartSurfing
700:     return true
701:   end
702:   return false
703: end
