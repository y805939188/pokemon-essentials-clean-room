2: # Entering/exiting cave animations
3: #===============================================================================
4: def pbCaveEntranceEx(exiting)
5:   # Create bitmap
6:   sprite = BitmapSprite.new(Graphics.width, Graphics.height)
7:   sprite.z = 100000
8:   # Define values used for the animation
9:   duration = 0.4
10:   totalBands = 15
11:   bandheight = ((Graphics.height / 2.0) - 10) / totalBands
12:   bandwidth  = ((Graphics.width / 2.0) - 12) / totalBands
13:   start_gray = (exiting) ? 0 : 255
14:   end_gray = (exiting) ? 255 : 0
15:   # Create initial array of band colors (black if exiting, white if entering)
16:   grays = Array.new(totalBands) { |i| start_gray }
17:   # Animate bands changing color
18:   timer_start = System.uptime
19:   until System.uptime - timer_start >= duration
20:     x = 0
21:     y = 0
22:     # Calculate color of each band
23:     totalBands.times do |k|
24:       grays[k] = lerp(start_gray, end_gray, duration, timer_start + (k * duration / totalBands), System.uptime)
25:     end
26:     # Draw gray rectangles
27:     rectwidth  = Graphics.width
28:     rectheight = Graphics.height
29:     totalBands.times do |i|
30:       currentGray = grays[i]
31:       sprite.bitmap.fill_rect(Rect.new(x, y, rectwidth, rectheight),
32:                               Color.new(currentGray, currentGray, currentGray))
33:       x += bandwidth
34:       y += bandheight
35:       rectwidth  -= bandwidth * 2
36:       rectheight -= bandheight * 2
37:     end
38:     Graphics.update
39:     Input.update
40:   end
41:   # Set the tone at end of band animation
42:   if exiting
43:     pbToneChangeAll(Tone.new(255, 255, 255), 0)
44:   else
45:     pbToneChangeAll(Tone.new(-255, -255, -255), 0)
46:   end
47:   # Animate fade to white (if exiting) or black (if entering)
48:   timer_start = System.uptime
49:   loop do
50:     sprite.color = Color.new(end_gray, end_gray, end_gray,
51:                              lerp(0, 255, duration, timer_start, System.uptime))
52:     Graphics.update
53:     Input.update
54:     break if sprite.color.alpha >= 255
55:   end
56:   # Set the tone at end of fading animation
57:   pbToneChangeAll(Tone.new(0, 0, 0), 8)
58:   # Pause briefly
59:   timer_start = System.uptime
60:   until System.uptime - timer_start >= 0.1
61:     Graphics.update
62:     Input.update
63:   end
64:   sprite.dispose
65: end

69:   pbCaveEntranceEx(false)
70: end

74:   pbCaveEntranceEx(true)
