264: # Returns a score for this Pokemon in the Bug-Catching Contest.
265: # Not exactly the HGSS calculation, but it should be decent enough.
266: def pbBugContestScore(pkmn)
267:   levelscore = pkmn.level * 4
268:   ivscore = 0
269:   pkmn.iv.each_value { |iv| ivscore += iv.to_f / Pokemon::IV_STAT_LIMIT }
270:   ivscore = (ivscore * 100).floor
271:   hpscore = (100.0 * pkmn.hp / pkmn.totalhp).floor
272:   catch_rate = pkmn.species_data.catch_rate
273:   rarescore = 60
274:   rarescore += 20 if catch_rate <= 120
275:   rarescore += 20 if catch_rate <= 60
276:   return levelscore + ivscore + hpscore + rarescore
277: end
