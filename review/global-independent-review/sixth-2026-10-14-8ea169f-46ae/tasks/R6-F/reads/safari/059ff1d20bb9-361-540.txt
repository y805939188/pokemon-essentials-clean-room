371: # Returns whether event is able to walk up to the player.
372: def pbEventCanReachPlayer?(event, player, distance)
373:   return false if !pbEventFacesPlayer?(event, player, distance)
374:   delta_x = (event.direction == 6) ? 1 : (event.direction == 4) ? -1 : 0
375:   delta_y = (event.direction == 2) ? 1 : (event.direction == 8) ? -1 : 0
376:   case event.direction
377:   when 2   # Down
378:     real_distance = player.y - event.y - 1
379:   when 4   # Left
380:     real_distance = event.x - player.x - 1
381:   when 6   # Right
382:     real_distance = player.x - event.x - event.width
383:   when 8   # Up
384:     real_distance = event.y - event.height - player.y
385:   end
386:   if real_distance > 0
387:     real_distance.times do |i|
388:       return false if !event.can_move_from_coordinate?(event.x + (i * delta_x), event.y + (i * delta_y), event.direction)
389:     end
390:   end
391:   return true
392: end

489: def pbMoveRoute(event, commands, waitComplete = false)
490:   route = RPG::MoveRoute.new
491:   route.repeat    = false
492:   route.skippable = true
493:   route.list.clear
494:   route.list.push(RPG::MoveCommand.new(PBMoveRoute::THROUGH_ON))
495:   i = 0
496:   while i < commands.length
497:     case commands[i]
498:     when PBMoveRoute::WAIT, PBMoveRoute::SWITCH_ON, PBMoveRoute::SWITCH_OFF,
499:        PBMoveRoute::CHANGE_SPEED, PBMoveRoute::CHANGE_FREQUENCY, PBMoveRoute::OPACITY,
500:        PBMoveRoute::BLENDING, PBMoveRoute::PLAY_SE, PBMoveRoute::SCRIPT
501:       route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1]]))
502:       i += 1
503:     when PBMoveRoute::SCRIPT_ASYNC
504:       route.list.push(RPG::MoveCommand.new(PBMoveRoute::SCRIPT, [commands[i + 1]]))
505:       route.list.push(RPG::MoveCommand.new(PBMoveRoute::WAIT, [0]))
506:       i += 1
507:     when PBMoveRoute::JUMP
508:       route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1], commands[i + 2]]))
509:       i += 2
510:     when PBMoveRoute::GRAPHIC
511:       route.list.push(RPG::MoveCommand.new(commands[i],
512:                                            [commands[i + 1], commands[i + 2],
513:                                             commands[i + 3], commands[i + 4]]))
514:       i += 4
515:     else
516:       route.list.push(RPG::MoveCommand.new(commands[i]))
517:     end
518:     i += 1
519:   end
520:   route.list.push(RPG::MoveCommand.new(PBMoveRoute::THROUGH_OFF))
521:   route.list.push(RPG::MoveCommand.new(0))
522:   event&.force_move_route(route)
523:   return route
524: end
