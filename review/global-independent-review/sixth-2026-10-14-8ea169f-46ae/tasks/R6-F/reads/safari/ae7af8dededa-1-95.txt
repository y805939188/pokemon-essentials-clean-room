32:       scene.pbDisplay(_INTL("There is no room left in the PC!")) if showMessages

40:     #       below is one half of making this happen; the other half is in def
41:     #       pbItemUsesAllActions?.
42:     if !firstAction
43:       scene.pbDisplay(_INTL("It's impossible to aim without being focused!")) if showMessages
44:       next false
45:     end
46:     if battler.semiInvulnerable?
47:       scene.pbDisplay(_INTL("It's no good! It's impossible to aim at a Pokémon that's not in sight!")) if showMessages
48:       next false
49:     end
50:     # NOTE: The code below stops you from throwing a Poké Ball if there is more
51:     #       than one unfainted opposing Pokémon. (Snag Balls can be thrown in
52:     #       this case, but only in trainer battles, and the trainer will deflect
53:     #       them if they are trying to catch a non-Shadow Pokémon.)
54:     if battle.pbOpposingBattlerCount > 1 && !(GameData::Item.get(item).is_snag_ball? && battle.trainerBattle?)

56:         scene.pbDisplay(_INTL("It's no good! It's impossible to aim when there are two Pokémon!")) if showMessages
57:       elsif showMessages
58:         scene.pbDisplay(_INTL("It's no good! It's impossible to aim when there is more than one Pokémon!"))
