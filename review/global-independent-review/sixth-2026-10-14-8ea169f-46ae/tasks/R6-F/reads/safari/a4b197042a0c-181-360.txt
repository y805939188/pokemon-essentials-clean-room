331:   # Save the result of the battle in a Game Variable (1 by default)
332:   #    0 - Undecided or aborted
333:   #    1 - Player won

336:   #    4 - Wild Pokémon was caught

338:   def set_outcome(outcome, outcome_variable = 1, trainer_battle = false)
339:     case outcome
340:     when 1, 4   # Won, caught
341:       $stats.wild_battles_won += 1 if !trainer_battle
342:       $stats.trainer_battles_won += 1 if trainer_battle
343:     when 2, 3, 5   # Lost, fled, draw
344:       $stats.wild_battles_lost += 1 if !trainer_battle
345:       $stats.trainer_battles_lost += 1 if trainer_battle
346:     end
347:     pbSet(outcome_variable, outcome)
348:   end
