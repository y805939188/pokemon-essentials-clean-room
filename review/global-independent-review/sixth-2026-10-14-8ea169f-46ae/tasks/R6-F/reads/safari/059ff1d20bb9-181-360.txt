326: # NOTE: Assumes the event is 1x1 tile in size. Only returns one tile.
327: def pbFacingTileRegular(direction = nil, event = nil)
328:   event = $game_player if !event

330:   x = event.x

338: # Returns whether event is in line with the player, is facing the player and is
339: # within distance tiles of the player.
340: def pbEventFacesPlayer?(event, player, distance)
341:   return false if !event || !player || distance <= 0
342:   x_min = x_max = y_min = y_max = -1
343:   case event.direction
344:   when 2   # Down
345:     x_min = event.x
346:     x_max = event.x + event.width - 1
347:     y_min = event.y + 1
348:     y_max = event.y + distance
349:   when 4   # Left
350:     x_min = event.x - distance
351:     x_max = event.x - 1
352:     y_min = event.y - event.height + 1
353:     y_max = event.y
354:   when 6   # Right

356:     x_max = event.x + event.width - 1 + distance
