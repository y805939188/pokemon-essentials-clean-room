615:       # Healer, Hydration, Shed Skin
616:       if battler.abilityActive?
617:         Battle::AbilityEffects.triggerEndOfRoundHealing(battler.ability, battler, self)
618:       end
619:       # Black Sludge, Leftovers
620:       if battler.itemActive?
621:         Battle::ItemEffects.triggerEndOfRoundHealing(battler.item, battler, self)
622:       end
623:     end

625:     if Settings::AFFECTION_EFFECTS && @internalBattle
626:       priority.each do |battler|

628:         next if !battler.pbOwnedByPlayer? || battler.affection_level < 4 || battler.mega?
629:         next if pbRandom(100) < 80
630:         old_status = battler.status
631:         battler.pbCureStatus(false)
632:         case old_status
633:         when :SLEEP

635:         when :POISON
636:           pbDisplay(_INTL("{1} managed to expel the poison so you wouldn't worry!", battler.pbThis))

638:           pbDisplay(_INTL("{1} healed its burn with its sheer determination so you wouldn't worry!", battler.pbThis))
639:         when :PARALYSIS
640:           pbDisplay(_INTL("{1} gathered all its energy to break through its paralysis so you wouldn't worry!", battler.pbThis))
641:         when :FROZEN
642:           pbDisplay(_INTL("{1} melted the ice with its fiery determination so you wouldn't worry!", battler.pbThis))
643:         end
644:       end
645:     end

656:     end

690:       end
691:       # Flame Orb, Sticky Barb, Toxic Orb
