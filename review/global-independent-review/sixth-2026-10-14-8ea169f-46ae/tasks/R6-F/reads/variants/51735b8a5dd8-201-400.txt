313:   # Called when one or more Pokémon switch in. Does a lot of things, including
314:   # entry hazards, form changes and items/abilities that trigger upon switching
315:   # in.
316:   def pbOnBattlerEnteringBattle(battler_index, skip_event_reset = false)
317:     battler_index = [battler_index] if !battler_index.is_a?(Array)
318:     battler_index.flatten!
319:     # NOTE: This isn't done for switch commands, because they previously call
320:     #       pbRecallAndReplace, which could cause Neutralizing Gas to end, which
321:     #       in turn could cause Intimidate to trigger another Pokémon's Eject
322:     #       Pack. That Eject Pack should trigger at the end of this method, but
323:     #       this resetting would prevent that from happening, so it is skipped
324:     #       and instead done earlier in def pbAttackPhaseSwitch.
325:     if !skip_event_reset
326:       allBattlers.each do |b|
327:         b.droppedBelowHalfHP = false
328:         b.statsDropped = false
329:       end
330:     end
331:     # For each battler that entered battle, in speed order
332:     pbPriority(true).each do |b|
333:       next if !battler_index.include?(b.index) || b.fainted?
334:       pbRecordBattlerAsParticipated(b)
335:       pbMessagesOnBattlerEnteringBattle(b)
336:       # Position/field effects triggered by the battler appearing
337:       pbEffectsOnBattlerEnteringPosition(b)   # Healing Wish/Lunar Dance
338:       pbEntryHazards(b)
339:       # Battler faints if it is knocked out because of an entry hazard above
340:       if b.fainted?
341:         b.pbFaint
342:         pbGainExp
343:         pbJudge
344:         next
345:       end
346:       b.pbCheckForm
347:       # Primal Revert upon entering battle
348:       pbPrimalReversion(b.index)
349:       # Ending primordial weather, checking Trace
350:       b.pbContinualAbilityChecks(true)
351:       # Abilities that trigger upon switching in
352:       if (!b.fainted? && b.unstoppableAbility?) || b.abilityActive?
353:         Battle::AbilityEffects.triggerOnSwitchIn(b.ability, b, self, true)
354:       end
355:       pbEndPrimordialWeather   # Checking this again just in case
356:       # Items that trigger upon switching in (Air Balloon message)
357:       if b.itemActive?
358:         Battle::ItemEffects.triggerOnSwitchIn(b.item, b, self)
359:       end
360:       # Berry check, status-curing ability check
361:       b.pbHeldItemTriggerCheck
362:       b.pbAbilityStatusCureCheck
363:     end
364:     # Check for triggering of Emergency Exit/Wimp Out/Eject Pack (only one will
365:     # be triggered)
366:     pbPriority(true).each do |b|
367:       break if b.pbItemOnStatDropped
368:       break if b.pbAbilitiesOnDamageTaken
369:     end
370:     allBattlers.each do |b|
371:       b.droppedBelowHalfHP = false
372:       b.statsDropped = false
373:     end
374:   end
