147: #===============================================================================
148: # When used against a sole wild Pokémon, makes target flee and ends the battle;
149: # fails if target is a higher level than the user.
150: # When used against a trainer's Pokémon, target switches out.
151: # For status moves. (Roar, Whirlwind)
152: #===============================================================================
153: class Battle::Move::SwitchOutTargetStatusMove < Battle::Move
154:   def ignoresSubstitute?(user); return true; end
155:   def canMagicCoat?;            return true; end
156: 
157:   def pbFailsAgainstTarget?(user, target, show_message)
158:     if target.hasActiveAbility?(:SUCTIONCUPS) && !@battle.moldBreaker
159:       if show_message
160:         @battle.pbShowAbilitySplash(target)
161:         if Battle::Scene::USE_ABILITY_SPLASH
162:           @battle.pbDisplay(_INTL("{1} anchors itself!", target.pbThis))
163:         else
164:           @battle.pbDisplay(_INTL("{1} anchors itself with {2}!", target.pbThis, target.abilityName))
165:         end
166:         @battle.pbHideAbilitySplash(target)
167:       end
168:       return true
169:     end
170:     if target.effects[PBEffects::Ingrain]
171:       @battle.pbDisplay(_INTL("{1} anchored itself with its roots!", target.pbThis)) if show_message
172:       return true
173:     end
174:     if target.wild? && target.allAllies.length == 0 && @battle.canRun
175:       # End the battle
176:       if target.level > user.level
177:         @battle.pbDisplay(_INTL("But it failed!")) if show_message
178:         return true
179:       end
180:     elsif !target.wild?
181:       # Switch target out
182:       canSwitch = false
183:       @battle.eachInTeamFromBattlerIndex(target.index) do |_pkmn, i|
184:         canSwitch = @battle.pbCanSwitchIn?(target.index, i)
185:         break if canSwitch
186:       end
187:       if !canSwitch
188:         @battle.pbDisplay(_INTL("But it failed!")) if show_message
189:         return true
190:       end
191:     else
192:       @battle.pbDisplay(_INTL("But it failed!")) if show_message
193:       return true
194:     end
195:     return false
196:   end
197: 
198:   def pbEffectAgainstTarget(user, target)
199:     @battle.decision = 3 if target.wild?   # Escaped from battle
200:   end
