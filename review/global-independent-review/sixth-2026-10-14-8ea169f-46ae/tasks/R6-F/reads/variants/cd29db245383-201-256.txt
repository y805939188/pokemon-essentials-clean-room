223: # When used against a sole wild Pokémon, makes target flee and ends the battle;
224: # fails if target is a higher level than the user.
225: # When used against a trainer's Pokémon, target switches out.
226: # For damaging moves. (Circle Throw, Dragon Tail)
227: #===============================================================================
228: class Battle::Move::SwitchOutTargetDamagingMove < Battle::Move
229:   def pbEffectAgainstTarget(user, target)
230:     if target.wild? && target.allAllies.length == 0 && @battle.canRun &&
231:        target.level <= user.level &&
232:        (target.effects[PBEffects::Substitute] == 0 || ignoresSubstitute?(user))
233:       @battle.decision = 3   # Escaped from battle
234:     end
235:   end
