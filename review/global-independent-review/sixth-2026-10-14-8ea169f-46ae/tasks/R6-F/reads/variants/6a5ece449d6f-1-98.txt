14:   def create_ai_objects
15:     # Initialize AI trainers
16:     @trainers = [[], []]
17:     @battle.player.each_with_index do |trainer, i|
18:       @trainers[0][i] = AITrainer.new(self, 0, i, trainer)
19:     end
20:     if @battle.wildBattle?
21:       @trainers[1][0] = AITrainer.new(self, 1, 0, nil)
22:     else
23:       @battle.opponent.each_with_index do |trainer, i|
24:         @trainers[1][i] = AITrainer.new(self, 1, i, trainer)
25:       end
26:     end
27:     # Initialize AI battlers
28:     @battlers = []
29:     @battle.battlers.each_with_index do |battler, i|
30:       @battlers[i] = AIBattler.new(self, i) if battler
31:     end
32:     # Initialize AI move object
33:     @move = AIMove.new(self)
34:   end

36:   # Set some class variables for the Pokémon whose action is being chosen
37:   def set_up(idxBattler)
38:     # Find the AI trainer choosing the action
39:     opposes = @battle.opposes?(idxBattler)
40:     trainer_index = @battle.pbGetOwnerIndexFromBattlerIndex(idxBattler)
41:     @trainer = @trainers[(opposes) ? 1 : 0][trainer_index]
42:     # Find the AI battler for which the action is being chosen
43:     @user = @battlers[idxBattler]
44:     @battlers.each { |b| b.refresh_battler if b }
45:   end
