34:   # index and the number of battlers on its side, assuming the battler has
35:   # metrics of 0 (those are added later).
36:   def self.pbBattlerPosition(index, sideSize = 1)
37:     # Start at the centre of the base for the appropriate side
38:     if (index & 1) == 0
39:       ret = [PLAYER_BASE_X, PLAYER_BASE_Y]
40:     else
41:       ret = [FOE_BASE_X, FOE_BASE_Y]
42:     end
43:     # Shift depending on index (no shifting needed for sideSize of 1)
44:     case sideSize
45:     when 2
46:       ret[0] += [-48, 48, 32, -32][index]
47:       ret[1] += [  0,  0, 16, -16][index]
48:     when 3
49:       ret[0] += [-80, 80,  0,  0, 80, -80][index]
50:       ret[1] += [  0,  0,  8, -8, 16, -16][index]

88:       shouldCompact = false

90:         a.update

92:         a.dispose
