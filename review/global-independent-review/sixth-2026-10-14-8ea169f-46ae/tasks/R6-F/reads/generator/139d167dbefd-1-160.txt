115:     pieces = insp.split(/\s*;\s*/)
116:     species = (GameData::Species.exists?(pieces[0])) ? GameData::Species.get(pieces[0]).id : nil
117:     item = (GameData::Item.exists?(pieces[1])) ? GameData::Item.get(pieces[1]).id : nil
118:     nature = (GameData::Nature.exists?(pieces[2])) ? GameData::Nature.get(pieces[2]).id : nil
119:     ev = pieces[3].split(/\s*,\s*/)
120:     ev_array = []
121:     ev.each do |stat|
122:       case stat.upcase
123:       when "HP"          then ev_array.push(:HP)
124:       when "ATK"         then ev_array.push(:ATTACK)
125:       when "DEF"         then ev_array.push(:DEFENSE)
126:       when "SA", "SPATK" then ev_array.push(:SPECIAL_ATTACK)
127:       when "SD", "SPDEF" then ev_array.push(:SPECIAL_DEFENSE)
128:       when "SPD"         then ev_array.push(:SPEED)
129:       end
130:     end
131:     moves = pieces[4].split(/\s*,\s*/)
132:     moveid = []
133:     Pokemon::MAX_MOVES.times do |i|
134:       move_data = GameData::Move.try_get(moves[i])
135:       moveid.push(move_data.id) if move_data
136:     end
137:     moveid.push(GameData::Move.keys.first) if moveid.length == 0   # Get any one move
138:     return self.new(species, item, nature, moveid[0], moveid[1], moveid[2], moveid[3], ev_array)
139:   end
