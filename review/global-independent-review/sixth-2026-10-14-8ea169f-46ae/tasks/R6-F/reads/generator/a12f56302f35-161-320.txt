181:   def winChancePercent
182:     return @estimatedRating if @estimatedRating
183:     if @deviation > 100
184:       # https://www.smogon.com/forums/threads/make-sense-of-your-shoddy-battle-rating.55764/
185:       otherRating = 1500.0
186:       otherDeviation = 350.0
187:       s = Math.sqrt(100_000.0 + (@deviation * @deviation) + (otherDeviation * otherDeviation))
188:       g = 10.0**((otherRating - @rating) * 0.79 / s)
189:       @estimatedRating = (1.0 / (1.0 + g)) * 100.0   # Percent chance that I win against opponent
190:     else
191:       # GLIXARE method
192:       rds = @deviation * @deviation
193:       sqr = Math.sqrt(15.905694331435 * (rds + 221_781.21786254))
194:       inner = (1500.0 - @rating) * Math::PI / sqr
195:       @estimatedRating = ((10_000.0 / (1.0 + (10.0**inner))) + 0.5) / 100.0
196:     end
197:     return @estimatedRating
198:   end

200:   def update(matches, system = 1.2)
201:     volatility = volatility2
202:     deviation = deviation2
203:     rating = rating2
204:     if matches.length == 0
205:       setDeviation2(Math.sqrt((deviation * deviation) + (volatility * volatility)))
206:       return
207:     end
208:     g = []
209:     e = []
210:     score = []
211:     matches.length.times do |i|
212:       match = matches[i]
213:       g[i] = getGFactor(match.opponentDeviation)
214:       e[i] = getEFactor(rating, match.opponentRating, g[i])
215:       score[i] = match.score
216:     end
217:     # Estimated variance
218:     variance = 0.0
219:     matches.length.times do |i|
220:       variance += g[i] * g[i] * e[i] * (1 - e[i])
221:     end
222:     variance = 1.0 / variance
223:     # Improvement sum
224:     sum = 0.0
225:     matches.length.times do |i|
226:       v = score[i]
227:       sum += g[i] * (v.to_f - e[i]) if v != -1
228:     end
229:     volatility = getUpdatedVolatility(volatility, deviation, variance, sum, system)
230:     # Update deviation
231:     t = (deviation * deviation) + (volatility * volatility)
232:     deviation = 1.0 / Math.sqrt((1.0 / t) + (1.0 / variance))
233:     # Update rating
234:     rating += deviation * deviation * sum
235:     setRating2(rating)
236:     setDeviation2(deviation)
237:     setVolatility2(volatility)
238:   end

283:   def getUpdatedVolatility(volatility, deviation, variance, improvementSum, system)
284:     improvement = improvementSum * variance
285:     a = Math.log(volatility * volatility)
286:     squSystem = system * system
287:     squDeviation = deviation * deviation
288:     squVariance = variance + variance
289:     squDevplusVar = squDeviation + variance
290:     x0 = a
291:     100.times do   # Up to 100 iterations to avoid potentially infinite loops
292:       e = Math.exp(x0)
293:       d = squDevplusVar + e
294:       squD = d * d
295:       i = improvement / d
296:       h1 = (-(x0 - a) / squSystem) - (0.5 * e * i * i)
297:       h2 = (-1.0 / squSystem) - (0.5 * e * squDevplusVar / squD)
298:       h2 += 0.5 * squVariance * e * (squDevplusVar - e) / (squD * d)
299:       x1 = x0
300:       x0 -= h1 / h2
301:       break if (x1 - x0).abs < 0.000001
302:     end
303:     return Math.exp(x0 / 2.0)
304:   end
