2: # Given an array of trainers and the number of wins the player already has,
3: # returns a random trainer index. The more wins, the later in the list the
4: # trainer comes from.

6: def pbBattleChallengeTrainer(win_count, bttrainers)
7:   # This table's start points and lengths are based on a bttrainers size of 300.
8:   # They are scaled based on the actual size of bttrainers later.
9:   table = [   # Each value is [minimum win count, range start point, range length]
10:     [ 0,   0, 100],   # 0-100
11:     [ 6,  80,  40],   # 80-120
12:     [ 7,  80,  40],   # 80-120
13:     [13, 120,  20],   # 120-140
14:     [14, 100,  40],   # 100-140
15:     [20, 140,  20],   # 140-160
16:     [21, 120,  40],   # 120-160
17:     [27, 160,  20],   # 160-180
18:     [28, 140,  40],   # 140-180
19:     [34, 180,  20],   # 180-200
20:     [35, 160,  40],   # 160-200
21:     [41, 200,  20],   # 200-220
22:     [42, 180,  40],   # 180-220
23:     [48, 220,  40],   # 220-260
24:     [49, 200, 100]    # 200-300 - This line is used for all higher win_counts
25:   ]
26:   slot = nil
27:   table.each { |val| slot = val if val[0] <= win_count && (!slot || slot[0] < val[0]) }
28:   return 0 if !slot
29:   # Scale the start point and length based on how many trainers are in bttrainers
30:   offset = slot[1] * bttrainers.length / 300
31:   length = slot[2] * bttrainers.length / 300
32:   # Return a random trainer index from the chosen range
33:   return offset + rand(length)
34: end

39: def pbGenerateBattleTrainer(idxTrainer, rules)
40:   bttrainers = pbGetBTTrainers(pbBattleChallenge.currentChallenge)
41:   btpokemon = pbGetBTPokemon(pbBattleChallenge.currentChallenge)
42:   level = rules.ruleset.suggestedLevel
43:   # Create the trainer
44:   trainerdata = bttrainers[idxTrainer]
45:   opponent = NPCTrainer.new(
46:     pbGetMessageFromHash(MessageTypes::TRAINER_NAMES, trainerdata[1]),
47:     trainerdata[0]
48:   )
49:   # Determine how many IVs the trainer's Pokémon will have
50:   indvalues = 31
51:   indvalues = 21 if idxTrainer < 220
52:   indvalues = 18 if idxTrainer < 200
53:   indvalues = 15 if idxTrainer < 180
54:   indvalues = 12 if idxTrainer < 160
55:   indvalues = 9 if idxTrainer < 140
56:   indvalues = 6 if idxTrainer < 120
57:   indvalues = 3 if idxTrainer < 100
58:   # Get the indices within bypokemon of the Pokémon the trainer may have
59:   pokemonnumbers = trainerdata[5]
60:   # The number of possible Pokémon is <= the required number; make them
61:   # all Pokémon and use them
62:   if pokemonnumbers.length <= rules.ruleset.suggestedNumber
63:     pokemonnumbers.each do |n|
64:       rndpoke = btpokemon[n]
65:       pkmn = rndpoke.createPokemon(level, indvalues, opponent)
66:       opponent.party.push(pkmn)
67:     end
68:     return opponent
69:   end
70:   # There are more possible Pokémon than there are spaces available in the
71:   # trainer's party; randomly choose Pokémon
72:   loop do
73:     opponent.party.clear
74:     while opponent.party.length < rules.ruleset.suggestedNumber
75:       rnd = pokemonnumbers[rand(pokemonnumbers.length)]
76:       rndpoke = btpokemon[rnd]
77:       pkmn = rndpoke.createPokemon(level, indvalues, opponent)
78:       opponent.party.push(pkmn)
79:     end
80:     break if rules.ruleset.isValid?(opponent.party)
81:   end
82:   return opponent
83: end

88: def pbBattleFactoryPokemon(rules, win_count, swap_count, rentals)
89:   btpokemon = pbGetBTPokemon(pbBattleChallenge.currentChallenge)
90:   level = rules.ruleset.suggestedLevel
91:   pokemonNumbers = [0, 0]   # Start and end indices in btpokemon
92:   ivs = [0, 0]   # Lower and higher IV values for Pokémon to use
93:   iv_threshold = 6   # Number of Pokémon that use the lower IV
94:   set = [win_count / 7, 7].min   # The set of 7 battles win_count is part of (minus 1)
95:   # Choose a range of Pokémon in btpokemon to randomly select from. The higher
96:   # the set number, the later the range lies within btpokemon (typically).
97:   # This table's start point and end point values are based on a btpokemon size
98:   # of 881. They are scaled based on the actual size of btpokemon.
99:   # Group 1 is 0 - 173. Group 2 is 174 - 371. Group 3 is 372 - 881.
100:   if level == GameData::GrowthRate.max_level   # Open Level (Level 100)
101:     table = [
102:       [372, 491],   # Group 3 (first quarter)
103:       [492, 610],   # Group 3 (second quarter)
104:       [611, 729],   # Group 3 (third quarter)
105:       [730, 849],   # Group 3 (fourth quarter)
106:       [372, 881],   # All of Group 3
107:       [372, 881],   # All of Group 3
108:       [372, 881],   # All of Group 3
109:       [372, 881]    # This line is used for all higher sets (all of Group 3)
110:     ]
111:   else
112:     table = [
113:       [  0, 173],   # Group 1
114:       [174, 272],   # Group 2 (first half)
115:       [273, 371],   # Group 2 (second half)
116:       [372, 491],   # Group 3 (first quarter)
117:       [492, 610],   # Group 3 (second quarter)
118:       [611, 729],   # Group 3 (third quarter)
119:       [730, 849],   # Group 3 (fourth quarter)
120:       [372, 881]    # This line is used for all higher sets (all of Group 3)
121:     ]
122:   end
123:   pokemonNumbers[0] = table[set][0] * btpokemon.length / 881
124:   pokemonNumbers[1] = table[set][1] * btpokemon.length / 881
125:   # Choose two IV values for Pokémon to use (the one for the current set, and
126:   # the one for the next set). The iv_threshold below determines which of these
127:   # two values a given Pokémon uses. The higher the set number, the higher these
128:   # values are.
129:   ivtable = [3, 6, 9, 12, 15, 21, 31, 31]   # Last value is used for all higher sets
130:   ivs = [ivtable[set], ivtable[[set + 1, 7].min]]
131:   # Choose a threshold, which is the number of Pokémon with the lower IV out of
132:   # the two chosen above. The higher the swap_count, the lower this threshold
133:   # (i.e. the more Pokémon will have the higher IV).
134:   thresholds = [   # Each value is [minimum swap count, threshold value]
135:     [ 0, 6],
136:     [15, 5],
137:     [22, 4],
138:     [29, 3],
139:     [36, 2],
140:     [43, 1]
141:   ]
142:   thresholds.each { |val| iv_threshold = val[1] if swap_count >= val[0] }
143:   # Randomly choose Pokémon from the range to fill the party with
144:   old_min = rules.ruleset.minLength
145:   old_max = rules.ruleset.maxLength
146:   if rentals.length == 0
147:     rules.ruleset.setNumber(6)   # Rentals
148:   else
149:     rules.ruleset.setNumber(old_max + rentals.length)   # Opponent
150:   end
151:   party = []
152:   loop do
153:     party.clear
154:     while party.length < ((rentals.length == 0) ? 6 : old_max)
155:       rnd = pokemonNumbers[0] + rand(pokemonNumbers[1] - pokemonNumbers[0] + 1)
156:       rndpoke = btpokemon[rnd]
157:       indvalue = (party.length < iv_threshold) ? ivs[0] : ivs[1]
158:       party.push(rndpoke.createPokemon(level, indvalue, nil))
159:     end
160:     break if rules.ruleset.isValid?([].concat(party).concat(rentals))
161:   end
162:   rules.ruleset.setNumberRange(old_min, old_max)
163:   return party
164: end
