111:   # This method is how each Pokémon is compiled from the PBS files listing
112:   # Battle Tower/Cup Pokémon.
113:   def self.fromInspected(str)
114:     insp = str.gsub(/^\s+/, "").gsub(/\s+$/, "")
115:     pieces = insp.split(/\s*;\s*/)
116:     species = (GameData::Species.exists?(pieces[0])) ? GameData::Species.get(pieces[0]).id : nil
117:     item = (GameData::Item.exists?(pieces[1])) ? GameData::Item.get(pieces[1]).id : nil
118:     nature = (GameData::Nature.exists?(pieces[2])) ? GameData::Nature.get(pieces[2]).id : nil
119:     ev = pieces[3].split(/\s*,\s*/)
120:     ev_array = []
121:     ev.each do |stat|
122:       case stat.upcase
123:       when "HP"          then ev_array.push(:HP)
124:       when "ATK"         then ev_array.push(:ATTACK)
125:       when "DEF"         then ev_array.push(:DEFENSE)
126:       when "SA", "SPATK" then ev_array.push(:SPECIAL_ATTACK)
127:       when "SD", "SPDEF" then ev_array.push(:SPECIAL_DEFENSE)
128:       when "SPD"         then ev_array.push(:SPEED)
129:       end
130:     end
131:     moves = pieces[4].split(/\s*,\s*/)
132:     moveid = []
133:     Pokemon::MAX_MOVES.times do |i|
134:       move_data = GameData::Move.try_get(moves[i])
135:       moveid.push(move_data.id) if move_data
136:     end
137:     moveid.push(GameData::Move.keys.first) if moveid.length == 0   # Get any one move
138:     return self.new(species, item, nature, moveid[0], moveid[1], moveid[2], moveid[3], ev_array)
139:   end

141:   def self.fromPokemon(pkmn)
142:     mov1 = (pkmn.moves[0]) ? pkmn.moves[0].id : nil
143:     mov2 = (pkmn.moves[1]) ? pkmn.moves[1].id : nil
144:     mov3 = (pkmn.moves[2]) ? pkmn.moves[2].id : nil
145:     mov4 = (pkmn.moves[3]) ? pkmn.moves[3].id : nil
146:     ev_array = []
147:     GameData::Stat.each_main do |s|
148:       ev_array.push(s.id) if pkmn.ev[s.id] > 60
149:     end
150:     return self.new(pkmn.species, pkmn.item_id, pkmn.nature,
151:                     mov1, mov2, mov3, mov4, ev_array)
152:   end

201:   def createPokemon(level, iv, trainer)
202:     pkmn = Pokemon.new(@species, level, trainer, false)
203:     pkmn.item = @item
204:     pkmn.personalID = rand(2**16) | (rand(2**16) << 16)
205:     pkmn.nature = nature
206:     pkmn.happiness = 0
207:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move1)))
208:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move2))) if @move2
209:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move3))) if @move3
210:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move4))) if @move4
211:     pkmn.moves.compact!
212:     if ev.length > 0
213:       ev.each { |stat| pkmn.ev[stat] = Pokemon::EV_LIMIT / ev.length }
214:     end
215:     GameData::Stat.each_main { |s| pkmn.iv[s.id] = iv }
216:     pkmn.calc_stats
217:     return pkmn
