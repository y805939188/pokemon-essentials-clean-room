4: def _pbNextComb(comb, length)
5:   i = comb.length - 1
6:   loop do
7:     valid = true
8:     (i...comb.length).each do |j|
9:       if j == i
10:         comb[j] += 1
11:       else
12:         comb[j] = comb[i] + (j - i)
13:       end
14:       if comb[j] >= length
15:         valid = false
16:         break
17:       end
18:     end
19:     return true if valid
20:     i -= 1
21:     break unless i >= 0
22:   end
23:   return false
24: end

26: # Iterates through the array and yields each combination of _num_ elements in
27: # the array.
28: def pbEachCombination(array, num)
29:   return if array.length < num || num <= 0
30:   if array.length == num
31:     yield array
32:     return
33:   elsif num == 1
34:     array.each do |x|
35:       yield [x]
36:     end
37:     return
38:   end
39:   currentComb = []
40:   arr = []
41:   num.times { |i| currentComb[i] = i }
42:   loop do
43:     num.times { |i| arr[i] = array[currentComb[i]] }
44:     yield arr
45:     break unless _pbNextComb(currentComb, array.length)
46:   end
47: end
