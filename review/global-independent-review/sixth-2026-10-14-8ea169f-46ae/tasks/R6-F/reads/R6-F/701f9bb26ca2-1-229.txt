100: class TotalLevelAdjustment < LevelAdjustment
101:   def initialize(minLevel, maxLevel, totalLevel)
102:     super(LevelAdjustment::ENEMY_TEAM)
103:     @minLevel = minLevel.clamp(1, GameData::GrowthRate.max_level)
104:     @maxLevel = maxLevel.clamp(1, GameData::GrowthRate.max_level)
105:     @totalLevel = totalLevel
106:   end
107: 
108:   def getAdjustment(thisTeam, _otherTeam)
109:     ret = []
110:     total = 0
111:     thisTeam.each_with_index do |pkmn, i|
112:       ret[i] = @minLevel
113:       total += @minLevel
114:     end
115:     loop do
116:       work = false
117:       thisTeam.each_with_index do |pkmn, i|
118:         next if ret[i] >= @maxLevel || total >= @totalLevel
119:         ret[i] += 1
120:         total += 1
121:         work = true
122:       end
123:       break if !work
124:     end
125:     return ret
126:   end
127: end
