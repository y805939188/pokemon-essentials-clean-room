232:       total2 += ratings2[i]

236:     textpos = [
237:       [battler1.name, 64, 6, :center, Color.new(248, 0, 0), Color.new(208, 208, 200)],
238:       [_INTL("VS"), 144, 6, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
239:       [battler2.name, 224, 6, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
240:       [_INTL("Mind"), 144, 54, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
241:       [_INTL("Skill"), 144, 86, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
242:       [_INTL("Body"), 144, 118, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
243:       [total1.to_s, 64, 166, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
244:       [_INTL("Judgment"), 144, 166, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)],
245:       [total2.to_s, 224, 166, :center, Color.new(72, 72, 72), Color.new(208, 208, 200)]
246:     ]

248:     images = []
249:     phase.times do |i|

252:       images.push(["Graphics/UI/Battle/judgment", 64 - 16, y, x, 0, 32, 32])
253:       x = (ratings1[i] == ratings2[i]) ? 64 : ((ratings1[i] < ratings2[i]) ? 0 : 32)
254:       images.push(["Graphics/UI/Battle/judgment", 224 - 16, y, x, 0, 32, 32])
255:     end
256:     pbDrawImagePositions(window.contents, images)
257:     window.contents.fill_rect(16, 150, 256, 4, Color.new(80, 80, 80))
258:   end

265:   def pbBattleArenaJudgment(battler1, battler2, ratings1, ratings2)
266:     msgwindow  = nil
267:     dimmingvp  = nil
268:     infowindow = nil
269:     begin
270:       msgwindow = pbCreateMessageWindow
271:       dimmingvp = Viewport.new(0, 0, Graphics.width, Graphics.height - msgwindow.height)
272:       pbMessageDisplay(msgwindow,
273:                        _INTL("REFEREE: That's it! We will now go to judging to determine the winner!") + "\\wtnp[10]") do
274:         pbBattleArenaUpdate
275:         dimmingvp.update
276:       end
277:       dimmingvp.z = 99999
278:       infowindow = SpriteWindow_Base.new(80, 0, 320, 224)
279:       infowindow.contents = Bitmap.new(infowindow.width - infowindow.borderX,
280:                                        infowindow.height - infowindow.borderY)
281:       infowindow.z        = 99999
282:       infowindow.visible  = false
283:       11.times do |i|
284:         pbGraphicsUpdate
285:         pbInputUpdate
286:         msgwindow.update
287:         dimmingvp.update
288:         dimmingvp.color = Color.new(0, 0, 0, i * 128 / 10)
289:       end
290:       updateJudgment(infowindow, 0, battler1, battler2, ratings1, ratings2)
291:       infowindow.visible = true
292:       11.times do |i|
293:         pbGraphicsUpdate
294:         pbInputUpdate
295:         msgwindow.update
296:         dimmingvp.update
297:         infowindow.update
298:       end
299:       updateJudgment(infowindow, 1, battler1, battler2, ratings1, ratings2)
300:       pbMessageDisplay(msgwindow,
301:                        _INTL("REFEREE: Judging category 1, Mind!\nThe Pokémon showing the most guts!") + "\\wtnp[20]") do
302:         pbBattleArenaUpdate
303:         dimmingvp.update
304:         infowindow.update
305:       end
306:       updateJudgment(infowindow, 2, battler1, battler2, ratings1, ratings2)
307:       pbMessageDisplay(msgwindow,
308:                        _INTL("REFEREE: Judging category 2, Skill!\nThe Pokémon using moves the best!") + "\\wtnp[20]") do
309:         pbBattleArenaUpdate
310:         dimmingvp.update
311:         infowindow.update
312:       end
313:       updateJudgment(infowindow, 3, battler1, battler2, ratings1, ratings2)
314:       pbMessageDisplay(msgwindow,
315:                        _INTL("REFEREE: Judging category 3, Body!\nThe Pokémon with the most vitality!") + "\\wtnp[20]") do
316:         pbBattleArenaUpdate
317:         dimmingvp.update
318:         infowindow.update
319:       end
320:       total1 = 0
321:       total2 = 0
322:       3.times do |i|
323:         total1 += ratings1[i]
324:         total2 += ratings2[i]
325:       end
326:       if total1 == total2
327:         pbMessageDisplay(msgwindow,
328:                          _INTL("REFEREE: Judgment: {1} to {2}!\nWe have a draw!", total1, total2) + "\\wtnp[20]") do
329:           pbBattleArenaUpdate
330:           dimmingvp.update
331:           infowindow.update
332:         end
333:       elsif total1 > total2
334:         pbMessageDisplay(msgwindow,
335:                          _INTL("REFEREE: Judgment: {1} to {2}!\nThe winner is {3}'s {4}!",
336:                                total1, total2, @battle.pbGetOwnerName(battler1.index), battler1.name) + "\\wtnp[20]") do
337:           pbBattleArenaUpdate
338:           dimmingvp.update
339:           infowindow.update
340:         end
341:       else
342:         pbMessageDisplay(msgwindow,
343:                          _INTL("REFEREE: Judgment: {1} to {2}!\nThe winner is {3}!",
344:                                total1, total2, battler2.name) + "\\wtnp[20]") do
345:           pbBattleArenaUpdate
346:           dimmingvp.update
347:           infowindow.update
348:         end
349:       end
350:       infowindow.visible = false
351:       msgwindow.visible  = false
352:       11.times do |i|
353:         pbGraphicsUpdate
354:         pbInputUpdate
355:         msgwindow.update
356:         dimmingvp.update
357:         dimmingvp.color = Color.new(0, 0, 0, (10 - i) * 128 / 10)
358:       end
359:     ensure
360:       pbDisposeMessageWindow(msgwindow)
361:       dimmingvp.dispose
362:       infowindow&.contents&.dispose
363:       infowindow&.dispose
364:     end
365:   end
