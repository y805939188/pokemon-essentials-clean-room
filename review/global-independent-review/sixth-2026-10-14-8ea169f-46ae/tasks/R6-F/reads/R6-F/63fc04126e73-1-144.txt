40: def pbOrganizedBattleEx(opponent, challengedata)
41:   # Skip battle if holding Ctrl in Debug mode
42:   if Input.press?(Input::CTRL) && $DEBUG
43:     pbMessage(_INTL("SKIPPING BATTLE..."))
44:     pbMessage(_INTL("AFTER WINNING..."))
45:     pbMessage(opponent.lose_text || "...")
46:     $game_temp.last_battle_record = nil
47:     pbMEStop
48:     return true
49:   end
50:   $player.heal_party
51:   # Remember original data, to be restored after battle
52:   challengedata = PokemonChallengeRules.new if !challengedata
53:   oldlevels = challengedata.adjustLevels($player.party, opponent.party)
54:   olditems  = $player.party.transform { |p| p.item_id }
55:   olditems2 = opponent.party.transform { |p| p.item_id }
56:   # Create the battle scene (the visual side of it)
57:   scene = BattleCreationHelperMethods.create_battle_scene
58:   # Create the battle class (the mechanics side of it)
59:   battle = challengedata.createBattle(scene, $player, opponent)
60:   battle.internalBattle = false
61:   # Set various other properties in the battle class
62:   BattleCreationHelperMethods.prepare_battle(battle)
63:   # Perform the battle itself
64:   decision = 0
65:   pbBattleAnimation(pbGetTrainerBattleBGM(opponent)) do
66:     pbSceneStandby { decision = battle.pbStartBattle }
67:   end
68:   Input.update
69:   # Restore both parties to their original levels
70:   challengedata.unadjustLevels($player.party, opponent.party, oldlevels)
71:   # Heal both parties and restore their original items
72:   $player.party.each_with_index do |pkmn, i|
73:     pkmn.heal
74:     pkmn.makeUnmega
75:     pkmn.makeUnprimal
76:     pkmn.item = olditems[i]
77:   end
78:   opponent.party.each_with_index do |pkmn, i|
79:     pkmn.heal
80:     pkmn.makeUnmega
81:     pkmn.makeUnprimal
82:     pkmn.item = olditems2[i]
83:   end
84:   # Save the record of the battle
85:   $game_temp.last_battle_record = nil
86:   if [1, 2, 5].include?(decision)   # if win, loss or draw
87:     $game_temp.last_battle_record = battle.pbDumpRecord
88:   end
89:   case decision
90:   when 1   # Won
91:     $stats.trainer_battles_won += 1
92:   when 2, 3, 5   # Lost, fled, draw
93:     $stats.trainer_battles_lost += 1
94:   end
95:   # Return true if the player won the battle, and false if any other result
96:   return (decision == 1)
97: end

111: def pbPlayBattle(battledata)
112:   return if !battledata
113:   scene = BattleCreationHelperMethods.create_battle_scene
114:   scene.abortable = true
115:   lastbattle = Marshal.restore(battledata)
116:   case lastbattle[0]
117:   when BattleChallenge::BATTLE_TOWER_ID
118:     battleplayer = RecordedBattle::PlaybackBattle.new(scene, lastbattle)
119:   when BattleChallenge::BATTLE_PALACE_ID
120:     battleplayer = RecordedBattle::BattlePalacePlaybackBattle.new(scene, lastbattle)
121:   when BattleChallenge::BATTLE_ARENA_ID
122:     battleplayer = RecordedBattle::BattleArenaPlaybackBattle.new(scene, lastbattle)
123:   end
124:   bgm = RecordedBattle::PlaybackHelper.pbGetBattleBGM(lastbattle)
125:   pbBattleAnimation(bgm) do
126:     pbSceneStandby { battleplayer.pbStartBattle }
127:   end
128: end
