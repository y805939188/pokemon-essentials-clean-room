83:   def pbGetCommands(list, choices)
84:     red_text_tag = shadowc3tag(RED_TEXT_BASE, RED_TEXT_SHADOW)
85:     commands = []

87:       pkmn = list[i]

89:       cmd = _INTL("{1} - {2} Pokémon", pkmn.speciesName, category)

99:       loop do
100:         Graphics.update

104:           return -1
105:         elsif Input.trigger?(Input::USE)
106:           index = @sprites["list"].index

108:             return -1

115:       end
116:     end
