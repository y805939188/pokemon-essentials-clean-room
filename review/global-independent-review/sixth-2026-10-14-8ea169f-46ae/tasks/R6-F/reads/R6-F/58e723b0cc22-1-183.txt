59:     return if favorDraws && @decision == 5
60:     return if !favorDraws && @decision > 0
61:     pbJudge

63:     2.times do |side|
64:       next if !@battlers[side].fainted?
65:       next if @partyindexes[side] + 1 >= self.pbParty(side).length
66:       @partyindexes[side] += 1
67:       newpoke = @partyindexes[side]

96:   def pbMindScore(move)
97:     if move.function_code == "ProtectUser" ||   # Detect/Protect
98:        move.function_code == "UserEnduresFaintingThisTurn" ||   # Endure
99:        move.function_code == "FlinchTargetFailsIfNotUserFirstTurn"   # Fake Out
100:       return -1
101:     end
102:     if move.function_code == "CounterPhysicalDamage" ||   # Counter
103:        move.function_code == "CounterSpecialDamage" ||   # Mirror Coat
104:        move.function_code == "MultiTurnAttackBideThenReturnDoubleDamage"   # Bide
105:       return 0
106:     end
107:     return 0 if move.statusMove?
108:     return 1
109:   end

111:   def pbCommandPhase
112:     if @battlersChanged
113:       @scene.pbBattleArenaBattlers(@battlers[0], @battlers[1])
114:       @battlersChanged = false
115:       @count = 0
116:     end
117:     super
118:     return if @decision != 0
119:     # Update mind rating (asserting that a move was chosen)
120:     2.times do |side|
121:       if @choices[side][2] && @choices[side][0] == :UseMove
122:         @mind[side] += pbMindScore(@choices[side][2])
123:       end
124:     end

127:   def pbEndOfRoundPhase
128:     super
129:     return if @decision != 0
130:     # Update skill rating
131:     2.times do |side|
132:       @skill[side] += self.successStates[side].skill
133:     end
134: #    PBDebug.log("[Mind: #{@mind.inspect}, Skill: #{@skill.inspect}]")
135:     # Increment turn counter
136:     @count += 1
137:     return if @count < 3
138:     # Half all multi-turn moves
139:     @battlers[0].pbCancelMoves(true)
140:     @battlers[1].pbCancelMoves(true)
141:     # Calculate scores in each category
142:     ratings1 = [0, 0, 0]

144:     if @mind[0] == @mind[1]

149:     else
150:       ratings2[0] = 2
151:     end
152:     if @skill[0] == @skill[1]
153:       ratings1[1] = 1
154:       ratings2[1] = 1

157:     else
158:       ratings2[1] = 2
159:     end
160:     body = [0, 0]
161:     body[0] = ((@battlers[0].hp * 100) / [@starthp[0], 1].max).floor
162:     body[1] = ((@battlers[1].hp * 100) / [@starthp[1], 1].max).floor
163:     if body[0] == body[1]
164:       ratings1[2] = 1
165:       ratings2[2] = 1
166:     elsif body[0] > body[1]
167:       ratings1[2] = 2
168:     else
169:       ratings2[2] = 2
170:     end
171:     # Show scores

180:                       @battlers[0].name, @battlers[1].name))
