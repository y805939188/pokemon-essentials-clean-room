4:   def get_all_PBS_file_paths(game_data)
5:     ret = []
6:     game_data.each { |element| ret.push(element.pbs_file_suffix) if !ret.include?(element.pbs_file_suffix) }
7:     ret.each_with_index do |element, i|
8:       ret[i] = [sprintf("PBS/%s.txt", game_data::PBS_BASE_FILENAME), element]
9:       if !nil_or_empty?(element)
10:         ret[i][0] = sprintf("PBS/%s_%s.txt", game_data::PBS_BASE_FILENAME, element)
11:       end
12:     end
13:     return ret
14:   end

16:   def add_PBS_header_to_file(file)
17:     file.write(0xEF.chr)
18:     file.write(0xBB.chr)
19:     file.write(0xBF.chr)
20:     file.write("\# " + _INTL("See the documentation on the wiki to learn how to edit this file.") + "\r\n")
21:   end

23:   def write_PBS_file_generic(game_data)
24:     paths = get_all_PBS_file_paths(game_data)
25:     schema = game_data.schema
26:     idx = 0
27:     paths.each do |path|
28:       write_pbs_file_message_start(path[0])
29:       File.open(path[0], "wb") do |f|
30:         add_PBS_header_to_file(f)
31:         # Write each element in turn
32:         game_data.each do |element|
33:           next if element.pbs_file_suffix != path[1]
34:           echo "." if idx % 100 == 0
35:           Graphics.update if idx % 500 == 0
36:           idx += 1
37:           f.write("\#-------------------------------\r\n")
38:           if schema["SectionName"]
39:             f.write("[")
40:             pbWriteCsvRecord(element.get_property_for_PBS("SectionName"), f, schema["SectionName"])
41:             f.write("]\r\n")
42:           else
43:             f.write("[#{element.id}]\r\n")
44:           end
45:           schema.each_key do |key|
46:             next if key == "SectionName"
47:             val = element.get_property_for_PBS(key)
48:             next if val.nil?
49:             if schema[key][1][0] == "^" && val.is_a?(Array)
50:               val.each do |sub_val|
51:                 f.write(sprintf("%s = ", key))
52:                 pbWriteCsvRecord(sub_val, f, schema[key])
53:                 f.write("\r\n")
54:               end
55:             else
56:               f.write(sprintf("%s = ", key))
57:               pbWriteCsvRecord(val, f, schema[key])
58:               f.write("\r\n")
59:             end
60:           end
61:         end
62:       end
63:       process_pbs_file_message_end
64:     end
65:   end

77:   def normalize_connection(conn)
78:     ret = conn.clone
79:     if conn[1].negative? != conn[4].negative?   # Exactly one is negative
80:       ret[4] = -conn[1]
81:       ret[1] = -conn[4]
82:     end
83:     if conn[2].negative? != conn[5].negative?   # Exactly one is negative
84:       ret[5] = -conn[2]
85:       ret[2] = -conn[5]
86:     end
87:     return ret
88:   end

90:   def get_connection_text(map1, x1, y1, map2, x2, y2)
91:     dims1 = MapFactoryHelper.getMapDims(map1)
92:     dims2 = MapFactoryHelper.getMapDims(map2)
93:     if x1 == 0 && x2 == dims2[0]
94:       return sprintf("%d,West,%d,%d,East,%d", map1, y1, map2, y2)
95:     elsif y1 == 0 && y2 == dims2[1]
96:       return sprintf("%d,North,%d,%d,South,%d", map1, x1, map2, x2)
97:     elsif x1 == dims1[0] && x2 == 0
98:       return sprintf("%d,East,%d,%d,West,%d", map1, y1, map2, y2)
99:     elsif y1 == dims1[1] && y2 == 0
100:       return sprintf("%d,South,%d,%d,North,%d", map1, x1, map2, x2)
101:     end
102:     return sprintf("%d,%d,%d,%d,%d,%d", map1, x1, y1, map2, x2, y2)
103:   end

105:   def write_connections(path = "PBS/map_connections.txt")
106:     conndata = load_data("Data/map_connections.dat")
107:     return if !conndata
108:     write_pbs_file_message_start(path)
109:     mapinfos = pbLoadMapInfos
110:     File.open(path, "wb") do |f|
111:       add_PBS_header_to_file(f)
112:       f.write("\#-------------------------------\r\n")
113:       conndata.each do |conn|
114:         if mapinfos
115:           # Skip if map no longer exists
116:           next if !mapinfos[conn[0]] || !mapinfos[conn[3]]
117:           f.write(sprintf("# %s (%d) - %s (%d)\r\n",
118:                           (mapinfos[conn[0]]) ? mapinfos[conn[0]].name : "???", conn[0],
119:                           (mapinfos[conn[3]]) ? mapinfos[conn[3]].name : "???", conn[3]))
120:         end
121:         if conn[1].is_a?(String) || conn[4].is_a?(String)
122:           f.write(sprintf("%d,%s,%d,%d,%s,%d", conn[0], conn[1], conn[2],
123:                           conn[3], conn[4], conn[5]))
124:         else
125:           ret = normalize_connection(conn)
126:           f.write(get_connection_text(ret[0], ret[1], ret[2], ret[3], ret[4], ret[5]))
127:         end
128:         f.write("\r\n")
129:       end
130:     end
