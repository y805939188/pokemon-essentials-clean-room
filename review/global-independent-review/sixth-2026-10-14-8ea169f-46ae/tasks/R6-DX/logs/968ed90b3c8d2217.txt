993:   viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)

998:     Graphics.width / 2, 64, Graphics.width / 2, Graphics.height - 64, viewport
999:   )
1000:   info.z = 2
1001:   dex.compact!
1002:   ret = dex.clone
1003:   commands = []
1004:   refresh_list = true
1005:   cmd = [0, 0]   # [action, index in list]
1006:   loop do
1007:     # Populate commands
1008:     if refresh_list
1009:       loop do
1010:         break if dex.length == 0 || dex[-1]
1011:         dex.slice!(-1)
1012:       end
1013:       commands = []
1014:       dex.each_with_index do |species, i|
1015:         text = (species) ? GameData::Species.get(species).real_name : "----------"
1016:         commands.push(sprintf("%03d: %s", i + 1, text))
1017:       end
1018:       commands.push(sprintf("%03d: ----------", commands.length + 1))
1019:       cmd[1] = [cmd[1], commands.length - 1].min
1020:       refresh_list = false
1021:     end
1022:     # Choose to do something
1023:     cmd = pbCommands3(cmd_window, commands, -1, cmd[1], true)
1024:     case cmd[0]
1025:     when 1   # Swap entry up
1026:       if cmd[1] < dex.length - 1
1027:         dex[cmd[1] + 1], dex[cmd[1]] = dex[cmd[1]], dex[cmd[1] + 1]
1028:         refresh_list = true
1029:       end
1030:     when 2   # Swap entry down
1031:       if cmd[1] > 0
1032:         dex[cmd[1] - 1], dex[cmd[1]] = dex[cmd[1]], dex[cmd[1] - 1]
1033:         refresh_list = true
1034:       end
1035:     when 3   # Delete spot
1036:       if cmd[1] < dex.length
1037:         dex.delete_at(cmd[1])
1038:         refresh_list = true
1039:       end
1040:     when 4   # Insert spot
1041:       if cmd[1] < dex.length
1042:         dex.insert(cmd[1], nil)
1043:         refresh_list = true
1044:       end
1045:     when 5   # Clear spot
1046:       if dex[cmd[1]]
1047:         dex[cmd[1]] = nil
1048:         refresh_list = true
1049:       end
1050:     when 0
1051:       if cmd[1] >= 0   # Edit entry
1052:         case pbMessage("\\ts[]" + _INTL("Do what with this entry?"),
1053:                        [_INTL("Change species"), _INTL("Clear"),
1054:                         _INTL("Insert entry"), _INTL("Delete entry"),
1055:                         _INTL("Cancel")], 5)
1056:         when 0   # Change species
1057:           species = pbChooseSpeciesList(dex[cmd[1]])
1058:           if species
1059:             dex[cmd[1]] = species
1060:             dex.each_with_index { |s, i| dex[i] = nil if i != cmd[1] && s == species }
1061:             refresh_list = true
1062:           end
1063:         when 1   # Clear spot
1064:           if dex[cmd[1]]
1065:             dex[cmd[1]] = nil
1066:             refresh_list = true
1067:           end
1068:         when 2   # Insert spot
1069:           if cmd[1] < dex.length
1070:             dex.insert(cmd[1], nil)
1071:             refresh_list = true
1072:           end
1073:         when 3   # Delete spot
1074:           if cmd[1] < dex.length
1075:             dex.delete_at(cmd[1])
1076:             refresh_list = true
1077:           end
1078:         end
1079:       else   # Cancel
1080:         case pbMessage(_INTL("Save changes?"),
1081:                        [_INTL("Yes"), _INTL("No"), _INTL("Cancel")], 3)
1082:         when 0   # Save all changes to Dex
1083:           dex.slice!(-1) until dex[-1]

1085:           break
1086:         when 1   # Just quit
1087:           break
1088:         end

1090:     end
1091:   end
