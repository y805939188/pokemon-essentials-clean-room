251: class StringListProperty
252:   def self.set(_setting_name, old_setting)
253:     old_setting = [] if !old_setting
254:     real_cmds = []
255:     real_cmds.push([_INTL("[ADD VALUE]"), -1])
256:     old_setting.length.times do |i|
257:       real_cmds.push([old_setting[i], 0])
258:     end
259:     # Edit list
260:     cmdwin = pbListWindow([], 200)
261:     oldsel = nil
262:     ret = old_setting
263:     cmd = 0
264:     commands = []
265:     do_refresh = true
266:     loop do
267:       if do_refresh
268:         commands = []
269:         real_cmds.each_with_index do |entry, i|
270:           commands.push(entry[0])
271:           cmd = i if oldsel && entry[0] == oldsel
272:         end
273:       end
274:       do_refresh = false
275:       oldsel = nil
276:       cmd = pbCommands2(cmdwin, commands, -1, cmd, true)
277:       if cmd >= 0   # Chose a value
278:         entry = real_cmds[cmd]
279:         if entry[1] == -1   # Add new value
280:           new_value = pbMessageFreeText(_INTL("Enter the new value."),
281:                                         "", false, 250, Graphics.width)
282:           if !nil_or_empty?(new_value)
283:             if real_cmds.any? { |e| e[0] == new_value }
284:               oldsel = new_value   # Already have value; just move cursor to it
285:             else
286:               real_cmds.push([new_value, 0])
287:             end
288:             do_refresh = true
289:           end
290:         else   # Edit value
291:           case pbMessage("\\ts[]" + _INTL("Do what with this value?"),
292:                          [_INTL("Edit"), _INTL("Delete"), _INTL("Cancel")], 3)
293:           when 0   # Edit
294:             new_value = pbMessageFreeText(_INTL("Enter the new value."),
295:                                           entry[0], false, 250, Graphics.width)
296:             if !nil_or_empty?(new_value)
297:               if real_cmds.any? { |e| e[0] == new_value }   # Already have value; delete this one
298:                 real_cmds.delete_at(cmd)
299:                 cmd = [cmd, real_cmds.length - 1].min
300:               else   # Change value
301:                 entry[0] = new_value
302:               end
303:               oldsel = new_value
304:               do_refresh = true
305:             end
306:           when 1   # Delete
307:             real_cmds.delete_at(cmd)
308:             cmd = [cmd, real_cmds.length - 1].min
309:             do_refresh = true
310:           end
311:         end
312:       else   # Cancel/quit
313:         case pbMessage(_INTL("Keep changes?"), [_INTL("Yes"), _INTL("No"), _INTL("Cancel")], 3)
314:         when 0
315:           real_cmds.length.times do |i|
316:             real_cmds[i] = (real_cmds[i][1] == -1) ? nil : real_cmds[i][0]
317:           end
318:           real_cmds.compact!
319:           ret = real_cmds
320:           break
321:         when 1
322:           break
323:         end
324:       end
325:     end
326:     cmdwin.dispose
327:     return ret
328:   end
329: 
330:   def self.defaultValue
331:     return []
332:   end
333: 
334:   def self.format(value)
335:     return (value) ? value.join(",") : ""
336:   end
337: end

342: class GameDataProperty
343:   def initialize(value)
344:     raise _INTL("Couldn't find class {1} in module GameData.", value.to_s) if !GameData.const_defined?(value.to_sym)
345:     @module = GameData.const_get(value.to_sym)
346:   end
347: 
348:   def set(settingname, oldsetting)
349:     commands = []
350:     i = 0
351:     @module.each do |data|
352:       if data.respond_to?("id_number")
353:         commands.push([data.id_number, data.name, data.id])
354:       else
355:         commands.push([i, data.name, data.id])
356:       end
357:       i += 1
358:     end
359:     return pbChooseList(commands, oldsetting, oldsetting, -1)
360:   end
361: 
362:   def defaultValue
363:     return nil
364:   end
365: 
366:   def format(value)
367:     return (value && @module.exists?(value)) ? @module.get(value).real_name : "-"
368:   end
369: end
