668: def pbDebugFixInvalidTiles
669:   total_errors = 0
670:   num_error_maps = 0
671:   tilesets = $data_tilesets
672:   mapData = Compiler::MapData.new
673:   t = System.uptime
674:   Graphics.update
675:   total_maps = mapData.mapinfos.keys.length
676:   Console.echo_h1(_INTL("Checking {1} maps for invalid tiles", total_maps))
677:   mapData.mapinfos.keys.sort.each do |id|
678:     if System.uptime - t >= 5
679:       t += 5
680:       Graphics.update
681:     end
682:     map_errors = 0
683:     map = mapData.getMap(id)
684:     next if !map || !mapData.mapinfos[id]
685:     passages = mapData.getTilesetPassages(map, id)
686:     # Check all tiles in map for non-existent tiles
687:     map.data.xsize.times do |x|
688:       map.data.ysize.times do |y|
689:         map.data.zsize.times do |i|
690:           tile_id = map.data[x, y, i]
691:           next if pbCheckTileValidity(tile_id, map, tilesets, passages)
692:           map.data[x, y, i] = 0
693:           map_errors += 1
694:         end
695:       end
696:     end
697:     # Check all events in map for page graphics using a non-existent tile
698:     map.events.each_key do |key|
699:       event = map.events[key]
700:       event.pages.each do |page|
701:         next if page.graphic.tile_id <= 0
702:         next if pbCheckTileValidity(page.graphic.tile_id, map, tilesets, passages)
703:         page.graphic.tile_id = 0
704:         map_errors += 1
705:       end
706:     end
707:     next if map_errors == 0
708:     # Map was changed; save it
709:     Console.echoln_li(_INTL("{1} error tile(s) found on map {2}: {3}.", map_errors, id, mapData.mapinfos[id].name))
710:     total_errors += map_errors
711:     num_error_maps += 1
712:     mapData.saveMap(id)
713:   end
714:   if num_error_maps == 0
715:     Console.echo_h2(_INTL("Done. No errors found."), text: :green)
716:     pbMessage(_INTL("No invalid tiles were found."))
717:   else
718:     echoln ""
719:     Console.echo_h2(_INTL("Done. {1} errors found and fixed.", total_errors), text: :green)
720:     Console.echo_warn(_INTL("RMXP data was altered. Close RMXP now to ensure changes are applied."))
721:     echoln ""
722:     pbMessage(_INTL("{1} error(s) were found across {2} map(s) and fixed.", total_errors, num_error_maps))
723:     pbMessage(_INTL("Close RPG Maker XP to ensure the changes are applied properly."))
724:   end
725: end
