609:   def update
610:     return if !@_iconBitmap
611:     @updating = true
612:     # Update bitmap
613:     @_iconBitmap.update
614:     self.bitmap = @_iconBitmap.bitmap
615:     # Pokémon sprite bobbing while Pokémon is selected
616:     @spriteYExtra = 0
617:     if @selected == 1 && COMMAND_BOBBING_DURATION    # When choosing commands for this Pokémon
618:       bob_delta = System.uptime % COMMAND_BOBBING_DURATION   # 0-COMMAND_BOBBING_DURATION
619:       bob_frame = (4 * bob_delta / COMMAND_BOBBING_DURATION).floor
620:       case bob_frame
621:       when 1 then @spriteYExtra = 2
622:       when 3 then @spriteYExtra = -2
623:       end
624:     end
625:     self.x       = self.x
626:     self.y       = self.y
627:     self.visible = @spriteVisible
628:     # Pokémon sprite blinking when targeted
629:     if @selected == 2 && @spriteVisible && TARGET_BLINKING_DURATION
630:       blink_delta = System.uptime % TARGET_BLINKING_DURATION   # 0-TARGET_BLINKING_DURATION
631:       blink_frame = (3 * blink_delta / TARGET_BLINKING_DURATION).floor
632:       self.visible = (blink_frame != 0)
633:     end
634:     @updating = false
635:   end
