291:   def write_pokemon_metrics
292:     paths = []
293:     GameData::SpeciesMetrics.each do |element|
294:       next if element.form == 0
295:       paths.push(element.pbs_file_suffix) if !paths.include?(element.pbs_file_suffix)
296:     end
297:     paths.each_with_index do |element, i|
298:       paths[i] = [sprintf("PBS/%s.txt", GameData::SpeciesMetrics::PBS_BASE_FILENAME), element]
299:       if !nil_or_empty?(element)
300:         paths[i][0] = sprintf("PBS/%s_%s.txt", GameData::SpeciesMetrics::PBS_BASE_FILENAME, element)
301:       end
302:     end
303:     schema = GameData::SpeciesMetrics.schema
304:     idx = 0
305:     paths.each do |path|
306:       write_pbs_file_message_start(path[0])
307:       File.open(path[0], "wb") do |f|
308:         add_PBS_header_to_file(f)
309:         # Write each element in turn
310:         GameData::SpeciesMetrics.each do |element|
311:           next if element.pbs_file_suffix != path[1]
312:           if element.form > 0
313:             base_element = GameData::SpeciesMetrics.get(element.species)
314:             next if element.back_sprite == base_element.back_sprite &&
315:                     element.front_sprite == base_element.front_sprite &&
316:                     element.front_sprite_altitude == base_element.front_sprite_altitude &&
317:                     element.shadow_x == base_element.shadow_x &&
318:                     element.shadow_size == base_element.shadow_size
319:           end
320:           echo "." if idx % 100 == 0
321:           Graphics.update if idx % 500 == 0
322:           idx += 1
323:           f.write("\#-------------------------------\r\n")
324:           if schema["SectionName"]
325:             f.write("[")
326:             pbWriteCsvRecord(element.get_property_for_PBS("SectionName"), f, schema["SectionName"])
327:             f.write("]\r\n")
328:           else
329:             f.write("[#{element.id}]\r\n")
330:           end
331:           schema.each_key do |key|
332:             next if key == "SectionName"
333:             val = element.get_property_for_PBS(key)
334:             next if val.nil?
335:             if schema[key][1][0] == "^" && val.is_a?(Array)
336:               val.each do |sub_val|
337:                 f.write(sprintf("%s = ", key))
338:                 pbWriteCsvRecord(sub_val, f, schema[key])
339:                 f.write("\r\n")
340:               end
341:             else
342:               f.write(sprintf("%s = ", key))
343:               pbWriteCsvRecord(val, f, schema[key])
344:               f.write("\r\n")
345:             end
346:           end
347:         end
348:       end
349:       process_pbs_file_message_end
350:     end
