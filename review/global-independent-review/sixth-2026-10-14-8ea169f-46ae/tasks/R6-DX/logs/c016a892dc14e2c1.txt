195:     def register(species, gender = 0, form = 0, shiny = false, should_refresh_dexes = true)
196:       if species.is_a?(Pokemon)
197:         species_data = species.species_data
198:         gender = species.gender
199:         shiny = species.shiny?
200:       else
201:         species_data = GameData::Species.get_species_form(species, form)
202:       end
203:       species = species_data.species
204:       gender = 0 if gender >= 2
205:       form = species_data.form
206:       shin = (shiny) ? 1 : 0
207:       if form != species_data.pokedex_form
208:         species_data = GameData::Species.get_species_form(species, species_data.pokedex_form)
209:         form = species_data.form
210:       end
211:       form = 0 if species_data.form_name.nil? || species_data.form_name.empty?
212:       # Register as seen
213:       @seen[species] = true
214:       @seen_forms[species] ||= [[[], []], [[], []]]
215:       @seen_forms[species][gender][shin][form] = true
216:       @last_seen_forms[species] ||= []
217:       @last_seen_forms[species] = [gender, form, shiny] if @last_seen_forms[species] == []
218:       self.refresh_accessible_dexes if should_refresh_dexes
219:     end
220: 
221:     # @param pkmn [Pokemon] Pokemon to register as most recently seen
222:     def register_last_seen(pkmn)
223:       validate pkmn => Pokemon
224:       species_data = pkmn.species_data
225:       form = species_data.pokedex_form
226:       form = 0 if species_data.form_name.nil? || species_data.form_name.empty?
227:       @last_seen_forms[pkmn.species] = [pkmn.gender, form, pkmn.shiny?]
228:     end
