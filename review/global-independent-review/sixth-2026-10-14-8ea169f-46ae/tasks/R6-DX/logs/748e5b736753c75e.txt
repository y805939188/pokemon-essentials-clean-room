541:     @sprite.z = 2
542:     @pkmnList = Window_UnformattedTextPokemon.newWithSize(

544:     )

546:     @selection = selection

548:     @ids = []
549:     @includeNew = includeNew
550:     @index = 0
551:   end
552: 
553:   def dispose
554:     @sprite.bitmap&.dispose
555:     @sprite.dispose
556:     @pkmnList.dispose
557:   end
558: 
559:   def setViewport(viewport)
560:     @sprite.viewport   = viewport
561:     @pkmnList.viewport = viewport
562:   end
563: 
564:   def startIndex
565:     return @index
566:   end
567: 
568:   def commands
569:     @commands.clear
570:     @ids.clear
571:     cmds = []
572:     idx = 1
573:     GameData::Trainer.each do |trainer|
574:       cmds.push([idx, trainer.trainer_type, trainer.real_name, trainer.version])
575:       idx += 1
576:     end
577:     cmds.sort! do |a, b|
578:       if a[1] == b[1]
579:         if a[2] == b[2]
580:           (a[3] == b[3]) ? a[0] <=> b[0] : a[3] <=> b[3]
581:         else
582:           a[2].downcase <=> b[2].downcase
583:         end
584:       else
585:         a[1].to_s.downcase <=> b[1].to_s.downcase
586:       end
587:     end
588:     if @includeNew
589:       @commands.push(_INTL("[NEW TRAINER BATTLE]"))
590:       @ids.push(true)
591:     end
592:     cmds.each do |t|
593:       if t[3] > 0
594:         @commands.push(_INTL("{1} {2} ({3}) x{4}",
595:                              GameData::TrainerType.get(t[1]).name, t[2], t[3],
596:                              GameData::Trainer.get(t[1], t[2], t[3]).pokemon.length))
597:       else
598:         @commands.push(_INTL("{1} {2} x{3}",
599:                              GameData::TrainerType.get(t[1]).name, t[2],
600:                              GameData::Trainer.get(t[1], t[2], t[3]).pokemon.length))
601:       end
602:       @ids.push([t[1], t[2], t[3]])
603:     end
604:     @index = @selection
605:     @index = @commands.length - 1 if @index >= @commands.length
606:     @index = 0 if @index < 0
607:     return @commands
608:   end
609:

611:     return nil if index < 0
612:     return @ids[index]
613:   end
614: 
615:   def refresh(index)
616:     # Refresh trainer sprite
617:     @sprite.bitmap&.dispose
618:     return if index < 0
619:     begin
620:       if @ids[index].is_a?(Array)
621:         @sprite.setBitmap(GameData::TrainerType.front_sprite_filename(@ids[index][0]), 0)
622:       else
623:         @sprite.setBitmap(nil)
624:       end
625:     rescue
626:       @sprite.setBitmap(nil)
627:     end
628:     if @sprite.bitmap
629:       @sprite.ox = @sprite.bitmap.width / 2
630:       @sprite.oy = @sprite.bitmap.height
631:     end
632:     # Refresh list of Pokémon
633:     text = ""
634:     if !@includeNew || index > 0
635:       tr_data = GameData::Trainer.get(@ids[index][0], @ids[index][1], @ids[index][2])
636:       if tr_data
637:         tr_data.pokemon.each_with_index do |pkmn, i|
638:           text += "\n" if i > 0
639:           text += sprintf("%s Lv.%d", GameData::Species.get(pkmn[:species]).real_name, pkmn[:level])
640:         end
641:       end
642:     end
643:     @pkmnList.text = text
644:     @pkmnList.resizeHeightToFit(text, Graphics.width / 2)
645:     @pkmnList.y = Graphics.height - @pkmnList.height
646:   end
