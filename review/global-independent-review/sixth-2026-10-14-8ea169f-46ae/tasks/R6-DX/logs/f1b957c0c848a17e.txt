52:     def self.check_file(tr_type, path, optional_suffix = "", suffix = "")
53:       tr_type_data = self.try_get(tr_type)
54:       return nil if tr_type_data.nil?
55:       # Check for files
56:       if optional_suffix && !optional_suffix.empty?
57:         ret = path + tr_type_data.id.to_s + optional_suffix + suffix
58:         return ret if pbResolveBitmap(ret)
59:       end
60:       ret = path + tr_type_data.id.to_s + suffix
61:       return (pbResolveBitmap(ret)) ? ret : nil
62:     end
