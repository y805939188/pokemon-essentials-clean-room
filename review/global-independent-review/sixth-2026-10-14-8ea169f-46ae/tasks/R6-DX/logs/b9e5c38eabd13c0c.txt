429: #   echoln "mousemove (#{mapid},#{x},#{y})"
430:     if @dragging

432:         sprite = getMapSprite(@dragmapid)

434:         y += @dragOffsetY
435:         sprite.x = x & ~3
436:         sprite.y = y & ~3
437:         @sprites["title"].text = _ISPRINTF("D: Help [{1:03d}: {2:s}]", mapid, @mapinfos[@dragmapid].name)
438:       else
439:         xpos = x - @dragOffsetX
440:         ypos = y - @dragOffsetY
441:         @mapspritepos.each_key do |i|
442:           sprite = getMapSprite(i)
443:           sprite.x = (@mapspritepos[i][0] + xpos) & ~3
444:           sprite.y = (@mapspritepos[i][1] + ypos) & ~3
445:         end
446:         @sprites["title"].text = _INTL("D: Help")
447:       end
448:     elsif mapid >= 0
449:       @sprites["title"].text = _ISPRINTF("D: Help [{1:03d}: {2:s}]", mapid, @mapinfos[mapid].name)
450:     else
451:       @sprites["title"].text = _INTL("D: Help")
452:     end

455:   def hittest(x, y)
456:     @mapsprites.each_key do |i|
457:       sx = @mapsprites[i].x
458:       sy = @mapsprites[i].y
459:       sr = sx + @mapsprites[i].bitmap.width
460:       sb = sy + @mapsprites[i].bitmap.height
461:       return i if x >= sx && x < sr && y >= sy && y < sb
462:     end
463:     return -1
464:   end

470:   def update
471:     mousepos = Mouse.getMousePos
472:     if mousepos
473:       hitmap = hittest(mousepos[0], mousepos[1])
474:       if Input.trigger?(Input::MOUSELEFT)
475:         onClick(hitmap, mousepos[0], mousepos[1])
476:       elsif Input.trigger?(Input::MOUSERIGHT)
477:         onRightClick(hitmap, mousepos[0], mousepos[1])
478:       elsif Input.release?(Input::MOUSELEFT)
479:         onMouseUp(hitmap)
480:       elsif Input.release?(Input::MOUSERIGHT)
481:         onRightMouseUp(hitmap)
482:       else
483:         if @lasthitmap != hitmap
484:           onMouseOver(hitmap, mousepos[0], mousepos[1])
485:           @lasthitmap = hitmap
486:         end
487:         if @oldmousex != mousepos[0] || @oldmousey != mousepos[1]
488:           onMouseMove(hitmap, mousepos[0], mousepos[1])
489:           @oldmousex = mousepos[0]
490:           @oldmousey = mousepos[1]
491:         end
492:       end
493:     end
494:     if Input.press?(Input::UP)
495:       @mapsprites.each do |i|
496:         i[1].y += 4 if i
497:       end
498:     end
499:     if Input.press?(Input::DOWN)
500:       @mapsprites.each do |i|
501:         i[1].y -= 4 if i
502:       end
503:     end
504:     if Input.press?(Input::LEFT)
505:       @mapsprites.each do |i|
506:         i[1].x += 4 if i
507:       end
508:     end
509:     if Input.press?(Input::RIGHT)
510:       @mapsprites.each do |i|
511:         i[1].x -= 4 if i
512:       end
513:     end
514:     if Input.triggerex?(:A)
515:       id = chooseMapScreen(_INTL("Add Map"), @currentmap)
516:       if id > 0
517:         addSprite(id)
518:         setTopSprite(id)
519:         @mapconns = generateConnectionData
520:       end
521:     elsif Input.triggerex?(:S)
522:       id = chooseMapScreen(_INTL("Go to Map"), @currentmap)
523:       if id > 0
524:         @mapconns = generateConnectionData
525:         pbDisposeSpriteHash(@mapsprites)
526:         @mapsprites.clear
527:         @sprites["selsprite"].othersprite = nil
528:         @selmapid = -1
529:         putSprite(id)
530:         @currentmap = id
531:       end
532:     elsif Input.triggerex?(:DELETE)
533:       if @mapsprites.keys.length > 1 && @selmapid >= 0
534:         @mapsprites[@selmapid].bitmap.dispose
535:         @mapsprites[@selmapid].dispose
536:         @mapsprites.delete(@selmapid)
537:         @sprites["selsprite"].othersprite = nil
538:         @selmapid = -1
539:       end
540:     elsif Input.triggerex?(:D)
541:       helpWindow
542:     end
543:     pbUpdateSpriteHash(@sprites)
544:   end
