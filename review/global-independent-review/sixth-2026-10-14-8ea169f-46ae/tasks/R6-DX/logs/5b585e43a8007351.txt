292:   def validate_compiled_pokemon(hash)
293:     # Convert base stats array to a hash
294:     if hash[:base_stats].is_a?(Array)
295:       new_stats = {}
296:       GameData::Stat.each_main do |s|
297:         new_stats[s.id] = (hash[:base_stats][s.pbs_order] || 1) if s.pbs_order >= 0
298:       end
299:       hash[:base_stats] = new_stats
300:     end
301:     # Convert EVs array to a hash
302:     if hash[:evs].is_a?(Array)
303:       new_evs = {}
304:       hash[:evs].each { |val| new_evs[val[0]] = val[1] }
305:       GameData::Stat.each_main { |s| new_evs[s.id] ||= 0 }
306:       hash[:evs] = new_evs
307:     end
308:     # Convert height and weight to integer values of tenths of a unit
309:     hash[:height] = [(hash[:height] * 10).round, 1].max if hash[:height]
310:     hash[:weight] = [(hash[:weight] * 10).round, 1].max if hash[:weight]
311:     # Ensure evolutions have a parameter if they need one (don't need to ensure
312:     # the parameter makes sense; that happens below)
313:     if hash[:evolutions]
314:       hash[:evolutions].each do |evo|
315:         FileLineData.setSection(hash[:id].to_s, "Evolution", "Evolution = #{evo[0]},#{evo[1]}")   # For error reporting
316:         param_type = GameData::Evolution.get(evo[1]).parameter
317:         next if evo[2] || param_type.nil?
318:         raise _INTL("Evolution method {1} requires a parameter, but none was given.", evo[1]) + "\n" + FileLineData.linereport
319:       end
320:     end
321:     # Record all evolutions as not being prevolutions
322:     if hash[:evolutions].is_a?(Array)
323:       hash[:evolutions].each { |evo| evo[3] = false }
324:     end
325:     # Remove duplicate types
326:     if hash[:types].is_a?(Array)
327:       hash[:types].uniq!
328:       hash[:types].compact!
329:     end
330:   end
