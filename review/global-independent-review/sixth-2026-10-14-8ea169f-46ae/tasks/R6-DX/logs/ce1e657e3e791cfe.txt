624:   def pbEditBG(canvas, timing)
625:     ret = false
626:     maxsizewindow = ControlWindow.new(0, 0, 320, 32 * 11)
627:     maxsizewindow.addSlider(_INTL("Duration:"), 0, 50, timing.duration)
628:     maxsizewindow.addOptionalSlider(_INTL("X:"), -500, 500, timing.bgX || 0)
629:     maxsizewindow.addOptionalSlider(_INTL("Y:"), -500, 500, timing.bgY || 0)
630:     maxsizewindow.addOptionalSlider(_INTL("Opacity:"), 0, 255, timing.opacity || 0)
631:     maxsizewindow.addOptionalSlider(_INTL("Red:"), 0, 255, timing.colorRed || 0)
632:     maxsizewindow.addOptionalSlider(_INTL("Green:"), 0, 255, timing.colorGreen || 0)
633:     maxsizewindow.addOptionalSlider(_INTL("Blue:"), 0, 255, timing.colorBlue || 0)
634:     maxsizewindow.addOptionalSlider(_INTL("Alpha:"), 0, 255, timing.colorAlpha || 0)
635:     maxsizewindow.addButton(_INTL("OK"))
636:     maxsizewindow.addButton(_INTL("Cancel"))
637:     maxsizewindow.controls[1].checked = !timing.bgX.nil?
638:     maxsizewindow.controls[2].checked = !timing.bgY.nil?
639:     maxsizewindow.controls[3].checked = !timing.opacity.nil?
640:     maxsizewindow.controls[4].checked = !timing.colorRed.nil?
641:     maxsizewindow.controls[5].checked = !timing.colorGreen.nil?
642:     maxsizewindow.controls[6].checked = !timing.colorBlue.nil?
643:     maxsizewindow.controls[7].checked = !timing.colorAlpha.nil?
644:     maxsizewindow.opacity = 200
645:     maxsizewindow.viewport = canvas.viewport
646:     loop do
647:       Graphics.update
648:       Input.update
649:       maxsizewindow.update
650:       if maxsizewindow.changed?(8)   # OK
651:         if maxsizewindow.controls[1].checked ||
652:            maxsizewindow.controls[2].checked ||
653:            maxsizewindow.controls[3].checked ||
654:            maxsizewindow.controls[4].checked ||
655:            maxsizewindow.controls[5].checked ||
656:            maxsizewindow.controls[6].checked ||
657:            maxsizewindow.controls[7].checked
658:           timing.duration = maxsizewindow.value(0)
659:           timing.bgX = maxsizewindow.value(1)
660:           timing.bgY = maxsizewindow.value(2)
661:           timing.opacity = maxsizewindow.value(3)
662:           timing.colorRed = maxsizewindow.value(4)
663:           timing.colorGreen = maxsizewindow.value(5)
664:           timing.colorBlue = maxsizewindow.value(6)
665:           timing.colorAlpha = maxsizewindow.value(7)
666:           ret = true
667:         end
668:         break
669:       end
670:       break if maxsizewindow.changed?(9)   # Cancel
671:       if Input.trigger?(Input::BACK)
672:         break
673:       end
674:     end
675:     maxsizewindow.dispose
676:     return ret
677:   end
