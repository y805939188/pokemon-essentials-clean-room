194:   def getCatmullRomPoint(src, t)
195:     x = 0, y = 0
196:     t *= 3.0
197:     if t < 1.0
198:       x = catmullRom(src[0].x, src[0].x, src[1].x, src[2].x, t)
199:       y = catmullRom(src[0].y, src[0].y, src[1].y, src[2].y, t)
200:     elsif t < 2.0
201:       t -= 1.0
202:       x = catmullRom(src[0].x, src[1].x, src[2].x, src[3].x, t)
203:       y = catmullRom(src[0].y, src[1].y, src[2].y, src[3].y, t)
204:     else
205:       t -= 2.0
206:       x = catmullRom(src[1].x, src[2].x, src[3].x, src[3].x, t)
207:       y = catmullRom(src[1].y, src[2].y, src[3].y, src[3].y, t)
208:     end
209:     return [x, y]
210:   end

212:   def getCurvePoint(src, t)
213:     return getCatmullRomPoint(src, t)
214:   end

216:   def curveToPointPath(curve, numpoints)
217:     return nil if numpoints < 2
218:     path = PointPath.new
219:     step = 1.0 / (numpoints - 1)
220:     t = 0.0
221:     numpoints.times do
222:       point = getCurvePoint(curve, t)
223:       path.addPoint(point[0], point[1])
224:       t += step
225:     end
226:     return path
227:   end
