672:       cel = @animation[self.currentframe]
673:       return false if !cel

678:     def deletable?(celindex)
679:       cel = @animation[self.currentframe]
680:       return true if !cel
681:       cel = cel[celindex]
682:       return true if !cel
683:       return false if cel[AnimFrame::LOCKED] != 0
684:       if cel[AnimFrame::PATTERN] < 0
685:         count = 0
686:         pattern = cel[AnimFrame::PATTERN]
687:         PBAnimation::MAX_SPRITES.times do |i|
688:           othercel = @animation[self.currentframe][i]
689:           count += 1 if othercel && othercel[AnimFrame::PATTERN] == pattern
690:         end
691:         return false if count <= 1
692:       end
693:       return true
694:     end
