1090:   end

1092:   # @return [Integer] the specified stat of this Pokémon (not used for total HP)
1093:   def calcStat(base, level, iv, ev, nat)
1094:     iv = ev = 0 if Settings::DISABLE_IVS_AND_EVS
1095:     return (((((base * 2) + iv + (ev / 4)) * level / 100).floor + 5) * nat / 100).floor
1096:   end

1098:   # Recalculates this Pokémon's stats.
1099:   def calc_stats
1100:     base_stats = self.baseStats
1101:     this_level = self.level
1102:     this_IV    = self.calcIV
1103:     # Format stat multipliers due to nature
1104:     nature_mod = {}
1105:     GameData::Stat.each_main { |s| nature_mod[s.id] = 100 }
1106:     this_nature = self.nature_for_stats
1107:     if this_nature
1108:       this_nature.stat_changes.each { |change| nature_mod[change[0]] += change[1] }
1109:     end
1110:     # Calculate stats
1111:     stats = {}
1112:     GameData::Stat.each_main do |s|
1113:       if s.id == :HP
1114:         stats[s.id] = calcHP(base_stats[s.id], this_level, this_IV[s.id], @ev[s.id])
1115:       else
1116:         stats[s.id] = calcStat(base_stats[s.id], this_level, this_IV[s.id], @ev[s.id], nature_mod[s.id])
1117:       end
1118:     end
1119:     hp_difference = stats[:HP] - @totalhp
1120:     @totalhp = stats[:HP]
1121:     self.hp = [@hp + hp_difference, 1].max if @hp > 0 || hp_difference > 0
1122:     @attack  = stats[:ATTACK]
1123:     @defense = stats[:DEFENSE]
1124:     @spatk   = stats[:SPECIAL_ATTACK]
1125:     @spdef   = stats[:SPECIAL_DEFENSE]
1126:     @speed   = stats[:SPEED]
1127:   end
