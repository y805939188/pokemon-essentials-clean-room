746: def chooseMapPoint(map, rgnmap = false)
747:   viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
748:   viewport.z = 99999
749:   title = Window_UnformattedTextPokemon.newWithSize(
750:     _INTL("Click a point on the map."), 0, Graphics.height - 64, Graphics.width, 64, viewport
751:   )
752:   title.z = 2
753:   if rgnmap
754:     sprite = RegionMapSprite.new(map, viewport)
755:   else
756:     sprite = MapSprite.new(map, viewport)
757:   end
758:   sprite.z = 2
759:   ret = nil
760:   loop do
761:     Graphics.update
762:     Input.update
763:     xy = sprite.getXY
764:     if xy
765:       ret = xy
766:       break
767:     end
768:     if Input.trigger?(Input::BACK)
769:       ret = nil
770:       break
771:     end
772:   end
773:   sprite.dispose
774:   title.dispose
775:   return ret
