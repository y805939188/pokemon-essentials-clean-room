201:   def createPokemon(level, iv, trainer)
202:     pkmn = Pokemon.new(@species, level, trainer, false)
203:     pkmn.item = @item
204:     pkmn.personalID = rand(2**16) | (rand(2**16) << 16)
205:     pkmn.nature = nature
206:     pkmn.happiness = 0
207:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move1)))
208:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move2))) if @move2
209:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move3))) if @move3
210:     pkmn.moves.push(Pokemon::Move.new(self.convertMove(@move4))) if @move4
211:     pkmn.moves.compact!
212:     if ev.length > 0
213:       ev.each { |stat| pkmn.ev[stat] = Pokemon::EV_LIMIT / ev.length }
214:     end
215:     GameData::Stat.each_main { |s| pkmn.iv[s.id] = iv }
216:     pkmn.calc_stats
217:     return pkmn
