401: def pbRaiseEffortValues(pkmn, stat, evGain = 10, no_ev_cap = false)
402:   stat = GameData::Stat.get(stat).id
403:   return 0 if !no_ev_cap && pkmn.ev[stat] >= 100
404:   evTotal = 0
405:   GameData::Stat.each_main { |s| evTotal += pkmn.ev[s.id] }
406:   evGain = evGain.clamp(0, Pokemon::EV_STAT_LIMIT - pkmn.ev[stat])
407:   evGain = evGain.clamp(0, 100 - pkmn.ev[stat]) if !no_ev_cap
408:   evGain = evGain.clamp(0, Pokemon::EV_LIMIT - evTotal)
409:   if evGain > 0
410:     pkmn.ev[stat] += evGain
411:     pkmn.calc_stats
412:   end
413:   return evGain
414: end

416: def pbMaxUsesOfEVRaisingItem(stat, amt_per_use, pkmn, no_ev_cap = false)
417:   max_per_stat = (no_ev_cap) ? Pokemon::EV_STAT_LIMIT : 100
418:   amt_can_gain = max_per_stat - pkmn.ev[stat]
419:   ev_total = 0
420:   GameData::Stat.each_main { |s| ev_total += pkmn.ev[s.id] }
421:   amt_can_gain = [amt_can_gain, Pokemon::EV_LIMIT - ev_total].min
422:   return [(amt_can_gain.to_f / amt_per_use).ceil, 1].max
423: end
