13: def pbListScreen(title, lister)
14:   viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
15:   viewport.z = 99999
16:   list = pbListWindow([])
17:   list.viewport = viewport
18:   list.z        = 2
19:   title = Window_UnformattedTextPokemon.newWithSize(
20:     title, Graphics.width / 2, 0, Graphics.width / 2, 64, viewport
21:   )
22:   title.z = 2
23:   lister.setViewport(viewport)
24:   selectedmap = -1
25:   commands = lister.commands
26:   selindex = lister.startIndex
27:   if commands.length == 0
28:     value = lister.value(-1)
29:     lister.dispose
30:     title.dispose
31:     list.dispose
32:     viewport.dispose
33:     return value
34:   end
35:   list.commands = commands
36:   list.index    = selindex
37:   loop do
38:     Graphics.update
39:     Input.update
40:     list.update
41:     if list.index != selectedmap
42:       lister.refresh(list.index)
43:       selectedmap = list.index
44:     end
45:     if Input.trigger?(Input::BACK)
46:       selectedmap = -1
47:       break
48:     elsif Input.trigger?(Input::USE)
49:       break
50:     end
51:   end
52:   value = lister.value(selectedmap)
53:   lister.dispose
54:   title.dispose
55:   list.dispose
56:   viewport.dispose
57:   Input.update
58:   return value
59: end

65:   list.viewport = viewport

67:   title = Window_UnformattedTextPokemon.newWithSize(
68:     title, Graphics.width / 2, 0, Graphics.width / 2, 64, viewport
69:   )
70:   title.z = 2
71:   lister.setViewport(viewport)
72:   selectedmap = -1
73:   commands = lister.commands
74:   selindex = lister.startIndex
75:   if commands.length == 0
76:     value = lister.value(-1)
77:     lister.dispose
78:     title.dispose
79:     list.dispose
80:     viewport.dispose
81:     return value
82:   end
83:   list.commands = commands
84:   list.index = selindex
85:   loop do
86:     Graphics.update
87:     Input.update
88:     list.update
89:     if list.index != selectedmap
90:       lister.refresh(list.index)
91:       selectedmap = list.index
92:     end
93:     if Input.trigger?(Input::ACTION)
94:       yield(Input::ACTION, lister.value(selectedmap))
95:       list.commands = lister.commands
96:       list.index = list.commands.length if list.index == list.commands.length
97:       lister.refresh(list.index)
98:     elsif Input.trigger?(Input::BACK)
99:       break
100:     elsif Input.trigger?(Input::USE)
101:       yield(Input::USE, lister.value(selectedmap))
102:       list.commands = lister.commands
103:       list.index = list.commands.length if list.index == list.commands.length
104:       lister.refresh(list.index)
105:     end
106:   end
107:   lister.dispose
108:   title.dispose
109:   list.dispose
110:   viewport.dispose
111:   Input.update
112: end

117: class GraphicsLister
118:   def initialize(folder, selection)
119:     @sprite = IconSprite.new(0, 0)
120:     @sprite.bitmap = nil
121:     @sprite.x      = Graphics.width * 3 / 4
122:     @sprite.y      = ((Graphics.height - 64) / 2) + 64
123:     @sprite.z      = 2
124:     @folder = folder
125:     @selection = selection
126:     @commands = []
127:     @index = 0
128:   end
129: 
130:   def dispose
131:     @sprite.bitmap&.dispose
132:     @sprite.dispose
133:   end
134: 
135:   def setViewport(viewport)
136:     @sprite.viewport = viewport
137:   end
138: 
139:   def startIndex
140:     return @index
141:   end
142: 
143:   def commands
144:     @commands.clear
145:     Dir.chdir(@folder) do
146:       Dir.glob("*.png") { |f| @commands.push(f) }
147:       Dir.glob("*.gif") { |f| @commands.push(f) }
148: #      Dir.glob("*.jpg") { |f| @commands.push(f) }
149: #      Dir.glob("*.jpeg") { |f| @commands.push(f) }
150: #      Dir.glob("*.bmp") { |f| @commands.push(f) }
151:     end
152:     @commands.sort!
153:     @commands.length.times do |i|
154:       @index = i if @commands[i] == @selection
155:     end
156:     pbMessage(_INTL("There are no files.")) if @commands.length == 0
157:     return @commands
158:   end
159: 
160:   def value(index)
161:     return (index < 0) ? "" : @commands[index]
162:   end
163: 
164:   def refresh(index)
165:     return if index < 0
166:     @sprite.setBitmap(@folder + @commands[index])
167:     sprite_width = @sprite.bitmap.width
168:     sprite_height = @sprite.bitmap.height
169:     @sprite.ox = sprite_width / 2
170:     @sprite.oy = sprite_height / 2
171:     scale_x = (Graphics.width / 2).to_f / sprite_width
172:     scale_y = (Graphics.height - 64).to_f / sprite_height
173:     if scale_x < 1.0 || scale_y < 1.0
174:       min_scale = [scale_x, scale_y].min
175:       @sprite.zoom_x = @sprite.zoom_y = min_scale
176:     else
177:       @sprite.zoom_x = @sprite.zoom_y = 1.0
178:     end
179:   end
180: end

185: class MusicFileLister
186:   def initialize(bgm, setting)
187:     @oldbgm = getPlayingBGM
188:     @commands = []
189:     @bgm = bgm
190:     @setting = setting
191:     @index = 0
192:   end
193: 
194:   def dispose
195:     pbPlayBGM(@oldbgm)
196:   end
197: 
198:   def setViewport(viewport); end
199: 
200:   def getPlayingBGM
201:     ($game_system) ? $game_system.getPlayingBGM : nil
202:   end
203: 
204:   def pbPlayBGM(bgm)
205:     (bgm) ? pbBGMPlay(bgm) : pbBGMStop
206:   end
207: 
208:   def startIndex
209:     return @index
210:   end
211: 
212:   def commands
213:     folder = (@bgm) ? "Audio/BGM/" : "Audio/ME/"
214:     @commands.clear
215:     Dir.chdir(folder) do
216:       Dir.glob("*.wav") { |f| @commands.push(f) }
217:       Dir.glob("*.ogg") { |f| @commands.push(f) }
218:       Dir.glob("*.mp3") { |f| @commands.push(f) }
219:       Dir.glob("*.midi") { |f| @commands.push(f) }
220:       Dir.glob("*.mid") { |f| @commands.push(f) }
221:       Dir.glob("*.wma") { |f| @commands.push(f) }
222:     end
223:     @commands.uniq!
224:     @commands.sort! { |a, b| a.downcase <=> b.downcase }
225:     @commands.length.times do |i|
226:       @index = i if @commands[i] == @setting
227:     end
228:     pbMessage(_INTL("There are no files.")) if @commands.length == 0
229:     return @commands
230:   end
231: 
232:   def value(index)
233:     return (index < 0) ? "" : @commands[index]
234:   end
235: 
236:   def refresh(index)
237:     return if index < 0
238:     if @bgm
239:       pbPlayBGM(@commands[index])
240:     else
241:       pbPlayBGM("../../Audio/ME/" + @commands[index])
242:     end
243:   end
244: end

249: class MetadataLister
250:   def initialize(sel_player_id = -1, new_player = false)
251:     @index = 0
252:     @commands = []
253:     @player_ids = []
254:     GameData::PlayerMetadata.each do |player|
255:       @index = @commands.length + 1 if sel_player_id > 0 && player.id == sel_player_id
256:       @player_ids.push(player.id)
257:     end
258:     @new_player = new_player
259:   end
260: 
261:   def dispose; end
262: 
263:   def setViewport(viewport); end
264: 
265:   def startIndex
266:     return @index
267:   end
268: 
269:   def commands
270:     @commands.clear
271:     @commands.push(_INTL("[GLOBAL METADATA]"))
272:     @player_ids.each { |id| @commands.push(_INTL("Player {1}", id)) }
273:     @commands.push(_INTL("[ADD NEW PLAYER]")) if @new_player
274:     return @commands
275:   end
276: 
277:   # Cancel: -1
278:   # New player: -2
279:   # Global metadata: 0
280:   # Player character: 1+ (the player ID itself)
281:   def value(index)
282:     return index if index < 1
283:     return -2 if @new_player && index == @commands.length - 1
284:     return @player_ids[index - 1]
285:   end
286: 
287:   def refresh(index); end
288: end

293: class MapLister
294:   def initialize(selmap, addGlobal = false)
295:     @sprite = Sprite.new
296:     @sprite.bitmap = nil
297:     @sprite.x      = Graphics.width * 3 / 4
298:     @sprite.y      = ((Graphics.height - 64) / 2) + 64
299:     @sprite.z      = -2
300:     @commands = []
301:     @maps = pbMapTree
302:     @addGlobalOffset = (addGlobal) ? 1 : 0
303:     @index = 0
304:     @maps.length.times do |i|
305:       @index = i + @addGlobalOffset if @maps[i][0] == selmap
306:     end
307:   end
308: 
309:   def dispose
310:     @sprite.bitmap&.dispose
311:     @sprite.dispose
312:   end
313: 
314:   def setViewport(viewport)
315:     @sprite.viewport = viewport
316:   end
317: 
318:   def startIndex
319:     return @index
320:   end
321: 
322:   def commands
323:     @commands.clear
324:     @commands.push(_INTL("[GLOBAL]")) if @addGlobalOffset == 1
325:     @maps.length.times do |i|
326:       @commands.push(sprintf("%s%03d %s", ("  " * @maps[i][2]), @maps[i][0], @maps[i][1]))
327:     end
328:     return @commands
329:   end
330: 
331:   def value(index)
332:     return 0 if @addGlobalOffset == 1 && index == 0
333:     return (index < 0) ? -1 : @maps[index - @addGlobalOffset][0]
334:   end
335: 
336:   def refresh(index)
337:     @sprite.bitmap&.dispose
338:     return if index < 0
339:     return if index == 0 && @addGlobalOffset == 1
340:     @sprite.bitmap = createMinimap(@maps[index - @addGlobalOffset][0])
