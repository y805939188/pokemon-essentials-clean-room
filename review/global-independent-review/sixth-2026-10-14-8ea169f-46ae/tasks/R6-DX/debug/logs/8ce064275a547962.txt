5: def pbSetLotteryNumber(variable = 1)
6:   t = pbGetTimeNow
7:   hash = t.day + (t.month << 5) + (t.year << 9)
8:   srand(hash)                      # seed RNG with fixed value depending on date
9:   lottery = rand(65_536)           # get a number
10:   srand                            # reseed RNG
11:   pbSet(variable, sprintf("%05d", lottery))
12: end
