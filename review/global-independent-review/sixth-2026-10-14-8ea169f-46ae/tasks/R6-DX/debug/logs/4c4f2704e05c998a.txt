180: # Screen for listing the above battle variables for modifying.
181: #===============================================================================
182: class SpriteWindow_DebugBattleFieldEffects < Window_DrawableCommand
183:   BASE_TEXT_COLOR   = Color.new(96, 96, 96)
184:   RED_TEXT_COLOR    = Color.new(168, 48, 56)
185:   GREEN_TEXT_COLOR  = Color.new(0, 144, 0)
186:   TEXT_SHADOW_COLOR = Color.new(208, 208, 200)
187: 
188:   def initialize(viewport, battle, variables, variables_data)
189:     @battle         = battle
190:     @variables      = variables
191:     @variables_data = variables_data
192:     super(0, 0, Graphics.width, Graphics.height, viewport)
193:   end
194: 
195:   def itemCount
196:     return @variables_data.length
197:   end
198: 
199:   def shadowtext(x, y, w, h, t, align = 0, colors = 0)
200:     width = self.contents.text_size(t).width
201:     case align
202:     when 1   # Right aligned
203:       x += w - width
204:     when 2   # Centre aligned
205:       x += (w - width) / 2
206:     end
207:     base_color = BASE_TEXT_COLOR
208:     case colors
209:     when 1 then base_color = RED_TEXT_COLOR
210:     when 2 then base_color = GREEN_TEXT_COLOR
211:     end
212:     pbDrawShadowText(self.contents, x, y, [width, w].max, h, t, base_color, TEXT_SHADOW_COLOR)
213:   end
214: 
215:   def drawItem(index, _count, rect)
216:     pbSetNarrowFont(self.contents)
217:     variable_data = @variables_data[@variables_data.keys[index]]
218:     variable = @variables[@variables_data.keys[index]]
219:     # Variables which aren't their default value are colored differently
220:     default = variable_data[:default]
221:     default = -1 if default == -2
222:     different = (variable || default) != default
223:     color = (different) ? 2 : 0
224:     # Draw cursor
225:     rect = drawCursor(index, rect)
226:     # Get value's text to draw
227:     variable_text = variable.to_s
228:     case variable_data[:default]
229:     when -1   # Battler
230:       if variable >= 0
231:         battler_name = @battle.battlers[variable].name
232:         battler_name = "-" if nil_or_empty?(battler_name)
233:         variable_text = sprintf("[%d] %s", variable, battler_name)
234:       else
235:         variable_text = _INTL("[None]")
236:       end
237:     when nil   # Move, item
238:       variable_text = _INTL("[None]") if !variable
239:     end
240:     # Draw text
