960: MenuHandlers.add(:debug_menu, :edit_phone_contacts, {
961:   "name"        => _INTL("Edit phone and contacts"),
962:   "parent"      => :player_menu,
963:   "description" => _INTL("Edit properties of the phone and of contacts registered in it."),
964:   "effect"      => proc {
965:     if !$PokemonGlobal.phone
966:       pbMessage(_INTL("The phone is not defined."))
967:       next
968:     end
969:     cmd = 0
970:     loop do
971:       cmds = []
972:       time = $PokemonGlobal.phone.time_to_next_call.to_i   # time is in seconds
973:       min = time / 60
974:       sec = time % 60
975:       cmds.push(_INTL("Time until next call: {1}m {2}s", min, sec))
976:       cmds.push((Phone.rematches_enabled ? "[Y]" : "[  ]") + " " + _INTL("Rematches possible"))
977:       cmds.push(_INTL("Maximum rematch version : {1}", Phone.rematch_variant))
978:       if $PokemonGlobal.phone.contacts.length > 0
979:         cmds.push(_INTL("Make all contacts ready for a rematch"))
980:         cmds.push(_INTL("Edit individual contacts: {1}", $PokemonGlobal.phone.contacts.length))
981:       end
982:       cmd = pbShowCommands(nil, cmds, -1, cmd)
983:       break if cmd < 0
984:       case cmd
985:       when 0   # Time until next call
986:         params = ChooseNumberParams.new
987:         params.setRange(0, 99999)
988:         params.setDefaultValue(min)
989:         params.setCancelValue(-1)
990:         new_time = pbMessageChooseNumber(_INTL("Set the time (in minutes) until the next phone call."), params)
991:         $PokemonGlobal.phone.time_to_next_call = new_time * 60 if new_time >= 0
992:       when 1   # Rematches possible
993:         Phone.rematches_enabled = !Phone.rematches_enabled
994:       when 2   # Maximum rematch version
995:         params = ChooseNumberParams.new
996:         params.setRange(0, 99)
997:         params.setDefaultValue(Phone.rematch_variant)
998:         new_version = pbMessageChooseNumber(_INTL("Set the maximum version number a trainer contact can reach."), params)
999:         Phone.rematch_variant = new_version
1000:       when 3   # Make all contacts ready for a rematch
1001:         $PokemonGlobal.phone.contacts.each do |contact|
1002:           next if !contact.trainer?
1003:           contact.rematch_flag = 1
1004:           contact.set_trainer_event_ready_for_rematch
1005:         end
1006:         pbMessage(_INTL("All trainers in the phone are now ready to rebattle."))
1007:       when 4   # Edit individual contacts
1008:         contact_cmd = 0
1009:         loop do
1010:           contact_cmds = []
1011:           $PokemonGlobal.phone.contacts.each do |contact|
1012:             visible_string = (contact.visible?) ? "[Y]" : "[  ]"
1013:             if contact.trainer?
1014:               battle_string = (contact.can_rematch?) ? "(can battle)" : ""
1015:               contact_cmds.push(sprintf("%s %s (%i) %s", visible_string, contact.display_name, contact.variant, battle_string))
1016:             else
1017:               contact_cmds.push(sprintf("%s %s", visible_string, contact.display_name))
1018:             end
1019:           end
1020:           contact_cmd = pbShowCommands(nil, contact_cmds, -1, contact_cmd)
1021:           break if contact_cmd < 0
1022:           contact = $PokemonGlobal.phone.contacts[contact_cmd]
1023:           edit_cmd = 0
1024:           loop do
1025:             edit_cmds = []
1026:             edit_cmds.push((contact.visible? ? "[Y]" : "[  ]") + " " + _INTL("Contact visible"))
1027:             if contact.trainer?
1028:               edit_cmds.push((contact.can_rematch? ? "[Y]" : "[  ]") + " " + _INTL("Can battle"))
1029:               ready_time = contact.time_to_ready   # time is in seconds
1030:               ready_min = ready_time / 60
1031:               ready_sec = ready_time % 60
1032:               edit_cmds.push(_INTL("Time until ready to battle: {1}m {2}s", ready_min, ready_sec))
1033:               edit_cmds.push(_INTL("Last defeated version: {1}", contact.variant))
1034:             end
1035:             break if edit_cmds.length == 0
1036:             edit_cmd = pbShowCommands(nil, edit_cmds, -1, edit_cmd)
1037:             break if edit_cmd < 0
1038:             case edit_cmd
1039:             when 0   # Visibility
1040:               contact.visible = !contact.visible if contact.can_hide?
1041:             when 1   # Can battle
1042:               contact.rematch_flag = (contact.can_rematch?) ? 0 : 1
1043:               contact.time_to_ready = 0 if contact.can_rematch?
1044:             when 2   # Time until ready to battle
1045:               params = ChooseNumberParams.new
1046:               params.setRange(0, 99999)
1047:               params.setDefaultValue(ready_min)
1048:               params.setCancelValue(-1)
1049:               new_time = pbMessageChooseNumber(_INTL("Set the time (in minutes) until this trainer is ready to battle."), params)
1050:               contact.time_to_ready = new_time * 60 if new_time >= 0
1051:             when 3   # Last defeated version
1052:               params = ChooseNumberParams.new
1053:               params.setRange(0, 99)
1054:               params.setDefaultValue(contact.variant)
1055:               new_version = pbMessageChooseNumber(_INTL("Set the last defeated version number of this trainer."), params)
1056:               contact.version = contact.start_version + new_version
1057:             end
1058:           end
1059:         end
1060:       end
1061:     end
1062:   }
1063: })
