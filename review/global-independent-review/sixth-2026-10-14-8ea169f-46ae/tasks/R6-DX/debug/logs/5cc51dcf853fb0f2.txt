4: class CharacterEntryHelper
5:   attr_accessor :text
6:   attr_accessor :maxlength
7:   attr_reader   :passwordChar
8:   attr_accessor :cursor
9: 
10:   def initialize(text)
11:     @maxlength = -1
12:     @text = text
13:     @passwordChar = ""
14:     @cursor = text.scan(/./m).length
15:   end
16: 
17:   def textChars
18:     chars = text.scan(/./m)
19:     if @passwordChar != ""
20:       chars.length.times { |i| chars[i] = @passwordChar }
21:     end
22:     return chars
23:   end
24: 
25:   def passwordChar=(value)
26:     @passwordChar = value || ""
27:   end
28: 
29:   def length
30:     return self.text.scan(/./m).length
31:   end
32: 
33:   def canInsert?
34:     chars = self.text.scan(/./m)
35:     return false if @maxlength >= 0 && chars.length >= @maxlength
36:     return true
37:   end
38: 
39:   def insert(ch)
40:     chars = self.text.scan(/./m)
41:     return false if @maxlength >= 0 && chars.length >= @maxlength
42:     chars.insert(@cursor, ch)
43:     @text = ""
44:     chars.each { |char| @text += char if char }
45:     @cursor += 1
46:     return true
47:   end
48: 
49:   def canDelete?
50:     chars = self.text.scan(/./m)
51:     return false if chars.length <= 0 || @cursor <= 0
52:     return true
53:   end
54: 
55:   def delete
56:     chars = self.text.scan(/./m)
57:     return false if chars.length <= 0 || @cursor <= 0
58:     chars.delete_at(@cursor - 1)
59:     @text = ""
60:     chars.each do |ch|
61:       @text += ch if ch
62:     end
63:     @cursor -= 1
64:     return true
65:   end
66: 
67:   #-----------------------------------------------------------------------------
68: 
69:   private
70: 
71:   def ensure
72:     return if @maxlength < 0
73:     chars = self.text.scan(/./m)
74:     chars = chars[0, @maxlength] if chars.length > @maxlength && @maxlength >= 0
75:     @text = ""
76:     chars.each do |ch|
77:       @text += ch if ch
78:     end
79:   end
80: end
