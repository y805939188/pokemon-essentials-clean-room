31: MenuHandlers.add(:pokemon_debug_menu, :set_status, {
32:   "name"   => _INTL("Set status"),
33:   "parent" => :hp_status_menu,
34:   "effect" => proc { |pkmn, pkmnid, heldpoke, settingUpBattle, screen|
35:     if pkmn.egg?
36:       screen.pbDisplay(_INTL("{1} is an egg.", pkmn.name))
37:     elsif pkmn.hp <= 0
38:       screen.pbDisplay(_INTL("{1} is fainted, can't change status.", pkmn.name))
39:     else
40:       cmd = 0
41:       commands = [_INTL("[Cure]")]
42:       ids = [:NONE]
43:       GameData::Status.each do |s|
44:         next if s.id == :NONE
45:         commands.push(_INTL("Set {1}", s.name))
46:         ids.push(s.id)
47:       end
48:       loop do
49:         msg = _INTL("Current status: {1}", GameData::Status.get(pkmn.status).name)
50:         if pkmn.status == :SLEEP
51:           msg = _INTL("Current status: {1} (turns: {2})",
52:                       GameData::Status.get(pkmn.status).name, pkmn.statusCount)
53:         end
54:         cmd = screen.pbShowCommands(msg, commands, cmd)
55:         break if cmd < 0
56:         case cmd
57:         when 0   # Cure
58:           pkmn.heal_status
59:           screen.pbRefreshSingle(pkmnid)
60:         else   # Give status problem
61:           count = 0
62:           cancel = false
63:           if ids[cmd] == :SLEEP
64:             params = ChooseNumberParams.new
65:             params.setRange(0, 9)
66:             params.setDefaultValue(3)
67:             count = pbMessageChooseNumber(
68:               _INTL("Set the Pokémon's sleep count."), params
69:             ) { screen.pbUpdate }
70:             cancel = true if count <= 0
71:           end
72:           if !cancel
73:             pkmn.status      = ids[cmd]
74:             pkmn.statusCount = count
75:             screen.pbRefreshSingle(pkmnid)
76:           end
77:         end
78:       end
79:     end
80:     next false
81:   }
82: })

129:       when 0   # Give random strain
130:         pkmn.givePokerus
131:         screen.pbRefreshSingle(pkmnid)
132:       when 1   # Make not infectious
133:         if pokerus > 0
134:           strain = pokerus / 16
135:           p = strain << 4
136:           pkmn.pokerus = p
137:           screen.pbRefreshSingle(pkmnid)
138:         end
139:       when 2   # Clear Pokérus
140:         pkmn.pokerus = 0

208: MenuHandlers.add(:pokemon_debug_menu, :hidden_values, {
209:   "name"   => _INTL("EV/IV/personal ID..."),
210:   "parent" => :level_stats,
211:   "effect" => proc { |pkmn, pkmnid, heldpoke, settingUpBattle, screen|
212:     cmd = 0
213:     loop do
214:       persid = sprintf("0x%08X", pkmn.personalID)
215:       cmd = screen.pbShowCommands(_INTL("Personal ID is {1}.", persid),
216:                                   [_INTL("Set EVs"),
217:                                    _INTL("Set IVs"),
218:                                    _INTL("Randomise pID")], cmd)
219:       break if cmd < 0
220:       case cmd
221:       when 0   # Set EVs
222:         cmd2 = 0
223:         loop do
224:           totalev = 0
225:           evcommands = []
226:           ev_id = []
227:           GameData::Stat.each_main do |s|
228:             evcommands.push(s.name + " (#{pkmn.ev[s.id]})")
229:             ev_id.push(s.id)
230:             totalev += pkmn.ev[s.id]
231:           end
232:           evcommands.push(_INTL("Randomise all"))
233:           evcommands.push(_INTL("Max randomise all"))
234:           cmd2 = screen.pbShowCommands(_INTL("Change which EV?\nTotal: {1}/{2} ({3}%)",
235:                                              totalev, Pokemon::EV_LIMIT,
236:                                              100 * totalev / Pokemon::EV_LIMIT), evcommands, cmd2)
237:           break if cmd2 < 0
238:           if cmd2 < ev_id.length
239:             params = ChooseNumberParams.new
240:             upperLimit = 0
241:             GameData::Stat.each_main { |s| upperLimit += pkmn.ev[s.id] if s.id != ev_id[cmd2] }
242:             upperLimit = Pokemon::EV_LIMIT - upperLimit
243:             upperLimit = [upperLimit, Pokemon::EV_STAT_LIMIT].min
244:             thisValue = [pkmn.ev[ev_id[cmd2]], upperLimit].min
245:             params.setRange(0, upperLimit)
246:             params.setDefaultValue(thisValue)
247:             params.setCancelValue(thisValue)
248:             f = pbMessageChooseNumber(_INTL("Set the EV for {1} (max. {2}).",
249:                                             GameData::Stat.get(ev_id[cmd2]).name, upperLimit), params) { screen.pbUpdate }
250:             if f != pkmn.ev[ev_id[cmd2]]
251:               pkmn.ev[ev_id[cmd2]] = f
252:               pkmn.calc_stats
253:               screen.pbRefreshSingle(pkmnid)
254:             end
255:           else   # (Max) Randomise all
256:             evTotalTarget = Pokemon::EV_LIMIT
257:             if cmd2 == evcommands.length - 2   # Randomize all (not max)
258:               evTotalTarget = rand(Pokemon::EV_LIMIT)
259:             end
260:             GameData::Stat.each_main { |s| pkmn.ev[s.id] = 0 }
261:             while evTotalTarget > 0
262:               r = rand(ev_id.length)
263:               next if pkmn.ev[ev_id[r]] >= Pokemon::EV_STAT_LIMIT
264:               addVal = 1 + rand(Pokemon::EV_STAT_LIMIT / 4)
265:               addVal = addVal.clamp(0, evTotalTarget)
266:               addVal = addVal.clamp(0, Pokemon::EV_STAT_LIMIT - pkmn.ev[ev_id[r]])
267:               next if addVal == 0
268:               pkmn.ev[ev_id[r]] += addVal
269:               evTotalTarget -= addVal
270:             end
271:             pkmn.calc_stats
272:             screen.pbRefreshSingle(pkmnid)
273:           end
274:         end
275:       when 1   # Set IVs
276:         cmd2 = 0
277:         loop do
278:           hiddenpower = pbHiddenPower(pkmn)
279:           totaliv = 0
280:           ivcommands = []
281:           iv_id = []
282:           GameData::Stat.each_main do |s|
283:             ivcommands.push(s.name + " (#{pkmn.iv[s.id]})")
284:             iv_id.push(s.id)
285:             totaliv += pkmn.iv[s.id]
286:           end
287:           msg = _INTL("Change which IV?\nHidden Power:\n{1}, power {2}\nTotal: {3}/{4} ({5}%)",
288:                       GameData::Type.get(hiddenpower[0]).name, hiddenpower[1], totaliv,
289:                       iv_id.length * Pokemon::IV_STAT_LIMIT, 100 * totaliv / (iv_id.length * Pokemon::IV_STAT_LIMIT))
290:           ivcommands.push(_INTL("Randomise all"))
