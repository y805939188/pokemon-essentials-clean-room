104:   def compile_connections(*paths)
105:     hashenum = {
106:       "N" => "N", "North" => "N",
107:       "E" => "E", "East"  => "E",
108:       "S" => "S", "South" => "S",
109:       "W" => "W", "West"  => "W"
110:     }
111:     schema = [nil, "iyiiyi", nil, hashenum, nil, nil, hashenum]
112:     records = []
113:     paths.each do |path|
114:       compile_pbs_file_message_start(path)
115:       pbCompilerEachPreppedLine(path) do |line, lineno|
116:         FileLineData.setLine(line, lineno)
117:         record = get_csv_record(line, schema)
118:         if !pbRgssExists?(sprintf("Data/Map%03d.rxdata", record[0]))
119:           print _INTL("Warning: Map {1}, as mentioned in the map connection data, was not found.", record[0]) + "\n" + FileLineData.linereport
120:         elsif !pbRgssExists?(sprintf("Data/Map%03d.rxdata", record[3]))
121:           print _INTL("Warning: Map {1}, as mentioned in the map connection data, was not found.", record[3]) + "\n" + FileLineData.linereport
122:         end
123:         case record[1]
124:         when "N"
125:           raise _INTL("North side of first map must connect with south side of second map.") + "\n" + FileLineData.linereport if record[4] != "S"
126:         when "S"
127:           raise _INTL("South side of first map must connect with north side of second map.") + "\n" + FileLineData.linereport if record[4] != "N"
128:         when "E"
129:           raise _INTL("East side of first map must connect with west side of second map.") + "\n" + FileLineData.linereport if record[4] != "W"
130:         when "W"
131:           raise _INTL("West side of first map must connect with east side of second map.") + "\n" + FileLineData.linereport if record[4] != "E"
132:         end
133:         records.push(record)
134:       end
135:       process_pbs_file_message_end
136:     end
137:     save_data(records, "Data/map_connections.dat")
138:   end
