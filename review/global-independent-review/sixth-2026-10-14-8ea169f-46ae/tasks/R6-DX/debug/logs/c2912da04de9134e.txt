277: def pbHasEgg?(species)
278:   species_data = GameData::Species.try_get(species)
279:   return false if !species_data
280:   species = species_data.species
281:   # species may be unbreedable, so check its evolution's compatibilities
282:   evoSpecies = species_data.get_evolutions(true)
283:   compatSpecies = (evoSpecies && evoSpecies[0]) ? evoSpecies[0][0] : species
284:   species_data = GameData::Species.try_get(compatSpecies)
285:   compat = species_data.egg_groups
286:   return false if compat.include?(:Undiscovered) || compat.include?(:Ditto)
287:   baby = GameData::Species.get(species).get_baby_species
288:   return true if species == baby   # Is a basic species
289:   baby = GameData::Species.get(species).get_baby_species(true)
290:   return true if species == baby   # Is an egg species without incense
291:   return false
292: end
