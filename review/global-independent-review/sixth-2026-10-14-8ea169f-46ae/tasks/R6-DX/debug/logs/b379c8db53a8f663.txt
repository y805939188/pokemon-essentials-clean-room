172:   # Command phase
173:   #=============================================================================
174:   def pbCommandPhase
175:     @command_phase = true
176:     @scene.pbBeginCommandPhase
177:     # Reset choices if commands can be shown
178:     @battlers.each_with_index do |b, i|
179:       next if !b
180:       pbClearChoice(i) if pbCanShowCommands?(i)
181:     end
182:     # Reset choices to perform Mega Evolution if it wasn't done somehow
183:     2.times do |side|
184:       @megaEvolution[side].each_with_index do |megaEvo, i|
185:         @megaEvolution[side][i] = -1 if megaEvo >= 0
186:       end
187:     end
188:     # Choose actions for the round (player first, then AI)
189:     pbCommandPhaseLoop(true)    # Player chooses their actions
190:     if @decision != 0   # Battle ended, stop choosing actions
191:       @command_phase = false
192:       return
193:     end
194:     pbCommandPhaseLoop(false)   # AI chooses their actions
195:     @command_phase = false
196:   end
197: 
198:   def pbCommandPhaseLoop(isPlayer)
199:     # NOTE: Doing some things (e.g. running, throwing a Poké Ball) takes up all
200:     #       your actions in a round.
201:     actioned = []
202:     idxBattler = -1
203:     loop do
204:       break if @decision != 0   # Battle ended, stop choosing actions
205:       idxBattler += 1
206:       break if idxBattler >= @battlers.length
207:       next if !@battlers[idxBattler] || pbOwnedByPlayer?(idxBattler) != isPlayer
208:       if @choices[idxBattler][0] != :None || !pbCanShowCommands?(idxBattler)
209:         # Action is forced, can't choose one
210:         PBDebug.log_ai("#{@battlers[idxBattler].pbThis} (#{idxBattler}) is forced to use a multi-turn move")
211:         next
212:       end
213:       # AI controls this battler
214:       if @controlPlayer || !pbOwnedByPlayer?(idxBattler)
215:         @battleAI.pbDefaultChooseEnemyCommand(idxBattler)
216:         next
217:       end
218:       # Player chooses an action
219:       actioned.push(idxBattler)
220:       commandsEnd = false   # Whether to cancel choosing all other actions this round
221:       loop do
222:         cmd = pbCommandMenu(idxBattler, actioned.length == 1)
223:         # If being Sky Dropped, can't do anything except use a move
224:         if cmd > 0 && @battlers[idxBattler].effects[PBEffects::SkyDrop] >= 0
225:           pbDisplay(_INTL("Sky Drop won't let {1} go!", @battlers[idxBattler].pbThis(true)))
226:           next
227:         end
228:         case cmd
229:         when 0    # Fight
230:           break if pbFightMenu(idxBattler)
231:         when 1    # Bag
232:           if pbItemMenu(idxBattler, actioned.length == 1)
233:             commandsEnd = true if pbItemUsesAllActions?(@choices[idxBattler][1])
234:             break
235:           end
236:         when 2    # Pokémon
237:           break if pbPartyMenu(idxBattler)
238:         when 3    # Run
239:           # NOTE: "Run" is only an available option for the first battler the
240:           #       player chooses an action for in a round. Attempting to run
241:           #       from battle prevents you from choosing any other actions in
242:           #       that round.
243:           if pbRunMenu(idxBattler)
244:             commandsEnd = true
245:             break
246:           end
247:         when 4    # Call
248:           break if pbCallMenu(idxBattler)
249:         when -2   # Debug
250:           pbDebugMenu
251:           next
252:         when -1   # Go back to previous battler's action choice
253:           next if actioned.length <= 1
254:           actioned.pop   # Forget this battler was done
255:           idxBattler = actioned.last - 1
256:           pbCancelChoice(idxBattler + 1)   # Clear the previous battler's choice
257:           actioned.pop   # Forget the previous battler was done
258:           break
259:         end
260:         pbCancelChoice(idxBattler)
261:       end
262:       break if commandsEnd
263:     end
264:   end
265: end
