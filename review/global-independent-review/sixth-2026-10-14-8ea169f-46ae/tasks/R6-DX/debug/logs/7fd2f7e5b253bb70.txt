245:   def self.can_add?(items, max_slots, max_per_slot, item, qty)
246:     raise "Invalid value for qty: #{qty}" if qty < 0
247:     return true if qty == 0
248:     max_slots.times do |i|
249:       item_slot = items[i]
250:       if !item_slot
251:         qty -= [qty, max_per_slot].min
252:         return true if qty == 0
253:       elsif item_slot[0] == item && item_slot[1] < max_per_slot
254:         new_amt = item_slot[1]
255:         new_amt = [new_amt + qty, max_per_slot].min
256:         qty -= (new_amt - item_slot[1])
257:         return true if qty == 0
258:       end
259:     end
260:     return false
261:   end

263:   def self.add(items, max_slots, max_per_slot, item, qty)
264:     raise "Invalid value for qty: #{qty}" if qty < 0
265:     return true if qty == 0
266:     max_slots.times do |i|
267:       item_slot = items[i]
268:       if !item_slot
269:         items[i] = [item, [qty, max_per_slot].min]
270:         qty -= items[i][1]
271:         return true if qty == 0
272:       elsif item_slot[0] == item && item_slot[1] < max_per_slot
273:         new_amt = item_slot[1]
274:         new_amt = [new_amt + qty, max_per_slot].min
275:         qty -= (new_amt - item_slot[1])
276:         item_slot[1] = new_amt
277:         return true if qty == 0
278:       end
279:     end
280:     return false
281:   end

283:   # Deletes an item (items array, max. size per slot, item, no. of items to delete)
284:   def self.remove(items, item, qty)
285:     raise "Invalid value for qty: #{qty}" if qty < 0
286:     return true if qty == 0
287:     ret = false
288:     items.each_with_index do |item_slot, i|
289:       next if !item_slot || item_slot[0] != item
290:       amount = [qty, item_slot[1]].min
291:       item_slot[1] -= amount
292:       qty -= amount
293:       items[i] = nil if item_slot[1] == 0
294:       next if qty > 0
295:       ret = true
296:       break
297:     end
298:     items.compact!
299:     return ret
300:   end
