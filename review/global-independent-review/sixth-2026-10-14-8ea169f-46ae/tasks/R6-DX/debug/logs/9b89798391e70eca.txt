96:   def pbPartyScreen(idxBattler, canCancel = false)
97:     replacements = []
98:     @battle.eachInTeamFromBattlerIndex(idxBattler) do |_b, idxParty|
99:       replacements.push(idxParty) if !@battle.pbFindBattler(idxParty, idxBattler)

101:     return if replacements.length == 0
