367:       map = getMap(mapID)
368:       return false if !map
369:       return false if x < 0 || x >= map.width || y < 0 || y >= map.height
370:       passages   = getTilesetPassages(map, mapID)
371:       priorities = getTilesetPriorities(map, mapID)
372:       [2, 1, 0].each do |i|
373:         tile_id = map.data[x, y, i]
374:         return false if tile_id.nil?
375:         passage = passages[tile_id]
376:         if !passage
377:           raise "The tile used on map #{mapID} at coordinates (#{x}, #{y}) on layer #{i + 1} doesn't exist in the tileset. " +
378:                 "It should be deleted to prevent errors."
379:         end
380:         return false if passage & 0x0f == 0x0f
381:         return true if priorities[tile_id] == 0
382:       end

384:     end
