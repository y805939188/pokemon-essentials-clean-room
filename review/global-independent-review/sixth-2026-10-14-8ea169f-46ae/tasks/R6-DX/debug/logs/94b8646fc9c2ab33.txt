471:     pbUpdateSpriteHash(sprites)
472:     if cmdwindow.index < cmdwindow.roamerCount
473:       pkmn = Settings::ROAMING_SPECIES[cmdwindow.index]
474:     else
475:       pkmn = nil
476:     end
477:     if Input.trigger?(Input::ACTION) && cmdwindow.index < cmdwindow.roamerCount &&
478:        (pkmn[2] <= 0 || $game_switches[pkmn[2]]) &&
479:        $PokemonGlobal.roamPokemon[cmdwindow.index] != true
480:       # Roam selected Pokémon
481:       pbPlayDecisionSE
482:       if Input.press?(Input::CTRL)   # Roam to current map
483:         if $PokemonGlobal.roamPosition[cmdwindow.index] == pbDefaultMap
484:           $PokemonGlobal.roamPosition[cmdwindow.index] = nil
485:         else
486:           $PokemonGlobal.roamPosition[cmdwindow.index] = pbDefaultMap
487:         end
488:       else   # Roam to a random other map
489:         oldmap = $PokemonGlobal.roamPosition[cmdwindow.index]
490:         pbRoamPokemonOne(cmdwindow.index)
491:         if $PokemonGlobal.roamPosition[cmdwindow.index] == oldmap
492:           $PokemonGlobal.roamPosition[cmdwindow.index] = nil
493:           pbRoamPokemonOne(cmdwindow.index)
494:         end
495:         $PokemonGlobal.roamedAlready = false
496:       end
497:       cmdwindow.refresh
498:     elsif Input.trigger?(Input::BACK)
499:       pbPlayCancelSE
500:       break
501:     elsif Input.trigger?(Input::USE)
502:       if cmdwindow.index < cmdwindow.roamerCount
503:         pbPlayDecisionSE
504:         # Toggle through roaming, not roaming, defeated
505:         if pkmn[2] > 0 && !$game_switches[pkmn[2]]
506:           # not roaming -> roaming
507:           $game_switches[pkmn[2]] = true
508:         elsif $PokemonGlobal.roamPokemon[cmdwindow.index] != true
509:           # roaming -> defeated
510:           $PokemonGlobal.roamPokemon[cmdwindow.index] = true
511:           $PokemonGlobal.roamPokemonCaught[cmdwindow.index] = false
512:         elsif $PokemonGlobal.roamPokemon[cmdwindow.index] == true &&
513:               !$PokemonGlobal.roamPokemonCaught[cmdwindow.index]
514:           # defeated -> caught
515:           $PokemonGlobal.roamPokemonCaught[cmdwindow.index] = true
516:         elsif pkmn[2] > 0
517:           # caught -> not roaming (or roaming if Switch ID is 0)
518:           $game_switches[pkmn[2]] = false if pkmn[2] > 0
519:           $PokemonGlobal.roamPokemon[cmdwindow.index] = nil
520:           $PokemonGlobal.roamPokemonCaught[cmdwindow.index] = false
521:         end
522:         cmdwindow.refresh
523:       elsif cmdwindow.index == cmdwindow.itemCount - 2   # All roam
524:         if Settings::ROAMING_SPECIES.length == 0
525:           pbPlayBuzzerSE
526:         else
527:           pbPlayDecisionSE
528:           pbRoamPokemon
529:           $PokemonGlobal.roamedAlready = false
530:           cmdwindow.refresh
531:         end
532:       else   # Clear all roaming locations
533:         if Settings::ROAMING_SPECIES.length == 0
534:           pbPlayBuzzerSE
535:         else
536:           pbPlayDecisionSE
537:           Settings::ROAMING_SPECIES.length.times do |i|
538:             $PokemonGlobal.roamPosition[i] = nil
539:           end
540:           $PokemonGlobal.roamedAlready = false
541:           cmdwindow.refresh
542:         end
543:       end
544:     end
545:   end
546:   pbDisposeSpriteHash(sprites)
547:   viewport.dispose
548: end
