4: def pbGetLegalMoves(species)
5:   species_data = GameData::Species.get(species)
6:   moves = []
7:   return moves if !species_data
8:   species_data.moves.each { |m| moves.push(m[1]) }

10:   babyspecies = species_data.get_baby_species
11:   GameData::Species.get(babyspecies).egg_moves.each { |m| moves.push(m) }
12:   moves |= []   # Remove duplicates

16: def pbSafeCopyFile(x, y, z = nil)
17:   if FileTest.exist?(x)
18:     safetocopy = true
19:     filedata = nil
20:     if FileTest.exist?(y)
21:       different = false
22:       if FileTest.size(x) == FileTest.size(y)
23:         filedata2 = ""
24:         File.open(x, "rb") { |f| filedata  = f.read }
25:         File.open(y, "rb") { |f| filedata2 = f.read }
26:         different = true if filedata != filedata2
27:       else
28:         different = true
29:       end
30:       if different
31:         safetocopy = pbConfirmMessage(_INTL("A different file named '{1}' already exists. Overwrite it?", y))
32:       else
33:         # No need to copy
34:         return
35:       end
36:     end
37:     if safetocopy
38:       if !filedata
39:         File.open(x, "rb") { |f| filedata = f.read }
40:       end
41:       File.open((z) ? z : y, "wb") { |f| f.write(filedata) }
42:     end
43:   end
44: end
