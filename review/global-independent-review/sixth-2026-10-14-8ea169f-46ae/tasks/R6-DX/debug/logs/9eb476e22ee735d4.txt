593:   def compile_regional_dexes(*paths)
594:     dex_lists = []
595:     paths.each do |path|
596:       compile_pbs_file_message_start(path)
597:       section = nil
598:       pbCompilerEachPreppedLine(path) do |line, line_no|
599:         Graphics.update if line_no % 200 == 0
600:         if line[/^\s*\[\s*(\d+)\s*\]\s*$/]
601:           section = $~[1].to_i
602:           if dex_lists[section]
603:             raise _INTL("Dex list number {1} is defined at least twice.", section) + "\n" + FileLineData.linereport
604:           end
605:           dex_lists[section] = []
606:         else
607:           raise _INTL("Expected a section at the beginning of the file.") + "\n" + FileLineData.linereport if !section
608:           species_list = line.split(",")
609:           species_list.each do |species|
610:             next if !species || species.empty?
611:             s = parseSpecies(species)
612:             dex_lists[section].push(s)
613:           end
614:         end
615:       end
616:       process_pbs_file_message_end
617:     end
618:     # Check for duplicate species in a Regional Dex
619:     dex_lists.each_with_index do |list, index|
620:       unique_list = list.uniq
621:       next if list == unique_list
622:       list.each_with_index do |s, i|
623:         next if unique_list[i] == s
624:         raise _INTL("Dex list number {1} has species {2} listed twice.", index, s) + "\n" + FileLineData.linereport
625:       end
626:     end
627:     # Save all data
628:     save_data(dex_lists, "Data/regional_dexes.dat")
629:   end
