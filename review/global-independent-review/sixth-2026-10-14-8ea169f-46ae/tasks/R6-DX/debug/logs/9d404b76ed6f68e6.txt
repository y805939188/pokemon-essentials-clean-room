76: MenuHandlers.add(:battle_debug_menu, :pokemon_teams, {
77:   "name"        => _INTL("Pokémon teams"),
78:   "parent"      => :main,
79:   "description" => _INTL("Look at and edit all Pokémon in each team."),
80:   "effect"      => proc { |battle|
81:     player_party_starts = battle.pbPartyStarts(0)
82:     foe_party_starts = battle.pbPartyStarts(1)
83:     cmd = 0
84:     loop do
85:       # Find all teams and how many Pokémon they have
86:       commands = []
87:       team_indices = []
88:       if battle.opponent
89:         battle.opponent.each_with_index do |trainer, i|
90:           first_index = foe_party_starts[i]
91:           last_index = (i < foe_party_starts.length - 1) ? foe_party_starts[i + 1] : battle.pbParty(1).length
92:           num_pkmn = last_index - first_index
93:           commands.push(_INTL("Opponent {1}: {2} ({3} Pokémon)", i + 1, trainer.full_name, num_pkmn))
94:           team_indices.push([1, i, first_index])
95:         end
96:       else
97:         commands.push(_INTL("Opponent: {1} wild Pokémon", battle.pbParty(1).length))
98:         team_indices.push([1, 0, 0])
99:       end
100:       battle.player.each_with_index do |trainer, i|
101:         first_index = player_party_starts[i]
102:         last_index = (i < player_party_starts.length - 1) ? player_party_starts[i + 1] : battle.pbParty(0).length
103:         num_pkmn = last_index - first_index
104:         if i == 0   # Player
105:           commands.push(_INTL("You: {1} ({2} Pokémon)", trainer.full_name, num_pkmn))
106:         else
107:           commands.push(_INTL("Ally {1}: {2} ({3} Pokémon)", i, trainer.full_name, num_pkmn))
108:         end
109:         team_indices.push([0, i, first_index])
110:       end
111:       # Choose a team
112:       cmd = pbMessage("\\ts[]" + _INTL("Choose a team."), commands, -1, nil, cmd)
113:       break if cmd < 0
114:       # Pick a Pokémon to look at
115:       pkmn_cmd = 0
116:       loop do
117:         pkmn = []
118:         pkmn_cmds = []
119:         battle.eachInTeam(team_indices[cmd][0], team_indices[cmd][1]) do |p|
120:           pkmn.push(p)
121:           pkmn_cmds.push("[#{pkmn_cmds.length + 1}] #{p.name} Lv.#{p.level} (HP: #{p.hp}/#{p.totalhp})")
122:         end
123:         pkmn_cmd = pbMessage("\\ts[]" + _INTL("Choose a Pokémon."), pkmn_cmds, -1, nil, pkmn_cmd)
124:         break if pkmn_cmd < 0
125:         battle.pbBattlePokemonDebug(pkmn[pkmn_cmd],
126:                                     battle.pbFindBattler(team_indices[cmd][2] + pkmn_cmd, team_indices[cmd][0]))
127:       end
128:     end
129:   }
130: })

141: MenuHandlers.add(:battle_debug_menu, :trainer_items, {
142:   "name"        => _INTL("NPC trainer items"),
143:   "parent"      => :trainers,
144:   "description" => _INTL("View and change the items each NPC trainer has access to."),
145:   "effect"      => proc { |battle|
146:     cmd = 0
147:     loop do
148:       # Find all NPC trainers and their items
149:       commands = []
150:       item_arrays = []
151:       trainer_indices = []
152:       if battle.opponent
153:         battle.opponent.each_with_index do |trainer, i|
154:           items = battle.items ? battle.items[i].clone : []
155:           commands.push(_INTL("Opponent {1}: {2} ({3} items)", i + 1, trainer.full_name, items.length))
156:           item_arrays.push(items)
157:           trainer_indices.push([1, i])
158:         end
159:       end
160:       if battle.player.length > 1
161:         battle.player.each_with_index do |trainer, i|
162:           next if i == 0   # Player
163:           items = battle.ally_items ? battle.ally_items[i].clone : []
164:           commands.push(_INTL("Ally {1}: {2} ({3} items)", i, trainer.full_name, items.length))
165:           item_arrays.push(items)
166:           trainer_indices.push([0, i])
167:         end
168:       end
169:       if commands.length == 0
170:         pbMessage("\\ts[]" + _INTL("There are no NPC trainers in this battle."))
171:         break
172:       end
173:       # Choose a trainer
174:       cmd = pbMessage("\\ts[]" + _INTL("Choose a trainer."), commands, -1, nil, cmd)
175:       break if cmd < 0
176:       # Get trainer's items
177:       items = item_arrays[cmd]
178:       indices = trainer_indices[cmd]
179:       # Edit trainer's items
180:       item_list_property = GameDataPoolProperty.new(:Item)
181:       new_items = item_list_property.set(nil, items)
182:       if indices[0] == 0   # Ally
183:         battle.ally_items = [] if !battle.ally_items
184:         battle.ally_items[indices[1]] = new_items
185:       else   # Opponent
186:         battle.items = [] if !battle.items
187:         battle.items[indices[1]] = new_items
188:       end
189:     end
190:   }
191: })

193: MenuHandlers.add(:battle_debug_menu, :mega_evolution, {
194:   "name"        => _INTL("Mega Evolution"),
195:   "parent"      => :trainers,
196:   "description" => _INTL("Whether each trainer is allowed to Mega Evolve."),
197:   "effect"      => proc { |battle|
198:     cmd = 0
199:     loop do
200:       commands = []
201:       cmds = []
202:       battle.megaEvolution.each_with_index do |side_values, side|
203:         trainers = (side == 0) ? battle.player : battle.opponent
204:         next if !trainers
205:         side_values.each_with_index do |value, i|
206:           next if !trainers[i]
207:           text = (side == 0) ? "Your side:" : "Foe side:"
208:           text += sprintf(" %d: %s", i, trainers[i].name)
209:           text += " [ABLE]" if value == -1
210:           text += " [UNABLE]" if value == -2
211:           commands.push(text)
212:           cmds.push([side, i])
213:         end
214:       end
215:       cmd = pbMessage("\\ts[]" + _INTL("Choose trainer to toggle whether they can Mega Evolve."),
216:                       commands, -1, nil, cmd)
217:       break if cmd < 0
218:       real_cmd = cmds[cmd]
219:       if battle.megaEvolution[real_cmd[0]][real_cmd[1]] == -1
220:         battle.megaEvolution[real_cmd[0]][real_cmd[1]] = -2   # Make unable
221:       else
222:         battle.megaEvolution[real_cmd[0]][real_cmd[1]] = -1   # Make able
223:       end
224:     end
225:   }
226: })
