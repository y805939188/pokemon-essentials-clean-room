364:   def write_regional_dexes(path = "PBS/regional_dexes.txt")
365:     write_pbs_file_message_start(path)
366:     dex_lists = pbLoadRegionalDexes
367:     File.open(path, "wb") do |f|
368:       add_PBS_header_to_file(f)
369:       # Write each Dex list in turn
370:       dex_lists.each_with_index do |list, index|
371:         f.write("\#-------------------------------\r\n")
372:         f.write("[#{index}]")
373:         comma = false
374:         current_family = nil
375:         list.each do |species|
376:           next if !species
377:           if current_family&.include?(species)
378:             f.write(",") if comma
379:           else
380:             current_family = GameData::Species.get(species).get_family_species
381:             comma = false
382:             f.write("\r\n")
383:           end
384:           f.write(species)
385:           comma = true
386:         end
387:         f.write("\r\n")
388:       end
389:     end
390:     process_pbs_file_message_end
391:   end
