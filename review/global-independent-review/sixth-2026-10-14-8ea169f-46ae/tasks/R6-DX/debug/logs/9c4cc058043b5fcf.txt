4: class StandardRestriction
5:   def isValid?(pkmn)
6:     return false if !pkmn || pkmn.egg?
7:     # Species with disadvantageous abilities are not banned
8:     pkmn.species_data.abilities.each do |a|
9:       return true if [:TRUANT, :SLOWSTART].include?(a)
10:     end
11:     # Certain named species are not banned
12:     return true if [:DRAGONITE, :SALAMENCE, :TYRANITAR].include?(pkmn.species)
13:     # Certain named species are banned
14:     return false if [:WYNAUT, :WOBBUFFET].include?(pkmn.species)
15:     # Species with total base stat 600 or more are banned
16:     bst = 0
17:     pkmn.baseStats.each_value { |s| bst += s }
18:     return false if bst >= 600
19:     # Is valid
20:     return true
21:   end
22: end

100: module NicknameChecker
101:   @@names = {}
102: 
103:   def getName(species)
104:     n = @@names[species]
105:     return n if n
106:     n = GameData::Species.get(species).name
107:     @@names[species] = n.upcase
108:     return n
109:   end
110: 
111:   def check(name, species)
112:     name = name.upcase
113:     return true if name == getName(species)
114:     return false if @@names.values.include?(name)
115:     GameData::Species.each_species do |species_data|
116:       return false if species_data.species != species && getName(species_data.id) == name
117:     end
118:     return true
119:   end
120: end

122: #===============================================================================
123: # No two Pokemon can have the same nickname.
124: # No nickname can be the same as the (real) name of another Pokemon character.
125: #===============================================================================
126: class NicknameClause
127:   def isValid?(team)
128:     (team.length - 1).times do |i|
129:       (i + 1...team.length).each do |j|
130:         return false if team[i].name == team[j].name
131:         return false if !NicknameChecker.check(team[i].name, team[i].species)
132:       end
133:     end
134:     return true
135:   end
136: 
137:   def errorMessage
138:     return _INTL("No identical nicknames.")
139:   end
140: end

197: class RestrictedSpeciesRestriction
198:   def initialize(maxValue, *specieslist)
199:     @specieslist = specieslist.clone
200:     @maxValue = maxValue
201:   end
202: 
203:   def isSpecies?(species, specieslist)
204:     return specieslist.include?(species)
205:   end
206: 
207:   def isValid?(team)
208:     count = 0
209:     team.each do |pkmn|
210:       count += 1 if pkmn && isSpecies?(pkmn.species, @specieslist)
211:     end
212:     return count <= @maxValue
213:   end
214: end
