232: def pbDebugDayCare
233:   day_care = $PokemonGlobal.day_care
234:   cmd_window = Window_CommandPokemonEx.newEmpty(0, 0, Graphics.width, Graphics.height)
235:   commands = []
236:   cmd = 0
237:   compat = 0
238:   need_refresh = true
239:   loop do
240:     if need_refresh
241:       commands.clear
242:       day_care.slots.each_with_index do |slot, i|
243:         if slot.filled?
244:           pkmn = slot.pokemon
245:           msg = _INTL("{1} ({2})", pkmn.name, pkmn.speciesName)
246:           if pkmn.male?
247:             msg += ", ♂"
248:           elsif pkmn.female?
249:             msg += ", ♀"
250:           end
251:           if slot.level_gain > 0
252:             msg += ", " + _INTL("Lv.{1} (+{2})", pkmn.level, slot.level_gain)
253:           else
254:             msg += ", " + _INTL("Lv.{1}", pkmn.level)
255:           end
256:           commands.push(_INTL("[Slot {1}] {2}", i, msg))
257:         else
258:           commands.push(_INTL("[Slot {1}] Empty", i))
259:         end
260:       end
261:       compat = $PokemonGlobal.day_care.get_compatibility
262:       if day_care.egg_generated
263:         commands.push(_INTL("[Egg available]"))
264:       elsif compat > 0
265:         commands.push(_INTL("[Can produce egg]"))
266:       else
267:         commands.push(_INTL("[Cannot breed]"))
268:       end
269:       commands.push(_INTL("[Steps to next cycle: {1}]", 256 - day_care.step_counter))
270:       cmd_window.commands = commands
271:       need_refresh = false
272:     end
273:     cmd = pbCommands2(cmd_window, commands, -1, cmd, true)
274:     break if cmd < 0
275:     if cmd == commands.length - 2   # Egg
276:       compat = $PokemonGlobal.day_care.get_compatibility
277:       if compat == 0
278:         pbMessage(_INTL("Pokémon cannot breed."))
279:       else
280:         msg = _INTL("Pokémon can breed (compatibility = {1}).", compat)
281:         # Show compatibility
282:         if day_care.egg_generated
283:           case pbMessage("\\ts[]" + msg,
284:                          [_INTL("Collect egg"), _INTL("Clear egg"), _INTL("Cancel")], 3)
285:           when 0   # Collect egg
286:             if $player.party_full?
287:               pbMessage(_INTL("Party is full, can't collect the egg."))
288:             else
289:               DayCare.collect_egg
290:               pbMessage(_INTL("Collected the {1} egg.", $player.last_party.speciesName))
291:               need_refresh = true
292:             end
293:           when 1   # Clear egg
294:             day_care.egg_generated = false
295:             need_refresh = true
296:           end
297:         else
298:           case pbMessage("\\ts[]" + msg, [_INTL("Make egg available"), _INTL("Cancel")], 2)
299:           when 0   # Make egg available
300:             day_care.egg_generated = true
301:             need_refresh = true
302:           end
303:         end
304:       end
305:     elsif cmd == commands.length - 1   # Steps to next cycle
306:       case pbMessage("\\ts[]" + _INTL("Change number of steps to next cycle?"),
307:                      [_INTL("Set to 1"), _INTL("Set to 256"), _INTL("Set to other value"), _INTL("Cancel")], 4)
308:       when 0   # Set to 1
309:         day_care.step_counter = 255
310:         need_refresh = true
311:       when 1   # Set to 256
312:         day_care.step_counter = 0
313:         need_refresh = true
314:       when 2   # Set to other value
315:         params = ChooseNumberParams.new
316:         params.setDefaultValue(day_care.step_counter)
317:         params.setRange(1, 256)
318:         new_counter = pbMessageChooseNumber(_INTL("Set steps until next cycle (1-256)."), params)
319:         if new_counter != 256 - day_care.step_counter
320:           day_care.step_counter = 256 - new_counter
321:           need_refresh = true
322:         end
323:       end
324:     else   # Slot
325:       slot = day_care[cmd]
326:       if slot.filled?
327:         pkmn = slot.pokemon
328:         msg = _INTL("Cost: ${1}", slot.cost)
329:         if pkmn.level < GameData::GrowthRate.max_level
330:           end_exp = pkmn.growth_rate.minimum_exp_for_level(pkmn.level + 1)
331:           msg += "\\n" + _INTL("Steps to next level: {1}", end_exp - pkmn.exp)
332:         end
333:         # Show level change and cost
334:         case pbMessage("\\ts[]" + msg,
335:                        [_INTL("Summary"), _INTL("Withdraw"), _INTL("Cancel")], 3)
336:         when 0   # Summary
337:           pbFadeOutIn do
338:             scene = PokemonSummary_Scene.new
339:             screen = PokemonSummaryScreen.new(scene, false)
340:             screen.pbStartScreen([pkmn], 0)
341:             need_refresh = true
342:           end
343:         when 1   # Withdraw
344:           if $player.party_full?
345:             pbMessage(_INTL("Party is full, can't withdraw Pokémon."))
346:           else
347:             $player.party.push(pkmn)
348:             slot.reset
349:             day_care.reset_egg_counters
350:             need_refresh = true
351:           end
352:         end
353:       else
354:         case pbMessage("\\ts[]" + _INTL("This slot is empty."),
355:                        [_INTL("Deposit"), _INTL("Cancel")], 2)
356:         when 0   # Deposit
357:           if $player.party.empty?
358:             pbMessage(_INTL("Party is empty, can't deposit Pokémon."))
359:           else
360:             pbChooseNonEggPokemon(1, 3)
361:             party_index = pbGet(1)
362:             if party_index >= 0
363:               pkmn = $player.party[party_index]
364:               slot.deposit(pkmn)
365:               $player.party.delete_at(party_index)
366:               day_care.reset_egg_counters
367:               need_refresh = true
368:             end
369:           end
370:         end
371:       end
372: 
373:     end
374:   end
375:   cmd_window.dispose
376: end

381: class SpriteWindow_DebugRoamers < Window_DrawableCommand
382:   def initialize(viewport)
383:     super(0, 0, Graphics.width, Graphics.height, viewport)
384:   end
385: 
386:   def roamerCount
387:     return Settings::ROAMING_SPECIES.length
388:   end
389: 
390:   def itemCount
391:     return self.roamerCount + 2
392:   end
393: 
394:   def shadowtext(t, x, y, w, h, align = 0, colors = 0)
395:     y += 8   # TEXT OFFSET
396:     width = self.contents.text_size(t).width
397:     case align
398:     when 1
399:       x += (w - width)             # Right aligned
400:     when 2
401:       x += (w / 2) - (width / 2)   # Centre aligned
402:     end
403:     base = Color.new(96, 96, 96)
404:     case colors
405:     when 1
406:       base = Color.new(168, 48, 56)   # Red
407:     when 2
408:       base = Color.new(0, 144, 0)     # Green
409:     end
410:     pbDrawShadowText(self.contents, x, y, [width, w].max, h, t, base, Color.new(208, 208, 200))
411:   end
412: 
413:   def drawItem(index, _count, rect)
414:     pbSetNarrowFont(self.contents)
415:     rect = drawCursor(index, rect)
416:     nameWidth   = rect.width * 50 / 100
417:     statusWidth = rect.width * 50 / 100
418:     if index == self.itemCount - 2
419:       # Advance roaming
420:       self.shadowtext(_INTL("[All roam to new locations]"), rect.x, rect.y, nameWidth, rect.height)
421:     elsif index == self.itemCount - 1
422:       # Advance roaming
423:       self.shadowtext(_INTL("[Clear all current roamer locations]"), rect.x, rect.y, nameWidth, rect.height)
424:     else
425:       pkmn = Settings::ROAMING_SPECIES[index]
426:       name = GameData::Species.get(pkmn[0]).name + " (Lv. #{pkmn[1]})"
427:       status = ""
428:       statuscolor = 0
429:       if pkmn[2] <= 0 || $game_switches[pkmn[2]]
430:         status = $PokemonGlobal.roamPokemon[index]
431:         if status == true
432:           if $PokemonGlobal.roamPokemonCaught[index]
433:             status = "[CAUGHT]"
434:           else
435:             status = "[DEFEATED]"
436:           end
437:           statuscolor = 1
438:         else
439:           # roaming
440:           curmap = $PokemonGlobal.roamPosition[index]
441:           if curmap
442:             mapinfos = pbLoadMapInfos
443:             status = "[ROAMING][#{curmap}: #{mapinfos[curmap].name}]"
444:           else
445:             status = "[ROAMING][map not set]"
446:           end
447:           statuscolor = 2
448:         end
449:       else
450:         status = "[NOT ROAMING][Switch #{pkmn[2]} is off]"
451:       end
452:       self.shadowtext(name, rect.x, rect.y, nameWidth, rect.height)
453:       self.shadowtext(status, rect.x + nameWidth, rect.y, statusWidth, rect.height, 1, statuscolor)
454:     end
455:   end
