3:   # The player chooses a main command for a Pokémon
4:   # Return values: -1=Cancel, 0=Fight, 1=Bag, 2=Pokémon, 3=Run, 4=Call
5:   #=============================================================================
6:   def pbCommandMenu(idxBattler, firstAction)
7:     shadowTrainer = (GameData::Type.exists?(:SHADOW) && @battle.trainerBattle?)
8:     cmds = [
9:       _INTL("What will\n{1} do?", @battle.battlers[idxBattler].name),
10:       _INTL("Fight"),
11:       _INTL("Bag"),
12:       _INTL("Pokémon"),
13:       (shadowTrainer) ? _INTL("Call") : (firstAction) ? _INTL("Run") : _INTL("Cancel")
14:     ]
15:     ret = pbCommandMenuEx(idxBattler, cmds, (shadowTrainer) ? 2 : (firstAction) ? 0 : 1)
16:     ret = 4 if ret == 3 && shadowTrainer   # Convert "Run" to "Call"
17:     ret = -1 if ret == 3 && !firstAction   # Convert "Run" to "Cancel"
18:     return ret
19:   end
20: 
21:   # Mode: 0 = regular battle with "Run" (first choosable action in the round only)
22:   #       1 = regular battle with "Cancel"
23:   #       2 = regular battle with "Call" (for Shadow Pokémon battles)
24:   #       3 = Safari Zone
25:   #       4 = Bug-Catching Contest
26:   def pbCommandMenuEx(idxBattler, texts, mode = 0)
27:     pbShowWindow(COMMAND_BOX)
28:     cw = @sprites["commandWindow"]
29:     cw.setTexts(texts)
30:     cw.setIndexAndMode(@lastCmd[idxBattler], mode)
31:     pbSelectBattler(idxBattler)
32:     ret = -1
33:     loop do
34:       oldIndex = cw.index
35:       pbUpdate(cw)
36:       # Update selected command
37:       if Input.trigger?(Input::LEFT)
38:         cw.index -= 1 if (cw.index & 1) == 1
39:       elsif Input.trigger?(Input::RIGHT)
40:         cw.index += 1 if (cw.index & 1) == 0
41:       elsif Input.trigger?(Input::UP)
42:         cw.index -= 2 if (cw.index & 2) == 2
43:       elsif Input.trigger?(Input::DOWN)
44:         cw.index += 2 if (cw.index & 2) == 0
45:       end
46:       pbPlayCursorSE if cw.index != oldIndex
47:       # Actions
48:       if Input.trigger?(Input::USE)                 # Confirm choice
49:         pbPlayDecisionSE
50:         ret = cw.index
51:         @lastCmd[idxBattler] = ret
52:         break
53:       elsif Input.trigger?(Input::BACK) && mode == 1   # Cancel
54:         pbPlayCancelSE
55:         break
56:       elsif Input.trigger?(Input::F9) && $DEBUG    # Debug menu
57:         pbPlayDecisionSE
58:         ret = -2
59:         break
60:       end
61:     end
62:     return ret
63:   end
