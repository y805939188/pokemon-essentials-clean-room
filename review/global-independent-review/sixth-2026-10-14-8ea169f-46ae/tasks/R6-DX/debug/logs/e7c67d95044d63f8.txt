951:   def compile_trainer_lists(path = "PBS/battle_facility_lists.txt")
952:     compile_pbs_file_message_start(path)
953:     btTrainersRequiredTypes = {
954:       "Trainers"   => [0, "s"],
955:       "Pokemon"    => [1, "s"],
956:       "Challenges" => [2, "*s"]
957:     }
958:     if !FileTest.exist?(path)
959:       File.open(path, "wb") do |f|
960:         f.write(0xEF.chr)
961:         f.write(0xBB.chr)
962:         f.write(0xBF.chr)
963:         f.write("[DefaultTrainerList]\r\n")
964:         f.write("Trainers = battle_tower_trainers.txt\r\n")
965:         f.write("Pokemon = battle_tower_pokemon.txt\r\n")
966:       end
967:     end
968:     sections = []
969:     MessageTypes.setMessagesAsHash(MessageTypes::FRONTIER_INTRO_SPEECHES, [])
970:     MessageTypes.setMessagesAsHash(MessageTypes::FRONTIER_END_SPEECHES_WIN, [])
971:     MessageTypes.setMessagesAsHash(MessageTypes::FRONTIER_END_SPEECHES_LOSE, [])
972:     File.open(path, "rb") do |f|
973:       FileLineData.file = path
974:       idx = 0
975:       pbEachFileSection(f) do |section, name|
976:         echo "."
977:         idx += 1
978:         Graphics.update
979:         next if name != "DefaultTrainerList" && name != "TrainerList"
980:         rsection = []
981:         section.each_key do |key|
982:           FileLineData.setSection(name, key, section[key])
983:           schema = btTrainersRequiredTypes[key]
984:           next if key == "Challenges" && name == "DefaultTrainerList"
985:           next if !schema
986:           record = get_csv_record(section[key], schema)
987:           rsection[schema[0]] = record
988:         end
989:         if !rsection[0]
990:           raise _INTL("No trainer data file given in section {1}.", name) + "\n" + FileLineData.linereport
991:         end
992:         if !rsection[1]
993:           raise _INTL("No trainer data file given in section {1}.", name) + "\n" + FileLineData.linereport
994:         end
995:         rsection[3] = rsection[0]
996:         rsection[4] = rsection[1]
997:         rsection[5] = (name == "DefaultTrainerList")
998:         if FileTest.exist?("PBS/" + rsection[0])
999:           rsection[0] = compile_battle_tower_trainers("PBS/" + rsection[0])
1000:         else
1001:           rsection[0] = []
1002:         end
1003:         if FileTest.exist?("PBS/" + rsection[1])
1004:           filename = "PBS/" + rsection[1]
1005:           rsection[1] = []
1006:           pbCompilerEachCommentedLine(filename) do |line, _lineno|
1007:             rsection[1].push(PBPokemon.fromInspected(line))
1008:           end
1009:         else
1010:           rsection[1] = []
1011:         end
1012:         rsection[2] = [] if !rsection[2]
1013:         while rsection[2].include?("")
1014:           rsection[2].delete("")
1015:         end
1016:         rsection[2].compact!
1017:         sections.push(rsection)
1018:       end
1019:     end
1020:     save_data(sections, "Data/trainer_lists.dat")
1021:     process_pbs_file_message_end
1022:   end
