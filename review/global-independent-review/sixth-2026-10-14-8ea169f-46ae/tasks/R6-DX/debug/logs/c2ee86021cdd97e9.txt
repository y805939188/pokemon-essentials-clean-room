50: def pbGetLegalMoves2(species, maxlevel)
51:   species_data = GameData::Species.get(species)
52:   moves = []
53:   return moves if !species_data
54:   # Populate available moves array (moves)
55:   species_data.moves.each { |m| addMove(moves, m[1], 2) if m[0] <= maxlevel }
56:   if !$tmMoves
57:     $tmMoves = []
58:     GameData::Item.each { |i| $tmMoves.push(i.move) if i.is_machine? }
59:   end
60:   species_data.tutor_moves.each { |m| addMove(moves, m, 0) if $tmMoves.include?(m) }
61:   babyspecies = babySpecies(species)
62:   GameData::Species.get(babyspecies).egg_moves.each { |m| addMove(moves, m, 2) }
63:   movedatas = []
64:   moves.each do |move|
65:     movedatas.push([move, GameData::Move.get(move)])
66:   end
67:   # Delete less powerful moves
68:   deleteAll = proc { |a, item|
69:     while a.include?(item)
70:       a.delete(item)
71:     end
72:   }
73:   moves.each do |move|
74:     md = GameData::Move.get(move)
75:     movedatas.each do |move2|
76:       # If we have a move that always hits, remove all other moves with no
77:       # effect of the same type and <= base power
78:       if md.accuracy == 0 && move2[1].function_code == "None" &&
79:          md.type == move2[1].type && md.power >= move2[1].power
80:         deleteAll.call(moves, move2[0])
81:       # If we have two status moves that have the same function code, delete the
82:       # one with lower accuracy (Supersonic vs. Confuse Ray, etc.)
83:       elsif md.function_code == move2[1].function_code && md.power == 0 &&
84:             move2[1].power == 0 && md.accuracy > move2[1].accuracy
85:         deleteAll.call(moves, move2[0])
86:       # Delete poison-causing moves if we have a move that causes toxic
87:       elsif md.function_code == "BadPoisonTarget" && move2[1].function_code == "PoisonTarget"
88:         deleteAll.call(moves, move2[0])
89:       # If we have two moves with the same function code and type, and one of
90:       # them is damaging and has 10/15/the same PP as the other move and EITHER
91:       # does more damage than the other move OR does the same damage but is more
92:       # accurate, delete the other move (Surf, Flamethrower, Thunderbolt, etc.)
93:       elsif md.function_code == move2[1].function_code && md.power != 0 &&
94:             md.type == move2[1].type &&
95:             (md.total_pp == 15 || md.total_pp == 10 || md.total_pp == move2[1].total_pp) &&
96:             (md.power > move2[1].power ||
97:             (md.power == move2[1].power && md.accuracy > move2[1].accuracy))
98:         deleteAll.call(moves, move2[0])
99:       end
100:     end
101:   end
102:   return moves
103: end

105: def addMove(moves, move, base)
106:   data = GameData::Move.get(move)
107:   return if moves.include?(data.id)
108:   return if [:BUBBLE, :BUBBLEBEAM].include?(data.id)   # Never add these moves
109:   count = base + 1   # Number of times to add move to moves
110:   count = base if data.function_code == "None" && data.power <= 40
111:   if data.power <= 30 || [:GROWL, :TAILWHIP, :LEER].include?(data.id)
112:     count = base
113:   end
114:   if data.power >= 60 ||
115:      [:REFLECT, :LIGHTSCREEN, :SAFEGUARD, :SUBSTITUTE, :FAKEOUT].include?(data.id)
116:     count = base + 2
117:   end
118:   count = base + 3 if data.power >= 80 && data.type == :NORMAL
119:   if [:PROTECT, :DETECT, :TOXIC, :AERIALACE, :WILLOWISP, :SPORE, :THUNDERWAVE,
120:       :HYPNOSIS, :CONFUSERAY, :ENDURE, :SWORDSDANCE].include?(data.id)
121:     count = base + 3
122:   end
123:   count.times { moves.push(data.id) }
124: end
