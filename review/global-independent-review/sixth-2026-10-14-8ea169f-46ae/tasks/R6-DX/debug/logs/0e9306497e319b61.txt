711:     num_error_maps += 1
712:     mapData.saveMap(id)
713:   end
714:   if num_error_maps == 0
715:     Console.echo_h2(_INTL("Done. No errors found."), text: :green)
716:     pbMessage(_INTL("No invalid tiles were found."))
717:   else
718:     echoln ""
719:     Console.echo_h2(_INTL("Done. {1} errors found and fixed.", total_errors), text: :green)
720:     Console.echo_warn(_INTL("RMXP data was altered. Close RMXP now to ensure changes are applied."))
721:     echoln ""
722:     pbMessage(_INTL("{1} error(s) were found across {2} map(s) and fixed.", total_errors, num_error_maps))
723:     pbMessage(_INTL("Close RPG Maker XP to ensure the changes are applied properly."))
724:   end
725: end

727: def pbCheckTileValidity(tile_id, map, tilesets, passages)
728:   return false if !tile_id
729:   if tile_id > 0 && tile_id < 384
730:     # Check for defined autotile
731:     autotile_id = (tile_id / 48) - 1
732:     autotile_name = tilesets[map.tileset_id].autotile_names[autotile_id]
733:     return true if autotile_name && autotile_name != ""
734:   else
735:     # Check for tileset data
736:     return true if passages[tile_id]
737:   end
738:   return false
739: end
