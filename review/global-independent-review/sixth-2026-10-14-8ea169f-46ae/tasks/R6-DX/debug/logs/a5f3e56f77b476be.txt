4: class RuledTeam
5:   attr_accessor :team
6: 
7:   def initialize(party, rule)
8:     count = rule.ruleset.suggestedNumber
9:     @team = []
10:     retnum = []
11:     loop do
12:       count.times do |i|
13:         retnum[i] = rand(party.length)
14:         @team[i] = party[retnum[i]]
15:         party.delete_at(retnum[i])
16:       end
17:       break if rule.ruleset.isValid?(@team)
18:     end
19:     @totalGames = 0
20:     @rating = PlayerRating.new
21:     @history = MatchHistory.new(@rating)
22:   end
23: 
24:   def [](i)
25:     @team[i]
26:   end
27: 
28:   def length
29:     return @team.length
30:   end
31: 
32:   def rating
33:     @rating.winChancePercent
34:   end
35: 
36:   def ratingData
37:     @rating
38:   end
39: 
40:   def ratingRaw
41:     [@rating.rating, @rating.deviation, @rating.volatility, @rating.winChancePercent]
42:   end
43: 
44:   def compare(other)
45:     @rating.compare(other.ratingData)
46:   end
47: 
48:   def totalGames
49:     (@totalGames || 0) + self.games
50:   end
51: 
52:   def addMatch(other, score)
53:     @history.addMatch(other.ratingData, score)
54:   end
55: 
56:   def games
57:     @history.length
58:   end
59: 
60:   def updateRating
61:     @totalGames = 0 if !@totalGames
62:     oldgames = self.games
63:     @history.updateAndClear
64:     newgames = self.games
65:     @totalGames += (oldgames - newgames)
66:   end
67: 
68:   def toStr
69:     return "[" + @rating.to_i.to_s + "," + @games.to_i.to_s + "]"
70:   end
71: 
72:   def load(party)
73:     ret = []
74:     team.length.times do |i|
75:       ret.push(party[team[i]])
76:     end
77:     return ret
78:   end

135: class PlayerRatingElo

137: 
138:   K_VALUE = 16
139: 
140:   def initialize
141:     @rating          = 1600.0
142:     @deviation       = 0
143:     @volatility      = 0
144:     @estimatedRating = nil
145:   end
146: 
147:   def winChancePercent
148:     return @estimatedRating if @estimatedRating
149:     x = (1 + (10.0**((@rating - 1600.0) / 400.0)))
150:     @estimatedRating = (x == 0 ? 1.0 : 1.0 / x)
151:     return @estimatedRating
152:   end
153: 
154:   def update(matches)
155:     return if matches.length == 0
156:     stake = 0
157:     matches.length.times do
158:       score = (match.score == -1) ? 0.5 : match.score
159:       e = (1 + (10.0**((@rating - match.opponentRating) / 400.0)))
160:       stake += match.kValue * (score - e)
161:     end
162:     @rating += stake
163:   end
164: end

178:     @estimatedRating = nil

181:   def winChancePercent
182:     return @estimatedRating if @estimatedRating
183:     if @deviation > 100
184:       # https://www.smogon.com/forums/threads/make-sense-of-your-shoddy-battle-rating.55764/
185:       otherRating = 1500.0
186:       otherDeviation = 350.0
187:       s = Math.sqrt(100_000.0 + (@deviation * @deviation) + (otherDeviation * otherDeviation))
188:       g = 10.0**((otherRating - @rating) * 0.79 / s)
189:       @estimatedRating = (1.0 / (1.0 + g)) * 100.0   # Percent chance that I win against opponent
190:     else
191:       # GLIXARE method
192:       rds = @deviation * @deviation
193:       sqr = Math.sqrt(15.905694331435 * (rds + 221_781.21786254))
194:       inner = (1500.0 - @rating) * Math::PI / sqr
195:       @estimatedRating = ((10_000.0 / (1.0 + (10.0**inner))) + 0.5) / 100.0
196:     end
197:     return @estimatedRating
198:   end
