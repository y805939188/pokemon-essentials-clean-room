4: module FilenameUpdater
5:   module_function
6: 
7:   def readDirectoryFiles(directory, formats)
8:     files = []
9:     Dir.chdir(directory) do
10:       formats.each do |format|
11:         Dir.glob(format) { |f| files.push(f) }
12:       end
13:     end
14:     return files
15:   end
16: 
17:   def rename_berry_plant_charsets
18:     src_dir = "Graphics/Characters/"
19:     return false if !FileTest.directory?(src_dir)
20:     Console.echo_li(_INTL("Renaming berry tree charsets..."))
21:     ret = false
22:     # generates a list of all graphic files
23:     files = readDirectoryFiles(src_dir, ["berrytree*.png"])
24:     # starts automatic renaming
25:     files.each_with_index do |file, i|
26:       next if file[/^berrytree_/]
27:       next if ["berrytreewet", "berrytreedamp", "berrytreedry", "berrytreeplanted"].include?(file.split(".")[0])
28:       new_file = file.gsub("berrytree", "berrytree_")
29:       File.move(src_dir + file, src_dir + new_file)
30:       ret = true
31:     end
32:     Console.echo_done(true)
33:     return ret
34:   end
35: 
36:   def update_berry_tree_event_charsets
37:     ret = []
38:     mapData = Compiler::MapData.new
39:     t = System.uptime
40:     Graphics.update
41:     Console.echo_li(_INTL("Checking {1} maps for used berry tree charsets...", mapData.mapinfos.keys.length))
42:     idx = 0
43:     mapData.mapinfos.keys.sort.each do |id|
44:       echo "." if idx % 20 == 0
45:       idx += 1
46:       Graphics.update if idx % 250 == 0
47:       map = mapData.getMap(id)
48:       next if !map || !mapData.mapinfos[id]
49:       changed = false
50:       map.events.each_key do |key|
51:         if System.uptime - t >= 5
52:           t += 5
53:           Graphics.update
54:         end
55:         map.events[key].pages.each do |page|
56:           next if nil_or_empty?(page.graphic.character_name)
57:           char_name = page.graphic.character_name
58:           next if !char_name[/^berrytree[^_]+/]
59:           next if ["berrytreewet", "berrytreedamp", "berrytreedry", "berrytreeplanted"].include?(char_name.split(".")[0])
60:           new_file = page.graphic.character_name.gsub("berrytree", "berrytree_")
61:           page.graphic.character_name = new_file
62:           changed = true
63:         end
64:       end
65:       next if !changed
66:       mapData.saveMap(id)
67:       ret.push(_INTL("Map {1}: '{2}' was modified and saved.", id, mapData.mapinfos[id].name))
68:     end
69:     Console.echo_done(true)
70:     return ret
71:   end
72: 
73:   def rename_files
74:     Console.echo_h1(_INTL("Updating file names and locations"))
75:     change_record = []
76:     # Add underscore to berry plant charsets
77:     if rename_berry_plant_charsets
78:       Console.echo_warn(_INTL("Berry plant charset files were renamed."))
79:     end
80:     change_record += update_berry_tree_event_charsets
81:     # Warn if any map data has been changed
82:     if !change_record.empty?
83:       change_record.each { |msg| Console.echo_warn(msg) }
84:       Console.echo_warn(_INTL("RMXP data was altered. Close RMXP now to ensure changes are applied."))
85:     end
86:     echoln ""
87:     Console.echo_h2(_INTL("Finished updating file names and locations"), text: :green)
88:   end
89: end
