201:       return false
202:     end
203:     team.each do |pkmn|
204:       next if isPokemonValid?(pkmn)
205:       if pkmn
206:         error&.push(_INTL("{1} is not allowed.", pkmn.name))
207:       elsif error
208:         error.push(_INTL("This team is not allowed."))
209:       end
210:       return false
211:     end
212:     @teamRules.each do |rule|
213:       next if rule.isValid?(team)
214:       error&.push(rule.errorMessage)
215:       return false
216:     end
217:     @subsetRules.each do |rule|
218:       next if rule.isValid?(team)
219:       error&.push(rule.errorMessage)
220:       return false
221:     end
222:     return true
223:   end
224: end
