582: def pbLearnMove(pkmn, move, ignore_if_known = false, by_machine = false, &block)
583:   return false if !pkmn
584:   move = GameData::Move.get(move).id
585:   if pkmn.egg? && !$DEBUG
586:     pbMessage(_INTL("Eggs can't be taught any moves."), &block)
587:     return false
588:   elsif pkmn.shadowPokemon?
589:     pbMessage(_INTL("Shadow Pokémon can't be taught any moves."), &block)
590:     return false
591:   end
592:   pkmn_name = pkmn.name
593:   move_name = GameData::Move.get(move).name
594:   if pkmn.hasMove?(move)
595:     pbMessage(_INTL("{1} already knows {2}.", pkmn_name, move_name), &block) if !ignore_if_known
596:     return false
597:   elsif pkmn.numMoves < Pokemon::MAX_MOVES
598:     pkmn.learn_move(move)
599:     pbMessage("\\se[]" + _INTL("{1} learned {2}!", pkmn_name, move_name) + "\\se[Pkmn move learnt]", &block)
600:     return true
601:   end
602:   pbMessage(_INTL("{1} wants to learn {2}, but it already knows {3} moves.",
603:                   pkmn_name, move_name, pkmn.numMoves.to_word) + "\1", &block)
604:   if pbConfirmMessage(_INTL("Should {1} forget a move to learn {2}?", pkmn_name, move_name), &block)
605:     loop do
606:       move_index = pbForgetMove(pkmn, move)
607:       if move_index >= 0
608:         old_move_name = pkmn.moves[move_index].name
609:         oldmovepp = pkmn.moves[move_index].pp
610:         pkmn.moves[move_index] = Pokemon::Move.new(move)   # Replaces current/total PP
611:         if by_machine && Settings::TAUGHT_MACHINES_KEEP_OLD_PP
612:           pkmn.moves[move_index].pp = [oldmovepp, pkmn.moves[move_index].total_pp].min
613:         end
614:         pbMessage(_INTL("1, 2, and...\\wt[16] ...\\wt[16] ...\\wt[16] Ta-da!") + "\\se[Battle ball drop]\1", &block)
615:         pbMessage(_INTL("{1} forgot how to use {2}.\\nAnd..." + "\1", pkmn_name, old_move_name), &block)
616:         pbMessage("\\se[]" + _INTL("{1} learned {2}!", pkmn_name, move_name) + "\\se[Pkmn move learnt]", &block)
617:         pkmn.changeHappiness("machine") if by_machine
618:         return true
619:       elsif pbConfirmMessage(_INTL("Give up on learning {1}?", move_name), &block)
620:         pbMessage(_INTL("{1} did not learn {2}.", pkmn_name, move_name), &block)
621:         return false
622:       end
623:     end
624:   else
625:     pbMessage(_INTL("{1} did not learn {2}.", pkmn_name, move_name), &block)
626:   end
627:   return false
628: end
