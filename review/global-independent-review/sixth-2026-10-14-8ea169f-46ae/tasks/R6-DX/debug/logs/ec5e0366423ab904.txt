227:   def update
228:     cursor_to_show = ((System.uptime - @cursor_timer_start) / 0.35).to_i.even?
229:     if cursor_to_show != @cursor_shown
230:       @cursor_shown = cursor_to_show
231:       refresh
232:     end
233:     return if !self.active
234:     # Moving cursor
235:     if Input.triggerex?(:LEFT) || Input.repeatex?(:LEFT)
236:       if @helper.cursor > 0
237:         @helper.cursor -= 1
238:         @cursor_timer_start = System.uptime
239:         @cursor_shown = true
240:         self.refresh
241:       end
242:       return
243:     elsif Input.triggerex?(:RIGHT) || Input.repeatex?(:RIGHT)
244:       if @helper.cursor < self.text.scan(/./m).length
245:         @helper.cursor += 1
246:         @cursor_timer_start = System.uptime
247:         @cursor_shown = true
248:         self.refresh
249:       end
250:       return
251:     elsif Input.triggerex?(:BACKSPACE) || Input.repeatex?(:BACKSPACE)
252:       self.delete if @helper.cursor > 0
253:       return
254:     elsif Input.triggerex?(:RETURN) || Input.triggerex?(:ESCAPE)
255:       return
256:     end
257:     Input.gets.each_char { |c| insert(c) }
258:   end
259: end
