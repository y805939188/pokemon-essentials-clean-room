89:   def pbConvertAnimToNewFormat(textdata)
90:     needconverting = false
91:     textdata.length.times do |i|
92:       next if !textdata[i]
93:       PBAnimation::MAX_SPRITES.times do |j|
94:         next if !textdata[i][j]
95:         needconverting = true if textdata[i][j][AnimFrame::FOCUS].nil?
96:         break if needconverting
97:       end
98:       break if needconverting
99:     end
100:     if needconverting
101:       textdata.length.times do |i|
102:         next if !textdata[i]
103:         PBAnimation::MAX_SPRITES.times do |j|
104:           next if !textdata[i][j]
105:           textdata[i][j][AnimFrame::PRIORITY] = 1 if textdata[i][j][AnimFrame::PRIORITY].nil?
106:           case j
107:           when 0      # User battler
108:             textdata[i][j][AnimFrame::FOCUS] = 2
109:             textdata[i][j][AnimFrame::X] = Battle::Scene::FOCUSUSER_X
110:             textdata[i][j][AnimFrame::Y] = Battle::Scene::FOCUSUSER_Y
111:           when 1   # Target battler
112:             textdata[i][j][AnimFrame::FOCUS] = 1
113:             textdata[i][j][AnimFrame::X] = Battle::Scene::FOCUSTARGET_X
114:             textdata[i][j][AnimFrame::Y] = Battle::Scene::FOCUSTARGET_Y
115:           else
116:             textdata[i][j][AnimFrame::FOCUS] = (textdata.position || 4)
117:             if textdata.position == 1
118:               textdata[i][j][AnimFrame::X] += Battle::Scene::FOCUSTARGET_X
119:               textdata[i][j][AnimFrame::Y] += Battle::Scene::FOCUSTARGET_Y - 2
120:             end
121:           end
122:         end
123:       end
124:     end
125:     return needconverting
126:   end
