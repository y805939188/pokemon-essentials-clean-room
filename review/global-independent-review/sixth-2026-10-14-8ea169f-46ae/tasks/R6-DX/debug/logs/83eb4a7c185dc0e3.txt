531:   #=============================================================================
532:   # Save Battle Tower trainer data to PBS file
533:   #=============================================================================
534:   def write_battle_tower_trainers(bttrainers, filename)
535:     return if !bttrainers || !filename
536:     btTrainersRequiredTypes = {
537:       "Type"          => [0, "e", nil],   # Specifies a trainer
538:       "Name"          => [1, "s"],
539:       "BeginSpeech"   => [2, "s"],
540:       "EndSpeechWin"  => [3, "s"],
541:       "EndSpeechLose" => [4, "s"],
542:       "PokemonNos"    => [5, "*u"]
543:     }
544:     File.open(filename, "wb") do |f|
545:       add_PBS_header_to_file(f)
546:       bttrainers.length.times do |i|
547:         next if !bttrainers[i]
548:         f.write("\#-------------------------------\r\n")
549:         f.write(sprintf("[%03d]\r\n", i))
550:         btTrainersRequiredTypes.each_key do |key|
551:           schema = btTrainersRequiredTypes[key]
552:           record = bttrainers[i][schema[0]]
553:           next if record.nil?
554:           f.write(sprintf("%s = ", key))
555:           case key
556:           when "Type"
557:             f.write(record.to_s)
558:           when "PokemonNos"
559:             f.write(record.join(","))   # pbWriteCsvRecord somehow won't work here
560:           else
561:             pbWriteCsvRecord(record, f, schema)
562:           end
563:           f.write("\r\n")
564:         end
565:       end
566:     end
567:     Graphics.update
568:   end

570:   #=============================================================================
571:   # Save Battle Tower Pokémon data to PBS file
572:   #=============================================================================
573:   def write_battle_tower_pokemon(btpokemon, filename)
574:     return if !btpokemon || !filename
575:     species = {}
576:     moves   = {}
577:     items   = {}
578:     natures = {}
579:     evs = {
580:       :HP              => "HP",
581:       :ATTACK          => "ATK",
582:       :DEFENSE         => "DEF",
583:       :SPECIAL_ATTACK  => "SA",
584:       :SPECIAL_DEFENSE => "SD",
585:       :SPEED           => "SPD"
586:     }
587:     File.open(filename, "wb") do |f|
588:       add_PBS_header_to_file(f)
589:       f.write("\#-------------------------------\r\n")
590:       btpokemon.length.times do |i|
591:         Graphics.update if i % 500 == 0
592:         pkmn = btpokemon[i]
593:         c1 = (species[pkmn.species]) ? species[pkmn.species] : (species[pkmn.species] = GameData::Species.get(pkmn.species).species.to_s)
594:         c2 = nil
595:         if pkmn.item && GameData::Item.exists?(pkmn.item)
596:           c2 = (items[pkmn.item]) ? items[pkmn.item] : (items[pkmn.item] = GameData::Item.get(pkmn.item).id.to_s)
597:         end
598:         c3 = (natures[pkmn.nature]) ? natures[pkmn.nature] : (natures[pkmn.nature] = GameData::Nature.get(pkmn.nature).id.to_s)
599:         evlist = ""
600:         pkmn.ev.each_with_index do |stat, j|
601:           evlist += "," if j > 0
602:           evlist += evs[stat]
603:         end
604:         c4 = c5 = c6 = c7 = ""
605:         [pkmn.move1, pkmn.move2, pkmn.move3, pkmn.move4].each_with_index do |move, j|
606:           next if !move
607:           text = (moves[move]) ? moves[move] : (moves[move] = GameData::Move.get(move).id.to_s)
608:           case j
609:           when 0 then c4 = text
610:           when 1 then c5 = text
611:           when 2 then c6 = text
612:           when 3 then c7 = text
613:           end
614:         end
615:         f.write("#{c1};#{c2};#{c3};#{evlist};#{c4},#{c5},#{c6},#{c7}\r\n")
616:       end
617:     end
618:     Graphics.update
619:   end
