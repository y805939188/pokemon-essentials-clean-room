804:   "effect"      => proc {
805:     params = ChooseNumberParams.new
806:     params.setRange(1, Settings::BAG_MAX_PER_SLOT)
807:     params.setInitialValue(1)
808:     params.setCancelValue(0)
809:     qty = pbMessageChooseNumber(_INTL("Choose the number of items."), params)
810:     if qty > 0
811:       $bag.clear
812:       # NOTE: This doesn't simply use $bag.add for every item in turn, because
813:       #       that's really slow when done in bulk.
814:       pocket_sizes = Settings::BAG_MAX_POCKET_SIZE
815:       bag = $bag.pockets   # Called here so that it only rearranges itself once
816:       GameData::Item.each do |i|
817:         next if !pocket_sizes[i.pocket - 1] || pocket_sizes[i.pocket - 1] == 0
818:         next if pocket_sizes[i.pocket - 1] > 0 && bag[i.pocket].length >= pocket_sizes[i.pocket - 1]
819:         item_qty = (i.is_important?) ? 1 : qty
820:         bag[i.pocket].push([i.id, item_qty])
821:       end
822:       # NOTE: Auto-sorting pockets don't need to be sorted afterwards, because
823:       #       items are added in the same order they would be sorted into.
824:       pbMessage(_INTL("The Bag was filled with {1} of each item.", qty))
825:     end
826:   }
