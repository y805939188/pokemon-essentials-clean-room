967:   def get_all_pbs_files_to_compile
968:     # Get the GameData classes and their respective base PBS filenames
969:     ret = GameData.get_all_pbs_base_filenames
970:     ret.merge!({
971:       :BattleFacility => "battle_facility_lists",
972:       :Connection     => "map_connections",
973:       :RegionalDex    => "regional_dexes"
974:     })
975:     ret.each { |key, val| ret[key] = [val] }   # [base_filename, ["PBS/file.txt", etc.]]
976:     # Look through all PBS files and match them to a GameData class based on
977:     # their base filenames
978:     text_files_keys = ret.keys.sort! { |a, b| ret[b][0].length <=> ret[a][0].length }
979:     Dir.chdir("PBS/") do
980:       Dir.glob("*.txt") do |f|
981:         base_name = File.basename(f, ".txt")
982:         text_files_keys.each do |key|
983:           next if base_name != ret[key][0] && !f.start_with?(ret[key][0] + "_")
984:           ret[key][1] ||= []
985:           ret[key][1].push("PBS/" + f)
986:           break
987:         end
988:       end
989:     end
990:     return ret
991:   end
