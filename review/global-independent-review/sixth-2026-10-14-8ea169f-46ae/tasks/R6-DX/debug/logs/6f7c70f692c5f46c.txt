1: #===============================================================================
2: # Nicknaming and storing Pokémon
3: #===============================================================================
4: def pbBoxesFull?
5:   return ($player.party_full? && $PokemonStorage.full?)
6: end
7: 
8: def pbNickname(pkmn)
9:   return if $PokemonSystem.givenicknames != 0
10:   species_name = pkmn.speciesName
11:   if pbConfirmMessage(_INTL("Would you like to give a nickname to {1}?", species_name))
12:     pkmn.name = pbEnterPokemonName(_INTL("{1}'s nickname?", species_name),
13:                                    0, Pokemon::MAX_NAME_SIZE, "", pkmn)
14:   end
15: end
16: 
17: def pbStorePokemon(pkmn)
18:   if pbBoxesFull?
19:     pbMessage(_INTL("There's no more room for Pokémon!") + "\1")
20:     pbMessage(_INTL("The Pokémon Boxes are full and can't accept any more!"))
21:     return
22:   end
23:   pkmn.record_first_moves
24:   if $player.party_full?
25:     stored_box = $PokemonStorage.pbStoreCaught(pkmn)
26:     box_name   = $PokemonStorage[stored_box].name
27:     pbMessage(_INTL("{1} has been sent to Box \"{2}\"!", pkmn.name, box_name))
28:   else
29:     $player.party[$player.party.length] = pkmn
30:   end
31: end
32: 
33: def pbNicknameAndStore(pkmn)
34:   if pbBoxesFull?
35:     pbMessage(_INTL("There's no more room for Pokémon!") + "\1")
36:     pbMessage(_INTL("The Pokémon Boxes are full and can't accept any more!"))
37:     return
38:   end
39:   $player.pokedex.set_seen(pkmn.species)
40:   $player.pokedex.set_owned(pkmn.species)
41:   pbNickname(pkmn)
42:   pbStorePokemon(pkmn)
43: end
44: 
45: #===============================================================================
46: # Giving Pokémon to the player (will send to storage if party is full)
47: #===============================================================================
48: def pbAddPokemon(pkmn, level = 1, see_form = true)
49:   return false if !pkmn
50:   if pbBoxesFull?
51:     pbMessage(_INTL("There's no more room for Pokémon!") + "\1")
52:     pbMessage(_INTL("The Pokémon Boxes are full and can't accept any more!"))
53:     return false
54:   end
55:   pkmn = Pokemon.new(pkmn, level) if !pkmn.is_a?(Pokemon)
56:   species_name = pkmn.speciesName
57:   pbMessage(_INTL("{1} obtained {2}!", $player.name, species_name) + "\\me[Pkmn get]\\wtnp[80]")
58:   was_owned = $player.owned?(pkmn.species)
59:   $player.pokedex.set_seen(pkmn.species)
60:   $player.pokedex.set_owned(pkmn.species)
61:   $player.pokedex.register(pkmn) if see_form
62:   # Show Pokédex entry for new species if it hasn't been owned before
63:   if Settings::SHOW_NEW_SPECIES_POKEDEX_ENTRY_MORE_OFTEN && see_form && !was_owned &&
64:      $player.has_pokedex && $player.pokedex.species_in_unlocked_dex?(pkmn.species)
65:     pbMessage(_INTL("{1}'s data was added to the Pokédex.", species_name))
66:     $player.pokedex.register_last_seen(pkmn)
67:     pbFadeOutIn do
68:       scene = PokemonPokedexInfo_Scene.new
69:       screen = PokemonPokedexInfoScreen.new(scene)
70:       screen.pbDexEntry(pkmn.species)
71:     end
72:   end
73:   # Nickname and add the Pokémon
74:   pbNicknameAndStore(pkmn)
75:   return true
76: end

78: def pbAddPokemonSilent(pkmn, level = 1, see_form = true)
79:   return false if !pkmn || pbBoxesFull?
80:   pkmn = Pokemon.new(pkmn, level) if !pkmn.is_a?(Pokemon)
81:   $player.pokedex.set_seen(pkmn.species)
82:   $player.pokedex.set_owned(pkmn.species)
83:   $player.pokedex.register(pkmn) if see_form
84:   pkmn.record_first_moves
85:   if $player.party_full?
86:     $PokemonStorage.pbStoreCaught(pkmn)
87:   else
88:     $player.party[$player.party.length] = pkmn
89:   end
90:   return true
91: end
