11: MenuHandlers.add(:debug_menu, :warp, {
12:   "name"        => _INTL("Warp to map"),
13:   "parent"      => :field_menu,
14:   "description" => _INTL("Instantly warp to another map of your choice."),
15:   "effect"      => proc { |sprites, viewport|
16:     map = pbWarpToMap
17:     next false if !map
18:     pbFadeOutAndHide(sprites)
19:     pbDisposeMessageWindow(sprites["textbox"])
20:     pbDisposeSpriteHash(sprites)
21:     viewport.dispose
22:     if $scene.is_a?(Scene_Map)
23:       $game_temp.player_new_map_id    = map[0]
24:       $game_temp.player_new_x         = map[1]
25:       $game_temp.player_new_y         = map[2]
26:       $game_temp.player_new_direction = 2
27:       $scene.transfer_player
28:     else
29:       pbCancelVehicles
30:       $map_factory.setup(map[0])
31:       $game_player.moveto(map[1], map[2])
32:       $game_player.turn_down
33:       $game_map.update
34:       $game_map.autoplay
35:     end
36:     $game_map.refresh
37:     next true   # Closes the debug menu to allow the warp
38:   }
39: })

68: MenuHandlers.add(:debug_menu, :safari_zone_and_bug_contest, {
69:   "name"        => _INTL("Safari Zone and Bug-Catching Contest"),
70:   "parent"      => :field_menu,
71:   "description" => _INTL("Edit steps/time remaining and number of usable Poké Balls."),
72:   "effect"      => proc {
73:     if pbInSafari?
74:       safari = pbSafariState
75:       cmd = 0
76:       loop do
77:         cmds = [_INTL("Steps remaining: {1}", (Settings::SAFARI_STEPS > 0) ? safari.steps : _INTL("infinite")),
78:                 GameData::Item.get(:SAFARIBALL).name_plural + ": " + safari.ballcount.to_s]
79:         cmd = pbShowCommands(nil, cmds, -1, cmd)
80:         break if cmd < 0
81:         case cmd
82:         when 0   # Steps remaining
83:           if Settings::SAFARI_STEPS > 0
84:             params = ChooseNumberParams.new
85:             params.setRange(0, 99999)
86:             params.setDefaultValue(safari.steps)
87:             safari.steps = pbMessageChooseNumber(_INTL("Set the steps remaining in this Safari game."), params)
88:           end
89:         when 1   # Safari Balls
90:           params = ChooseNumberParams.new
91:           params.setRange(0, 99999)
92:           params.setDefaultValue(safari.ballcount)
93:           safari.ballcount = pbMessageChooseNumber(
94:             _INTL("Set the quantity of {1}.", GameData::Item.get(:SAFARIBALL).name_plural), params)
95:         end
96:       end
97:     elsif pbInBugContest?
98:       contest = pbBugContestState
99:       cmd = 0
100:       loop do
101:         cmds = []
102:         if Settings::BUG_CONTEST_TIME > 0
103:           time_left = Settings::BUG_CONTEST_TIME - (System.uptime - contest.timer_start).to_i
104:           time_left = 0 if time_left < 0
105:           min = time_left / 60
106:           sec = time_left % 60
107:           time_string = _ISPRINTF("{1:02d}m {2:02d}s", min, sec)
108:         else
109:           time_string = _INTL("infinite")
110:         end
111:         cmds.push(_INTL("Time remaining: {1}", time_string))
112:         cmds.push(GameData::Item.get(:SPORTBALL).name_plural + ": " + contest.ballcount.to_s)
113:         cmd = pbShowCommands(nil, cmds, -1, cmd)
114:         break if cmd < 0
115:         case cmd
116:         when 0   # Steps remaining
117:           if Settings::BUG_CONTEST_TIME > 0
118:             params = ChooseNumberParams.new
119:             params.setRange(0, 99999)
120:             params.setDefaultValue(min)
121:             new_time = pbMessageChooseNumber(_INTL("Set the time remaining (in minutes) in this Bug-Catching Contest."), params)
122:             contest.timer_start += (new_time - min) * 60
123:             $scene.spriteset.usersprites.each do |sprite|
124:               next if !sprite.is_a?(TimerDisplay)
125:               sprite.start_time = contest.timer_start
126:               break
127:             end
128:           end
129:         when 1   # Sport Balls
130:           params = ChooseNumberParams.new
131:           params.setRange(0, 99999)
132:           params.setDefaultValue(contest.ballcount)
133:           contest.ballcount = pbMessageChooseNumber(
134:             _INTL("Set the quantity of {1}.", GameData::Item.get(:SPORTBALL).name_plural), params)
135:         end
136:       end
137:     else
138:       pbMessage(_INTL("You aren't in the Safari Zone or a Bug-Catching Contest!"))
139:     end
140:   }
141: })

217: MenuHandlers.add(:debug_menu, :storage_wallpapers, {
218:   "name"        => _INTL("Toggle storage wallpapers"),
219:   "parent"      => :field_menu,
220:   "description" => _INTL("Unlock and lock special wallpapers used in Pokémon storage."),
221:   "effect"      => proc {
222:     w = $PokemonStorage.allWallpapers
223:     if w.length <= PokemonStorage::BASICWALLPAPERQTY
224:       pbMessage(_INTL("There are no special wallpapers defined."))
225:     else
226:       paperscmd = 0
227:       unlockarray = $PokemonStorage.unlockedWallpapers
228:       loop do
229:         paperscmds = []
230:         paperscmds.push(_INTL("Unlock all"))
231:         paperscmds.push(_INTL("Lock all"))
232:         (PokemonStorage::BASICWALLPAPERQTY...w.length).each do |i|
233:           paperscmds.push((unlockarray[i] ? "[Y]" : "[  ]") + " " + w[i])
234:         end
235:         paperscmd = pbShowCommands(nil, paperscmds, -1, paperscmd)
236:         break if paperscmd < 0
237:         case paperscmd
238:         when 0   # Unlock all
239:           (PokemonStorage::BASICWALLPAPERQTY...w.length).each do |i|
240:             unlockarray[i] = true
241:           end
242:         when 1   # Lock all
243:           (PokemonStorage::BASICWALLPAPERQTY...w.length).each do |i|
244:             unlockarray[i] = false
245:           end
246:         else
247:           paperindex = paperscmd - 2 + PokemonStorage::BASICWALLPAPERQTY
248:           unlockarray[paperindex] = !$PokemonStorage.unlockedWallpapers[paperindex]
249:         end
250:       end
251:     end
252:   }
253: })
