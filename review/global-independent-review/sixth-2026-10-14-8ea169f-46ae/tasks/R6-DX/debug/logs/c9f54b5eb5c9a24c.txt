166: def pbChooseNumber(msgwindow, params)
167:   return 0 if !params
168:   ret = 0
169:   maximum = params.maxNumber
170:   minimum = params.minNumber
171:   defaultNumber = params.initialNumber
172:   cancelNumber = params.cancelNumber
173:   cmdwindow = Window_InputNumberPokemon.new(params.maxDigits)
174:   cmdwindow.z = 99999
175:   cmdwindow.visible = true
176:   cmdwindow.setSkin(params.skin) if params.skin
177:   cmdwindow.sign = params.negativesAllowed # must be set before number
178:   cmdwindow.number = defaultNumber
179:   pbPositionNearMsgWindow(cmdwindow, msgwindow, :right)
180:   loop do
181:     Graphics.update
182:     Input.update
183:     pbUpdateSceneMap
184:     cmdwindow.update
185:     msgwindow&.update
186:     yield if block_given?
187:     if Input.trigger?(Input::USE)
188:       ret = cmdwindow.number
189:       if ret > maximum
190:         pbPlayBuzzerSE
191:       elsif ret < minimum
192:         pbPlayBuzzerSE
193:       else
194:         pbPlayDecisionSE
195:         break
196:       end
197:     elsif Input.trigger?(Input::BACK)
198:       pbPlayCancelSE
199:       ret = cancelNumber
200:       break
201:     end
202:   end
203:   cmdwindow.dispose
204:   Input.update
205:   return ret
206: end
