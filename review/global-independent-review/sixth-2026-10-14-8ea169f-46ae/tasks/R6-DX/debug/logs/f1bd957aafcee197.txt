706: def pbReceiveItem(item, quantity = 1)
707:   item = GameData::Item.get(item)
708:   return false if !item || quantity < 1
709:   itemname = (quantity > 1) ? item.portion_name_plural : item.portion_name
710:   pocket = item.pocket
711:   move = item.move
712:   meName = (item.is_key_item?) ? "Key item get" : "Item get"
713:   if item == :DNASPLICERS
714:     pbMessage("\\me[#{meName}]" + _INTL("You obtained \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
715:   elsif item.is_machine?   # TM or HM
716:     if quantity > 1
717:       pbMessage("\\me[Machine get]" + _INTL("You obtained {1} \\c[1]{2} {3}\\c[0]!",
718:                                             quantity, itemname, GameData::Move.get(move).name) + "\\wtnp[70]")
719:     else
720:       pbMessage("\\me[Machine get]" + _INTL("You obtained \\c[1]{1} {2}\\c[0]!",
721:                                             itemname, GameData::Move.get(move).name) + "\\wtnp[70]")
722:     end
723:   elsif quantity > 1
724:     pbMessage("\\me[#{meName}]" + _INTL("You obtained {1} \\c[1]{2}\\c[0]!", quantity, itemname) + "\\wtnp[40]")
725:   elsif itemname.starts_with_vowel?
726:     pbMessage("\\me[#{meName}]" + _INTL("You obtained an \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
727:   else
728:     pbMessage("\\me[#{meName}]" + _INTL("You obtained a \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
729:   end
730:   if $bag.add(item, quantity)   # If item can be added
731:     pbMessage(_INTL("You put the {1} in\\nyour Bag's <icon=bagPocket{2}>\\c[1]{3}\\c[0] pocket.",
732:                     itemname, pocket, PokemonBag.pocket_names[pocket - 1]))
733:     return true
734:   end
735:   return false   # Can't add the item
736: end
