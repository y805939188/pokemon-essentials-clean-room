84:   def update
85:     @loop_count = 0
86:     loop do
87:       @loop_count += 1
88:       if @loop_count > 100   # Call Graphics.update for freeze prevention
89:         Graphics.update
90:         @loop_count = 0
91:       end
92:       # If this interpreter's map isn't the current map or connected to it,
93:       # forget this interpreter's event ID
94:       if $game_map.map_id != @map_id && !$map_factory.areConnected?($game_map.map_id, @map_id)
95:         @event_id = 0
96:       end
97:       # Update child interpreter if one exists
98:       if @child_interpreter
99:         @child_interpreter.update
100:         @child_interpreter = nil if !@child_interpreter.running?
101:         return if @child_interpreter
102:       end
103:       # Do nothing if a message is being shown
104:       return if @message_waiting
105:       # Do nothing if any event or the player is in the middle of a move route
106:       if @move_route_waiting
107:         return if $game_player.move_route_forcing
108:         $game_map.events.each_value do |event|
109:           return if event.move_route_forcing
110:         end
111:         $game_temp.followers.each_follower do |event, follower|
112:           return if event.move_route_forcing
113:         end
114:         @move_route_waiting = false
115:       end
116:       # Do nothing if the player is jumping out of surfing
117:       return if $game_temp.ending_surf
118:       # Do nothing while waiting
119:       if @wait_count > 0
120:         return if System.uptime - @wait_start < @wait_count
121:         @wait_count = 0
122:         @wait_start = nil
123:       end
124:       # Do nothing if the pause menu is going to open
125:       return if $game_temp.menu_calling
126:       # If there are no commands in the list, try to find something that wants to run
127:       if @list.nil?
128:         setup_starting_event if @main
129:         return if @list.nil?   # Couldn't find anything that wants to run
130:       end
131:       # Execute the next command
132:       return if execute_command == false
133:       # Move to the next @index
134:       @index += 1
135:     end
136:   end
