474:   def playTiming(frame, bgGraphic, bgColor, foGraphic, foColor, oldbg = [], oldfo = [], user = nil)
475:     @timing.each do |i|
476:       next if !i.duration || i.duration <= 0
477:       next if i.frame + i.duration < frame || i.frame >= frame
478:       fraction = (frame - i.frame).to_f / i.duration
479:       case i.timingType
480:       when 2
481:         if bgGraphic.bitmap.nil?
482:           bgColor.opacity = oldbg[2] + ((i.opacity - oldbg[2]) * fraction) if i.opacity
483:           cr = (i.colorRed) ? oldbg[3].red + ((i.colorRed - oldbg[3].red) * fraction) : oldbg[3].red
484:           cg = (i.colorGreen) ? oldbg[3].green + ((i.colorGreen - oldbg[3].green) * fraction) : oldbg[3].green
485:           cb = (i.colorBlue) ? oldbg[3].blue + ((i.colorBlue - oldbg[3].blue) * fraction) : oldbg[3].blue
486:           ca = (i.colorAlpha) ? oldbg[3].alpha + ((i.colorAlpha - oldbg[3].alpha) * fraction) : oldbg[3].alpha
487:           bgColor.color = Color.new(cr, cg, cb, ca)
488:         else
489:           bgGraphic.ox      = oldbg[0] - ((i.bgX - oldbg[0]) * fraction) if i.bgX
490:           bgGraphic.oy      = oldbg[1] - ((i.bgY - oldbg[1]) * fraction) if i.bgY
491:           bgGraphic.opacity = oldbg[2] + ((i.opacity - oldbg[2]) * fraction) if i.opacity
492:           cr = (i.colorRed) ? oldbg[3].red + ((i.colorRed - oldbg[3].red) * fraction) : oldbg[3].red
493:           cg = (i.colorGreen) ? oldbg[3].green + ((i.colorGreen - oldbg[3].green) * fraction) : oldbg[3].green
494:           cb = (i.colorBlue) ? oldbg[3].blue + ((i.colorBlue - oldbg[3].blue) * fraction) : oldbg[3].blue
495:           ca = (i.colorAlpha) ? oldbg[3].alpha + ((i.colorAlpha - oldbg[3].alpha) * fraction) : oldbg[3].alpha
496:           bgGraphic.color = Color.new(cr, cg, cb, ca)
497:         end
498:       when 4
499:         if foGraphic.bitmap.nil?
500:           foColor.opacity = oldfo[2] + ((i.opacity - oldfo[2]) * fraction) if i.opacity
501:           cr = (i.colorRed) ? oldfo[3].red + ((i.colorRed - oldfo[3].red) * fraction) : oldfo[3].red
502:           cg = (i.colorGreen) ? oldfo[3].green + ((i.colorGreen - oldfo[3].green) * fraction) : oldfo[3].green
503:           cb = (i.colorBlue) ? oldfo[3].blue + ((i.colorBlue - oldfo[3].blue) * fraction) : oldfo[3].blue
504:           ca = (i.colorAlpha) ? oldfo[3].alpha + ((i.colorAlpha - oldfo[3].alpha) * fraction) : oldfo[3].alpha
505:           foColor.color = Color.new(cr, cg, cb, ca)
506:         else
507:           foGraphic.ox      = oldfo[0] - ((i.bgX - oldfo[0]) * fraction) if i.bgX
508:           foGraphic.oy      = oldfo[1] - ((i.bgY - oldfo[1]) * fraction) if i.bgY
509:           foGraphic.opacity = oldfo[2] + ((i.opacity - oldfo[2]) * fraction) if i.opacity
510:           cr = (i.colorRed) ? oldfo[3].red + ((i.colorRed - oldfo[3].red) * fraction) : oldfo[3].red
511:           cg = (i.colorGreen) ? oldfo[3].green + ((i.colorGreen - oldfo[3].green) * fraction) : oldfo[3].green
512:           cb = (i.colorBlue) ? oldfo[3].blue + ((i.colorBlue - oldfo[3].blue) * fraction) : oldfo[3].blue
513:           ca = (i.colorAlpha) ? oldfo[3].alpha + ((i.colorAlpha - oldfo[3].alpha) * fraction) : oldfo[3].alpha
514:           foGraphic.color = Color.new(cr, cg, cb, ca)
515:         end
516:       end
517:     end
518:     @timing.each do |i|
519:       next if i.frame != frame
520:       case i.timingType
521:       when 0   # Play SE
522:         if i.name && i.name != ""
523:           pbSEPlay("Anim/" + i.name, i.volume, i.pitch)
524:         elsif user&.pokemon
525:           name = GameData::Species.cry_filename_from_pokemon(user.pokemon)
526:           pbSEPlay(name, i.volume, i.pitch) if name
527:         end
528: #        if sprite
529: #          sprite.flash(i.flashColor, i.flashDuration * 2) if i.flashScope == 1
530: #          sprite.flash(nil, i.flashDuration * 2) if i.flashScope == 3
531: #        end
532:       when 1   # Set background graphic (immediate)
533:         if i.name && i.name != ""
534:           bgGraphic.setBitmap("Graphics/Animations/" + i.name)
535:           bgGraphic.ox      = -i.bgX || 0
536:           bgGraphic.oy      = -i.bgY || 0
537:           bgGraphic.color   = Color.new(i.colorRed || 0, i.colorGreen || 0, i.colorBlue || 0, i.colorAlpha || 0)
538:           bgGraphic.opacity = i.opacity || 0
539:           bgColor.opacity = 0
540:         else
541:           bgGraphic.setBitmap(nil)
542:           bgGraphic.opacity = 0
543:           bgColor.color   = Color.new(i.colorRed || 0, i.colorGreen || 0, i.colorBlue || 0, i.colorAlpha || 0)
544:           bgColor.opacity = i.opacity || 0
545:         end
546:       when 2   # Move/recolour background graphic
547:         if bgGraphic.bitmap.nil?
548:           oldbg[0] = 0

592: def pbSpriteSetAnimFrame(sprite, frame, user = nil, target = nil, inEditor = false)
593:   return if !sprite
594:   if !frame
595:     sprite.visible  = false
596:     sprite.src_rect = Rect.new(0, 0, 1, 1)
597:     return
598:   end
599:   sprite.blend_type = frame[AnimFrame::BLENDTYPE]
600:   sprite.angle      = frame[AnimFrame::ANGLE]
601:   sprite.mirror     = (frame[AnimFrame::MIRROR] > 0)
602:   sprite.opacity    = frame[AnimFrame::OPACITY]
603:   sprite.visible    = true
604:   if !frame[AnimFrame::VISIBLE] == 1 && inEditor
605:     sprite.opacity /= 2
606:   else
607:     sprite.visible = (frame[AnimFrame::VISIBLE] == 1)
608:   end
609:   pattern = frame[AnimFrame::PATTERN]
610:   if pattern >= 0
611:     animwidth = 192
612:     sprite.src_rect.set((pattern % 5) * animwidth, (pattern / 5) * animwidth,
613:                         animwidth, animwidth)
614:   else
615:     sprite.src_rect.set(0, 0,
616:                         (sprite.bitmap) ? sprite.bitmap.width : 128,
617:                         (sprite.bitmap) ? sprite.bitmap.height : 128)
618:   end
619:   sprite.zoom_x = frame[AnimFrame::ZOOMX] / 100.0
620:   sprite.zoom_y = frame[AnimFrame::ZOOMY] / 100.0
621:   sprite.color.set(
622:     frame[AnimFrame::COLORRED],
623:     frame[AnimFrame::COLORGREEN],
624:     frame[AnimFrame::COLORBLUE],
625:     frame[AnimFrame::COLORALPHA]
626:   )
627:   sprite.tone.set(
628:     frame[AnimFrame::TONERED],
629:     frame[AnimFrame::TONEGREEN],
630:     frame[AnimFrame::TONEBLUE],
631:     frame[AnimFrame::TONEGRAY]
632:   )
633:   sprite.ox = sprite.src_rect.width / 2
634:   sprite.oy = sprite.src_rect.height / 2
635:   sprite.x  = frame[AnimFrame::X]
636:   sprite.y  = frame[AnimFrame::Y]
637:   if sprite != user && sprite != target
638:     case frame[AnimFrame::PRIORITY]
639:     when 0   # Behind everything
640:       sprite.z = 10
641:     when 1   # In front of everything
642:       sprite.z = 80
643:     when 2   # Just behind focus
644:       case frame[AnimFrame::FOCUS]
645:       when 1   # Focused on target
646:         sprite.z = (target) ? target.z - 1 : 20
647:       when 2   # Focused on user
648:         sprite.z = (user) ? user.z - 1 : 20
649:       else     # Focused on user and target, or screen
650:         sprite.z = 20
651:       end
652:     when 3   # Just in front of focus
653:       case frame[AnimFrame::FOCUS]
654:       when 1   # Focused on target
655:         sprite.z = (target) ? target.z + 1 : 80
656:       when 2   # Focused on user
657:         sprite.z = (user) ? user.z + 1 : 80
658:       else     # Focused on user and target, or screen
659:         sprite.z = 80
660:       end
661:     else
662:       sprite.z = 80
663:     end
664:   end
665: end
