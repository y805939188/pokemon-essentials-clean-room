686:     cancelbutton = sliderwin2.addButton(_INTL("Cancel"))
687:     sliderwin2.opacity = 200

693:         startvalue = sliderwin2.value(0) - 1
694:         endvalue = sliderwin2.value(1) - 1
695:         dstvalue = sliderwin2.value(2) - 1
696:         length = (endvalue - startvalue) + 1
697:         if length > 0   # Ensure correct overlap handling
698:           if startvalue < dstvalue
699:             startvalue += length
700:             dstvalue += length
701:             while length != 0
702:               canvas.copyFrame(startvalue - 1, dstvalue - 1)
703:               startvalue -= 1
704:               dstvalue -= 1
705:               length -= 1
706:             end
707:           elsif startvalue != dstvalue
708:             while length != 0
709:               canvas.copyFrame(startvalue, dstvalue)
710:               startvalue += 1
711:               dstvalue += 1
712:               length -= 1
713:             end
714:           end
715:         end
716:         break
717:       end
718:       if sliderwin2.changed?(cancelbutton) || Input.trigger?(Input::BACK)
719:         break
720:       end
721:     end
