252:   class InvalidatableSprite < Sprite
253:     def initialize(viewport = nil)
254:       super(viewport)
255:       @invalid = false
256:     end
257: 
258:     # Marks that the control must be redrawn to reflect current logic.
259:     def invalidate
260:       @invalid = true
261:     end
262: 
263:     # Determines whether the control is invalid
264:     def invalid?
265:       return @invalid
266:     end
267: 
268:     # Marks that the control is valid.  Normally called only by repaint.
269:     def validate
270:       @invalid = false
271:     end
272: 
273:     # Redraws the sprite only if it is invalid, and then revalidates the sprite
274:     def repaint
275:       if self.invalid?
276:         refresh
277:         validate
278:       end
279:     end
280: 
281:     # Redraws the sprite.  This method should not check whether
282:     # the sprite is invalid, to allow it to be explicitly called.
283:     def refresh; end
284:   end

289:   class SpriteFrame < InvalidatableSprite
290:     attr_reader :id
291:     attr_reader :locked
292:     attr_reader :selected
293:     attr_reader :sprite
294: 
295:     NUM_ROWS = (PBAnimation::MAX_SPRITES.to_f / 10).ceil   # 10 frame number icons in each row
296: 
297:     def initialize(id, sprite, viewport, previous = false)
298:       super(viewport)
299:       @id = id
300:       @sprite = sprite
301:       @previous = previous
302:       @locked = false
303:       @selected = false
304:       @selcolor = Color.black
305:       @unselcolor = Color.new(220, 220, 220)
306:       @prevcolor = Color.new(64, 128, 192)
307:       @contents = Bitmap.new(64, 64)
308:       self.z = (@previous) ? 49 : 50
309:       @iconbitmap = AnimatedBitmap.new("Graphics/UI/Debug/anim_editor_frame_icons")
310:       self.bitmap = @contents
311:       self.invalidate
312:     end
313: 
314:     def dispose
315:       @contents.dispose
316:       super
317:     end
318: 
319:     def sprite=(value)
320:       @sprite = value
321:       self.invalidate
322:     end
323: 
324:     def locked=(value)
325:       @locked = value
326:       self.invalidate
327:     end
328: 
329:     def selected=(value)
330:       @selected = value
331:       self.invalidate
332:     end
333: 
334:     def refresh
335:       @contents.clear
336:       self.z = (@previous) ? 49 : (@selected) ? 51 : 50
337:       # Draw frame
338:       color = (@previous) ? @prevcolor : (@selected) ? @selcolor : @unselcolor
339:       @contents.fill_rect(0, 0, 64, 1, color)
340:       @contents.fill_rect(0, 63, 64, 1, color)
341:       @contents.fill_rect(0, 0, 1, 64, color)
342:       @contents.fill_rect(63, 0, 1, 64, color)
343:       # Determine frame number graphic to use from @iconbitmap
344:       yoffset = (@previous) ? (NUM_ROWS + 1) * 16 : 0   # 1 is for padlock icon
345:       bmrect = Rect.new((@id % 10) * 16, yoffset + ((@id / 10) * 16), 16, 16)
346:       @contents.blt(0, 0, @iconbitmap.bitmap, bmrect)
347:       # Draw padlock if frame is locked
348:       if @locked && !@previous
349:         bmrect = Rect.new(0, NUM_ROWS * 16, 16, 16)
350:         @contents.blt(16, 0, @iconbitmap.bitmap, bmrect)
351:       end
352:     end
353:   end

358:   class AnimationCanvas < Sprite
359:     attr_reader :viewport
360:     attr_reader :sprites
361:     attr_reader :currentframe # Currently active frame
362:     attr_reader :currentcel
363:     attr_reader :animation # Currently selected animation
364:     attr_reader :animbitmap # Currently selected animation bitmap
365:     attr_accessor :pattern  # Currently selected pattern
366: 
367:     BORDERSIZE = 64
368: 
369:     def initialize(animation, viewport = nil)
370:       super(viewport)
371:       @currentframe = 0
372:       @currentcel = -1
373:       @pattern = 0
374:       @sprites = {}
375:       @celsprites = []
376:       @framesprites = []
377:       @lastframesprites = []
378:       @dirty = []
379:       @viewport = viewport
380:       @selecting = false
381:       @selectOffsetX = 0
382:       @selectOffsetY = 0
383:       @playing = false
384:       @playingframe = 0
385:       @player = nil
386:       @battle = MiniBattle.new
387:       @user = AnimatedBitmap.new("Graphics/UI/Debug/anim_editor_battler_back").deanimate
388:       @target = AnimatedBitmap.new("Graphics/UI/Debug/anim_editor_battler_front").deanimate
389:       @testscreen = AnimatedBitmap.new("Graphics/UI/Debug/anim_editor_battle_bg")
390:       self.bitmap = @testscreen.bitmap
391:       PBAnimation::MAX_SPRITES.times do |i|
392:         @lastframesprites[i] = SpriteFrame.new(i, @celsprites[i], viewport, true)
393:         @lastframesprites[i].selected = false
394:         @lastframesprites[i].visible = false
395:       end
396:       PBAnimation::MAX_SPRITES.times do |i|
397:         @celsprites[i] = Sprite.new(viewport)
398:         @celsprites[i].visible = false
399:         @celsprites[i].src_rect = Rect.new(0, 0, 0, 0)
400:         @celsprites[i].bitmap = nil
401:         @framesprites[i] = SpriteFrame.new(i, @celsprites[i], viewport)
402:         @framesprites[i].selected = false
403:         @framesprites[i].visible = false
404:         @dirty[i] = true
405:       end
406:       loadAnimation(animation)
407:     end
408: 
409:     def loadAnimation(anim)
410:       @animation = anim
411:       @animbitmap&.dispose
412:       if @animation.graphic == ""
413:         @animbitmap = nil
414:       else
415:         begin
416:           @animbitmap = AnimatedBitmap.new("Graphics/Animations/" + @animation.graphic,
417:                                            @animation.hue).deanimate
418:         rescue
419:           @animbitmap = nil
420:         end
421:       end
422:       @currentcel = -1
423:       self.currentframe = 0
424:       @selecting = false
425:       @pattern = 0
426:       self.invalidate
427:     end
428: 
429:     def animbitmap=(value)
430:       @animbitmap&.dispose
431:       @animbitmap = value
432:       (2...PBAnimation::MAX_SPRITES).each do |i|
433:         @celsprites[i].bitmap = @animbitmap if @celsprites[i]
434:       end
435:       self.invalidate
436:     end
437: 
438:     def dispose
439:       @user.dispose
440:       @target.dispose
441:       @animbitmap&.dispose
442:       @selectedbitmap&.dispose
443:       @celbitmap&.dispose
444:       self.bitmap&.dispose
445:       PBAnimation::MAX_SPRITES.times do |i|
446:         @celsprites[i]&.dispose
447:       end
448:       super
449:     end
450: 
451:     def play(oppmove = false)
452:       if !@playing
453:         @sprites["pokemon_0"] = Sprite.new(@viewport)
454:         @sprites["pokemon_0"].bitmap = @user
455:         @sprites["pokemon_0"].z = 21
456:         @sprites["pokemon_1"] = Sprite.new(@viewport)
457:         @sprites["pokemon_1"].bitmap = @target
458:         @sprites["pokemon_1"].z = 16
459:         pbSpriteSetAnimFrame(@sprites["pokemon_0"],
460:                              pbCreateCel(Battle::Scene::FOCUSUSER_X,
461:                                          Battle::Scene::FOCUSUSER_Y, -1, 2),
462:                              @sprites["pokemon_0"], @sprites["pokemon_1"])
463:         pbSpriteSetAnimFrame(@sprites["pokemon_1"],
464:                              pbCreateCel(Battle::Scene::FOCUSTARGET_X,
465:                                          Battle::Scene::FOCUSTARGET_Y, -2, 1),
466:                              @sprites["pokemon_0"], @sprites["pokemon_1"])
467:         usersprite = @sprites["pokemon_#{oppmove ? 1 : 0}"]
468:         targetsprite = @sprites["pokemon_#{oppmove ? 0 : 1}"]
469:         olduserx = usersprite ? usersprite.x : 0
470:         oldusery = usersprite ? usersprite.y : 0
471:         oldtargetx = targetsprite ? targetsprite.x : 0
472:         oldtargety = targetsprite ? targetsprite.y : 0
473:         @player = PBAnimationPlayerX.new(@animation,
474:                                          @battle.battlers[oppmove ? 1 : 0],
475:                                          @battle.battlers[oppmove ? 0 : 1],
476:                                          self, oppmove, true)
477:         @player.setLineTransform(
478:           Battle::Scene::FOCUSUSER_X, Battle::Scene::FOCUSUSER_Y,
479:           Battle::Scene::FOCUSTARGET_X, Battle::Scene::FOCUSTARGET_Y,
480:           olduserx, oldusery,
481:           oldtargetx, oldtargety
482:         )
483:         @player.start
484:         @playing = true
485:         @sprites["pokemon_0"].x += BORDERSIZE
486:         @sprites["pokemon_0"].y += BORDERSIZE
487:         @sprites["pokemon_1"].x += BORDERSIZE
488:         @sprites["pokemon_1"].y += BORDERSIZE
489:         oldstate = []
490:         PBAnimation::MAX_SPRITES.times do |i|
491:           oldstate.push([@celsprites[i].visible, @framesprites[i].visible, @lastframesprites[i].visible])
492:           @celsprites[i].visible = false
493:           @framesprites[i].visible = false
494:           @lastframesprites[i].visible = false
495:         end
496:         loop do
497:           Graphics.update
498:           self.update
499:           break if !@playing
500:         end
