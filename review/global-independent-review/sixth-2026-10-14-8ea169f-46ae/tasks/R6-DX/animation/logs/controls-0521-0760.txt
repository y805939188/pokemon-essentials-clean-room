631:   class TextSlider < UIControl
632:     attr_reader :minvalue
633:     attr_reader :maxvalue
634:     attr_reader :curvalue
635:     attr_accessor :label
636:     attr_accessor :options
637:     attr_accessor :maxoptionwidth
638: 
639:     def curvalue=(value)
640:       @curvalue = value
641:       @curvalue = self.minvalue if self.minvalue && @curvalue < self.minvalue
642:       @curvalue = self.maxvalue if self.maxvalue && @curvalue > self.maxvalue
643:       self.invalidate
644:     end
645: 
646:     def minvalue=(value)
647:       @minvalue = value
648:       @curvalue = self.minvalue if self.minvalue && @curvalue < self.minvalue
649:       @curvalue = self.maxvalue if self.maxvalue && @curvalue > self.maxvalue
650:       self.invalidate
651:     end
652: 
653:     def maxvalue=(value)
654:       @maxvalue = value
655:       @curvalue = self.minvalue if self.minvalue && @curvalue < self.minvalue
656:       @curvalue = self.maxvalue if self.maxvalue && @curvalue > self.maxvalue
657:       self.invalidate
658:     end
659: 
660:     def initialize(label, options, curval)
661:       super(label)
662:       @label = label
663:       @options = options
664:       @minvalue = 0
665:       @maxvalue = options.length - 1
666:       @curvalue = curval
667:       @leftarrow = Rect.new(0, 0, 0, 0)
668:       @rightarrow = Rect.new(0, 0, 0, 0)
669:       self.minvalue = @minvalue
670:       self.maxvalue = @maxvalue
671:       self.curvalue = @curvalue
672:     end
673: 
674:     def update
675:       mousepos = Mouse.getMousePos
676:       self.changed = false
677:       if self.minvalue < self.maxvalue && self.curvalue < self.minvalue
678:         self.curvalue = self.minvalue
679:       end
680:       return false if self.disabled
681:       return false if !Input.repeat?(Input::MOUSELEFT)
682:       return false if !mousepos
683:       left = toAbsoluteRect(@leftarrow)
684:       right = toAbsoluteRect(@rightarrow)
685:       oldvalue = self.curvalue
686:       repeattime = Input.time?(Input::MOUSELEFT)
687:       # Left arrow
688:       if left.contains?(mousepos[0], mousepos[1])
689:         if repeattime > 3.0
690:           self.curvalue -= 10
691:         elsif repeattime > 1.5
692:           self.curvalue -= 5
693:         else
694:           self.curvalue -= 1
695:         end
696:         self.changed = (self.curvalue != oldvalue)
697:         self.invalidate
698:       end
699:       # Right arrow
700:       if right.contains?(mousepos[0], mousepos[1])
701:         if repeattime > 3.0
702:           self.curvalue += 10
703:         elsif repeattime > 1.5
704:           self.curvalue += 5
705:         else
706:           self.curvalue += 1
707:         end
708:         self.changed = (self.curvalue != oldvalue)
709:         self.invalidate
710:       end
711:     end
712: 
713:     def refresh
714:       bitmap = self.bitmap
715:       if @maxoptionwidth.nil?
716:         @options.length.times do |i|
717:           w = self.bitmap.text_size(" " + @options[i] + " ").width
718:           @maxoptionwidth = w if !@maxoptionwidth || @maxoptionwidth < w
719:         end
720:       end
721:       x = self.x
722:       y = self.y
723:       width = self.width
724:       height = self.height
725:       color = Color.new(120, 120, 120)
726:       bitmap.fill_rect(x, y, width, height, Color.new(0, 0, 0, 0))
727:       size = bitmap.text_size(self.label).width
728:       leftarrows = bitmap.text_size(" <<")
729:       rightarrows = bitmap.text_size(">> ")
730:       bitmap.font.color = color
731:       shadowtext(bitmap, x, y, size, height, self.label)
732:       x += size
733:       shadowtext(bitmap, x, y, leftarrows.width, height, " <<",
734:                  self.disabled || self.curvalue == self.minvalue)
735:       @leftarrow = Rect.new(x, y, leftarrows.width, height)
736:       x += leftarrows.width
737:       if !self.disabled
738:         bitmap.font.color = color
739:         shadowtext(bitmap, x, y, @maxoptionwidth, height, " #{@options[self.curvalue]} ", false, 1)
740:       end
741:       x += @maxoptionwidth
742:       shadowtext(bitmap, x, y, rightarrows.width, height, ">> ",
743:                  self.disabled || self.curvalue == self.maxvalue)
744:       @rightarrow = Rect.new(x, y, rightarrows.width, height)
745:     end
746:   end
