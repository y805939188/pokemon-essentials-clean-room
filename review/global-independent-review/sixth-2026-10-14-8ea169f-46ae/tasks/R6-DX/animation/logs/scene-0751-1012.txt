751:         height *= sprite.zoom_y
752:       else
753:         spritex = sprite.x - sprite.ox
754:         spritey = sprite.y - sprite.oy
755:         spritex += xwidth / 2
756:         spritey += xheight / 2
757:       end
758:       if !(x >= spritex && x <= spritex + width && y >= spritey && y <= spritey + height)
759:         return false
760:       end
761:       if usealpha
762:         # TODO: This should account for sprite.angle as well.
763:         bitmapX = sprite.src_rect.x
764:         bitmapY = sprite.src_rect.y
765:         bitmapX += sprite.ox
766:         bitmapY += sprite.oy
767:         bitmapX += (x - sprite.x) / sprite.zoom_x if sprite.zoom_x > 0
768:         bitmapY += (y - sprite.y) / sprite.zoom_y if sprite.zoom_y > 0
769:         bitmapX = bitmapX.round
770:         bitmapY = bitmapY.round
771:         if sprite.mirror
772:           xmirror = bitmapX - sprite.src_rect.x
773:           bitmapX = sprite.src_rect.x + 192 - xmirror
774:         end
775:         color = sprite.bitmap.get_pixel(bitmapX, bitmapY)
776:         return false if color.alpha == 0
777:       end
778:       return true
779:     end

781:     def updateInput
782:       cel = currentCel
783:       mousepos = Mouse.getMousePos
784:       if Input.trigger?(Input::MOUSELEFT) && mousepos &&
785:          pbSpriteHitTest(self, mousepos[0], mousepos[1], false, true)
786:         selectedcel = -1
787:         usealpha = Input.press?(Input::ALT)
788:         PBAnimation::MAX_SPRITES.times do |j|
789:           if pbSpriteHitTest(@celsprites[j], mousepos[0], mousepos[1], usealpha, false)
790:             selectedcel = j
791:           end
792:         end
793:         if selectedcel < 0
794:           if @animbitmap && addSprite(mousepos[0] - BORDERSIZE, mousepos[1] - BORDERSIZE)
795:             @selecting = true if !self.locked?(@currentcel)
796:             @selectOffsetX = 0
797:             @selectOffsetY = 0
798:             cel = currentCel
799:             invalidate
800:           end
801:         else
802:           @currentcel = selectedcel
803:           @selecting = true if !self.locked?(@currentcel)
804:           cel = currentCel
805:           @selectOffsetX = cel[AnimFrame::X] - mousepos[0] + BORDERSIZE
806:           @selectOffsetY = cel[AnimFrame::Y] - mousepos[1] + BORDERSIZE
807:           invalidate
808:         end
809:       end
810:       currentFrame = getCurrentFrame
811:       if currentFrame && !@selecting &&
812:          (Input.triggerex?(:TAB) || Input.repeatex?(:TAB))
813:         currentFrame.length.times do
814:           @currentcel += 1
815:           @currentcel = 0 if @currentcel >= currentFrame.length
816:           break if currentFrame[@currentcel]
817:         end
818:         invalidate
819:         return
820:       end
821:       if cel && @selecting && mousepos
822:         cel[AnimFrame::X] = mousepos[0] - BORDERSIZE + @selectOffsetX
823:         cel[AnimFrame::Y] = mousepos[1] - BORDERSIZE + @selectOffsetY
824:         @dirty[@currentcel] = true
825:       end
826:       if !Input.press?(Input::MOUSELEFT) && @selecting
827:         @selecting = false
828:       end
829:       if cel
830:         if (Input.triggerex?(:DELETE) || Input.repeatex?(:DELETE)) && self.deletable?(@currentcel)
831:           @animation[@currentframe][@currentcel] = nil
832:           @dirty[@currentcel] = true
833:           return
834:         end
835:         if Input.triggerex?(:P) || Input.repeatex?(:P)   # Properties
836:           BattleAnimationEditor.pbCellProperties(self)
837:           @dirty[@currentcel] = true
838:           return
839:         end
840:         if Input.triggerex?(:L) || Input.repeatex?(:L)   # Lock
841:           cel[AnimFrame::LOCKED] = (cel[AnimFrame::LOCKED] == 0) ? 1 : 0
842:           @dirty[@currentcel] = true
843:         end
844:         if Input.triggerex?(:R) || Input.repeatex?(:R)   # Rotate right
845:           cel[AnimFrame::ANGLE] += 10
846:           cel[AnimFrame::ANGLE] %= 360
847:           @dirty[@currentcel] = true
848:         end
849:         if Input.triggerex?(:E) || Input.repeatex?(:E)   # Rotate left
850:           cel[AnimFrame::ANGLE] -= 10
851:           cel[AnimFrame::ANGLE] %= 360
852:           @dirty[@currentcel] = true
853:         end
854:         if Input.triggerex?(:KP_PLUS) || Input.repeatex?(:KP_PLUS)   # Zoom in
855:           cel[AnimFrame::ZOOMX] += 10
856:           cel[AnimFrame::ZOOMX] = 1000 if cel[AnimFrame::ZOOMX] > 1000
857:           cel[AnimFrame::ZOOMY] += 10
858:           cel[AnimFrame::ZOOMY] = 1000 if cel[AnimFrame::ZOOMY] > 1000
859:           @dirty[@currentcel] = true
860:         end
861:         if Input.triggerex?(:KP_MINUS) || Input.repeatex?(:KP_MINUS)   # Zoom out
862:           cel[AnimFrame::ZOOMX] -= 10
863:           cel[AnimFrame::ZOOMX] = 10 if cel[AnimFrame::ZOOMX] < 10
864:           cel[AnimFrame::ZOOMY] -= 10
865:           cel[AnimFrame::ZOOMY] = 10 if cel[AnimFrame::ZOOMY] < 10
866:           @dirty[@currentcel] = true
867:         end
868:         if !self.locked?(@currentcel)
869:           if Input.trigger?(Input::UP) || Input.repeat?(Input::UP)
870:             increment = (Input.press?(Input::ALT)) ? 1 : 8
871:             cel[AnimFrame::Y] -= increment
872:             @dirty[@currentcel] = true
873:           end
874:           if Input.trigger?(Input::DOWN) || Input.repeat?(Input::DOWN)
875:             increment = (Input.press?(Input::ALT)) ? 1 : 8
876:             cel[AnimFrame::Y] += increment
877:             @dirty[@currentcel] = true
878:           end
879:           if Input.trigger?(Input::LEFT) || Input.repeat?(Input::LEFT)
880:             increment = (Input.press?(Input::ALT)) ? 1 : 8
881:             cel[AnimFrame::X] -= increment
882:             @dirty[@currentcel] = true
883:           end
884:           if Input.trigger?(Input::RIGHT) || Input.repeat?(Input::RIGHT)
885:             increment = (Input.press?(Input::ALT)) ? 1 : 8
886:             cel[AnimFrame::X] += increment
887:             @dirty[@currentcel] = true
888:           end
889:         end
890:       end
891:     end

949:         @bitmapname = value

955:       if @hue != value
956:         @hue = value
957:         refresh

961:     def refresh
962:       self.contents.clear
963:       bmap = AnimatedBitmap.new("Graphics/Animations/" + @bitmapname, @hue).deanimate
964:       return if !bmap
965:       ww = bmap.width
966:       wh = bmap.height
967:       sx = self.contents.width / ww.to_f
968:       sy = self.contents.height / wh.to_f
969:       if sx > sy
970:         ww = sy * ww
971:         wh = self.contents.height
972:       else
973:         wh = sx * wh
974:         ww = self.contents.width
975:       end
976:       dest = Rect.new((self.contents.width - ww) / 2,
977:                       (self.contents.height - wh) / 2,
978:                       ww, wh)
979:       src = Rect.new(0, 0, bmap.width, bmap.height)
980:       self.contents.stretch_blt(dest, bmap, src)
981:       bmap.dispose
982:     end
