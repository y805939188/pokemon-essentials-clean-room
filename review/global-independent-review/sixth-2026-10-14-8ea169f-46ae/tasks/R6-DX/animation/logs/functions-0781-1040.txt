784:           diffY = cel2[AnimFrame::Y] - cel1[AnimFrame::Y]
785:           diffZoomX = cel2[AnimFrame::ZOOMX] - cel1[AnimFrame::ZOOMX]
786:           diffZoomY = cel2[AnimFrame::ZOOMY] - cel1[AnimFrame::ZOOMY]
787:           diffAngle = cel2[AnimFrame::ANGLE] - cel1[AnimFrame::ANGLE]
788:           diffOpacity = cel2[AnimFrame::OPACITY] - cel1[AnimFrame::OPACITY]
789:           diffBlend = cel2[AnimFrame::BLENDTYPE] - cel1[AnimFrame::BLENDTYPE]
790:           startPattern = cel1[AnimFrame::PATTERN]
791:           startX = cel1[AnimFrame::X]
792:           startY = cel1[AnimFrame::Y]
793:           startZoomX = cel1[AnimFrame::ZOOMX]
794:           startZoomY = cel1[AnimFrame::ZOOMY]
795:           startAngle = cel1[AnimFrame::ANGLE]
796:           startOpacity = cel1[AnimFrame::OPACITY]
797:           startBlend = cel1[AnimFrame::BLENDTYPE]
798:           (0..frames).each do |k|
799:             cel = canvas.animation[startframe + k][j]

803:               canvas.animation[startframe + k][j] = cel
804:             end
805:             if sliderwin2.value(set0) || !curcel
806:               cel[AnimFrame::PATTERN] = startPattern + (diffPattern * k / frames)
807:             end
808:             if sliderwin2.value(set1) || !curcel
809:               cel[AnimFrame::X] = startX + (diffX * k / frames)
810:               cel[AnimFrame::Y] = startY + (diffY * k / frames)
811:               cel[AnimFrame::ZOOMX] = startZoomX + (diffZoomX * k / frames)
812:               cel[AnimFrame::ZOOMY] = startZoomY + (diffZoomY * k / frames)
813:               cel[AnimFrame::ANGLE] = startAngle + (diffAngle * k / frames)
814:             end
815:             if sliderwin2.value(set2) || !curcel
816:               cel[AnimFrame::OPACITY] = startOpacity + (diffOpacity * k / frames)
817:               cel[AnimFrame::BLENDTYPE] = startBlend + (diffBlend * k / frames)
818:             end
819:           end
820:         end
821:         canvas.invalidate
822:         break
823:       end
824:       if sliderwin2.changed?(cancelbutton) || Input.trigger?(Input::BACK)
825:         break
826:       end
