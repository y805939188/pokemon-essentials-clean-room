29:   def pbSelectAnim(canvas, animwin)
30:     animfiles = []
31:     pbRgssChdir(File.join("Graphics", "Animations")) { animfiles.concat(Dir.glob("*.png")) }
32:     cmdwin = pbListWindow(animfiles, 320)
33:     cmdwin.opacity = 200
34:     cmdwin.height = 512
35:     bmpwin = BitmapDisplayWindow.new(320, 0, 320, 448)
36:     ctlwin = ControlWindow.new(320, 448, 320, 64)
37:     cmdwin.viewport = canvas.viewport
38:     bmpwin.viewport = canvas.viewport
39:     ctlwin.viewport = canvas.viewport
40:     ctlwin.addSlider(_INTL("Hue:"), 0, 359, 0)
41:     loop do
42:       bmpwin.bitmapname = cmdwin.commands[cmdwin.index]
43:       Graphics.update
44:       Input.update
45:       cmdwin.update
46:       bmpwin.update
47:       ctlwin.update
48:       bmpwin.hue = ctlwin.value(0) if ctlwin.changed?(0)
49:       if Input.trigger?(Input::USE) && animfiles.length > 0
50:         filename = cmdwin.commands[cmdwin.index]
51:         bitmap = AnimatedBitmap.new("Graphics/Animations/" + filename, ctlwin.value(0)).deanimate
52:         canvas.animation.graphic = File.basename(filename, ".*")
53:         canvas.animation.hue = ctlwin.value(0)
54:         canvas.animbitmap = bitmap
55:         animwin.animbitmap = bitmap
56:         break
57:       end
58:       if Input.trigger?(Input::BACK)
59:         break
60:       end
61:     end
62:     bmpwin.dispose
63:     cmdwin.dispose
64:     ctlwin.dispose
65:     return
66:   end

117:   def pbAnimList(animations, canvas, animwin)
118:     commands = []
119:     animations.length.times do |i|
120:       animations[i] = PBAnimation.new if !animations[i]
121:       commands[commands.length] = _INTL("{1} {2}", i, animations[i].name)
122:     end
123:     cmdwin = pbListWindow(commands, 320)
124:     cmdwin.height = 416
125:     cmdwin.opacity = 224
126:     cmdwin.index = animations.selected
127:     cmdwin.viewport = canvas.viewport
128:     helpwindow = Window_UnformattedTextPokemon.newWithSize(
129:       _INTL("Enter: Load/rename an animation\nEsc: Cancel"),
130:       320, 0, 320, 128, canvas.viewport
131:     )
132:     maxsizewindow = ControlWindow.new(0, 416, 320, 32 * 3)
133:     maxsizewindow.addSlider(_INTL("Total Animations:"), 1, 2000, animations.length)
134:     maxsizewindow.addButton(_INTL("Resize Animation List"))
135:     maxsizewindow.opacity = 224
136:     maxsizewindow.viewport = canvas.viewport
137:     loop do
138:       Graphics.update
139:       Input.update
140:       cmdwin.update
141:       maxsizewindow.update
142:       helpwindow.update
143:       if maxsizewindow.changed?(1)
144:         newsize = maxsizewindow.value(0)
145:         animations.resize(newsize)
146:         commands.clear
147:         animations.length.times do |i|
148:           commands[commands.length] = _INTL("{1} {2}", i, animations[i].name)
149:         end
150:         cmdwin.commands = commands
151:         cmdwin.index = animations.selected
152:         next
153:       end
154:       if Input.trigger?(Input::USE) && animations.length > 0
155:         cmd2 = pbShowCommands(helpwindow,
156:                               [_INTL("Load Animation"),
157:                                _INTL("Rename"),
158:                                _INTL("Delete")], -1)
159:         case cmd2
160:         when 0   # Load Animation
161:           canvas.loadAnimation(animations[cmdwin.index])
162:           animwin.animbitmap = canvas.animbitmap
163:           animations.selected = cmdwin.index
164:           break
165:         when 1   # Rename
166:           pbAnimName(animations[cmdwin.index], cmdwin)
167:           cmdwin.refresh
168:         when 2   # Delete
169:           if pbConfirmMessage(_INTL("Are you sure you want to delete this animation?"))
170:             animations[cmdwin.index] = PBAnimation.new
171:             cmdwin.commands[cmdwin.index] = _INTL("{1} {2}", cmdwin.index, animations[cmdwin.index].name)
172:             cmdwin.refresh
173:           end
174:         end
175:       end
176:       if Input.trigger?(Input::BACK)
177:         break
178:       end
179:     end
180:     helpwindow.dispose
181:     maxsizewindow.dispose
182:     cmdwin.dispose
183:   end

213:   def pbSetTone(cel, previewsprite)
214:     sliderwin2 = ControlWindow.new(0, 0, 320, 320)
215:     sliderwin2.z = 99999
216:     sliderwin2.addSlider(_INTL("Red Offset:"), -255, 255, cel[AnimFrame::TONERED])
217:     sliderwin2.addSlider(_INTL("Green Offset:"), -255, 255, cel[AnimFrame::TONEGREEN])
218:     sliderwin2.addSlider(_INTL("Blue Offset:"), -255, 255, cel[AnimFrame::TONEBLUE])
219:     sliderwin2.addSlider(_INTL("Gray Tone:"), 0, 255, cel[AnimFrame::TONEGRAY])
220:     okbutton = sliderwin2.addButton(_INTL("OK"))
221:     cancelbutton = sliderwin2.addButton(_INTL("Cancel"))
222:     loop do
223:       previewsprite.tone.set(sliderwin2.value(0), sliderwin2.value(1),
224:                              sliderwin2.value(2), sliderwin2.value(3))
225:       Graphics.update
226:       Input.update
227:       sliderwin2.update
228:       if sliderwin2.changed?(okbutton)
229:         cel[AnimFrame::TONERED] = sliderwin2.value(0)
230:         cel[AnimFrame::TONEGREEN] = sliderwin2.value(1)
231:         cel[AnimFrame::TONEBLUE] = sliderwin2.value(2)
232:         cel[AnimFrame::TONEGRAY] = sliderwin2.value(3)
233:         break
234:       end
235:       if sliderwin2.changed?(cancelbutton) || Input.trigger?(Input::BACK)
236:         break
237:       end
238:     end
239:     sliderwin2.dispose
240:     return
241:   end
