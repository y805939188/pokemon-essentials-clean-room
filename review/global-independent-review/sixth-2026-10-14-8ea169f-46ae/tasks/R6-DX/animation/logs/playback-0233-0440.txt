233: class PBAnimTiming
234:   attr_accessor :frame
235:   attr_writer   :timingType   # 0=play SE, 1=set bg, 2=bg mod
236:   attr_accessor :name         # Name of SE file or BG file
237:   attr_accessor :volume
238:   attr_accessor :pitch
239:   attr_accessor :bgX          # x coordinate of bg (or to move bg to)
240:   attr_accessor :bgY          # y coordinate of bg (or to move bg to)
241:   attr_accessor :opacity      # Opacity of bg (or to change bg to)
242:   attr_accessor :colorRed     # Color of bg (or to change bg to)
243:   attr_accessor :colorGreen   # Color of bg (or to change bg to)
244:   attr_accessor :colorBlue    # Color of bg (or to change bg to)
245:   attr_accessor :colorAlpha   # Color of bg (or to change bg to)
246:   attr_writer   :duration     # How long to spend changing to the new bg coords/color
247:   attr_accessor :flashScope
248:   attr_accessor :flashColor
249:   attr_accessor :flashDuration
250: 
251:   def initialize(type = 0)
252:     @frame         = 0
253:     @timingType    = type
254:     @name          = ""
255:     @volume        = 80
256:     @pitch         = 100
257:     @bgX           = nil
258:     @bgY           = nil
259:     @opacity       = nil
260:     @colorRed      = nil
261:     @colorGreen    = nil
262:     @colorBlue     = nil
263:     @colorAlpha    = nil
264:     @duration      = 5
265:     @flashScope    = 0
266:     @flashColor    = Color.white
267:     @flashDuration = 5
268:   end

270:   def timingType
271:     return @timingType || 0
272:   end
273:

275:     return @duration || 5
276:   end
277: 
278:   def to_s
279:     case self.timingType
280:     when 0
281:       return "[#{@frame + 1}] Play SE: #{name} (volume #{@volume}, pitch #{@pitch})"
282:     when 1
283:       text = sprintf("[%d] Set BG: \"%s\"", @frame + 1, name)
284:       text += sprintf(" (color=%s,%s,%s,%s)",
285:                       @colorRed || "-",
286:                       @colorGreen || "-",
287:                       @colorBlue || "-",
288:                       @colorAlpha || "-")
289:       text += sprintf(" (opacity=%d)", @opacity)
290:       text += sprintf(" (coords=%s,%s)", @bgX || "-", @bgY || "-")
291:       return text
292:     when 2
293:       text = sprintf("[%d] Change BG: @%d", @frame + 1, duration)
294:       if @colorRed || @colorGreen || @colorBlue || @colorAlpha
295:         text += sprintf(" (color=%s,%s,%s,%s)",
296:                         @colorRed || "-",
297:                         @colorGreen || "-",
298:                         @colorBlue || "-",
299:                         @colorAlpha || "-")
300:       end
301:       text += sprintf(" (opacity=%d)", @opacity) if @opacity
302:       text += sprintf(" (coords=%s,%s)", @bgX || "-", @bgY || "-") if @bgX || @bgY
303:       return text
304:     when 3
305:       text = sprintf("[%d] Set FG: \"%s\"", @frame + 1, name)
306:       text += sprintf(" (color=%s,%s,%s,%s)",
307:                       @colorRed || "-",
308:                       @colorGreen || "-",
309:                       @colorBlue || "-",
310:                       @colorAlpha || "-")
311:       text += sprintf(" (opacity=%d)", @opacity)
312:       text += sprintf(" (coords=%s,%s)", @bgX || "-", @bgY || "-")
313:       return text
314:     when 4
315:       text = sprintf("[%d] Change FG: @%d", @frame + 1, duration)
316:       if @colorRed || @colorGreen || @colorBlue || @colorAlpha
317:         text += sprintf(" (color=%s,%s,%s,%s)",
318:                         @colorRed || "-",
319:                         @colorGreen || "-",
320:                         @colorBlue || "-",
321:                         @colorAlpha || "-")
322:       end
323:       text += sprintf(" (opacity=%d)", @opacity) if @opacity
324:       text += sprintf(" (coords=%s,%s)", @bgX || "-", @bgY || "-") if @bgX || @bgY
325:       return text
326:     end
327:     return ""
328:   end

334: class PBAnimations < Array
335:   include Enumerable
336:   attr_reader   :array
337:   attr_accessor :selected
338: 
339:   def initialize(size = 1)
340:     @array = []
341:     @selected = 0
342:     size = 1 if size < 1   # Always create at least one animation
343:     size.times do
344:       @array.push(PBAnimation.new)
345:     end
346:   end
347: 
348:   def length
349:     return @array.length
350:   end
351: 
352:   def each
353:     @array.each { |i| yield i }
354:   end
355: 
356:   def [](i)
357:     return @array[i]
358:   end
359: 
360:   def []=(i, value)
361:     @array[i] = value
362:   end
363: 
364:   def get_from_name(name)
365:     @array.each { |i| return i if i&.name == name }
366:     return nil
367:   end
368: 
369:   def compact
370:     @array.compact!
371:   end
372: 
373:   def insert(index, val)
374:     @array.insert(index, val)
375:   end
376: 
377:   def delete_at(index)
378:     @array.delete_at(index)
379:   end
380: 
381:   def resize(len)
382:     idxStart = @array.length
383:     idxEnd   = len
384:     if idxStart > idxEnd
385:       (idxStart - idxEnd).times { @array.pop }
386:     else
387:       (idxEnd - idxStart).times { @array.push(PBAnimation.new) }
388:     end
389:     self.selected = len if self.selected >= len
390:   end
391: end
