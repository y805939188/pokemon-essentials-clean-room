284: def pbCommands3(cmdwindow, commands, cmdIfCancel, defaultindex = -1, noresize = false)
285:   cmdwindow.commands = commands
286:   cmdwindow.index    = defaultindex if defaultindex >= 0
287:   cmdwindow.x        = 0
288:   cmdwindow.y        = 0
289:   if noresize
290:     cmdwindow.height = Graphics.height
291:   else
292:     cmdwindow.width  = Graphics.width / 2
293:   end
294:   cmdwindow.height   = Graphics.height if cmdwindow.height > Graphics.height
295:   cmdwindow.z        = 99999
296:   cmdwindow.visible  = true
297:   cmdwindow.active   = true
298:   command = 0
299:   loop do
300:     Graphics.update
301:     Input.update
302:     cmdwindow.update
303:     if Input.trigger?(Input::SPECIAL)
304:       command = [5, cmdwindow.index]
305:       break
306:     elsif Input.press?(Input::ACTION)
307:       if Input.repeat?(Input::UP)
308:         command = [1, cmdwindow.index]
309:         break
310:       elsif Input.repeat?(Input::DOWN)
311:         command = [2, cmdwindow.index]
312:         break
313:       elsif Input.trigger?(Input::LEFT)
314:         command = [3, cmdwindow.index]
315:         break
316:       elsif Input.trigger?(Input::RIGHT)
317:         command = [4, cmdwindow.index]
318:         break
319:       end
320:     elsif Input.trigger?(Input::BACK)
321:       if cmdIfCancel > 0
322:         command = [0, cmdIfCancel - 1]
323:         break
324:       elsif cmdIfCancel < 0
325:         command = [0, cmdIfCancel]
326:         break
327:       end
328:     elsif Input.trigger?(Input::USE)
329:       command = [0, cmdwindow.index]
330:       break
331:     end
332:   end
333:   ret = command
334:   cmdwindow.active = false
335:   return ret
336: end
