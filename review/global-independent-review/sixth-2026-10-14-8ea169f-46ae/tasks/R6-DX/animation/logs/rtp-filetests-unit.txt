184:   def self.exists?(filename, extensions = [])
185:     return false if nil_or_empty?(filename)
186:     eachPathFor(filename) do |path|
187:       return true if FileTest.exist?(path)
188:       extensions.each do |ext|
189:         return true if FileTest.exist?(path + ext)
190:       end
191:     end
192:     return false
193:   end
194: 
195:   def self.getImagePath(filename)
196:     return self.getPath(filename, ["", ".png", ".gif"])   # ".jpg", ".jpeg", ".bmp"
197:   end
198: 
199:   def self.getAudioPath(filename)
200:     return self.getPath(filename, ["", ".wav", ".ogg", ".mp3", ".midi", ".mid", ".wma"])
201:   end
202: 
203:   def self.getPath(filename, extensions = [])
204:     return filename if nil_or_empty?(filename)
205:     eachPathFor(filename) do |path|
206:       return path if FileTest.exist?(path)
207:       extensions.each do |ext|
208:         file = path + ext
209:         return file if FileTest.exist?(file)
210:       end
211:     end
212:     return filename
213:   end
214: 
215:   # Gets the absolute RGSS paths for the given file name
216:   def self.eachPathFor(filename)
217:     return if !filename
218:     if filename[/^[A-Za-z]\:[\/\\]/] || filename[/^[\/\\]/]
219:       # filename is already absolute
220:       yield filename
221:     else
222:       # relative path
223:       RTP.eachPath do |path|
224:         if path == "./"
225:           yield filename
226:         else
227:           yield path + filename
228:         end
229:       end
230:     end
231:   end
232: 
233:   # Gets all RGSS search paths.
234:   # This function basically does nothing now, because
235:   # the passage of time and introduction of MKXP make
236:   # it useless, but leaving it for compatibility
237:   # reasons
238:   def self.eachPath
239:     # XXX: Use "." instead of Dir.pwd because of problems retrieving files if
240:     # the current directory contains an accent mark
241:     yield ".".gsub(/[\/\\]/, "/").gsub(/[\/\\]$/, "") + "/"
242:   end
