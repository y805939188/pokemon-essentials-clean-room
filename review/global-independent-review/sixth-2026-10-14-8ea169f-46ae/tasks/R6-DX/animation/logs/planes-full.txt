9: #===============================================================================
10: # A plane class that displays a single color.
11: #===============================================================================
12: class ColoredPlane < Plane
13:   def initialize(color, viewport = nil)
14:     super(viewport)
15:     self.bitmap = Bitmap.new(32, 32)
16:     set_plane_color(color)
17:   end
18: 
19:   def dispose
20:     self.bitmap&.dispose
21:     super
22:   end
23: 
24:   def set_plane_color(value)
25:     self.bitmap.fill_rect(0, 0, self.bitmap.width, self.bitmap.height, value)
26:     refresh
27:   end
28: end

30: #===============================================================================
31: # A plane class that supports animated images.
32: #===============================================================================
33: class AnimatedPlane < Plane
34:   def initialize(viewport)
35:     super(viewport)
36:     @bitmap = nil
37:   end
38: 
39:   def dispose
40:     clear_bitmap
41:     super
42:   end
43: 
44:   def setBitmap(file, hue = 0)
45:     clear_bitmap
46:     return if file.nil?
47:     @bitmap = AnimatedBitmap.new(file, hue)
48:     self.bitmap = @bitmap.bitmap if @bitmap
49:   end
50: 
51:   def set_panorama(file, hue = 0)
52:     if file.is_a?(String) && file.length > 0
53:       setBitmap("Graphics/Panoramas/" + file, hue)
54:     else
55:       clear_bitmap
56:     end
57:   end
58: 
59:   def set_fog(file, hue = 0)
60:     if file.is_a?(String) && file.length > 0
61:       setBitmap("Graphics/Fogs/" + file, hue)
62:     else
63:       clear_bitmap
64:     end
65:   end
66: 
67:   #-----------------------------------------------------------------------------
68: 
69:   private
70: 
71:   def clear_bitmap
72:     @bitmap&.dispose
73:     @bitmap = nil
74:     self.bitmap = nil if !self.disposed?
75:   end
76: end
