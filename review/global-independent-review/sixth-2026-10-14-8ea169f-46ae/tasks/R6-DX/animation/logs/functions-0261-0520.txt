376:   def pbTimingList(canvas)
377:     commands = []
378:     cmdNewSound = -1
379:     cmdNewBG = -1
380:     cmdEditBG = -1
381:     cmdNewFO = -1
382:     cmdEditFO = -1
383:     canvas.animation.timing.each { |i| commands.push(i.to_s) }
384:     commands[cmdNewSound = commands.length] = _INTL("Add: Play Sound...")
385:     commands[cmdNewBG = commands.length] = _INTL("Add: Set Background Graphic...")
386:     commands[cmdEditBG = commands.length] = _INTL("Add: Edit Background Color/Location...")
387:     commands[cmdNewFO = commands.length] = _INTL("Add: Set Foreground Graphic...")
388:     commands[cmdEditFO = commands.length] = _INTL("Add: Edit Foreground Color/Location...")
389:     cmdwin = pbListWindow(commands, 480)
390:     cmdwin.x = 0
391:     cmdwin.y = 0
392:     cmdwin.width = 640
393:     cmdwin.height = 384
394:     cmdwin.opacity = 200
395:     cmdwin.viewport = canvas.viewport
396:     framewindow = ControlWindow.new(0, 384, 640, 32 * 4)
397:     framewindow.addSlider(_INTL("Frame:"), 1, canvas.animation.length, canvas.currentframe + 1)
398:     framewindow.addButton(_INTL("Set Frame"))
399:     framewindow.addButton(_INTL("Delete Timing"))
400:     framewindow.opacity = 200
401:     framewindow.viewport = canvas.viewport
402:     loop do
403:       Graphics.update
404:       Input.update
405:       cmdwin.update
406:       framewindow.update
407:       if cmdwin.index != cmdNewSound &&
408:          cmdwin.index != cmdNewBG &&
409:          cmdwin.index != cmdEditBG &&
410:          cmdwin.index != cmdNewFO &&
411:          cmdwin.index != cmdEditFO
412:         if framewindow.changed?(1)   # Set Frame
413:           canvas.animation.timing[cmdwin.index].frame = framewindow.value(0) - 1
414:           cmdwin.commands[cmdwin.index] = canvas.animation.timing[cmdwin.index].to_s
415:           cmdwin.refresh
416:           next
417:         end
418:         if framewindow.changed?(2)   # Delete Timing
419:           canvas.animation.timing.delete_at(cmdwin.index)
420:           cmdwin.commands.delete_at(cmdwin.index)
421:           cmdNewSound -= 1 if cmdNewSound >= 0
422:           cmdNewBG -= 1 if cmdNewBG >= 0
423:           cmdEditBG -= 1 if cmdEditBG >= 0
424:           cmdNewFO -= 1 if cmdNewFO >= 0
425:           cmdEditFO -= 1 if cmdEditFO >= 0
426:           cmdwin.refresh
427:           next
428:         end
429:       end
430:       if Input.trigger?(Input::USE)
431:         redrawcmds = false
432:         if cmdwin.index == cmdNewSound   # Add new sound
433:           newaudio = PBAnimTiming.new(0)
434:           if pbSelectSE(canvas, newaudio)

436:             canvas.animation.timing.push(newaudio)
437:             redrawcmds = true
438:           end
439:         elsif cmdwin.index == cmdNewBG   # Add new background graphic set
440:           newtiming = PBAnimTiming.new(1)
441:           if pbSelectBG(canvas, newtiming)
442:             newtiming.frame = framewindow.value(0) - 1
443:             canvas.animation.timing.push(newtiming)
444:             redrawcmds = true
445:           end
446:         elsif cmdwin.index == cmdEditBG   # Add new background edit
447:           newtiming = PBAnimTiming.new(2)
448:           if pbEditBG(canvas, newtiming)
449:             newtiming.frame = framewindow.value(0) - 1
450:             canvas.animation.timing.push(newtiming)
451:             redrawcmds = true
452:           end
453:         elsif cmdwin.index == cmdNewFO   # Add new foreground graphic set
454:           newtiming = PBAnimTiming.new(3)

456:             newtiming.frame = framewindow.value(0) - 1
457:             canvas.animation.timing.push(newtiming)
458:             redrawcmds = true
459:           end
460:         elsif cmdwin.index == cmdEditFO   # Add new foreground edit
461:           newtiming = PBAnimTiming.new(4)
462:           if pbEditBG(canvas, newtiming)
463:             newtiming.frame = framewindow.value(0) - 1
464:             canvas.animation.timing.push(newtiming)
465:             redrawcmds = true
466:           end
467:         else
468:           # Edit a timing here
469:           case canvas.animation.timing[cmdwin.index].timingType
470:           when 0
471:             pbSelectSE(canvas, canvas.animation.timing[cmdwin.index])
472:           when 1, 3
473:             pbSelectBG(canvas, canvas.animation.timing[cmdwin.index])
474:           when 2, 4
475:             pbEditBG(canvas, canvas.animation.timing[cmdwin.index])
476:           end
477:           cmdwin.commands[cmdwin.index] = canvas.animation.timing[cmdwin.index].to_s
478:           cmdwin.refresh
479:         end
480:         if redrawcmds
481:           cmdwin.commands[cmdNewSound] = nil if cmdNewSound >= 0
482:           cmdwin.commands[cmdNewBG] = nil if cmdNewBG >= 0
483:           cmdwin.commands[cmdEditBG] = nil if cmdEditBG >= 0
484:           cmdwin.commands[cmdNewFO] = nil if cmdNewFO >= 0
485:           cmdwin.commands[cmdEditFO] = nil if cmdEditFO >= 0

489:           cmdwin.commands[cmdNewBG = cmdwin.commands.length] = _INTL("Add: Set Background Graphic...")
490:           cmdwin.commands[cmdEditBG = cmdwin.commands.length] = _INTL("Add: Edit Background Color/Location...")
491:           cmdwin.commands[cmdNewFO = cmdwin.commands.length] = _INTL("Add: Set Foreground Graphic...")
492:           cmdwin.commands[cmdEditFO = cmdwin.commands.length] = _INTL("Add: Edit Foreground Color/Location...")
493:           cmdwin.refresh
494:         end
495:       elsif Input.trigger?(Input::BACK)
496:         break
497:       end
498:     end

500:     framewindow.dispose
501:     return
