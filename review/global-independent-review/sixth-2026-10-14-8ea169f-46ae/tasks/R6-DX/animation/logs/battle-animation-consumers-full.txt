390:   def pbSaveShadows
391:     # Remember which shadows were visible
392:     shadows = Array.new(@battle.battlers.length) do |i|
393:       shadow = @sprites["shadow_#{i}"]
394:       ret = (shadow) ? shadow.visible : false
395:       shadow.visible = false if shadow
396:       next ret
397:     end
398:     # Yield to other code, i.e. playing an animation
399:     yield
400:     # Restore shadow visibility
401:     @battle.battlers.length.times do |i|
402:       shadow = @sprites["shadow_#{i}"]
403:       shadow.visible = shadows[i] if shadow
404:     end
405:   end

410:   # Returns the animation ID to use for a given move/user. Returns nil if that

412:   def pbFindMoveAnimDetails(move2anim, moveID, idxUser, hitNum = 0)
413:     real_move_id = GameData::Move.try_get(moveID)&.id || moveID
414:     noFlip = false
415:     if (idxUser & 1) == 0   # On player's side
416:       anim = move2anim[0][real_move_id]
417:     else                # On opposing side
418:       anim = move2anim[1][real_move_id]
419:       noFlip = true if anim
420:       anim = move2anim[0][real_move_id] if !anim
421:     end
422:     return [anim + hitNum, noFlip] if anim
423:     return nil
424:   end

426:   # Returns the animation ID to use for a given move. If the move has no
427:   # animations, tries to use a default move animation depending on the move's
428:   # type. If that default move animation doesn't exist, trues to use Tackle's
429:   # move animation. Returns nil if it can't find any of these animations to use.
430:   def pbFindMoveAnimation(moveID, idxUser, hitNum)
431:     begin
432:       move2anim = pbLoadMoveToAnim

434:       # looks for an OppMove version then a Move version)
435:       anim = pbFindMoveAnimDetails(move2anim, moveID, idxUser, hitNum)
436:       return anim if anim
437:       # Actual animation not found, get the default animation for the move's type
438:       moveData = GameData::Move.get(moveID)
439:       target_data = GameData::Target.get(moveData.target)
440:       moveType = moveData.type
441:       moveKind = moveData.category
442:       moveKind += 3 if target_data.num_targets > 1 || target_data.affects_foe_side
443:       moveKind += 3 if moveData.status? && target_data.num_targets > 0
444:       # [one target physical, one target special, user status,
445:       #  multiple targets physical, multiple targets special, non-user status]
446:       typeDefaultAnim = {
447:         :NORMAL   => [:TACKLE,       :SONICBOOM,    :DEFENSECURL, :EXPLOSION,  :SWIFT,        :TAILWHIP],

450:         :POISON   => [:POISONSTING,  :SLUDGE,       :ACIDARMOR,   nil,         :ACID,         :POISONPOWDER],
451:         :GROUND   => [:SANDTOMB,     :MUDSLAP,      nil,          :EARTHQUAKE, :EARTHPOWER,   :MUDSPORT],
452:         :ROCK     => [:ROCKTHROW,    :POWERGEM,     :ROCKPOLISH,  :ROCKSLIDE,  nil,           :SANDSTORM],

454:         :GHOST    => [:LICK,         :SHADOWBALL,   :GRUDGE,      nil,         nil,           :CONFUSERAY],
455:         :STEEL    => [:IRONHEAD,     :MIRRORSHOT,   :IRONDEFENSE, nil,         nil,           :METALSOUND],
456:         :FIRE     => [:FIREPUNCH,    :EMBER,        :SUNNYDAY,    nil,         :INCINERATE,   :WILLOWISP],
457:         :WATER    => [:CRABHAMMER,   :WATERGUN,     :AQUARING,    nil,         :SURF,         :WATERSPORT],
458:         :GRASS    => [:VINEWHIP,     :MEGADRAIN,    :COTTONGUARD, :RAZORLEAF,  nil,           :SPORE],
459:         :ELECTRIC => [:THUNDERPUNCH, :THUNDERSHOCK, :CHARGE,      nil,         :DISCHARGE,    :THUNDERWAVE],
460:         :PSYCHIC  => [:ZENHEADBUTT,  :CONFUSION,    :CALMMIND,    nil,         :SYNCHRONOISE, :MIRACLEEYE],
461:         :ICE      => [:ICEPUNCH,     :ICEBEAM,      :MIST,        nil,         :POWDERSNOW,   :HAIL],
462:         :DRAGON   => [:DRAGONCLAW,   :DRAGONRAGE,   :DRAGONDANCE, nil,         :TWISTER,      nil],

464:         :FAIRY    => [:TACKLE,       :FAIRYWIND,    :MOONLIGHT,   nil,         :SWIFT,        :SWEETKISS]
465:       }
466:       if typeDefaultAnim[moveType]
467:         anims = typeDefaultAnim[moveType]
468:         if GameData::Move.exists?(anims[moveKind])
469:           anim = pbFindMoveAnimDetails(move2anim, anims[moveKind], idxUser)
470:         end
471:         if !anim && moveKind >= 3 && GameData::Move.exists?(anims[moveKind - 3])
472:           anim = pbFindMoveAnimDetails(move2anim, anims[moveKind - 3], idxUser)

477:       end
478:       return anim if anim
479:       # Default animation for the move's type not found, use Tackle's animation
480:       if GameData::Move.exists?(:TACKLE)
481:         return pbFindMoveAnimDetails(move2anim, :TACKLE, idxUser)
482:       end
483:     rescue
484:     end
485:     return nil

492:   def pbAnimation(moveID, user, targets, hitNum = 0)
493:     animID = pbFindMoveAnimation(moveID, user.index, hitNum)
494:     return if !animID
495:     anim = animID[0]

498:     return if !animations

503:         pbAnimationCore(animations[anim], user, target)
504:       end
505:     end

508:   # Plays a common animation.
509:   def pbCommonAnimation(animName, user = nil, target = nil)
510:     return if nil_or_empty?(animName)
511:     target = target[0] if target.is_a?(Array)
512:     animations = pbLoadBattleAnimations
513:     return if !animations
514:     animations.each do |a|
515:       next if !a || a.name != "Common:" + animName
516:       pbAnimationCore(a, user, target || user)
517:       return
518:     end

521:   def pbAnimationCore(animation, user, target, oppMove = false)
522:     return if !animation
523:     @briefMessage = false
524:     userSprite   = (user) ? @sprites["pokemon_#{user.index}"] : nil
525:     targetSprite = (target) ? @sprites["pokemon_#{target.index}"] : nil
526:     # Remember the original positions of Pokémon sprites
527:     oldUserX = (userSprite) ? userSprite.x : 0
528:     oldUserY = (userSprite) ? userSprite.y : 0
529:     oldTargetX = (targetSprite) ? targetSprite.x : oldUserX
530:     oldTargetY = (targetSprite) ? targetSprite.y : oldUserY
531:     # Create the animation player
532:     animPlayer = PBAnimationPlayerX.new(animation, user, target, self, oppMove)
533:     # Apply a transformation to the animation based on where the user and target
534:     # actually are. Get the centres of each sprite.
535:     userHeight = (userSprite&.bitmap && !userSprite.bitmap.disposed?) ? userSprite.bitmap.height : 128
536:     if targetSprite
537:       targetHeight = (targetSprite.bitmap && !targetSprite.bitmap.disposed?) ? targetSprite.bitmap.height : 128
538:     else
539:       targetHeight = userHeight
540:     end
541:     animPlayer.setLineTransform(
542:       FOCUSUSER_X, FOCUSUSER_Y, FOCUSTARGET_X, FOCUSTARGET_Y,
543:       oldUserX, oldUserY - (userHeight / 2), oldTargetX, oldTargetY - (targetHeight / 2)
544:     )
545:     # Play the animation
546:     animPlayer.start
547:     loop do
548:       animPlayer.update
549:       pbUpdate
550:       break if animPlayer.animDone?
551:     end
552:     animPlayer.dispose
553:     # Return Pokémon sprites to their original positions
554:     if userSprite
555:       userSprite.x = oldUserX
556:       userSprite.y = oldUserY
557:       userSprite.pbSetOrigin
558:     end
559:     if targetSprite
560:       targetSprite.x = oldTargetX
561:       targetSprite.y = oldTargetY
562:       targetSprite.pbSetOrigin
563:     end
564:   end
