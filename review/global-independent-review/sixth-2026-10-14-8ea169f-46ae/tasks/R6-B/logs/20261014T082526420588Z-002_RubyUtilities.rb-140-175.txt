144: class File
145:   # Copies the source file to the destination path.
146:   def self.copy(source, destination)
147:     data = ""
148:     t = System.uptime
149:     File.open(source, "rb") do |f|
150:       loop do
151:         r = f.read(4096)
152:         break if !r
153:         if System.uptime - t >= 5
154:           t += 5
155:           Graphics.update
156:         end
157:         data += r
158:       end
159:     end
160:     File.delete(destination) if File.file?(destination)
161:     f = File.new(destination, "wb")
162:     f.write data
163:     f.close
164:   end
165: 
166:   # Copies the source to the destination and deletes the source.
167:   def self.move(source, destination)
168:     File.copy(source, destination)
169:     File.delete(source)
170:   end
171: end
