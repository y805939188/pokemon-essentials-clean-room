1: def pbStringToAudioFile(str)
2:   if str[/^(.*)\:\s*(\d+)\s*\:\s*(\d+)\s*$/]   # Of the format "XXX: ###: ###"
3:     file   = $1
4:     volume = $2.to_i
5:     pitch  = $3.to_i
6:     return RPG::AudioFile.new(file, volume, pitch)
7:   elsif str[/^(.*)\:\s*(\d+)\s*$/]             # Of the format "XXX: ###"
8:     file   = $1
9:     volume = $2.to_i
10:     return RPG::AudioFile.new(file, volume, 100)
11:   else
12:     return RPG::AudioFile.new(str, 100, 100)
13:   end
14: end
15: 
16: # Converts an object to an audio file.
17: # str -- Either a string showing the filename or an RPG::AudioFile object.
18: # Possible formats for _str_:
19: # filename                        volume and pitch 100
20: # filename:volume           pitch 100
21: # filename:volume:pitch
22: # volume -- Volume of the file, up to 100
23: # pitch -- Pitch of the file, normally 100
24: def pbResolveAudioFile(str, volume = nil, pitch = nil)
25:   if str.is_a?(String)
26:     str = pbStringToAudioFile(str)
27:     str.volume = volume || 100
28:     str.pitch  = pitch || 100
29:   end
30:   if str.is_a?(RPG::AudioFile)
31:     if volume || pitch
32:       return RPG::AudioFile.new(str.name, volume || str.volume || 100,
33:                                 pitch || str.pitch || 100)
34:     else
35:       return str
36:     end
37:   end
38:   return str
39: end

43: # Plays a BGM file.
44: # param -- Either a string showing the filename
45: # (relative to Audio/BGM/) or an RPG::AudioFile object.
46: # Possible formats for _param_:
47: # filename                        volume and pitch 100
48: # filename:volume           pitch 100
49: # filename:volume:pitch
50: # volume -- Volume of the file, up to 100
51: # pitch -- Pitch of the file, normally 100
52: def pbBGMPlay(param, volume = nil, pitch = nil)
53:   return if !param
54:   param = pbResolveAudioFile(param, volume, pitch)
55:   if param.name && param.name != ""
56:     if $game_system
57:       $game_system.bgm_play(param)
58:       return
59:     elsif (RPG.const_defined?(:BGM) rescue false)
60:       b = RPG::BGM.new(param.name, param.volume, param.pitch)
61:       if b.respond_to?("play")
62:         b.play
63:         return
64:       end
65:     end
66:     Audio.bgm_play(canonicalize("Audio/BGM/" + param.name), param.volume, param.pitch)
67:   end
68: end
69: 
70: # Fades out or stops BGM playback. 'x' is the time in seconds to fade out.
71: def pbBGMFade(x = 0.0); pbBGMStop(x); end
72: 
73: # Fades out or stops BGM playback. 'x' is the time in seconds to fade out.
74: def pbBGMStop(timeInSeconds = 0.0)
75:   if $game_system && timeInSeconds > 0.0
76:     $game_system.bgm_fade(timeInSeconds)
77:     return
78:   elsif $game_system
79:     $game_system.bgm_stop
80:     return
81:   elsif (RPG.const_defined?(:BGM) rescue false)
82:     begin
83:       (timeInSeconds > 0.0) ? RPG::BGM.fade((timeInSeconds * 1000).floor) : RPG::BGM.stop
84:       return
85:     rescue
86:     end
87:   end
88:   (timeInSeconds > 0.0) ? Audio.bgm_fade((timeInSeconds * 1000).floor) : Audio.bgm_stop
89: end
