1: animmaker.exe - Converts sprites to RPG Maker XP format animations
2: (C) 2008 Peter O.
3: 
4: Usage: animmaker xmlfile.xml
5: ('xmlfile.xml' is the path to any XML file, see below)
6: 
7: Example of XML file:
8: 
9: <animations>
10: <!--Animation file to create (in this case, animation.png)-->
11: <animation name="animation.png">
12:  <!--Pattern to add to the animation.  Consists of the image's
13:      filename plus the rectangle of the pattern to extract
14:      from the image (x, y, width, height) (up to 96x96 in size)
15:      Sprites listed here will be doubled in size in the animation-->
16:  <pattern filename="pkmnfrlg_effects.png" x="50" y="201" 
17:    width="56" height="68" />
18: </animation>
19: <!--Another animation file, animation2.png)-->
20: <animation name="animation2.png">
21:  <pattern filename="pkmnfrlg_effects.png" x="50" y="201" 
22:    width="56" height="68" />
23:  <!-- You can leave out x, y, etc.  to use the whole image-->
24:  <pattern filename="effects2.png" />
25: </animation>
26: </animations>
