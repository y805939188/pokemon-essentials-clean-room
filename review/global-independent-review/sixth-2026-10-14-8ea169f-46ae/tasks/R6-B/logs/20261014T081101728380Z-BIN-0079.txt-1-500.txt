273:     def initialize
274:       super
275:       @exp_params = [30,20,30,30]
276:       @params = Table.new(8,100)
277:       (1..99).each do |i|
278:         @params[0,i] = 400+i*50
279:         @params[1,i] = 80+i*10
280:         (2..5).each {|j| @params[j,i] = 15+i*5/4 }
281:         (6..7).each {|j| @params[j,i] = 30+i*5/2 }
282:       end
283:       @learnings = []
284:       @features.push(BaseItem::Feature.new(23, 0, 1))
285:       @features.push(BaseItem::Feature.new(22, 0, 0.95))
286:       @features.push(BaseItem::Feature.new(22, 1, 0.05))
287:       @features.push(BaseItem::Feature.new(22, 2, 0.04))
288:       @features.push(BaseItem::Feature.new(41, 1))
289:       @features.push(BaseItem::Feature.new(51, 1))
290:       @features.push(BaseItem::Feature.new(52, 1))
291:     end
292:     def exp_for_level(level)
293:       lv = level.to_f
294:       basis = @exp_params[0].to_f
295:       extra = @exp_params[1].to_f
296:       acc_a = @exp_params[2].to_f
297:       acc_b = @exp_params[3].to_f
298:       return (basis*((lv-1)**(0.9+acc_a/250))*lv*(lv+1)/
299:         (6+lv**2/50/acc_b)+(lv-1)*extra).round.to_i
300:     end
301:     attr_accessor :exp_params
302:     attr_accessor :params
303:     attr_accessor :learnings
304:   end

307:     class Damage
308:       def initialize
309:         @type = 0
310:         @element_id = 0
311:         @formula = '0'
312:         @variance = 20
313:         @critical = false
314:       end
315:       def none?
316:         @type == 0
317:       end
318:       def to_hp?
319:         [1,3,5].include?(@type)
320:       end
321:       def to_mp?
322:         [2,4,6].include?(@type)
323:       end
324:       def recover?
325:         [3,4].include?(@type)
326:       end
327:       def drain?
328:         [5,6].include?(@type)
329:       end
330:       def sign
331:         recover? ? -1 : 1
332:       end
333:       def eval(a, b, v)
334:         [Kernel.eval(@formula), 0].max * sign rescue 0
335:       end
336:       attr_accessor :type

338:       attr_accessor :formula
339:       attr_accessor :variance
340:       attr_accessor :critical
