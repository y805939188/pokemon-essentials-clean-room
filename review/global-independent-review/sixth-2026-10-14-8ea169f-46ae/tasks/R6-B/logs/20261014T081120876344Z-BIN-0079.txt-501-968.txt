886:       if @name.empty?
887:         Audio.bgm_stop
888:         @@last = BGM.new
889:       else
890:         Audio.bgm_play('Audio/BGM/' + @name, @volume, @pitch, pos)
891:         @@last = self.clone
892:       end
893:     end
894:     def replay
895:       play(@pos)
896:     end

904:     end
