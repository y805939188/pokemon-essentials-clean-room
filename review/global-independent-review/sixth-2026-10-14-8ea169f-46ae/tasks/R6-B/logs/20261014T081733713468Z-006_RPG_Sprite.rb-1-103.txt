1: #===============================================================================
2: # Additions to class Sprite that allows class AnimationContainerSprite to attach
3: # overworld animations to itself.
4: #===============================================================================
5: module RPG
6:   class Sprite < ::Sprite
7:     def initialize(viewport = nil)
8:       super(viewport)
9:       @_animation_duration = 0
10:       @_animation_frame = 0
11:       @animations = []
12:       @loopAnimations = []
13:     end
14: 
15:     def dispose
16:       dispose_animation
17:       dispose_loop_animation
18:       super
19:     end
20: 
21:     def dispose_animation
22:       @animations.each { |a| a&.dispose_animation }
23:       @animations.clear
24:     end
25: 
26:     def dispose_loop_animation
27:       @loopAnimations.each { |a| a&.dispose_loop_animation }
28:       @loopAnimations.clear
29:     end
30: 
31:     def x=(x)
32:       @animations.each { |a| a.x = x if a }
33:       @loopAnimations.each { |a| a.x = x if a }
34:       super
35:     end
36: 
37:     def y=(y)
38:       @animations.each { |a| a.y = y if a }
39:       @loopAnimations.each { |a| a.y = y if a }
40:       super
41:     end
42: 
43:     def pushAnimation(array, anim)
44:       array.length.times do |i|
45:         next if array[i]&.active?
46:         array[i] = anim
47:         return
48:       end
49:       array.push(anim)
50:     end
51: 
52:     def animation(animation, hit, height = 3)
53:       anim = SpriteAnimation.new(self)
54:       anim.animation(animation, hit, height)
55:       pushAnimation(@animations, anim)
56:     end
57: 
58:     def loop_animation(animation)
59:       anim = SpriteAnimation.new(self)
60:       anim.loop_animation(animation)
61:       pushAnimation(@loopAnimations, anim)
62:     end
63: 
64:     def effect?
65:       @animations.each { |a| return true if a.effect? }
66:       return false
67:     end
68: 
69:     def update_animation
70:       @animations.each { |a| a.update_animation if a&.active? }
71:     end
72: 
73:     def update_loop_animation
74:       @loopAnimations.each { |a| a.update_loop_animation if a&.active? }
75:     end
76: 
77:     def update
78:       super
79:       @animations.each { |a| a.update }
80:       @loopAnimations.each { |a| a.update }
81:       SpriteAnimation.clear
82:     end
83:   end
84: end

87: # A version of class Sprite that allows its coordinates to be floats rather than
88: # integers.
89: #===============================================================================
90: class FloatSprite < Sprite
91:   def x; return @float_x; end
92:   def y; return @float_y; end
93: 
94:   def x=(value)
95:     @float_x = value
96:     super
97:   end
98: 
99:   def y=(value)
100:     @float_y = value
101:     super
102:   end
103: end
