175:     ret.animation_name = otherAnim.animation_name.clone

177:     ret.position       = otherAnim.position

181:   def addSound(frame, se)
182:     timing = RPG::Animation::Timing.new
183:     timing.frame = frame
184:     timing.se    = RPG::AudioFile.new(se, 100)
185:     self.timings.push(timing)

188:   # frame is zero-based.
189:   def addAnimation(otherAnim, frame, x, y)
190:     if frame + otherAnim.frames.length >= self.frames.length
191:       totalframes = frame + otherAnim.frames.length + 1
192:       (totalframes - self.frames.length).times do
193:         self.frames.push(RPG::Animation::Frame.new)
194:       end
195:     end
196:     self.frame_max = self.frames.length
197:     otherAnim.frame_max.times do |i|
198:       thisframe = self.frames[frame + i]
199:       otherframe = otherAnim.frames[i]
200:       cellStart = thisframe.cell_max
201:       thisframe.cell_max += otherframe.cell_max
202:       thisframe.cell_data.resize(thisframe.cell_max, 8)
203:       otherframe.cell_max.times do |j|
204:         thisframe.cell_data[cellStart + j, 0] = otherframe.cell_data[j, 0]
205:         thisframe.cell_data[cellStart + j, 1] = otherframe.cell_data[j, 1] + x
206:         thisframe.cell_data[cellStart + j, 2] = otherframe.cell_data[j, 2] + y
207:         thisframe.cell_data[cellStart + j, 3] = otherframe.cell_data[j, 3]
208:         thisframe.cell_data[cellStart + j, 4] = otherframe.cell_data[j, 4]
209:         thisframe.cell_data[cellStart + j, 5] = otherframe.cell_data[j, 5]
210:         thisframe.cell_data[cellStart + j, 6] = otherframe.cell_data[j, 6]
211:         thisframe.cell_data[cellStart + j, 7] = otherframe.cell_data[j, 7]
212:       end
213:     end
214:     otherAnim.timings.each do |othertiming|
215:       timing = RPG::Animation::Timing.new
216:       timing.frame          = frame + othertiming.frame
217:       timing.se             = RPG::AudioFile.new(othertiming.se.name.clone,
218:                                                  othertiming.se.volume,
219:                                                  othertiming.se.pitch)
220:       timing.flash_scope    = othertiming.flash_scope
221:       timing.flash_color    = othertiming.flash_color.clone
222:       timing.flash_duration = othertiming.flash_duration
223:       timing.condition      = othertiming.condition
224:       self.timings.push(timing)
225:     end
226:     self.timings.sort! { |a, b| a.frame <=> b.frame }
227:   end
228: end
