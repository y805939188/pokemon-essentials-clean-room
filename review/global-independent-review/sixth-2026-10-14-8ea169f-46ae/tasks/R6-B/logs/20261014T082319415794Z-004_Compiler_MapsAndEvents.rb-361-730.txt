382:       end

478:     isFirstCommand = false
479:     # Find all the trainer comments in the event
480:     list.length.times do |i|
481:       next if list[i].code != 108   # Comment (first line)
482:       command = list[i].parameters[0]
483:       ((i + 1)...list.length).each do |j|
484:         break if list[j].code != 408   # Comment (continuation line)
485:         command += "\r\n" + list[j].parameters[0]

514:     doublebattle   = false

538:         doublebattle = true if value.upcase == "TRUE" || value.upcase == "YES"
539:         push_comment(firstpage.list, command) if rewriteComments

542:         continue = true if value.upcase == "TRUE" || value.upcase == "YES"

545:         endifswitch.push(($~[1].gsub(/^\s+/, "").gsub(/\s+$/, "")).to_i)
