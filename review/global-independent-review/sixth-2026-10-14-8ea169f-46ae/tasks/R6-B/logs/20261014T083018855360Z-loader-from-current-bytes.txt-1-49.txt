3: def traceback_report
4:   backtrace = $!.backtrace.clone
5:   backtrace.each{ |bt|
6:     bt.sub!(/\{(\d+)\}/) {"[#{$1}]#{$RGSS_SCRIPTS[$1.to_i][1]}"}
7:   }
8:   return $!.message + "\n\n" + backtrace.join("\n")
9: end
10: 
11: def raise_traceback_error
12:   if $!.message.size >= 900
13:     File.open('traceback.log', 'w') { |f| f.write($!) }
14:     raise 'Traceback is too big. Output in traceback.log'
15:   else
16:     raise
17:   end
18: end
19: 
20: def load_scripts_from_folder(path)
21:   files   = []
22:   folders = []
23:   Dir.foreach(path) do |f|
24:     next if f == '.' || f == '..'
25:     if File.directory?(path + "/" + f)
26:       folders.push(f) if !f[/^\./]
27:     else
28:       files.push(f) if f[/\.rb$/i]
29:     end
30:   end
31:   files.sort!
32:   files.each do |f|
33:     code = File.open(path + "/" + f, "r") { |file| file.read }
34:     begin
35:       eval(code, nil, f)
36:     rescue ScriptError
37:       raise ScriptError.new($!.message)
38:     rescue
39:       $!.message.sub!($!.message, traceback_report)
40:       raise_traceback_error
41:     end
42:   end
43:   folders.sort!
44:   folders.each do |folder|
45:     load_scripts_from_folder(path + "/" + folder)
46:   end
47: end
48: 
49: load_scripts_from_folder("Data/Scripts")
