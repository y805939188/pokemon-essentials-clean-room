501:     end
502:     def type=(type)
503:       return if @type == type
504:       @type = type
505:       case @type
506:       when 1
507:         bitmap = @rain_bitmap
508:       when 2
509:         bitmap = @storm_bitmap
510:       when 3
511:         bitmap = @snow_bitmap
512:       else
513:         bitmap = nil
514:       end
515:       for i in 1..40
516:         sprite = @sprites[i]
517:         if sprite != nil
518:           sprite.visible = (i <= @max)
519:           sprite.bitmap = bitmap
520:         end
521:       end
522:     end
523:     def ox=(ox)
524:       return if @ox == ox;
525:       @ox = ox
526:       for sprite in @sprites
527:         sprite.ox = @ox
528:       end
529:     end
530:     def oy=(oy)
531:       return if @oy == oy;
532:       @oy = oy
533:       for sprite in @sprites
534:         sprite.oy = @oy
535:       end
536:     end
537:     def max=(max)
538:       return if @max == max;
539:       @max = [[max, 0].max, 40].min
540:       for i in 1..40
541:         sprite = @sprites[i]
542:         if sprite != nil
543:           sprite.visible = (i <= @max)
544:         end
545:       end
546:     end
547:     def update
548:       return if @type == 0
549:       for i in 1..@max
550:         sprite = @sprites[i]
551:         if sprite == nil
552:           break
553:         end
554:         if @type == 1
555:           sprite.x -= 2
556:           sprite.y += 16
557:           sprite.opacity -= 8
558:         end
559:         if @type == 2
560:           sprite.x -= 8
561:           sprite.y += 16
562:           sprite.opacity -= 12
563:         end
564:         if @type == 3
565:           sprite.x -= 2
566:           sprite.y += 8
567:           sprite.opacity -= 8
568:         end
569:         x = sprite.x - @ox
570:         y = sprite.y - @oy
571:         if sprite.opacity < 64 or x < -50 or x > 750 or y < -300 or y > 500
572:           sprite.x = rand(800) - 50 + @ox
573:           sprite.y = rand(800) - 200 + @oy
574:           sprite.opacity = 255
575:         end
576:       end
577:     end
578:     attr_reader :type
579:     attr_reader :max
580:     attr_reader :ox
581:     attr_reader :oy
582:   end
