45:   def import_new_maps
46:     return false if !$DEBUG
47:     mapfiles = {}
48:     # Get IDs of all maps in the Data folder
49:     Dir.chdir("Data") do
50:       mapData = sprintf("Map*.rxdata")
51:       Dir.glob(mapData).each do |map|
52:         mapfiles[$1.to_i(10)] = true if map[/map(\d+)\.rxdata/i]
53:       end
54:     end
55:     mapinfos = pbLoadMapInfos
56:     maxOrder = 0
57:     # Exclude maps found in mapinfos
58:     mapinfos.each_key do |id|
59:       next if !mapinfos[id]
60:       mapfiles.delete(id) if mapfiles[id]
61:       maxOrder = [maxOrder, mapinfos[id].order].max
62:     end
63:     # Import maps not found in mapinfos
64:     maxOrder += 1
65:     imported = false
66:     count = 0
67:     mapfiles.each_key do |id|
68:       next if id == 999   # Ignore 999 (random dungeon map)
69:       mapinfo = RPG::MapInfo.new
70:       mapinfo.order = maxOrder
71:       mapinfo.name  = sprintf("MAP%03d", id)
72:       maxOrder += 1
73:       mapinfos[id] = mapinfo
74:       imported = true
75:       count += 1
76:     end
77:     if imported
78:       save_data(mapinfos, "Data/MapInfos.rxdata")
79:       $game_temp.map_infos = nil
80:       pbMessage(_INTL("{1} new map(s) copied to the Data folder were successfully imported.", count))
81:     end
82:     return imported
83:   end

88:   def generate_move_route(commands)
89:     route           = RPG::MoveRoute.new
90:     route.repeat    = false
91:     route.skippable = true
92:     route.list.clear
93:     i = 0
94:     while i < commands.length
95:       case commands[i]
96:       when PBMoveRoute::WAIT, PBMoveRoute::SWITCH_ON, PBMoveRoute::SWITCH_OFF,
97:            PBMoveRoute::CHANGE_SPEED, PBMoveRoute::CHANGE_FREQUENCY, PBMoveRoute::OPACITY,
98:            PBMoveRoute::BLENDING, PBMoveRoute::PLAY_SE, PBMoveRoute::SCRIPT
99:         route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1]]))
100:         i += 1
101:       when PBMoveRoute::SCRIPT_ASYNC
102:         route.list.push(RPG::MoveCommand.new(PBMoveRoute::SCRIPT, [commands[i + 1]]))
103:         route.list.push(RPG::MoveCommand.new(PBMoveRoute::WAIT, [0]))
104:         i += 1
105:       when PBMoveRoute::JUMP
106:         route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1], commands[i + 2]]))
107:         i += 2
108:       when PBMoveRoute::GRAPHIC
109:         route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1], commands[i + 2], commands[i + 3], commands[i + 4]]))
110:         i += 4
111:       else
112:         route.list.push(RPG::MoveCommand.new(commands[i]))
113:       end
114:       i += 1
115:     end
116:     route.list.push(RPG::MoveCommand.new(0))
117:     return route
118:   end

154:   def push_text(list, text, indent = 0)
155:     return if !text
156:     textsplit = text.split(/\\m/)
157:     textsplit.each do |t|
158:       first = true
159:       textsplit2 = t.split(/\n/)
160:       textsplit2.length.times do |i|
161:         textchunk = textsplit2[i].gsub(/\s+$/, "")
162:         if textchunk && textchunk != ""
163:           list.push(RPG::EventCommand.new((first) ? 101 : 401, indent, [textchunk]))
164:           first = false
165:         end
166:       end
167:     end
168:   end

254:   def safequote(x)
255:     x = x.gsub(/\"\#\'\\/) { |a| "\\" + a }
256:     x = x.gsub(/\t/, "\\t")
257:     x = x.gsub(/\r/, "\\r")
258:     x = x.gsub(/\n/, "\\n")
259:     return x
260:   end

325:     def getEventFromXY(mapID, x, y)
326:       return nil if x < 0 || y < 0
327:       mapPositions = @mapxy[mapID]
328:       return mapPositions[(y * @mapWidths[mapID]) + x] if mapPositions
329:       map = getMap(mapID)
330:       return nil if !map
331:       @mapWidths[mapID]  = map.width
332:       @mapHeights[mapID] = map.height
333:       mapPositions = []
334:       width = map.width
335:       map.events.each_value do |e|
336:         mapPositions[(e.y * width) + e.x] = e if e
337:       end
338:       @mapxy[mapID] = mapPositions
339:       return mapPositions[(y * width) + x]
340:     end

343:       map = getMap(mapID)
