1039:   def main
1040:     return if !$DEBUG
1041:     begin
1042:       mustCompile = false
1043:       # If no PBS file, create one and fill it, then recompile
1044:       if !FileTest.directory?("PBS")
1045:         Dir.mkdir("PBS") rescue nil
1046:         GameData.load_all
1047:         write_all
1048:         mustCompile = true
1049:       end
1050:       # Get all data files and PBS files to be checked for their last modified times
1051:       data_files = GameData.get_all_data_filenames
1052:       data_files += [   # Extra .dat files for data that isn't a GameData class
1053:         ["map_connections.dat", true],
1054:         ["regional_dexes.dat", true],
1055:         ["trainer_lists.dat", true]
1056:       ]
1057:       text_files = get_all_pbs_files_to_compile
1058:       latestDataTime = 0
1059:       latestTextTime = 0
1060:       # Should recompile if new maps were imported
1061:       mustCompile |= import_new_maps
1062:       # Check data files for their latest modify time
1063:       data_files.each do |filename|   # filename = [string, boolean (whether mandatory)]
1064:         if FileTest.exist?("Data/" + filename[0])
1065:           begin
1066:             File.open("Data/#{filename[0]}") do |file|
1067:               latestDataTime = [latestDataTime, file.mtime.to_i].max
1068:             end
1069:           rescue SystemCallError
1070:             mustCompile = true
1071:           end
1072:         elsif filename[1]
1073:           mustCompile = true
1074:           break
1075:         end
1076:       end
1077:       # Check PBS files for their latest modify time
1078:       text_files.each do |key, value|
1079:         next if !value || !value[1].is_a?(Array)
1080:         value[1].each do |filepath|
1081:           begin
1082:             File.open(filepath) { |file| latestTextTime = [latestTextTime, file.mtime.to_i].max }
1083:           rescue SystemCallError
1084:           end
1085:         end
1086:       end
1087:       # Decide to compile if a PBS file was edited more recently than any .dat files
1088:       mustCompile |= (latestTextTime >= latestDataTime)
1089:       # Should recompile if holding Ctrl
1090:       Input.update
1091:       mustCompile = true if $full_compile || Input.press?(Input::CTRL)
1092:       # Delete old data files in preparation for recompiling
1093:       if mustCompile
1094:         data_files.each do |filename|
1095:           begin
1096:             File.delete("Data/#{filename[0]}") if FileTest.exist?("Data/#{filename[0]}")
1097:           rescue SystemCallError
1098:           end
1099:         end
1100:       end
1101:       # Recompile all data
1102:       compile_all(mustCompile)
1103:     rescue Exception
1104:       e = $!
1105:       raise e if e.class.to_s == "Reset" || e.is_a?(Reset) || e.is_a?(SystemExit)
1106:       pbPrintException(e)
1107:       data_files.each do |filename|
1108:         begin
1109:           File.delete("Data/#{filename[0]}") if FileTest.exist?("Data/#{filename[0]}")
1110:         rescue SystemCallError
1111:         end
1112:       end
1113:       raise Reset.new if e.is_a?(Hangup)
1114:       raise "Unknown exception when compiling."
1115:     end
1116:   end
1117: end
