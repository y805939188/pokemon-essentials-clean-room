5:       path = folder_name + filename
6:       if not @cache.include?(path) or @cache[path].disposed?

8:           @cache[path] = Bitmap.new(path)
9:         else
10:           @cache[path] = Bitmap.new(32, 32)
11:         end

17:         if not @cache.include?(key) or @cache[key].disposed?
18:           @cache[key] = @cache[path].clone

149:       bitmap.draw_text(-1, 12+1, 160, 36, damage_string, 1)
150:       bitmap.draw_text(+1, 12+1, 160, 36, damage_string, 1)
151:       if value.is_a?(Numeric) and value < 0
152:         bitmap.font.color.set(176, 255, 144)

290:     def update
291:       super
292:       if @_whiten_duration > 0
293:         @_whiten_duration -= 1
294:         self.color.alpha = 128 - (16 - @_whiten_duration) * 10
295:       end
296:       if @_appear_duration > 0
297:         @_appear_duration -= 1
298:         self.opacity = (16 - @_appear_duration) * 16
299:       end
300:       if @_escape_duration > 0
301:         @_escape_duration -= 1
302:         self.opacity = 256 - (32 - @_escape_duration) * 10
303:       end
304:       if @_collapse_duration > 0
305:         @_collapse_duration -= 1
306:         self.opacity = 256 - (48 - @_collapse_duration) * 6
307:       end
308:       if @_damage_duration > 0
309:         @_damage_duration -= 1
310:         case @_damage_duration
311:         when 38..39
312:           @_damage_sprite.y -= 4
313:         when 36..37
314:           @_damage_sprite.y -= 2
315:         when 34..35
316:           @_damage_sprite.y += 2
317:         when 28..33
318:           @_damage_sprite.y += 4
319:         end
320:         @_damage_sprite.opacity = 256 - (12 - @_damage_duration) * 32
321:         if @_damage_duration == 0
322:           dispose_damage
323:         end
324:       end
325:       if @_animation != nil and (Graphics.frame_count % 2 == 0)
326:         @_animation_duration -= 1
327:         update_animation
328:       end
329:       if @_loop_animation != nil and (Graphics.frame_count % 2 == 0)
330:         update_loop_animation
331:         @_loop_animation_index += 1
332:         @_loop_animation_index %= @_loop_animation.frame_max
333:       end
334:       if @_blink
335:         @_blink_count = (@_blink_count + 1) % 32
336:         if @_blink_count < 16
337:           alpha = (16 - @_blink_count) * 6
338:         else
339:           alpha = (@_blink_count - 16) * 6
340:         end
341:         self.color.set(255, 255, 255, alpha)
342:       end

344:     end
345:     def update_animation
346:       if @_animation_duration > 0
347:         frame_index = @_animation.frame_max - @_animation_duration
348:         cell_data = @_animation.frames[frame_index].cell_data
349:         position = @_animation.position
350:         animation_set_sprites(@_animation_sprites, cell_data, position)
351:         for timing in @_animation.timings
352:           if timing.frame == frame_index
353:             animation_process_timing(timing, @_animation_hit)
354:           end
355:         end
356:       else
357:         dispose_animation
358:       end
359:     end
360:     def update_loop_animation
361:       frame_index = @_loop_animation_index
362:       cell_data = @_loop_animation.frames[frame_index].cell_data
363:       position = @_loop_animation.position
364:       animation_set_sprites(@_loop_animation_sprites, cell_data, position)
365:       for timing in @_loop_animation.timings
366:         if timing.frame == frame_index
367:           animation_process_timing(timing, true)
368:         end
369:       end
370:     end
371:     def animation_set_sprites(sprites, cell_data, position)
372:       for i in 0..15
373:         sprite = sprites[i]
374:         pattern = cell_data[i, 0]
375:         if sprite == nil or pattern == nil or pattern == -1
376:           sprite.visible = false if sprite != nil
377:           next
378:         end
379:         sprite.visible = true
380:         sprite.src_rect.set(pattern % 5 * 192, pattern / 5 * 192, 192, 192)
381:         if position == 3
382:           if self.viewport != nil
383:             sprite.x = self.viewport.rect.width / 2
384:             sprite.y = self.viewport.rect.height - 160
385:           else
386:             sprite.x = 320
387:             sprite.y = 240
388:           end
389:         else
390:           sprite.x = self.x - self.ox + self.src_rect.width / 2
391:           sprite.y = self.y - self.oy + self.src_rect.height / 2
392:           sprite.y -= self.src_rect.height / 4 if position == 0
393:           sprite.y += self.src_rect.height / 4 if position == 2
394:         end
395:         sprite.x += cell_data[i, 1]
396:         sprite.y += cell_data[i, 2]
397:         sprite.z = 2000
398:         sprite.ox = 96
399:         sprite.oy = 96
400:         sprite.zoom_x = cell_data[i, 3] / 100.0
401:         sprite.zoom_y = cell_data[i, 3] / 100.0
402:         sprite.angle = cell_data[i, 4]
403:         sprite.mirror = (cell_data[i, 5] == 1)
404:         sprite.opacity = cell_data[i, 6] * self.opacity / 255.0
405:         sprite.blend_type = cell_data[i, 7]
406:       end
407:     end
408:     def animation_process_timing(timing, hit)
409:       if (timing.condition == 0) or
410:          (timing.condition == 1 and hit == true) or
411:          (timing.condition == 2 and hit == false)
412:         if timing.se.name != ""
413:           se = timing.se
414:           Audio.se_play("Audio/SE/" + se.name, se.volume, se.pitch)
415:         end
416:         case timing.flash_scope
417:         when 1
418:           self.flash(timing.flash_color, timing.flash_duration * 2)
419:         when 2
420:           if self.viewport != nil
421:             self.viewport.flash(timing.flash_color, timing.flash_duration * 2)
422:           end
423:         when 3
424:           self.flash(nil, timing.flash_duration * 2)
425:         end
426:       end
427:     end
