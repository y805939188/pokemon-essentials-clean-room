816:     changed = false
817:     return false if event.is_a?(RPG::CommonEvent)
818:     # Check if event has 2+ pages and the last page meets all of these criteria:
819:     #   - Has a condition of a Switch being ON
820:     #   - The event has a charset graphic
821:     #   - There are more than 5 commands in that page, the first of which is a
822:     #     Conditional Branch
823:     lastPage = event.pages[event.pages.length - 1]
824:     if event.pages.length >= 2 &&
825:        lastPage.condition.switch1_valid &&
826:        lastPage.graphic.character_name != "" &&
827:        lastPage.list.length > 5 &&
828:        lastPage.list[0].code == 111
829:       # This bit of code is just in case Switch 22 has been renamed/repurposed,
830:       # which is highly unlikely. It changes the Switch used in the condition to
831:       # whichever is named 's:tsOff?("A")'.
832:       if lastPage.condition.switch1_id == 22 &&
833:          mapData.switchName(lastPage.condition.switch1_id) != 's:tsOff?("A")'
834:         lastPage.condition.switch1_id = mapData.registerSwitch('s:tsOff?("A")')
835:         changed = true
836:       end
837:       # If the last page's Switch condition uses a Switch named 's:tsOff?("A")',
838:       # check the penultimate page. If it contains exactly 1 "Transfer Player"
839:       # command and does NOT contain a "Change Transparent Flag" command, rewrite
840:       # both the penultimate page and the last page.
841:       if mapData.switchName(lastPage.condition.switch1_id) == 's:tsOff?("A")'
842:         list = event.pages[event.pages.length - 2].list
843:         transferCommand = list.find_all { |cmd| cmd.code == 201 }   # Transfer Player
844:         if transferCommand.length == 1 && list.none? { |cmd| cmd.code == 208 }   # Change Transparent Flag
845:           # Rewrite penultimate page
846:           list.clear
847:           push_move_route_and_wait(   # Move Route for door opening
848:             list, 0,
849:             [PBMoveRoute::PLAY_SE, RPG::AudioFile.new("Door enter"), PBMoveRoute::WAIT, 2,

857:           )
858:           push_event(list, 208, [0])   # Change Transparent Flag (invisible)
859:           push_script(list, "Followers.follow_into_door")
860:           push_event(list, 210, [])   # Wait for Move's Completion
