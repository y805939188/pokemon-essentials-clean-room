90:   def self.aggregate_from_folder(path, scripts, level = 0)
91:     files = []
92:     folders = []
93:     Dir.foreach(path) do |f|
94:       next if f == '.' || f == '..'
95:       if File.directory?(path + "/" + f)
96:         folders.push(f) if !f[/^\./]
97:       else
98:         files.push(f) if f[/\.rb$/i]
99:       end
100:     end
101:     # Aggregate individual script files into Scripts.rxdata
102:     files.sort!
103:     files.each do |f|
104:       section_name = filename_to_title(f)
105:       content = File.open(path + "/" + f, "rb") { |f2| f2.read }#.gsub(/\n/, "\r\n")
106:       scripts << [rand(999_999), section_name, Zlib::Deflate.deflate(content)]
107:     end
108:     # Check each subfolder for scripts to aggregate
109:     folders.sort!
110:     folders.each do |f|
111:       section_name = filename_to_title(f)
112:       scripts << [rand(999_999), "==================", Zlib::Deflate.deflate("")] if level == 0
113:       scripts << [rand(999_999), "", Zlib::Deflate.deflate("")] if level == 1
114:       scripts << [rand(999_999), "[[ " + section_name + " ]]", Zlib::Deflate.deflate("")]

119:   def self.filename_to_title(filename)
120:     filename = filename.bytes.pack('U*')
121:     title = ""
122:     if filename[/^[^_]*_(.+)$/]
123:       title = $~[1]
124:       title = title[0..-4] if title.end_with?(".rb")
125:       title = title.strip
126:     end
127:     title = "unnamed" if !title || title.empty?
128:     title.gsub!(/&bs;/, "\\")
129:     title.gsub!(/&fs;/, "/")
130:     title.gsub!(/&cn;/, ":")
131:     title.gsub!(/&as;/, "*")
132:     title.gsub!(/&qm;/, "?")
133:     title.gsub!(/&dq;/, "\"")
134:     title.gsub!(/&lt;/, "<")
135:     title.gsub!(/&gt;/, ">")
136:     title.gsub!(/&po;/, "|")
137:     return title
138:   end

144:     filename.gsub!(/:/, "&cn;")

147:     filename.gsub!(/"/, "&dq;")
