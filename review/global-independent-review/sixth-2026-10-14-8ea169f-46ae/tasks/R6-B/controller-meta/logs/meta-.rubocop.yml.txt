102: # Style
103: #===============================================================================
104: 
105: # List the attr_reader/writer/accessor variables however you want.
106: Style/AccessorGrouping:
107:   Enabled: false
108: 
109: # The assign_to_condition style looks awful, indenting loads of lines and
110: # increasing the separation between variable and value being assigned to it.

113: # disabled because such assignment is fine.
114: Style/ConditionalAssignment:
115:   Enabled: false
116:   EnforcedStyle: assign_inside_condition
117: 
118: # Check with yard instead.
119: Style/Documentation:
120:   Enabled: false

123: Style/EndlessMethod:
124:   EnforcedStyle: disallow
125: 
126: # It's a choice between format and sprintf. We already make use of sprintf and
127: # the translatable _ISPRINTF, so...
128: Style/FormatString:
129:   EnforcedStyle: sprintf
130: 
131: # Prefer sprintf("%s", "Hello") over sprintf("%<greeting>s", greeting: "Hello")
132: # because it should be easy enough to see which token is which, and it saves
133: # space.
134: Style/FormatStringToken:
135:   EnforcedStyle: unannotated
136: 
137: # String literals are not frozen by default, which makes this comment a
138: # pointless bit of boilerplate that we neither need nor want.
139: Style/FrozenStringLiteralComment:
140:   Enabled: false
141: 
142: # RMXP and Essentials use lots of global variables.
143: Style/GlobalVars:
144:   Enabled: false
145: 
146: # Mixing the styles within a hash just looks silly.
147: Style/HashSyntax:
148:   EnforcedStyle: no_mixed_keys
149: 
150: # Sometimes you want to clearly separate sets of code, one per "paradigm".
151: Style/IfInsideElse:
152:   Enabled: false
153:   AllowIfModifier: true
154: 
155: # The alernative is ->(x) { x } which is less English than "lambda". This style
156: # makes lambda definitions require the word "lambda".
157: Style/Lambda:
158:   EnforcedStyle: lambda
159: 
160: # unless just adds mental gymnastics trying to figure out what it actually
161: # means. I much prefer if !something.
162: Style/NegatedIf:

174:   Enabled: false

177: Style/PreferredHashMethods:
178:   Enabled: false
179:

181: Style/RedundantReturn:
182:   Enabled: false
183: 
184: # Enforcing the names of variables? To single letter ones? Just no.
185: Style/SingleLineBlockParams:
186:   Enabled: false
187: 
188: # Single line methods use up less space, and they're easier to list next to each
189: # other and see that they behave similarly.
190: Style/SingleLineMethods:
191:   Enabled: false
