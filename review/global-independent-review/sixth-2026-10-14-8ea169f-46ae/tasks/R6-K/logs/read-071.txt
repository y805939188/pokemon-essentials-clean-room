276	    r = self.red
277	    g = self.green
278	    b = self.blue
279	    yuv = [
280	      (r * 0.299) + (g * 0.587) + (b * 0.114),
281	      (r * -0.1687) + (g * -0.3313) + (b *  0.500) + 0.5,
282	      (r * 0.500) + (g * -0.4187) + (b * -0.0813) + 0.5
283	    ]
284	    if yuv[0] < 127.5
285	      yuv[0] += (255 - yuv[0]) / 2
286	    else
287	      yuv[0] = yuv[0] / 2
288	    end
289	    return Color.new(
290	      yuv[0] + (1.4075 * (yuv[2] - 0.5)),
291	      yuv[0] - (0.3455 * (yuv[1] - 0.5)) - (0.7169 * (yuv[2] - 0.5)),

336	# Wrap code blocks in a class which passes data accessible as instance variables
337	# within the code block.
338	#
339	# wrapper = CallbackWrapper.new { puts @test }
340	# wrapper.set(test: "Hi")
341	# wrapper.execute  #=>  "Hi"
342	#===============================================================================
343	class CallbackWrapper
344	  @params = {}
345	
346	  def initialize(&block)
347	    @code_block = block
348	  end
349	
350	  def execute(given_block = nil, *args)
351	    execute_block = given_block || @code_block
352	    @params.each do |key, value|
353	      args.instance_variable_set("@#{key}", value)
354	    end
355	    args.instance_eval(&execute_block)
356	  end
357	
358	  def set(params = {})
359	    @params = params
360	  end
361	end

395	# Linear interpolation between two values, given the duration of the change and
396	# either:
397	#   - the time passed since the start of the change (delta), or
398	#   - the start time of the change (delta) and the current time (now)
399	#===============================================================================
400	def lerp(start_val, end_val, duration, delta, now = nil)
401	  return end_val if duration <= 0
402	  delta = now - delta if now
403	  return start_val if delta <= 0
404	  return end_val if delta >= duration
405	  return start_val + ((end_val - start_val) * delta / duration.to_f)
