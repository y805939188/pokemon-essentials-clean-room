6	  def getMegaForm
7	    ret = 0
8	    GameData::Species.each do |data|
9	      next if data.species != @species || data.unmega_form != form_simple
10	      if data.mega_stone && hasItem?(data.mega_stone)
11	        ret = data.form
12	        break
13	      elsif data.mega_move && hasMove?(data.mega_move)
14	        ret = data.form
15	        break
16	      end
17	    end
18	    return ret   # form number, or 0 if no accessible Mega form
19	  end
