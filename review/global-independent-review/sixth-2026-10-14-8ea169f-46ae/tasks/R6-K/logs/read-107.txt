3	    def self.check_graphic_file(path, species, form = 0, gender = 0, shiny = false, shadow = false, subfolder = "")
4	      try_subfolder = sprintf("%s/", subfolder)
5	      try_species = species
6	      try_form    = (form > 0) ? sprintf("_%d", form) : ""
7	      try_gender  = (gender == 1) ? "_female" : ""
8	      try_shadow  = (shadow) ? "_shadow" : ""
9	      factors = []
10	      factors.push([4, sprintf("%s shiny/", subfolder), try_subfolder]) if shiny
11	      factors.push([3, try_shadow, ""]) if shadow
12	      factors.push([2, try_gender, ""]) if gender == 1
13	      factors.push([1, try_form, ""]) if form > 0
14	      factors.push([0, try_species, "000"])
15	      # Go through each combination of parameters in turn to find an existing sprite
16	      (2**factors.length).times do |i|
17	        # Set try_ parameters for this combination
18	        factors.each_with_index do |factor, index|
19	          value = ((i / (2**index)).even?) ? factor[1] : factor[2]
20	          case factor[0]
21	          when 0 then try_species   = value
22	          when 1 then try_form      = value
23	          when 2 then try_gender    = value
24	          when 3 then try_shadow    = value
25	          when 4 then try_subfolder = value   # Shininess
26	          end
27	        end
28	        # Look for a graphic matching this combination's parameters
29	        try_species_text = try_species
30	        ret = pbResolveBitmap(sprintf("%s%s%s%s%s%s", path, try_subfolder,
31	                                      try_species_text, try_form, try_gender, try_shadow))
32	        return ret if ret
33	      end
34	      return nil
35	    end

93	      species = pkmn.species if !species
94	      species = GameData::Species.get(species).species   # Just to be sure it's a symbol
95	      return self.egg_sprite_bitmap(species, pkmn.form) if pkmn.egg?
96	      if back
97	        ret = self.back_sprite_bitmap(species, pkmn.form, pkmn.gender, pkmn.shiny?, pkmn.shadowPokemon?)
98	      else
99	        ret = self.front_sprite_bitmap(species, pkmn.form, pkmn.gender, pkmn.shiny?, pkmn.shadowPokemon?)
100	      end
101	      alter_bitmap_function = MultipleForms.getFunction(species, "alterBitmap")
102	      if ret && alter_bitmap_function
103	        new_ret = ret.copy
104	        ret.dispose
105	        new_ret.each { |bitmap| alter_bitmap_function.call(pkmn, bitmap) }
106	        ret = new_ret
107	      end
108	      return ret
109	    end

224	    def self.cry_length(species, form = 0, pitch = 100, suffix = "")
225	      pitch ||= 100
226	      return 0 if !species || pitch <= 0
227	      pitch = pitch.to_f / 100
228	      ret = 0.0
229	      if species.is_a?(Pokemon)
230	        if !species.egg?
231	          filename = self.cry_filename_from_pokemon(species, suffix)
232	          filename = self.cry_filename_from_pokemon(species) if !filename && !nil_or_empty?(suffix)
233	          filename = pbResolveAudioSE(filename)
234	          ret = getPlayTime(filename) if filename
235	        end
236	      else
237	        filename = self.cry_filename(species, form, suffix)
238	        filename = self.cry_filename(species, form) if !filename && !nil_or_empty?(suffix)
239	        filename = pbResolveAudioSE(filename)
240	        ret = getPlayTime(filename) if filename
241	      end
242	      ret /= pitch   # Sound played at a lower pitch lasts longer
243	      return ret
