6	    # This move's ID.
7	    attr_reader :id
8	    # The amount of PP remaining for this move.
9	    attr_reader :pp
10	    # The number of PP Ups used on this move (each one adds 20% to the total PP).
11	    attr_reader :ppup
12	
13	    # Creates a new Move object.
14	    # @param move_id [Symbol, String, GameData::Move] move ID
15	    def initialize(move_id)
16	      @id   = GameData::Move.get(move_id).id
17	      @ppup = 0
18	      @pp   = total_pp
19	    end
20	
21	    # Sets this move's ID, and caps the PP amount if it is now greater than this
22	    # move's total PP.
23	    # @param value [Symbol, String, GameData::Move] the new move ID
24	    def id=(value)
25	      @id = GameData::Move.get(value).id
26	      @pp = @pp.clamp(0, total_pp)
27	    end
28	
29	    # Sets this move's PP, capping it at this move's total PP.
30	    # @param value [Integer] the new PP amount
31	    def pp=(value)
32	      @pp = value.clamp(0, total_pp)
33	    end
34	
35	    # Sets this move's PP Up count, and caps the PP if necessary.
36	    # @param value [Integer] the new PP Up value
37	    def ppup=(value)
38	      @ppup = value
39	      @pp = @pp.clamp(0, total_pp)
40	    end
41	
42	    # Returns the total PP of this move, taking PP Ups into account.
43	    # @return [Integer] total PP
44	    def total_pp
45	      max_pp = GameData::Move.get(@id).total_pp
46	      return max_pp + (max_pp * @ppup / 5)
47	    end
48	    alias totalpp total_pp
