57	  pbMessage(_INTL("{1} obtained {2}!", $player.name, species_name) + "\\me[Pkmn get]\\wtnp[80]")
58	  was_owned = $player.owned?(pkmn.species)
59	  $player.pokedex.set_seen(pkmn.species)
60	  $player.pokedex.set_owned(pkmn.species)
61	  $player.pokedex.register(pkmn) if see_form
62	  # Show Pokédex entry for new species if it hasn't been owned before
63	  if Settings::SHOW_NEW_SPECIES_POKEDEX_ENTRY_MORE_OFTEN && see_form && !was_owned &&
64	     $player.has_pokedex && $player.pokedex.species_in_unlocked_dex?(pkmn.species)
65	    pbMessage(_INTL("{1}'s data was added to the Pokédex.", species_name))
66	    $player.pokedex.register_last_seen(pkmn)
67	    pbFadeOutIn do
68	      scene = PokemonPokedexInfo_Scene.new
69	      screen = PokemonPokedexInfoScreen.new(scene)
70	      screen.pbDexEntry(pkmn.species)
71	    end
72	  end
73	  # Nickname and add the Pokémon
74	  pbNicknameAndStore(pkmn)
75	  return true

194	def pbBalancedLevel(party)
195	  return 1 if party.length == 0
196	  # Calculate the mean of all levels
197	  sum = 0
198	  party.each { |p| sum += p.level }
199	  return 1 if sum == 0
200	  mLevel = GameData::GrowthRate.max_level
201	  average = sum.to_f / party.length
202	  # Calculate the standard deviation
203	  varianceTimesN = 0
204	  party.each do |pkmn|
205	    deviation = pkmn.level - average
206	    varianceTimesN += deviation * deviation
207	  end
208	  # NOTE: This is the "population" standard deviation calculation, since no
209	  # sample is being taken.
210	  stdev = Math.sqrt(varianceTimesN / party.length)
211	  mean = 0
212	  weights = []
213	  # Skew weights according to standard deviation
214	  party.each do |pkmn|
215	    weight = pkmn.level.to_f / sum
216	    if weight < 0.5
217	      weight -= (stdev / mLevel.to_f)
218	      weight = 0.001 if weight <= 0.001
219	    else
220	      weight += (stdev / mLevel.to_f)
221	      weight = 0.999 if weight >= 0.999
222	    end
223	    weights.push(weight)
224	  end
225	  weightSum = 0
226	  weights.each { |w| weightSum += w }
227	  # Calculate the weighted mean, assigning each weight to each level's
228	  # contribution to the sum
229	  party.each_with_index { |pkmn, i| mean += pkmn.level * weights[i] }
230	  mean /= weightSum
231	  mean = mean.round
232	  mean = 1 if mean < 1
233	  # Add 2 to the mean to challenge the player
234	  mean += 2
235	  # Adjust level to maximum
236	  mean = mLevel if mean > mLevel
237	  return mean
238	end

242	#===============================================================================
243	def pbSize(pkmn)
244	  baseheight = pkmn.height
245	  hpiv = pkmn.iv[:HP] & 15
246	  ativ = pkmn.iv[:ATTACK] & 15
247	  dfiv = pkmn.iv[:DEFENSE] & 15
248	  saiv = pkmn.iv[:SPECIAL_ATTACK] & 15
249	  sdiv = pkmn.iv[:SPECIAL_DEFENSE] & 15
250	  spiv = pkmn.iv[:SPEED] & 15
251	  m = pkmn.personalID & 0xFF
252	  n = (pkmn.personalID >> 8) & 0xFF
253	  s = ((((ativ ^ dfiv) * hpiv) ^ m) * 256) + (((saiv ^ sdiv) * spiv) ^ n)
254	  xyz = [1700, 1, 65_510]
255	  case s
256	  when 0...10          then xyz = [ 290,   1,      0]
257	  when 10...110        then xyz = [ 300,   1,     10]
258	  when 110...310       then xyz = [ 400,   2,    110]
259	  when 310...710       then xyz = [ 500,   4,    310]
260	  when 710...2710      then xyz = [ 600,  20,    710]
261	  when 2710...7710     then xyz = [ 700,  50,   2710]
262	  when 7710...17_710   then xyz = [ 800, 100,   7710]
263	  when 17_710...32_710 then xyz = [ 900, 150, 17_710]
264	  when 32_710...47_710 then xyz = [1000, 150, 32_710]
265	  when 47_710...57_710 then xyz = [1100, 100, 47_710]
266	  when 57_710...62_710 then xyz = [1200,  50, 57_710]
267	  when 62_710...64_710 then xyz = [1300,  20, 62_710]
268	  when 64_710...65_210 then xyz = [1400,   5, 64_710]
269	  when 65_210...65_410 then xyz = [1500,   2, 65_210]
270	  end
271	  return ((((s - xyz[2]) / xyz[1]) + xyz[0]).floor * baseheight / 10).floor
272	end

275	# Returns true if the given species can be legitimately obtained as an egg
276	#===============================================================================
277	def pbHasEgg?(species)
278	  species_data = GameData::Species.try_get(species)
279	  return false if !species_data
280	  species = species_data.species
281	  # species may be unbreedable, so check its evolution's compatibilities
282	  evoSpecies = species_data.get_evolutions(true)
283	  compatSpecies = (evoSpecies && evoSpecies[0]) ? evoSpecies[0][0] : species
284	  species_data = GameData::Species.try_get(compatSpecies)
285	  compat = species_data.egg_groups
286	  return false if compat.include?(:Undiscovered) || compat.include?(:Ditto)
287	  baby = GameData::Species.get(species).get_baby_species
288	  return true if species == baby   # Is a basic species
289	  baby = GameData::Species.get(species).get_baby_species(true)
290	  return true if species == baby   # Is an egg species without incense
291	  return false
