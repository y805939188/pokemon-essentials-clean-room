452	def pbMoveTutorChoose(move, movelist = nil, bymachine = false, oneusemachine = false)
453	  ret = false
454	  move = GameData::Move.get(move).id
455	  if movelist.is_a?(Array)
456	    movelist.map! { |m| GameData::Move.get(m).id }
457	  end
458	  pbFadeOutIn do
459	    movename = GameData::Move.get(move).name
460	    annot = pbMoveTutorAnnotations(move, movelist)
461	    scene = PokemonParty_Scene.new
462	    screen = PokemonPartyScreen.new(scene, $player.party)
463	    screen.pbStartScene(_INTL("Teach which Pokémon?"), false, annot)
464	    loop do
465	      chosen = screen.pbChoosePokemon
466	      break if chosen < 0
467	      pokemon = $player.party[chosen]
468	      if pokemon.egg?
469	        pbMessage(_INTL("Eggs can't be taught any moves.")) { screen.pbUpdate }
470	      elsif pokemon.shadowPokemon?
471	        pbMessage(_INTL("Shadow Pokémon can't be taught any moves.")) { screen.pbUpdate }
472	      elsif movelist && movelist.none? { |j| j == pokemon.species }
473	        pbMessage(_INTL("{1} can't learn {2}.", pokemon.name, movename)) { screen.pbUpdate }
474	      elsif !pokemon.compatible_with_move?(move)
475	        pbMessage(_INTL("{1} can't learn {2}.", pokemon.name, movename)) { screen.pbUpdate }
476	      elsif pbLearnMove(pokemon, move, false, bymachine) { screen.pbUpdate }
477	        $stats.moves_taught_by_item += 1 if bymachine
478	        $stats.moves_taught_by_tutor += 1 if !bymachine
479	        pokemon.add_first_move(move) if oneusemachine
480	        ret = true
481	        break
482	      end
483	    end
484	    screen.pbEndScene
485	  end
486	  return ret   # Returns whether the move was learned by a Pokemon
487	end
