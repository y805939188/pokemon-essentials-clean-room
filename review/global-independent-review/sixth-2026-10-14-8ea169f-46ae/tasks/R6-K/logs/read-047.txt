8	  if !ret
9	    # Check map metadata

11	    ret = pbStringToAudioFile(music) if music && music != ""
12	  end

14	    # Check global metadata

63	def pbPlayTrainerIntroBGM(trainer_type)
64	  trainer_type_data = GameData::TrainerType.get(trainer_type)
65	  return if nil_or_empty?(trainer_type_data.intro_BGM)
66	  bgm = pbStringToAudioFile(trainer_type_data.intro_BGM)
67	  if !$game_temp.memorized_bgm
68	    if $game_temp.cue_bgm_delay
69	      $game_temp.cue_bgm_delay = nil
70	      $game_temp.memorized_bgm = $game_temp.cue_bgm
71	      $game_temp.memorized_bgm_position = 0
72	    else
73	      $game_temp.memorized_bgm = $game_system.getPlayingBGM
74	      $game_temp.memorized_bgm_position = (Audio.bgm_pos rescue 0)
75	    end
76	  end
77	  pbBGMPlay(bgm)
78	end
