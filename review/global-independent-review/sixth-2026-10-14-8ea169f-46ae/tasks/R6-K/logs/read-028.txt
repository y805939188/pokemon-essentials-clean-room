562	MultipleForms.register(:NECROZMA, {
563	  "getFormOnLeavingBattle" => proc { |pkmn, battle, usedInBattle, endBattle|
564	    # Fused forms are 1 and 2, Ultra form is 3 or 4 depending on which fusion
565	    next pkmn.form - 2 if pkmn.form >= 3 && (pkmn.fainted? || endBattle)
566	  },
567	  "onSetForm" => proc { |pkmn, form, oldForm|
568	    next if form > 2 || oldForm > 2   # Ultra form changes don't affect moveset
569	    form_moves = [
570	      :SUNSTEELSTRIKE,   # Dusk Mane (with Solgaleo) (form 1)
571	      :MOONGEISTBEAM     # Dawn Wings (with Lunala) (form 2)
572	    ]
573	    if form == 0   # Normal
574	      # Turned back into the base form; forget form-specific moves
575	      form_moves.each do |move|
576	        next if !pkmn.hasMove?(move)
577	        pkmn.forget_move(move)
578	        pbMessage(_INTL("{1} forgot {2}...", pkmn.name, GameData::Move.get(move).name))
579	      end
580	      pbLearnMove(pkmn, :CONFUSION) if pkmn.numMoves == 0
581	    else   # Dusk Mane, Dawn Wings
582	      # Turned into an alternate form; try learning that form's unique move
583	      new_move_id = form_moves[form - 1]
584	      pbLearnMove(pkmn, new_move_id, true)
585	    end
586	  }
587	})
