452	    def generate_layout(maxWidth, maxHeight)
453	      cellWidth = @parameters.cell_width
454	      cellHeight = @parameters.cell_height
455	      # Map is too small, make the whole map a room
456	      if maxWidth < cellWidth || maxHeight < cellHeight
457	        paint_ground_rect(@buffer_x, @buffer_y, maxWidth, maxHeight, :room)
458	        return
459	      end
460	      # Generate connections between cells
461	      maze = Maze.new(maxWidth / cellWidth, maxHeight / cellHeight, @parameters)
462	      maze.generate_layout
463	      # If no rooms were generated, make the whole map a room
464	      if maze.room_count == 0
465	        paint_ground_rect(@buffer_x, @buffer_y, maxWidth, maxHeight, :room)
466	        return
467	      end
468	      # Draw each cell's contents in turn (room and corridors)
469	      (maxHeight / cellHeight).times do |y|
470	        (maxWidth / cellWidth).times do |x|
471	          paint_node_contents(@buffer_x + (x * cellWidth), @buffer_y + (y * cellHeight), maze.get_node(x, y))
472	        end
473	      end
474	      check_for_isolated_rooms
