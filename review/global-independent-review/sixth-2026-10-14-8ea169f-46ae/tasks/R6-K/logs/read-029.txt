699	MultipleForms.register(:CALYREX, {
700	  "onSetForm" => proc { |pkmn, form, oldForm|
701	    form_moves = [
702	      :GLACIALLANCE,   # Ice Rider (with Glastrier) (form 1)
703	      :ASTRALBARRAGE   # Shadow Rider (with Spectrier) (form 2)
704	    ]
705	    if form == 0   # Normal
706	      # Forget special form moves
707	      form_moves.each do |move|
708	        next if !pkmn.hasMove?(move)
709	        pkmn.forget_move(move)
710	        pbMessage(_INTL("{1} forgot {2}...", pkmn.name, GameData::Move.get(move).name))
711	      end
712	      # Forget all other moves not accessible to the base form
713	      sp_data = pkmn.species_data
714	      pkmn.moves.each_with_index do |move, i|
715	        next if sp_data.moves.any? { |learn_move| learn_move[1] == move.id }
716	        next if sp_data.tutor_moves.include?(move.id)
717	        next if sp_data.egg_moves.include?(move.id)
718	        pbMessage(_INTL("{1} forgot {2}...", pkmn.name, move.name))
719	        pkmn.moves[i] = nil
720	      end
721	      pkmn.moves.compact!
722	      # Ensure pkmn has at least one move in the end
723	      pbLearnMove(pkmn, :CONFUSION) if pkmn.numMoves == 0
724	    else   # Ice Rider, Shadow Rider
725	      new_move = form_moves[form - 1]
726	      pbLearnMove(pkmn, new_move, true)
727	    end
728	  }
729	})

732	# Regional forms
733	# This code is for determining the form of a Pokémon in an egg created at the
734	# Day Care, where that Pokémon's species has regional forms. The regional form
735	# chosen depends on the region in which the egg was produced (not where it
736	# hatches).
737	#===============================================================================
738	
739	# The code in this proc assumes that the appropriate regional form for a Pokémon
740	# is equal to the region's number. This may not be true in your game.
741	# Note that this proc only produces a non-zero form number if the species has a
742	# defined form with that number, which means it can be used for both Alolan and
743	# Galarian forms separately (and for Meowth which has both).
744	MultipleForms.register(:RATTATA, {
745	  "getFormOnEggCreation" => proc { |pkmn|
746	    if $game_map
747	      map_pos = $game_map.metadata&.town_map_position
748	      next map_pos[0] if map_pos &&
749	                         GameData::Species.get_species_form(pkmn.species, map_pos[0]).form == map_pos[0]
750	    end
751	    next 0
752	  }
753	})
