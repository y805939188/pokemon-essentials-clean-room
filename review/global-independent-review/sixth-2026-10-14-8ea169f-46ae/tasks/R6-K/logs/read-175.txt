352	#===============================================================================
353	class Battle::Move::UserLosesHalfHP < Battle::Move::RecoilMove
354	  def pbRecoilDamage(user, target)
355	    return (user.hp / 2.0).round
356	  end
357	
358	  def pbEffectAfterAllHits(user, target)
359	    return if user.fainted? || target.damageState.unaffected
360	    # NOTE: This move's recoil is not prevented by Rock Head/Magic Guard.
361	    amt = pbRecoilDamage(user, target)
362	    amt = 1 if amt < 1
363	    user.pbReduceHP(amt, false)
364	    @battle.pbDisplay(_INTL("{1} is damaged by recoil!", user.pbThis))
365	    user.pbItemHPHealCheck
366	  end

394	      return true
