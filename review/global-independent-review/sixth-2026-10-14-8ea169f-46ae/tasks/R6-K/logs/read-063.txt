1	# The Deprecation module is used to warn game & plugin creators of deprecated
2	# methods.
3	module Deprecation
4	  module_function
5	
6	  # Sends a warning of a deprecated method into the debug console.
7	  # @param method_name [String] name of the deprecated method
8	  # @param removal_version [String] version the method is removed in
9	  # @param alternative [String] preferred alternative method
10	  def warn_method(method_name, removal_version = nil, alternative = nil)
11	    text = _INTL('Usage of deprecated method "{1}" or its alias.', method_name)
12	    unless removal_version.nil?
13	      text += "\n" + _INTL("The method is slated to be removed in Essentials {1}.", removal_version)
14	    end
15	    unless alternative.nil?
16	      text += "\n" + _INTL("Use \"{1}\" instead.", alternative)
17	    end
18	    Console.echo_warn text
19	  end
20	end

22	# The Module class is extended to allow easy deprecation of instance and class methods.
23	class Module
24	  private
25	
26	  # Creates a deprecated alias for a method.
27	  # Using it sends a warning to the debug console.
28	  # @param name [Symbol] name of the new alias
29	  # @param aliased_method [Symbol] name of the aliased method
30	  # @param removal_in [String] version the alias is removed in
31	  # @param class_method [Boolean] whether the method is a class method
32	  def deprecated_method_alias(name, aliased_method, removal_in: nil, class_method: false)
33	    validate name => Symbol, aliased_method => Symbol, removal_in => [NilClass, String],
34	             class_method => [TrueClass, FalseClass]
35	
36	    target = class_method ? self.class : self
37	    class_name = self.name
38	
39	    unless target.method_defined?(aliased_method)
40	      raise ArgumentError, "#{class_name} does not have method #{aliased_method} defined"
41	    end
42	
43	    delimiter = class_method ? "." : "#"
44	
45	    target.define_method(name) do |*args, **kvargs|
46	      alias_name = sprintf("%s%s%s", class_name, delimiter, name)
47	      aliased_method_name = sprintf("%s%s%s", class_name, delimiter, aliased_method)
48	      Deprecation.warn_method(alias_name, removal_in, aliased_method_name)
49	      method(aliased_method).call(*args, **kvargs)
50	    end
51	  end
52	end
