653	  def priv_add_to_hash(type, array, ret, map_id = 0)
654	    if type == MessageTypes::EVENT_TEXTS
655	      @default_core_messages[type] ||= []
656	      @default_core_messages[type][map_id] ||= {}
657	      default_keys = @default_core_messages[type][map_id].keys
658	    else
659	      @default_core_messages[type] ||= {}
660	      default_keys = @default_core_messages[type].keys
661	    end
662	    ret = {} if !ret
663	    array.each do |text|
664	      next if !text
665	      key = Translation.stringToKey(text)
666	      ret[key] = text if !default_keys.include?(key)
667	    end
668	    return ret

772	# Replaces first argument with a localized version and formats the other

774	def _INTL(*arg)
775	  begin
776	    string = MessageTypes.getFromHash(MessageTypes::SCRIPT_TEXTS, arg[0])
777	  rescue

787	# Replaces first argument with a localized version and formats the other
788	# parameters by replacing {1}, {2}, etc. with those placeholders.
789	# This version acts more like sprintf, supports e.g. {1:d} or {2:s}
790	def _ISPRINTF(*arg)

792	    string = MessageTypes.getFromHash(MessageTypes::SCRIPT_TEXTS, arg[0])
793	  rescue
794	    string = arg[0]
795	  end
796	  string = string.clone
797	  (1...arg.length).each do |i|
798	    string.gsub!(/\{#{i}\:([^\}]+?)\}/) { |m| next sprintf("%" + $1, arg[i]) }
799	  end
800	  return string
