1	# The SaveData module is used to manipulate save data. It contains the {Value}s
2	# that make up the save data and {Conversion}s for resolving incompatibilities
3	# between Essentials and game versions.
4	# @see SaveData.register
5	# @see SaveData.register_conversion
6	module SaveData
7	  # Contains the file path of the save file.
8	  FILE_PATH = if File.directory?(System.data_directory)
9	                System.data_directory + "/Game.rxdata"
10	              else
11	                "./Game.rxdata"
12	              end
13	
14	  # @return [Boolean] whether the save file exists
15	  def self.exists?
16	    return File.file?(FILE_PATH)
17	  end
18	
19	  # Fetches the save data from the given file.
20	  # Returns an Array in the case of a pre-v19 save file.
21	  # @param file_path [String] path of the file to load from
22	  # @return [Hash, Array] loaded save data
23	  # @raise [IOError, SystemCallError] if file opening fails
24	  def self.get_data_from_file(file_path)
25	    validate file_path => String
26	    save_data = nil
27	    File.open(file_path) do |file|
28	      data = Marshal.load(file)
29	      if data.is_a?(Hash)
30	        save_data = data
31	        next
32	      end
33	      save_data = [data]
34	      save_data << Marshal.load(file) until file.eof?
35	    end
36	    return save_data
37	  end
38	
39	  # Fetches save data from the given file. If it needed converting, resaves it.
40	  # @param file_path [String] path of the file to read from
41	  # @return [Hash] save data in Hash format
42	  # @raise (see .get_data_from_file)
43	  def self.read_from_file(file_path)
44	    validate file_path => String
45	    save_data = get_data_from_file(file_path)
46	    save_data = to_hash_format(save_data) if save_data.is_a?(Array)
47	    if !save_data.empty? && run_conversions(save_data)
48	      File.open(file_path, "wb") { |file| Marshal.dump(save_data, file) }
49	    end
50	    return save_data
51	  end
52	
53	  # Compiles the save data and saves a marshaled version of it into
54	  # the given file.
55	  # @param file_path [String] path of the file to save into
56	  # @raise [InvalidValueError] if an invalid value is being saved
57	  def self.save_to_file(file_path)
58	    validate file_path => String
59	    save_data = self.compile_save_hash
60	    File.open(file_path, "wb") { |file| Marshal.dump(save_data, file) }
61	  end
62	
63	  # Deletes the save file (and a possible .bak backup file if one exists)
64	  # @raise [Error::ENOENT]
65	  def self.delete_file
66	    File.delete(FILE_PATH)
67	    File.delete(FILE_PATH + ".bak") if File.file?(FILE_PATH + ".bak")
68	  end
69	
70	  # Converts the pre-v19 format data to the new format.
71	  # @param old_format [Array] pre-v19 format save data
72	  # @return [Hash] save data in new format
73	  def self.to_hash_format(old_format)
74	    validate old_format => Array
75	    hash = {}
76	    @values.each do |value|
77	      data = value.get_from_old_format(old_format)
78	      hash[value.id] = data unless data.nil?
79	    end
80	    return hash
81	  end
82	end
