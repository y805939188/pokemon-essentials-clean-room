1	# The pbs_order value determines the order in which the stats are written in
2	# several PBS files, where base stats/IVs/EVs/EV yields are defined. Only stats
3	# which are yielded by the "each_main" method can have stat numbers defined in
4	# those places. The values of pbs_order defined below should start with 0 and
5	# increase without skipping any numbers.
6	module GameData
7	  class Stat
8	    attr_reader :id
9	    attr_reader :real_name
10	    attr_reader :real_name_brief
11	    attr_reader :type
12	    attr_reader :pbs_order
13	
14	    DATA = {}
15	
16	    extend ClassMethodsSymbols
17	    include InstanceMethods
18	
19	    def self.load; end
20	    def self.save; end
21	
22	    # These stats are defined in PBS files, and should have the :pbs_order
23	    # property.
24	    def self.each_main
25	      self.each { |s| yield s if [:main, :main_battle].include?(s.type) }
26	    end
27	
28	    def self.each_main_battle
29	      self.each { |s| yield s if [:main_battle].include?(s.type) }
30	    end
31	
32	    # These stats have associated stat stages in battle.
33	    def self.each_battle
34	      self.each { |s| yield s if [:main_battle, :battle].include?(s.type) }
35	    end
36	
37	    def initialize(hash)
38	      @id              = hash[:id]
39	      @real_name       = hash[:name]       || "Unnamed"
40	      @real_name_brief = hash[:name_brief] || "None"
41	      @type            = hash[:type]       || :none
42	      @pbs_order       = hash[:pbs_order]  || -1
43	    end
44	
45	    # @return [String] the translated name of this stat
46	    def name
47	      return _INTL(@real_name)
48	    end
49	
50	    # @return [String] the translated brief name of this stat
51	    def name_brief
52	      return _INTL(@real_name_brief)
53	    end
54	  end
55	end
