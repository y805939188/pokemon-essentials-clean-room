361	  def pbPlace(pkmn, position)
362	    return false if !pkmn
363	    if pkmn.egg?
364	      @scene.pbDisplay(_INTL("Can't place an egg there."))
365	      return false
366	    end
367	    if position == 0
368	      if pkmn.shadowPokemon?
369	        # Remove from storage and place in set
370	        oldpkmn = PurifyChamberHelper.pbGetPokemon(@chamber, position)
371	        if oldpkmn
372	          @scene.pbShift(position, pkmn)
373	        else
374	          @scene.pbPlace(position, pkmn)
375	        end
376	        PurifyChamberHelper.pbSetPokemon(@chamber, position, pkmn)
377	        @scene.pbRefresh
378	      else
379	        @scene.pbDisplay(_INTL("Only a Shadow Pokémon can go there."))
380	        return false
381	      end
382	    elsif position >= 1
383	      if pkmn.shadowPokemon?
384	        @scene.pbDisplay(_INTL("Can't place a Shadow Pokémon there."))
385	        return false
386	      else
387	        oldpkmn = PurifyChamberHelper.pbGetPokemon(@chamber, position)
388	        if oldpkmn
389	          @scene.pbShift(position, pkmn)
390	        else
391	          @scene.pbPlace(position, pkmn)
392	        end
393	        PurifyChamberHelper.pbSetPokemon(@chamber, position, pkmn)
394	        @scene.pbRefresh
395	      end
396	    end
397	    return true
398	  end

401	    return false if !pos

403	    if pbPlace(pkmn, position)
404	      $PokemonStorage.pbDelete(pos[0], pos[1])
405	      return true
406	    end
407	    return false

421	  def pbOpenSetDetail
422	    chamber = @chamber
423	    @scene.pbOpenSetDetail(chamber.currentSet)
424	    heldpkmn = nil
425	    loop do
426	      # Commands
427	      # array[0]==0 - a position was chosen
428	      # array[0]==1 - a new set was chosen
429	      # array[0]==2 - choose Pokemon command
430	      cmd = @scene.pbSetScreen
431	      case cmd[0]
432	      when 0   # Place Pokemon in the set
433	        curpkmn = PurifyChamberHelper.pbGetPokemon(@chamber, cmd[1])
434	        if curpkmn || heldpkmn
435	          commands = [_INTL("MOVE"), _INTL("SUMMARY"), _INTL("WITHDRAW")]
436	          if curpkmn && heldpkmn
437	            commands[0] = _INTL("EXCHANGE")
438	          elsif heldpkmn
439	            commands[0] = _INTL("PLACE")
440	          end
441	          cmdReplace = -1
442	          cmdRotate = -1
443	          if !heldpkmn && curpkmn && cmd[1] == 0 &&
444	             @chamber[@chamber.currentSet].length > 0
445	            commands[cmdRotate = commands.length] = _INTL("ROTATE")
446	          end
447	          if !heldpkmn && curpkmn
448	            commands[cmdReplace = commands.length] = _INTL("REPLACE")
449	          end
450	          commands.push(_INTL("CANCEL"))
451	          choice = @scene.pbShowCommands(
452	            _INTL("What shall I do with this {1}?", heldpkmn ? heldpkmn.name : curpkmn.name),
453	            commands
454	          )
455	          if choice == 0
456	            if heldpkmn
457	              if pbPlace(heldpkmn, cmd[1]) # calls place or shift as appropriate
458	                if curpkmn
459	                  heldpkmn = curpkmn # Pokemon was shifted
460	                else
461	                  pbOnPlace(heldpkmn)
462	                  @scene.pbPositionHint(PurifyChamberHelper.adjustOnInsert(cmd[1]))
463	                  heldpkmn = nil # Pokemon was placed
464	                end
465	              end
466	            else
467	              @scene.pbMove(cmd[1])
468	              PurifyChamberHelper.pbSetPokemon(@chamber, cmd[1], nil)
469	              @scene.pbRefresh
470	              heldpkmn = curpkmn
471	            end
472	          elsif choice == 1

477	            elsif heldpkmn
478	              @scene.pbWithdraw(cmd[1], heldpkmn)
479	              $PokemonStorage.pbStoreCaught(heldpkmn)
480	              heldpkmn = nil
481	              @scene.pbRefresh
482	            else

484	              @scene.pbWithdraw(cmd[1], heldpkmn)

492	            if count > 0

511	              end
512	            end
513	          end
514	        else   # No current Pokemon

570	  def pbDoPurify
571	    purifiables = []
572	    PurifyChamber::NUMSETS.times do |set|
573	      if @chamber.isPurifiable?(set) # if ready for purification
574	        purifiables.push(set)
575	      end
576	    end
577	    purifiables.length.times do |i|
578	      set = purifiables[i]
579	      @chamber.currentSet = set
580	      @scene.pbOpenSet(set)
581	      @scene.pbPurify
582	      pbPurify(@chamber[set].shadow, self)
583	      pbStorePokemon(@chamber[set].shadow)
584	      @chamber.setShadow(set, nil) # Remove shadow Pokemon from set
585	      if (i + 1) != purifiables.length
586	        @scene.pbDisplay(_INTL("There is another Pokémon that is ready to open its heart!") + "\1")
587	        if !@scene.pbConfirm(_INTL("Would you like to switch sets?"))
588	          @scene.pbCloseSet
589	          break
590	        end
591	      end
592	      @scene.pbCloseSet
593	    end
594	  end
