2	# Walking charset, for use in text entry screens and load game screen
3	#===============================================================================
4	class TrainerWalkingCharSprite < Sprite
5	  attr_accessor :anim_duration
6	
7	  # Default time in seconds for one animation cycle of a charset. The icon for a
8	  # storage box is 0.4 instead (set manually).
9	  ANIMATION_DURATION = 0.5
10	
11	  def initialize(charset, viewport = nil)
12	    super(viewport)
13	    @animbitmap = nil
14	    self.charset = charset
15	    @current_frame = 0   # Current pattern
16	    @anim_duration = ANIMATION_DURATION
17	  end
18	
19	  def dispose
20	    @animbitmap&.dispose
21	    super
22	  end
23	
24	  def charset=(value)
25	    @animbitmap&.dispose
26	    @animbitmap = nil
27	    bitmapFileName = sprintf("Graphics/Characters/%s", value)
28	    @charset = pbResolveBitmap(bitmapFileName)
29	    if @charset
30	      @animbitmap = AnimatedBitmap.new(@charset)
31	      self.bitmap = @animbitmap.bitmap
32	      self.src_rect.set(0, 0, self.bitmap.width / 4, self.bitmap.height / 4)
33	    else
34	      self.bitmap = nil
35	    end
36	  end
37	
38	  # Used for the box icon in the naming screen.
39	  def altcharset=(value)
40	    @animbitmap&.dispose
41	    @animbitmap = nil
42	    @charset = pbResolveBitmap(value)
43	    if @charset
44	      @animbitmap = AnimatedBitmap.new(@charset)
45	      self.bitmap = @animbitmap.bitmap
46	      self.src_rect.set(0, 0, self.bitmap.width / 4, self.bitmap.height)
47	    else
48	      self.bitmap = nil
49	    end
50	  end
51	
52	  def update_frame
53	    @current_frame = (4 * (System.uptime % @anim_duration) / @anim_duration).floor
54	  end
55	
56	  def update
57	    @updating = true
58	    super
59	    if @animbitmap
60	      @animbitmap.update
61	      self.bitmap = @animbitmap.bitmap
62	    end
63	    # Update animation
64	    update_frame
65	    self.src_rect.x = self.src_rect.width * @current_frame
66	    @updating = false
67	  end
68	end
