4	module MultipleForms
5	  @@formSpecies = SpeciesHandlerHash.new
6	
7	  def self.copy(sym, *syms)
8	    @@formSpecies.copy(sym, *syms)
9	  end
10	
11	  def self.register(sym, hash)
12	    @@formSpecies.add(sym, hash)
13	  end
14	
15	  def self.registerIf(sym, cond, hash)
16	    @@formSpecies.addIf(sym, cond, hash)
17	  end
18	
19	  def self.hasFunction?(pkmn, func)
20	    spec = (pkmn.is_a?(Pokemon)) ? pkmn.species : pkmn
21	    sp = @@formSpecies[spec]
22	    return sp && sp[func]
23	  end
24	
25	  def self.getFunction(pkmn, func)
26	    spec = (pkmn.is_a?(Pokemon)) ? pkmn.species : pkmn
27	    sp = @@formSpecies[spec]
28	    return (sp && sp[func]) ? sp[func] : nil
29	  end
30	
31	  def self.call(func, pkmn, *args)
32	    sp = @@formSpecies[pkmn.species]
33	    return nil if !sp || !sp[func]
34	    return sp[func].call(pkmn, *args)
35	  end
36	end

41	def drawSpot(bitmap, spotpattern, x, y, red, green, blue)
42	  height = spotpattern.length
43	  width  = spotpattern[0].length
44	  height.times do |yy|
45	    spot = spotpattern[yy]
46	    width.times do |xx|
47	      next if spot[xx] != 1
48	      xOrg = (x + xx) * 2
49	      yOrg = (y + yy) * 2
50	      color = bitmap.get_pixel(xOrg, yOrg)
51	      r = color.red + red
52	      g = color.green + green
53	      b = color.blue + blue
54	      color.red   = [[r, 0].max, 255].min
55	      color.green = [[g, 0].max, 255].min
56	      color.blue  = [[b, 0].max, 255].min
57	      bitmap.set_pixel(xOrg, yOrg, color)
58	      bitmap.set_pixel(xOrg + 1, yOrg, color)
59	      bitmap.set_pixel(xOrg, yOrg + 1, color)
60	      bitmap.set_pixel(xOrg + 1, yOrg + 1, color)
61	    end
62	  end
63	end
