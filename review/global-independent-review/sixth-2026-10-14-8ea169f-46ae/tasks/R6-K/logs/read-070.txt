35	  def cut(bitmap, width)
36	    string = self
37	    width -= bitmap.text_size("...").width
38	    string_width = 0
39	    text = []
40	    string.scan(/./).each do |char|
41	      wdh = bitmap.text_size(char).width
42	      next if (wdh + string_width) > width
43	      string_width += wdh
44	      text.push(char)
45	    end
46	    text.push("...") if text.length < string.length
47	    new_string = ""
48	    text.each do |char|
49	      new_string += char
50	    end
51	    return new_string
52	  end

101	  def deep_merge(hash)
102	    merged_hash = self.clone
103	    merged_hash.deep_merge!(hash) if hash.is_a?(Hash)
104	    return merged_hash
105	  end
106	
107	  def deep_merge!(hash)
108	    # failsafe
109	    return unless hash.is_a?(Hash)
110	    hash.each do |key, val|
111	      if self[key].is_a?(Hash)
112	        self[key].deep_merge!(val)
113	      else
114	        self[key] = val
115	      end
116	    end
117	  end

146	  def self.copy(source, destination)
147	    data = ""
148	    t = System.uptime
149	    File.open(source, "rb") do |f|
150	      loop do
151	        r = f.read(4096)
152	        break if !r
153	        if System.uptime - t >= 5
154	          t += 5
155	          Graphics.update
156	        end
157	        data += r
158	      end
159	    end
160	    File.delete(destination) if File.file?(destination)
161	    f = File.new(destination, "wb")
162	    f.write data
163	    f.close
164	  end
165	
166	  # Copies the source to the destination and deletes the source.
167	  def self.move(source, destination)
168	    File.copy(source, destination)
169	    File.delete(source)
170	  end

180	  # New constructor, accepts RGB values as well as a hex number or string value.
181	  def initialize(*args)
182	    pbPrintException("Wrong number of arguments! At least 1 is needed!") if args.length < 1
183	    case args.length
184	    when 1
185	      case args.first
186	      when Integer
187	        hex = args.first.to_s(16)
188	      when String
189	        try_rgb_format = args.first.split(",")
190	        init_original(*try_rgb_format.map(&:to_i)) if try_rgb_format.length.between?(3, 4)
191	        hex = args.first.delete("#")
192	      end
193	      pbPrintException("Wrong type of argument given!") if !hex
194	      r = hex[0...2].to_i(16)
195	      g = hex[2...4].to_i(16)
196	      b = hex[4...6].to_i(16)
197	    when 3
198	      r, g, b = *args
199	    end
200	    init_original(r, g, b) if r && g && b
201	    init_original(*args)
202	  end
