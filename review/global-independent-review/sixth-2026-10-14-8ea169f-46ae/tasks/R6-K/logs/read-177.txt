384	# Applies to all items defined as an evolution stone.
385	# No need to add more code for new ones.
386	ItemHandlers::UseOnPokemon.addIf(:evolution_stones,
387	  proc { |item| GameData::Item.get(item).is_evolution_stone? },
388	  proc { |item, qty, pkmn, scene|
389	    if pkmn.shadowPokemon?
390	      scene.pbDisplay(_INTL("It won't have any effect."))
391	      next false
392	    end
393	    newspecies = pkmn.check_evolution_on_use_item(item)
394	    if newspecies
395	      pbFadeOutInWithMusic do
396	        evo = PokemonEvolutionScene.new
397	        evo.pbStartScreen(pkmn, newspecies)
398	        evo.pbEvolution(false)
399	        evo.pbEndScreen
400	        if scene.is_a?(PokemonPartyScreen)
401	          scene.pbRefreshAnnotations(proc { |p| !p.check_evolution_on_use_item(item).nil? })
402	          scene.pbRefresh
403	        end
404	      end
405	      next true
406	    end
407	    scene.pbDisplay(_INTL("It won't have any effect."))
408	    next false
409	  }
410	)
