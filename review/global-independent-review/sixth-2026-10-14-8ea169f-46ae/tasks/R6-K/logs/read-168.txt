55	  # Return value:
56	  # 0 - Item not used
57	  # 1 - Item used, don't end screen
58	  # 2 - Item used, end screen
59	  def self.triggerUseFromBag(item)
60	    return UseFromBag.trigger(item) if UseFromBag[item]
61	    # No UseFromBag handler exists; check the UseInField handler if present
62	    if UseInField[item]
63	      return (UseInField.trigger(item)) ? 1 : 0
64	    end
65	    return 0
66	  end

75	  # -1 - Item effect not found

77	  # 1  - Item used

80	    return (UseInField.trigger(item)) ? 1 : 0

132	def pbChangeLevel(pkmn, new_level, scene)
133	  new_level = new_level.clamp(1, GameData::GrowthRate.max_level)
134	  if pkmn.level == new_level
135	    if scene.is_a?(PokemonPartyScreen)
136	      scene.pbDisplay(_INTL("{1}'s level remained unchanged.", pkmn.name))
137	    else
138	      pbMessage(_INTL("{1}'s level remained unchanged.", pkmn.name))
139	    end
140	    return
141	  end
142	  old_level           = pkmn.level
143	  old_total_hp        = pkmn.totalhp
144	  old_attack          = pkmn.attack
145	  old_defense         = pkmn.defense
146	  old_special_attack  = pkmn.spatk
147	  old_special_defense = pkmn.spdef
148	  old_speed           = pkmn.speed
149	  pkmn.level = new_level
150	  pkmn.calc_stats
151	  pkmn.hp = 1 if new_level > old_level && pkmn.species_data.base_stats[:HP] == 1
152	  scene.pbRefresh
153	  if old_level > new_level
154	    if scene.is_a?(PokemonPartyScreen)
155	      scene.pbDisplay(_INTL("{1} dropped to Lv. {2}!", pkmn.name, pkmn.level))
156	    else
157	      pbMessage(_INTL("{1} dropped to Lv. {2}!", pkmn.name, pkmn.level))
158	    end
159	    total_hp_diff        = pkmn.totalhp - old_total_hp
160	    attack_diff          = pkmn.attack - old_attack
161	    defense_diff         = pkmn.defense - old_defense
162	    special_attack_diff  = pkmn.spatk - old_special_attack
163	    special_defense_diff = pkmn.spdef - old_special_defense
164	    speed_diff           = pkmn.speed - old_speed
165	    pbTopRightWindow(_INTL("Max. HP<r>{1}\nAttack<r>{2}\nDefense<r>{3}\nSp. Atk<r>{4}\nSp. Def<r>{5}\nSpeed<r>{6}",
166	                           total_hp_diff, attack_diff, defense_diff, special_attack_diff, special_defense_diff, speed_diff), scene)
167	    pbTopRightWindow(_INTL("Max. HP<r>{1}\nAttack<r>{2}\nDefense<r>{3}\nSp. Atk<r>{4}\nSp. Def<r>{5}\nSpeed<r>{6}",
168	                           pkmn.totalhp, pkmn.attack, pkmn.defense, pkmn.spatk, pkmn.spdef, pkmn.speed), scene)
169	  else
170	    pkmn.changeHappiness("vitamin")
171	    if scene.is_a?(PokemonPartyScreen)
172	      scene.pbDisplay(_INTL("{1} grew to Lv. {2}!", pkmn.name, pkmn.level))
173	    else
174	      pbMessage(_INTL("{1} grew to Lv. {2}!", pkmn.name, pkmn.level))
175	    end
176	    total_hp_diff        = pkmn.totalhp - old_total_hp
177	    attack_diff          = pkmn.attack - old_attack
178	    defense_diff         = pkmn.defense - old_defense
179	    special_attack_diff  = pkmn.spatk - old_special_attack
180	    special_defense_diff = pkmn.spdef - old_special_defense
181	    speed_diff           = pkmn.speed - old_speed
182	    pbTopRightWindow(_INTL("Max. HP<r>+{1}\nAttack<r>+{2}\nDefense<r>+{3}\nSp. Atk<r>+{4}\nSp. Def<r>+{5}\nSpeed<r>+{6}",
183	                           total_hp_diff, attack_diff, defense_diff, special_attack_diff, special_defense_diff, speed_diff), scene)
184	    pbTopRightWindow(_INTL("Max. HP<r>{1}\nAttack<r>{2}\nDefense<r>{3}\nSp. Atk<r>{4}\nSp. Def<r>{5}\nSpeed<r>{6}",
185	                           pkmn.totalhp, pkmn.attack, pkmn.defense, pkmn.spatk, pkmn.spdef, pkmn.speed), scene)
186	    # Learn new moves upon level up
187	    movelist = pkmn.getMoveList
188	    movelist.each do |i|
189	      next if i[0] <= old_level || i[0] > pkmn.level
190	      pbLearnMove(pkmn, i[1], true) { scene.pbUpdate }
191	    end
192	    # Check for evolution
193	    new_species = pkmn.check_evolution_on_level_up
194	    if new_species
195	      pbFadeOutInWithMusic do
196	        evo = PokemonEvolutionScene.new
197	        evo.pbStartScreen(pkmn, new_species)
198	        evo.pbEvolution
199	        evo.pbEndScreen
200	        scene.pbRefresh if scene.is_a?(PokemonPartyScreen)
201	      end
202	    end
203	  end
204	end

206	def pbTopRightWindow(text, scene = nil)
207	  window = Window_AdvancedTextPokemon.new(text)
208	  window.width = 198
209	  window.x     = Graphics.width - window.width
210	  window.y     = 0
211	  window.z     = 99999
212	  pbPlayDecisionSE
213	  loop do
214	    Graphics.update
215	    Input.update
216	    window.update
217	    scene&.pbUpdate
218	    break if Input.trigger?(Input::USE)
219	  end
220	  window.dispose
221	end
