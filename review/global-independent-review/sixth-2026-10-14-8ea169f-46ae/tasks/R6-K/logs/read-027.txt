221	MultipleForms.register(:ROTOM, {
222	  "onSetForm" => proc { |pkmn, form, oldForm|
223	    form_moves = [
224	      :OVERHEAT,    # Heat (microwave oven)
225	      :HYDROPUMP,   # Wash (washing machine)
226	      :BLIZZARD,    # Frost (refrigerator)
227	      :AIRSLASH,    # Fan (electric fan)
228	      :LEAFSTORM    # Mow (lawn mower)
229	    ]
230	    # Find a known move that should be forgotten
231	    old_move_index = -1
232	    pkmn.moves.each_with_index do |move, i|
233	      next if !form_moves.include?(move.id)
234	      old_move_index = i
235	      break
236	    end
237	    # Determine which new move to learn (if any)
238	    new_move_id = (form > 0) ? form_moves[form - 1] : nil
239	    new_move_id = nil if !GameData::Move.exists?(new_move_id)
240	    if new_move_id.nil? && old_move_index >= 0 && pkmn.numMoves == 1
241	      new_move_id = :THUNDERSHOCK
242	      new_move_id = nil if !GameData::Move.exists?(new_move_id)
243	      raise _INTL("Rotom is trying to forget its last move, but there isn't another move to replace it with.") if new_move_id.nil?
244	    end
245	    new_move_id = nil if pkmn.hasMove?(new_move_id)
246	    # Forget a known move (if relevant) and learn a new move (if relevant)
247	    if old_move_index >= 0
248	      old_move_name = pkmn.moves[old_move_index].name
249	      if new_move_id.nil?
250	        # Just forget the old move
251	        pkmn.forget_move_at_index(old_move_index)
252	        pbMessage(_INTL("{1} forgot {2}...", pkmn.name, old_move_name))
253	      else
254	        # Replace the old move with the new move (keeps the same index)
255	        pkmn.moves[old_move_index].id = new_move_id
256	        new_move_name = pkmn.moves[old_move_index].name
257	        pbMessage(_INTL("{1} forgot {2}...", pkmn.name, old_move_name) + "\1")
258	        pbMessage("\\se[]" + _INTL("{1} learned {2}!", pkmn.name, new_move_name) + "\\se[Pkmn move learnt]")
259	      end
260	    elsif !new_move_id.nil?
261	      # Just learn the new move
262	      pbLearnMove(pkmn, new_move_id, true)
263	    end
264	  }
265	})

330	MultipleForms.register(:KYUREM, {
331	  "getFormOnEnteringBattle" => proc { |pkmn, wild|
332	    next pkmn.form + 2 if pkmn.form == 1 || pkmn.form == 2
333	  },
334	  "getFormOnLeavingBattle" => proc { |pkmn, battle, usedInBattle, endBattle|
335	    next pkmn.form - 2 if pkmn.form >= 3   # Fused forms stop glowing
336	  },
337	  "onSetForm" => proc { |pkmn, form, oldForm|
338	    case form
339	    when 0   # Normal
340	      pkmn.moves.each_with_index do |move, i|
341	        case move.id
342	        when :ICEBURN, :FREEZESHOCK
343	          next if !GameData::Move.exists?(:GLACIATE)
344	          if pkmn.hasMove?(:GLACIATE)
345	            pkmn.moves[i] = nil
346	          else
347	            move.id = :GLACIATE
348	          end
349	        when :FUSIONFLARE, :FUSIONBOLT
350	          next if !GameData::Move.exists?(:SCARYFACE)
351	          if pkmn.hasMove?(:SCARYFACE)
352	            pkmn.moves[i] = nil
353	          else
354	            move.id = :SCARYFACE
355	          end
356	        end
357	        pkmn.moves.compact!
358	      end
359	    when 1   # White
360	      pkmn.moves.each do |move|
361	        case move.id
362	        when :GLACIATE
363	          next if !GameData::Move.exists?(:ICEBURN) || pkmn.hasMove?(:ICEBURN)
364	          move.id = :ICEBURN
365	        when :SCARYFACE
366	          next if !GameData::Move.exists?(:FUSIONFLARE) || pkmn.hasMove?(:FUSIONFLARE)
367	          move.id = :FUSIONFLARE
368	        end
369	      end
370	    when 2   # Black
371	      pkmn.moves.each do |move|
372	        case move.id
373	        when :GLACIATE
374	          next if !GameData::Move.exists?(:FREEZESHOCK) || pkmn.hasMove?(:FREEZESHOCK)
375	          move.id = :FREEZESHOCK
376	        when :SCARYFACE
377	          next if !GameData::Move.exists?(:FUSIONBOLT) || pkmn.hasMove?(:FUSIONBOLT)
378	          move.id = :FUSIONBOLT
379	        end
380	      end
381	    end
382	  }
383	})
