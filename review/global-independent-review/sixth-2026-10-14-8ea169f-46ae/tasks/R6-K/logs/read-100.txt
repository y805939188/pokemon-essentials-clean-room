1	# Category has the following effects:
2	#   - Determines the in-battle weather.
3	#   - Some abilities reduce the encounter rate in certain categories of weather.
4	#   - Some evolution methods check the current weather's category.
5	#   - The :Rain category treats the last listed particle graphic as a water splash rather
6	#     than a raindrop, which behaves differently.
7	#   - :Rain auto-waters berry plants.
8	# Delta values are per second.
9	# For the tone_proc, strength goes from 0 to RPG::Weather::MAX_SPRITES (60) and
10	# will typically be the maximum.
11	module GameData
12	  class Weather
13	    attr_reader :id
14	    attr_reader :id_number
15	    attr_reader :real_name
16	    attr_reader :category   # :None, :Rain, :Hail, :Sandstorm, :Sun, :Fog
17	    attr_reader :graphics   # [[particle file names], [tile file names]]
18	    attr_reader :particle_delta_x
19	    attr_reader :particle_delta_y
20	    attr_reader :particle_delta_opacity
21	    attr_reader :tile_delta_x
22	    attr_reader :tile_delta_y
23	    attr_reader :tone_proc
24	
25	    DATA = {}
26	
27	    extend ClassMethods
28	    include InstanceMethods
29	
30	    def self.load; end
31	    def self.save; end
32	
33	    def initialize(hash)
34	      @id                     = hash[:id]
35	      @id_number              = hash[:id_number]
36	      @real_name              = hash[:id].to_s                || "Unnamed"
37	      @category               = hash[:category]               || :None
38	      @particle_delta_x       = hash[:particle_delta_x]       || 0
39	      @particle_delta_y       = hash[:particle_delta_y]       || 0
40	      @particle_delta_opacity = hash[:particle_delta_opacity] || 0
41	      @tile_delta_x           = hash[:tile_delta_x]           || 0
42	      @tile_delta_y           = hash[:tile_delta_y]           || 0
43	      @graphics               = hash[:graphics]               || []
44	      @tone_proc              = hash[:tone_proc]
45	    end
46	
47	    alias name real_name
48	
49	    def has_particles?
50	      return @graphics[0] && @graphics[0].length > 0
51	    end
52	
53	    def has_tiles?
54	      return @graphics[1] && @graphics[1].length > 0
55	    end
56	
57	    def tone(strength)
58	      return (@tone_proc) ? @tone_proc.call(strength) : Tone.new(0, 0, 0, 0)
59	    end
60	  end
61	end
