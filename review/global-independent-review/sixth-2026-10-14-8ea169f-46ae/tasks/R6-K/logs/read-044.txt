247	def pbTrainerName(name = nil, outfit = 0)
248	  pbChangePlayer(1) if $player.character_ID < 1
249	  if name.nil?
250	    name = pbEnterPlayerName(_INTL("Your name?"), 0, Settings::MAX_PLAYER_NAME_SIZE)
251	    if name.nil? || name.empty?
252	      player_metadata = GameData::PlayerMetadata.get($player.character_ID)
253	      trainer_type = (player_metadata) ? player_metadata.trainer_type : nil
254	      gender = pbGetTrainerTypeGender(trainer_type)
255	      name = pbSuggestTrainerName(gender)
256	    end
257	  end
258	  $player.name   = name
259	  $player.outfit = outfit
260	end

263	  userName = pbGetUserName
264	  userName = userName.gsub(/\s+.*$/, "")
265	  if userName.length > 0 && userName.length < Settings::MAX_PLAYER_NAME_SIZE
266	    userName[0, 1] = userName[0, 1].upcase
267	    return userName
268	  end
269	  userName = userName.gsub(/\d+$/, "")
270	  if userName.length > 0 && userName.length < Settings::MAX_PLAYER_NAME_SIZE
271	    userName[0, 1] = userName[0, 1].upcase
272	    return userName
273	  end
274	  userName = System.user_name.capitalize
275	  userName = userName[0, Settings::MAX_PLAYER_NAME_SIZE]
276	  return userName

285	def getRandomNameEx(type, variable, upper, maxLength = 100)
286	  return "" if maxLength <= 0
287	  name = ""
288	  50.times do
289	    name = ""
290	    formats = []
291	    case type
292	    when 0 then formats = ["F5", "BvE", "FE", "FE5", "FEvE"]                    # Names for males
293	    when 1 then formats = ["vE6", "vEvE6", "BvE6", "B4", "v3", "vEv3", "Bv3"]   # Names for females
294	    when 2 then formats = ["WE", "WEU", "WEvE", "BvE", "BvEU", "BvEvE"]         # Neutral gender names
295	    else        return ""
296	    end
297	    format = formats.sample
298	    format.scan(/./) do |c|
299	      case c
300	      when "c"   # consonant
301	        set = ["b", "c", "d", "f", "g", "h", "j", "k", "l", "m", "n", "p", "r",
302	               "s", "t", "v", "w", "x", "z"]
303	        name += set.sample
304	      when "v"   # vowel
305	        set = ["a", "a", "a", "e", "e", "e", "i", "i", "i", "o", "o", "o", "u", "u", "u"]
306	        name += set.sample
307	      when "W"   # beginning vowel
308	        set = ["a", "a", "a", "e", "e", "e", "i", "i", "i", "o", "o", "o", "u",
309	               "u", "u", "au", "au", "ay", "ay", "ea", "ea", "ee", "ee", "oo",
310	               "oo", "ou", "ou"]
311	        name += set.sample
312	      when "U"   # ending vowel
313	        set = ["a", "a", "a", "a", "a", "e", "e", "e", "i", "i", "i", "o", "o",
314	               "o", "o", "o", "u", "u", "ay", "ay", "ie", "ie", "ee", "ue", "oo"]
315	        name += set.sample
316	      when "B"   # beginning consonant
317	        set1 = ["b", "c", "d", "f", "g", "h", "j", "k", "l", "l", "m", "n", "n",
318	                "p", "r", "r", "s", "s", "t", "t", "v", "w", "y", "z"]
319	        set2 = ["bl", "br", "ch", "cl", "cr", "dr", "fr", "fl", "gl", "gr", "kh",
320	                "kl", "kr", "ph", "pl", "pr", "sc", "sk", "sl", "sm", "sn", "sp",
321	                "st", "sw", "th", "tr", "tw", "vl", "zh"]
322	        name += (rand(3) > 0) ? set1.sample : set2.sample
323	      when "E"   # ending consonant
324	        set1 = ["b", "c", "d", "f", "g", "h", "j", "k", "k", "l", "l", "m", "n",
325	                "n", "p", "r", "r", "s", "s", "t", "t", "v", "z"]
326	        set2 = ["bb", "bs", "ch", "cs", "ds", "fs", "ft", "gs", "gg", "ld", "ls",
327	                "nd", "ng", "nk", "rn", "kt", "ks", "ms", "ns", "ph", "pt", "ps",
328	                "sk", "sh", "sp", "ss", "st", "rd", "rn", "rp", "rm", "rt", "rk",
329	                "ns", "th", "zh"]
330	        name += (rand(3) > 0) ? set1.sample : set2.sample
331	      when "f"   # consonant and vowel
332	        set = ["iz", "us", "or"]
333	        name += set.sample
334	      when "F"   # consonant and vowel
335	        set = ["bo", "ba", "be", "bu", "re", "ro", "si", "mi", "zho", "se", "nya",
336	               "gru", "gruu", "glee", "gra", "glo", "ra", "do", "zo", "ri", "di",
337	               "ze", "go", "ga", "pree", "pro", "po", "pa", "ka", "ki", "ku",
338	               "de", "da", "ma", "mo", "le", "la", "li"]
339	        name += set.sample
340	      when "2"
341	        set = ["c", "f", "g", "k", "l", "p", "r", "s", "t"]
342	        name += set.sample
343	      when "3"
344	        set = ["nka", "nda", "la", "li", "ndra", "sta", "cha", "chie"]
345	        name += set.sample
346	      when "4"
347	        set = ["una", "ona", "ina", "ita", "ila", "ala", "ana", "ia", "iana"]
348	        name += set.sample
349	      when "5"
350	        set = ["e", "e", "o", "o", "ius", "io", "u", "u", "ito", "io", "ius", "us"]
351	        name += set.sample
352	      when "6"
353	        set = ["a", "a", "a", "elle", "ine", "ika", "ina", "ita", "ila", "ala", "ana"]
354	        name += set.sample
355	      end
356	    end
357	    break if name.length <= maxLength
358	  end
359	  name = name[0, maxLength]
360	  case upper
361	  when 0 then name = name.upcase
362	  when 1 then name[0, 1] = name[0, 1].upcase
363	  end
364	  if $game_variables && variable
365	    $game_variables[variable] = name
366	    $game_map.need_refresh = true if $game_map
367	  end
368	  return name
369	end

386	# Returns the Regional Pokédex number of the given species in the given Regional
387	# Dex. The parameter "region" is zero-based. For example, if two regions are
388	# defined, they would each be specified as 0 and 1.
389	def pbGetRegionalNumber(region, species)
390	  dex_list = pbLoadRegionalDexes[region]
391	  return 0 if !dex_list || dex_list.length == 0
392	  species_data = GameData::Species.try_get(species)
393	  return 0 if !species_data
394	  dex_list.each_with_index do |s, index|
395	    return index + 1 if s == species_data.species
396	  end
397	  return 0
398	end

400	# Returns an array of all species in the given Regional Dex in that Dex's order.
401	def pbAllRegionalSpecies(region_dex)
402	  return nil if region_dex < 0
403	  dex_list = pbLoadRegionalDexes[region_dex]
404	  return nil if !dex_list || dex_list.length == 0
405	  return dex_list.clone
406	end

408	# Returns the number of species in the given Regional Dex. Returns 0 if that
409	# Regional Dex doesn't exist. If region_dex is a negative number, returns the
410	# number of species in the National Dex (i.e. all species).
411	def pbGetRegionalDexLength(region_dex)
412	  if region_dex < 0
413	    ret = 0
414	    GameData::Species.each_species { |s| ret += 1 }
415	    return ret
416	  end
417	  dex_list = pbLoadRegionalDexes[region_dex]
418	  return (dex_list) ? dex_list.length : 0
419	end
