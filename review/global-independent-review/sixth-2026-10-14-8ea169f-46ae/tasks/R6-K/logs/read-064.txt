17	  def draw_text(x, y, width, height = nil, text = "", align = 0)
18	    if x.is_a?(Rect)
19	      x.y -= (@text_offset_y || 0)
20	      # rect, string & alignment
21	      mkxp_draw_text(x, y, width)
22	    else
23	      y -= (@text_offset_y || 0)
24	      height = text_size(text).height
25	      mkxp_draw_text(x, y, width, height, text, align)
26	    end

30	def pbSetResizeFactor(factor)
31	  if !$ResizeInitialized
32	    Graphics.resize_screen(Settings::SCREEN_WIDTH, Settings::SCREEN_HEIGHT)
33	    $ResizeInitialized = true
34	  end
35	  if factor < 0 || factor == 4
36	    Graphics.fullscreen = true if !Graphics.fullscreen
37	  else
38	    Graphics.fullscreen = false if Graphics.fullscreen
39	    Graphics.scale = (factor + 1) * 0.5
40	    Graphics.center
41	  end
42	end
