101	SaveData.register_conversion(:v20_refactor_planted_berries_data) do
102	  essentials_version 20
103	  display_title "Updating berry plant data format"
104	  to_value :global_metadata do |global|
105	    if global.eventvars
106	      global.eventvars.each_pair do |key, value|
107	        next if !value || !value.is_a?(Array)
108	        case value.length
109	        when 6   # Old berry plant data
110	          data = BerryPlantData.new
111	          if value[1].is_a?(Symbol)
112	            plant_data = GameData::BerryPlant.get(value[1])
113	            data.new_mechanics      = false
114	            data.berry_id           = value[1]
115	            data.time_alive         = value[0] * plant_data.hours_per_stage * 3600
116	            data.time_last_updated  = value[3]
117	            data.growth_stage       = value[0]
118	            data.replant_count      = value[5]
119	            data.watered_this_stage = value[2]
120	            data.watering_count     = value[4]
121	          end
122	          global.eventvars[key] = data
123	        when 7, 8   # New berry plant data
124	          data = BerryPlantData.new
125	          if value[1].is_a?(Symbol)
126	            data.new_mechanics     = true
127	            data.berry_id          = value[1]
128	            data.mulch_id          = value[7] if value[7].is_a?(Symbol)
129	            data.time_alive        = value[2]
130	            data.time_last_updated = value[3]
131	            data.growth_stage      = value[0]
132	            data.replant_count     = value[5]
133	            data.moisture_level    = value[4]
134	            data.yield_penalty     = value[6]
135	          end
136	          global.eventvars[key] = data
137	        end
138	      end
139	    end
140	  end

301	SaveData.register_conversion(:v20_convert_pokemon_markings) do
302	  essentials_version 20
303	  display_title "Updating format of Pokémon markings"
304	  to_all do |save_data|
305	    # Create a lambda function that updates a Pokémon's markings
306	    update_markings = lambda do |pkmn|
307	      return if !pkmn || !pkmn.markings.is_a?(Integer)
308	      markings = []
309	      6.times { |i| markings[i] = ((pkmn.markings & (1 << i)) == 0) ? 0 : 1 }
310	      pkmn.markings = markings
311	    end
312	    # Party Pokémon
313	    save_data[:player].party.each { |pkmn| update_markings.call(pkmn) }
314	    # Pokémon storage
315	    save_data[:storage_system].boxes.each do |box|
316	      box.pokemon.each { |pkmn| update_markings.call(pkmn) if pkmn }
317	    end
318	    # NOTE: Pokémon in the Day Care have their markings converted above.
319	    # Partner trainer
320	    if save_data[:global_metadata].partner
321	      save_data[:global_metadata].partner[3].each { |pkmn| update_markings.call(pkmn) }
322	    end
323	    # Roaming Pokémon
324	    if save_data[:global_metadata].roamPokemon
325	      save_data[:global_metadata].roamPokemon.each { |pkmn| update_markings.call(pkmn) }
326	    end
327	    # Purify Chamber
328	    save_data[:global_metadata].purifyChamber.sets.each do |set|
329	      set.list.each { |pkmn| update_markings.call(pkmn) }
330	      update_markings.call(set.shadow) if set.shadow
331	    end
332	    # Hall of Fame records
333	    if save_data[:global_metadata].hallOfFame
334	      save_data[:global_metadata].hallOfFame.each do |team|
335	        next if !team
336	        team.each { |pkmn| update_markings.call(pkmn) }
337	      end
338	    end
339	    # Pokémon stored in Game Variables for some reason
340	    variables = save_data[:variables]
341	    (0..5000).each do |i|
342	      value = variables[i]
343	      case value
344	      when Array
345	        value.each { |value2| update_markings.call(value2) if value2.is_a?(Pokemon) }
346	      when Pokemon
347	        update_markings.call(value)
348	      end
349	    end
350	  end
351	end

374	            end
