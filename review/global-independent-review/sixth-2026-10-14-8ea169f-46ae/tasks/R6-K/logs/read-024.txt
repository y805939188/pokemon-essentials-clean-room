145	  def pbStartScene(bag, choosing = false, filterproc = nil, resetpocket = true)
146	    @viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
147	    @viewport.z = 99999
148	    @bag        = bag
149	    @choosing   = choosing
150	    @filterproc = filterproc
151	    pbRefreshFilter
152	    lastpocket = @bag.last_viewed_pocket
153	    numfilledpockets = @bag.pockets.length - 1
154	    if @choosing
155	      numfilledpockets = 0
156	      if @filterlist.nil?
157	        (1...@bag.pockets.length).each do |i|
158	          numfilledpockets += 1 if @bag.pockets[i].length > 0
159	        end
160	      else
161	        (1...@bag.pockets.length).each do |i|
162	          numfilledpockets += 1 if @filterlist[i].length > 0
163	        end
164	      end
165	      lastpocket = (resetpocket) ? 1 : @bag.last_viewed_pocket
166	      if (@filterlist && @filterlist[lastpocket].length == 0) ||
167	         (!@filterlist && @bag.pockets[lastpocket].length == 0)
168	        (1...@bag.pockets.length).each do |i|
169	          if @filterlist && @filterlist[i].length > 0
170	            lastpocket = i
171	            break
172	          elsif !@filterlist && @bag.pockets[i].length > 0
173	            lastpocket = i
174	            break
175	          end
176	        end
177	      end
178	    end
179	    @bag.last_viewed_pocket = lastpocket
180	    @sliderbitmap = AnimatedBitmap.new("Graphics/UI/Bag/icon_slider")
181	    @pocketbitmap = AnimatedBitmap.new("Graphics/UI/Bag/icon_pocket")
182	    @sprites = {}
183	    @sprites["background"] = IconSprite.new(0, 0, @viewport)
184	    @sprites["overlay"] = BitmapSprite.new(Graphics.width, Graphics.height, @viewport)
185	    pbSetSystemFont(@sprites["overlay"].bitmap)
186	    @sprites["bagsprite"] = IconSprite.new(30, 20, @viewport)
187	    @sprites["pocketicon"] = BitmapSprite.new(186, 32, @viewport)
188	    @sprites["pocketicon"].x = 0
189	    @sprites["pocketicon"].y = 224
190	    @sprites["leftarrow"] = AnimatedSprite.new("Graphics/UI/left_arrow", 8, 40, 28, 2, @viewport)
191	    @sprites["leftarrow"].x       = -4
192	    @sprites["leftarrow"].y       = 76
193	    @sprites["leftarrow"].visible = (!@choosing || numfilledpockets > 1)
194	    @sprites["leftarrow"].play
195	    @sprites["rightarrow"] = AnimatedSprite.new("Graphics/UI/right_arrow", 8, 40, 28, 2, @viewport)
196	    @sprites["rightarrow"].x       = 150
197	    @sprites["rightarrow"].y       = 76
198	    @sprites["rightarrow"].visible = (!@choosing || numfilledpockets > 1)
199	    @sprites["rightarrow"].play
200	    @sprites["itemlist"] = Window_PokemonBag.new(@bag, @filterlist, lastpocket, 168, -8, 314, 40 + 32 + (ITEMSVISIBLE * 32))
201	    @sprites["itemlist"].viewport    = @viewport
202	    @sprites["itemlist"].pocket      = lastpocket
203	    @sprites["itemlist"].index       = @bag.last_viewed_index(lastpocket)
204	    @sprites["itemlist"].baseColor   = ITEMLISTBASECOLOR
205	    @sprites["itemlist"].shadowColor = ITEMLISTSHADOWCOLOR
206	    @sprites["itemicon"] = ItemIconSprite.new(48, Graphics.height - 48, nil, @viewport)
207	    @sprites["itemtext"] = Window_UnformattedTextPokemon.newWithSize(
208	      "", 72, 272, Graphics.width - 72 - 24, 128, @viewport
209	    )

211	    @sprites["itemtext"].shadowColor = ITEMTEXTSHADOWCOLOR
212	    @sprites["itemtext"].visible     = true
213	    @sprites["itemtext"].windowskin  = nil
214	    @sprites["helpwindow"] = Window_UnformattedTextPokemon.new("")
215	    @sprites["helpwindow"].visible  = false
216	    @sprites["helpwindow"].viewport = @viewport
217	    @sprites["msgwindow"] = Window_AdvancedTextPokemon.new("")
218	    @sprites["msgwindow"].visible  = false
219	    @sprites["msgwindow"].viewport = @viewport
