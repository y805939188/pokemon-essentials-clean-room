212	  def canConsumeBerry?
213	    return false if @battle.pbCheckOpposingAbility([:UNNERVE, :ASONECHILLINGNEIGH, :ASONEGRIMNEIGH], @index)
214	    return true
215	  end
216	
217	  def canConsumePinchBerry?(check_gluttony = true)
218	    return false if !canConsumeBerry?
219	    return true if @hp <= @totalhp / 4
220	    return true if @hp <= @totalhp / 2 && (!check_gluttony || hasActiveAbility?(:GLUTTONY))
221	    return false
222	  end

225	  # Off.
226	  def pbRemoveItem(permanent = true)
227	    @effects[PBEffects::ChoiceBand] = nil if !hasActiveAbility?(:GORILLATACTICS)
228	    @effects[PBEffects::Unburden]   = true if self.item && hasActiveAbility?(:UNBURDEN)
229	    setInitialItem(nil) if permanent && self.item == self.initialItem
230	    self.item = nil
231	  end
