49	    def self.editor_properties
50	      field_use_array = [_INTL("Can't use in field")]
51	      self.schema["FieldUse"][2].each { |key, value| field_use_array[value] = key if !field_use_array[value] }
52	      battle_use_array = [_INTL("Can't use in battle")]
53	      self.schema["BattleUse"][2].each { |key, value| battle_use_array[value] = key if !battle_use_array[value] }
54	      return [
55	        ["ID",                ReadOnlyProperty,                        _INTL("ID of this item (used as a symbol like :XXX).")],
56	        ["Name",              ItemNameProperty,                        _INTL("Name of this item as displayed by the game.")],
57	        ["NamePlural",        ItemNameProperty,                        _INTL("Plural name of this item as displayed by the game.")],
58	        ["PortionName",       ItemNameProperty,                        _INTL("Name of a portion of this item as displayed by the game.")],
59	        ["PortionNamePlural", ItemNameProperty,                        _INTL("Name of 2 or more portions of this item as displayed by the game.")],
60	        ["Pocket",            PocketProperty,                          _INTL("Pocket in the Bag where this item is stored.")],
61	        ["Price",             LimitProperty.new(Settings::MAX_MONEY),  _INTL("Purchase price of this item.")],

74	    def self.icon_filename(item)
75	      return "Graphics/Items/back" if item.nil?
76	      item_data = self.try_get(item)
77	      return "Graphics/Items/000" if item_data.nil?
78	      # Check for files
79	      ret = sprintf("Graphics/Items/%s", item_data.id)
80	      return ret if pbResolveBitmap(ret)
81	      # Check for TM/HM type icons
82	      if item_data.is_machine?
83	        prefix = "machine"
84	        if item_data.is_HM?
85	          prefix = "machine_hm"
86	        elsif item_data.is_TR?
87	          prefix = "machine_tr"
88	        end
89	        move_type = GameData::Move.get(item_data.move).type
90	        type_data = GameData::Type.get(move_type)
91	        ret = sprintf("Graphics/Items/%s_%s", prefix, type_data.id)
92	        return ret if pbResolveBitmap(ret)
93	        if !item_data.is_TM?
94	          ret = sprintf("Graphics/Items/machine_%s", type_data.id)
95	          return ret if pbResolveBitmap(ret)
96	        end
97	      end
98	      return "Graphics/Items/000"
99	    end

252	    alias __orig__get_property_for_PBS get_property_for_PBS unless method_defined?(:__orig__get_property_for_PBS)
253	    def get_property_for_PBS(key)
254	      key = "SectionName" if key == "ID"
255	      ret = __orig__get_property_for_PBS(key)
256	      case key
257	      when "SellPrice"
258	        ret = nil if ret == @price / 2
259	      when "BPPrice"
260	        ret = nil if ret == 1
261	      when "FieldUse", "BattleUse"
262	        ret = nil if ret == 0
263	      when "Consumable"
264	        ret = @consumable
265	        ret = nil if ret || is_important?   # Only return false, only for non-important items
266	      end
267	      return ret
268	    end
