4	def pbPostData(url, postdata, filename = nil, depth = 0)
5	  if url[/^http:\/\/([^\/]+)(.*)$/]
6	    host = $1
7	#    path = $2
8	#    path = "/" if path.length == 0
9	    userAgent = "Mozilla/5.0 (Windows; U; Windows NT 5.1; en-US; rv:1.9.0.14) Gecko/2009082707 Firefox/3.0.14"
10	    body = postdata.map do |key, value|
11	      keyString   = key.to_s
12	      valueString = value.to_s
13	      keyString.gsub!(/[^a-zA-Z0-9_\.\-]/n) { |s| sprintf("%%%02x", s[0]) }
14	      valueString.gsub!(/[^a-zA-Z0-9_\.\-]/n) { |s| sprintf("%%%02x", s[0]) }
15	      next "#{keyString}=#{valueString}"
16	    end.join("&")
17	    ret = HTTPLite.post_body(
18	      url,
19	      body,
20	      "application/x-www-form-urlencoded",
21	      {
22	        "Host"             => host, # might not be necessary
23	        "Proxy-Connection" => "Close",
24	        "Content-Length"   => body.bytesize.to_s,
25	        "Pragma"           => "no-cache",
26	        "User-Agent"       => userAgent
27	      }
28	    ) rescue ""
29	    return ret if !ret.is_a?(Hash)
30	    return "" if ret[:status] != 200
31	    return ret[:body] if !filename
32	    File.open(filename, "wb") { |f| f.write(ret[:body]) }
33	    return ""
34	  end
35	  return ""
36	end

38	def pbDownloadData(url, filename = nil, authorization = nil, depth = 0, &block)
39	  headers = {
40	    "Proxy-Connection" => "Close",
41	    "Pragma"           => "no-cache",
42	    "User-Agent"       => "Mozilla/5.0 (Windows; U; Windows NT 5.1; en-US; rv:1.9.0.14) Gecko/2009082707 Firefox/3.0.14"
43	  }
44	  headers["authorization"] = authorization if authorization
45	  ret = HTTPLite.get(url, headers) rescue ""
46	  return ret if !ret.is_a?(Hash)
47	  return "" if ret[:status] != 200
48	  return ret[:body] if !filename
49	  File.open(filename, "wb") { |f| f.write(ret[:body]) }
50	  return ""
51	end
