313	# Gets at least the first byte of a file. Doesn't check RTP, but does check
314	# encrypted archives.
315	def pbGetFileChar(file)
316	  canon_file = canonicalize(file)
317	  if !FileTest.exist?("./Game.rgssad")
318	    return nil if !FileTest.exist?(canon_file)
319	    return nil if file.last == "/"   # Is a directory
320	    begin
321	      File.open(canon_file, "rb") { |f| return f.read(1) }   # read one byte
322	    rescue Errno::ENOENT, Errno::EINVAL, Errno::EACCES, Errno::EISDIR
323	      return nil
324	    end
325	  end
326	  str = nil
327	  begin
328	    str = load_data(canon_file, true)
329	  rescue Errno::ENOENT, Errno::EINVAL, Errno::EACCES, Errno::EISDIR, RGSSError, MKXPError
330	    str = nil
331	  end
332	  return str
333	end

366	class StringInput
367	  include Enumerable
368	
369	  attr_reader :lineno, :string
370	
371	  class << self
372	    def new(str)
373	      if block_given?
374	        begin
375	          f = super
376	          yield f
377	        ensure
378	          f&.close
379	        end
380	      else
381	        super
382	      end
383	    end
384	    alias open new
385	  end
386	
387	  def initialize(str)
388	    @string = str
389	    @pos = 0
390	    @closed = false
391	    @lineno = 0
392	  end
393	
394	  def inspect
395	    return "#<#{self.class}:#{@closed ? 'closed' : 'open'},src=#{@string[0, 30].inspect}>"
396	  end
397	
398	  def close
399	    raise IOError, "closed stream" if @closed
400	    @pos = nil
401	    @closed = true
402	  end
403	
404	  def closed?; @closed; end
405	
406	  def pos
407	    raise IOError, "closed stream" if @closed
408	    [@pos, @string.size].min
409	  end
410	
411	  alias tell pos
412	
413	  def rewind; seek(0); end
414	
415	  def pos=(value); seek(value); end
416	
417	  def seek(offset, whence = IO::SEEK_SET)
418	    raise IOError, "closed stream" if @closed
419	    case whence
420	    when IO::SEEK_SET then @pos = offset
421	    when IO::SEEK_CUR then @pos += offset
422	    when IO::SEEK_END then @pos = @string.size - offset
423	    else
424	      raise ArgumentError, "unknown seek flag: #{whence}"
425	    end
426	    @pos = 0 if @pos < 0
427	    @pos = [@pos, @string.size + 1].min
428	    offset
429	  end
430	
431	  def eof?
432	    raise IOError, "closed stream" if @closed
433	    @pos > @string.size
434	  end
435	
436	  def each(&block)
437	    raise IOError, "closed stream" if @closed
438	    begin
439	      @string.each(&block)
440	    ensure
441	      @pos = 0
442	    end
443	  end
444	
445	  def gets
446	    raise IOError, "closed stream" if @closed
447	    idx = @string.index("\n", @pos)
448	    if idx
449	      idx += 1  # "\n".size
450	      line = @string[@pos...idx]
451	      @pos = idx
452	      @pos += 1 if @pos == @string.size
453	    else
454	      line = @string[@pos..-1]
455	      @pos = @string.size + 1
456	    end
457	    @lineno += 1
458	    line
459	  end
460	
461	  def getc
462	    raise IOError, "closed stream" if @closed
463	    ch = @string[@pos]
464	    @pos += 1
465	    @pos += 1 if @pos == @string.size
466	    ch
467	  end
468	
469	  def read(len = nil)
470	    raise IOError, "closed stream" if @closed
471	    if !len
472	      return nil if eof?
473	      rest = @string[@pos...@string.size]
474	      @pos = @string.size + 1
475	      return rest
476	    end
477	    str = @string[@pos, len]
478	    @pos += len
479	    @pos += 1 if @pos == @string.size
480	    str
481	  end
482	  alias read_all read
483	  alias sysread read
