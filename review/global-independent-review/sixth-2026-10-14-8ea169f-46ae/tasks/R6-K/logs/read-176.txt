123	def pbRepel(item, steps)
124	  if $PokemonGlobal.repel > 0
125	    pbMessage(_INTL("But a repellent's effect still lingers from earlier."))
126	    return false
127	  end
128	  pbSEPlay("Repel")
129	  $stats.repel_count += 1
130	  pbUseItemMessage(item)
131	  $PokemonGlobal.repel = steps
132	  return true
133	end

148	  proc {
149	    next if $PokemonGlobal.repel <= 0 || $game_player.terrain_tag.ice   # Shouldn't count down if on ice
150	    $PokemonGlobal.repel -= 1
151	    next if $PokemonGlobal.repel > 0
152	    repels = []
153	    GameData::Item.each { |itm| repels.push(itm.id) if itm.has_flag?("Repel") }
154	    if repels.none? { |item| $bag.has?(item) }
155	      pbMessage(_INTL("The repellent's effect wore off!"))
156	      next
157	    end
158	    next if !pbConfirmMessage(_INTL("The repellent's effect wore off! Would you like to use another one?"))
159	    ret = nil
160	    pbFadeOutIn do
161	      scene = PokemonBag_Scene.new
162	      screen = PokemonBagScreen.new(scene, $bag)
163	      ret = screen.pbChooseItemScreen(proc { |item| repels.include?(item) })
164	    end
165	    pbUseItem($bag, ret) if ret
166	  }

318	ItemHandlers::UseInField.add(:ITEMFINDER, proc { |item|
319	  $stats.itemfinder_count += 1
320	  pbSEPlay("Itemfinder")
321	  event = pbClosestHiddenItem
322	  if !event
323	    pbMessage(_INTL("... \\wt[10]... \\wt[10]... \\wt[10]... \\wt[10]Nope! There's no response."))
324	    next true
325	  end
326	  offsetX = event.x - $game_player.x
327	  offsetY = event.y - $game_player.y
328	  if offsetX == 0 && offsetY == 0   # Standing on the item, spin around
329	    4.times do
330	      pbWait(0.2)
331	      $game_player.turn_right_90
332	    end
333	    pbWait(0.3)
334	    pbMessage(_INTL("The {1}'s indicating something right underfoot!", GameData::Item.get(item).name))
335	  else   # Item is nearby, face towards it
336	    direction = $game_player.direction
337	    if offsetX.abs > offsetY.abs
338	      direction = (offsetX < 0) ? 4 : 6
339	    else
340	      direction = (offsetY < 0) ? 8 : 2
341	    end
342	    case direction
343	    when 2 then $game_player.turn_down
344	    when 4 then $game_player.turn_left
345	    when 6 then $game_player.turn_right
346	    when 8 then $game_player.turn_up
347	    end
348	    pbWait(0.3)
349	    pbMessage(_INTL("Huh? The {1}'s responding!", GameData::Item.get(item).name) + "\1")
350	    pbMessage(_INTL("There's an item buried around here!"))
351	  end
352	  next true
353	})
