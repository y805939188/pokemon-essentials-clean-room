164	    def get_random_tile_of_type(tile_type, dungeon, x, y, layer)
165	      tiles = @tile_type_ids[tile_type]
166	      return 0 if !tiles || tiles.empty?
167	      ret = tiles.sample[0]
168	      if ret < 384   # Autotile
169	        nb = TileDrawingHelper.tableNeighbors(dungeon, x, y, layer)
170	        variant = TileDrawingHelper::NEIGHBORS_TO_AUTOTILE_INDEX[nb]
171	        ret += variant
172	      else
173	        case tile_type
174	        when :void
175	          if @large_void_tiles
176	            ret += 1 if x.odd?
177	            ret += 8 if y.odd?
178	          end
179	        when :floor
180	          if large_floor_tiles
181	            ret += 1 if x.odd?
182	            ret += 8 if y.odd?
183	          end
184	        when :wall_2, :wall_8, :wall_top
185	          ret += 1 if @large_wall_tiles && x.odd?
186	        when :wall_4, :wall_6
187	          ret += 8 if @large_wall_tiles && y.odd?
188	        end
189	        # Different wall tiles for northern walls if there's another wall directly
190	        # north of them (i.e. tree tiles that shouldn't have shaded grass because
191	        # there isn't a tree-enclosed area there)
192	        if @thin_north_wall_offset != 0 && [:wall_7, :wall_8, :wall_9].include?(tile_type)
193	          ret += @thin_north_wall_offset if dungeon.tile_is_wall?(dungeon[x, y - 1, 1])
194	        end
195	      end
196	      return ret
197	    end

200	    def get_property_for_PBS(key)
201	      ret = __orig__get_property_for_PBS(key)
202	      case key
203	      when "ThinNorthWallOffset"
204	        ret = nil if ret == 0
205	      when "Tile", "Autotile"
206	        ret = []
207	        @tile_type_ids.each do |tile_type, tile_ids|
208	          tile_ids.each do |tile|
209	            case key
210	            when "Tile"
211	              ret.push([tile[0] - 384, tile_type]) if !tile[1] && tile[0] >= 384
212	            when "Autotile"
213	              ret.push([tile[0] / 48, tile_type]) if !tile[1] && tile[0] < 384
214	            end
215	          end
216	        end
217	        ret = nil if ret.length == 0
218	      end
