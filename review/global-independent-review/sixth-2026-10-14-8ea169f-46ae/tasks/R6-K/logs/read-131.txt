73	module Effectiveness
74	  INEFFECTIVE                   = 0
75	  NOT_VERY_EFFECTIVE            = 1
76	  NORMAL_EFFECTIVE              = 2
77	  SUPER_EFFECTIVE               = 4
78	  INEFFECTIVE_MULTIPLIER        = INEFFECTIVE.to_f / NORMAL_EFFECTIVE
79	  NOT_VERY_EFFECTIVE_MULTIPLIER = NOT_VERY_EFFECTIVE.to_f / NORMAL_EFFECTIVE
80	  NORMAL_EFFECTIVE_MULTIPLIER   = 1.0
81	  SUPER_EFFECTIVE_MULTIPLIER    = SUPER_EFFECTIVE.to_f / NORMAL_EFFECTIVE
82	
83	  module_function
84	
85	  def ineffective?(value)
86	    return value == INEFFECTIVE_MULTIPLIER
87	  end
88	
89	  def not_very_effective?(value)
90	    return value > INEFFECTIVE_MULTIPLIER && value < NORMAL_EFFECTIVE_MULTIPLIER
91	  end
92	
93	  def resistant?(value)
94	    return value < NORMAL_EFFECTIVE_MULTIPLIER
95	  end
96	
97	  def normal?(value)
98	    return value == NORMAL_EFFECTIVE_MULTIPLIER
99	  end
100	
101	  def super_effective?(value)
102	    return value > NORMAL_EFFECTIVE_MULTIPLIER
103	  end
104	
105	  def ineffective_type?(attack_type, *defend_types)
106	    value = calculate(attack_type, *defend_types)
107	    return ineffective?(value)
108	  end
109	
110	  def not_very_effective_type?(attack_type, *defend_types)
111	    value = calculate(attack_type, *defend_types)
112	    return not_very_effective?(value)
113	  end
114	
115	  def resistant_type?(attack_type, *defend_types)
116	    value = calculate(attack_type, *defend_types)
117	    return resistant?(value)
118	  end
119	
120	  def normal_type?(attack_type, *defend_types)
121	    value = calculate(attack_type, *defend_types)
122	    return normal?(value)
123	  end
124	
125	  def super_effective_type?(attack_type, *defend_types)
126	    value = calculate(attack_type, *defend_types)
127	    return super_effective?(value)
128	  end
129	
130	  def get_type_effectiveness(attack_type, defend_type)
131	    return GameData::Type.get(defend_type).effectiveness(attack_type)
132	  end
133	
134	  def calculate(attack_type, *defend_types)
135	    ret = NORMAL_EFFECTIVE_MULTIPLIER
136	    defend_types.each do |type|
137	      ret *= get_type_effectiveness(attack_type, type) / NORMAL_EFFECTIVE.to_f
138	    end
139	    return ret
140	  end
