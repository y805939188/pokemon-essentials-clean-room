1	# NOTE: The order these shapes are registered are the order they are listed in
2	#       the Pokédex search screen.
3	#       "Graphics/UI/Pokedex/icon_shapes.png" contains icons for these
4	#       shapes.
5	module GameData
6	  class BodyShape
7	    attr_reader :id
8	    attr_reader :real_name
9	    attr_reader :icon_position   # Where this shape's icon is within icon_shapes.png
10	
11	    DATA = {}
12	
13	    extend ClassMethodsSymbols
14	    include InstanceMethods
15	
16	    def self.load; end
17	    def self.save; end
18	
19	    def initialize(hash)
20	      @id            = hash[:id]
21	      @real_name     = hash[:name]          || "Unnamed"
22	      @icon_position = hash[:icon_position] || 0
23	    end
24	
25	    # @return [String] the translated name of this body shape
26	    def name
27	      return _INTL(@real_name)
28	    end
29	  end
30	end
