329	      return false if !contact
330	      if !can_make?
331	        pbMessage(_INTL("There is no phone signal here..."))
332	        return false
333	      end
334	      return true if !contact.trainer?
335	      if contact.map_id == $game_map.map_id
336	        pbMessage(_INTL("The Trainer is close by.\nTalk to the Trainer in person!"))
337	        return false
338	      end
339	      caller_map_metadata = GameData::MapMetadata.try_get(contact.map_id)
340	      this_map_metadata = $game_map.metadata
341	      if !caller_map_metadata || !caller_map_metadata.town_map_position ||
342	         !this_map_metadata || !this_map_metadata.town_map_position ||
343	         caller_map_metadata.town_map_position[0] != this_map_metadata.town_map_position[0]
344	        pbMessage(_INTL("The Trainer is out of range."))
345	        return false
346	      end
347	      return true
348	    end

352	    def get_random_trainer_for_incoming_call
353	      player_location = $game_map.metadata&.town_map_position
354	      return nil if !player_location
355	      player_region = player_location[0]
356	      valid_contacts = []
357	      $PokemonGlobal.phone.contacts.each do |contact|
358	        next if !contact.trainer? || !contact.visible?
359	        next if contact.map_id == $game_map.map_id ||
360	                pbGetMapNameFromId(contact.map_id) == $game_map.name
361	        caller_map_metadata = GameData::MapMetadata.try_get(contact.map_id)
362	        next if !caller_map_metadata || !caller_map_metadata.town_map_position
363	        next if caller_map_metadata.town_map_position[0] != player_region
364	        valid_contacts.push(contact)
365	      end
366	      return valid_contacts.sample
367	    end

414	    def play(dialogue, contact)
415	      start_message(contact)
416	      contact_pokemon_species  = get_random_contact_pokemon_species(contact)
417	      random_encounter_species = get_random_encounter_species(contact)
418	      contact_map_name         = get_map_name(contact)
419	      gender_colour_text       = ""
420	      if Settings::COLOR_PHONE_CALL_MESSAGES_BY_CONTACT_GENDER && contact.trainer?
421	        data = GameData::TrainerType.try_get(contact.trainer_type)
422	        if data
423	          case data.gender
424	          when 0 then gender_colour_text = "\\b"
425	          when 1 then gender_colour_text = "\\r"
426	          end
427	        end
428	      end
429	      messages = dialogue.split("\\m")
430	      messages.each_with_index do |message, i|
431	        message.gsub!(/\\TN/, _INTL(contact.name))
432	        message.gsub!(/\\TP/, contact_pokemon_species)
433	        message.gsub!(/\\TE/, random_encounter_species)
434	        message.gsub!(/\\TM/, contact_map_name)
435	        message += "\1" if i < messages.length - 1
436	        pbMessage(gender_colour_text + message)
437	      end
438	      end_message(contact)
439	    end
