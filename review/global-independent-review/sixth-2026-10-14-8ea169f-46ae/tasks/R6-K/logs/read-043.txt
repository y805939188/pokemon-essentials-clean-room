4	def _pbNextComb(comb, length)
5	  i = comb.length - 1
6	  loop do
7	    valid = true
8	    (i...comb.length).each do |j|
9	      if j == i
10	        comb[j] += 1
11	      else
12	        comb[j] = comb[i] + (j - i)
13	      end
14	      if comb[j] >= length
15	        valid = false
16	        break
17	      end
18	    end
19	    return true if valid
20	    i -= 1
21	    break unless i >= 0
22	  end
23	  return false
24	end

26	# Iterates through the array and yields each combination of _num_ elements in
27	# the array.
28	def pbEachCombination(array, num)
29	  return if array.length < num || num <= 0
30	  if array.length == num
31	    yield array
32	    return
33	  elsif num == 1
34	    array.each do |x|
35	      yield [x]
36	    end
37	    return
38	  end
39	  currentComb = []
40	  arr = []
41	  num.times { |i| currentComb[i] = i }
42	  loop do
43	    num.times { |i| arr[i] = array[currentComb[i]] }
44	    yield arr
45	    break unless _pbNextComb(currentComb, array.length)
46	  end
47	end

74	# This class is designed to favor different values more than a uniform
75	# random generator does.
76	#===============================================================================
77	class AntiRandom
78	  def initialize(size)
79	    @old = []
80	    @new = Array.new(size) { |i| i }
81	  end
82	
83	  def get
84	    if @new.length == 0   # No new values
85	      @new = @old.clone
86	      @old.clear
87	    end
88	    if @old.length > 0 && rand(7) == 0   # Get old value
89	      return @old[rand(@old.length)]
90	    end
91	    if @new.length > 0   # Get new value
92	      ret = @new.delete_at(rand(@new.length))
93	      @old.push(ret)
94	      return ret
95	    end
96	    return @old[rand(@old.length)]   # Get old value
97	  end
98	end

137	def getConstantName(mod, value, raise_if_none = true)
138	  mod = Object.const_get(mod) if mod.is_a?(Symbol)
139	  mod.constants.each do |c|
140	    return c.to_s if mod.const_get(c.to_sym) == value
141	  end
142	  raise _INTL("Value {1} not defined by a constant in {2}", value, mod.name) if raise_if_none
143	  return nil

166	def pbTimeEventDays(variableNumber, days = 0)
167	  return if !$game_variables
168	  return if !variableNumber || variableNumber < 0
169	  days = 0 if days < 0
170	  timenow = pbGetTimeNow
171	  time = timenow.to_f
172	  expiry = (time % 86_400.0) + (days * 86_400.0)
173	  $game_variables[variableNumber] = [time, expiry - time]
174	  $game_map&.refresh
175	end

177	def pbTimeEventValid(variableNumber)
178	  return false if !$game_variables
179	  return false if !variableNumber || variableNumber < 0
180	  ret = false
181	  value = $game_variables[variableNumber]
182	  if value.is_a?(Array)
183	    timenow = pbGetTimeNow
184	    ret = (timenow.to_f - value[0] > value[1])   # value[1] is age in seconds
185	    ret = false if value[1] <= 0   # zero age
186	  end
187	  if !ret
188	    $game_variables[variableNumber] = 0
189	    $game_map&.refresh
190	  end
191	  return ret
192	end

195	  if event.is_a?(Array)
196	    sprite = nil
197	    done = []
198	    event.each do |i|
199	      next if done.include?(i.id)
200	      spriteset = $scene.spriteset(i.map_id)
201	      sprite ||= spriteset&.addUserAnimation(id, i.x, i.y, tinting, 2)
202	      done.push(i.id)
203	    end
204	  else
205	    spriteset = $scene.spriteset(event.map_id)
206	    sprite = spriteset&.addUserAnimation(id, event.x, event.y, tinting, 2)
207	  end
208	  until sprite.disposed?
209	    Graphics.update
210	    Input.update
211	    pbUpdateSceneMap
212	  end
