3	#===============================================================================
4	class ItemIconSprite < Sprite
5	  attr_reader :item
6	
7	  # Height in pixels the item's icon graphic must be for it to be animated by
8	  # being a horizontal set of frames.
9	  ANIM_ICON_SIZE = 48
10	  # Time in seconds for one animation cycle of this item icon.
11	  ANIMATION_DURATION = 1.0
12	
13	  def initialize(x, y, item, viewport = nil)
14	    super(viewport)
15	    @animbitmap = nil
16	    @frames_count = 1
17	    @current_frame = 0
18	    self.x = x
19	    self.y = y
20	    @blankzero = false
21	    @forceitemchange = true
22	    self.item = item
23	    @forceitemchange = false
24	  end
25	
26	  def dispose
27	    @animbitmap&.dispose
28	    super
29	  end
30	
31	  def width
32	    return 0 if !self.bitmap || self.bitmap.disposed?
33	    return (@frames_count == 1) ? self.bitmap.width : ANIM_ICON_SIZE
34	  end
35	
36	  def height
37	    return (self.bitmap && !self.bitmap.disposed?) ? self.bitmap.height : 0
38	  end
39	
40	  def blankzero=(val)
41	    @blankzero = val
42	    @forceitemchange = true
43	    self.item = @item
44	    @forceitemchange = false
45	  end
46	
47	  def setOffset(offset = PictureOrigin::CENTER)
48	    @offset = offset
49	    changeOrigin
50	  end
51	
52	  def changeOrigin
53	    @offset = PictureOrigin::CENTER if !@offset
54	    case @offset
55	    when PictureOrigin::TOP_LEFT, PictureOrigin::TOP, PictureOrigin::TOP_RIGHT
56	      self.oy = 0
57	    when PictureOrigin::LEFT, PictureOrigin::CENTER, PictureOrigin::RIGHT
58	      self.oy = self.height / 2
59	    when PictureOrigin::BOTTOM_LEFT, PictureOrigin::BOTTOM, PictureOrigin::BOTTOM_RIGHT
60	      self.oy = self.height
61	    end
62	    case @offset
63	    when PictureOrigin::TOP_LEFT, PictureOrigin::LEFT, PictureOrigin::BOTTOM_LEFT
64	      self.ox = 0
65	    when PictureOrigin::TOP, PictureOrigin::CENTER, PictureOrigin::BOTTOM
66	      self.ox = self.width / 2
67	    when PictureOrigin::TOP_RIGHT, PictureOrigin::RIGHT, PictureOrigin::BOTTOM_RIGHT
68	      self.ox = self.width
69	    end
70	  end
71	
72	  def item=(value)
73	    return if @item == value && !@forceitemchange
74	    @item = value
75	    @animbitmap&.dispose
76	    @animbitmap = nil
77	    if @item || !@blankzero
78	      @animbitmap = AnimatedBitmap.new(GameData::Item.icon_filename(@item))
79	      self.bitmap = @animbitmap.bitmap
80	      if self.bitmap.height == ANIM_ICON_SIZE
81	        @frames_count = [(self.bitmap.width / ANIM_ICON_SIZE).floor, 1].max
82	        self.src_rect = Rect.new(0, 0, ANIM_ICON_SIZE, ANIM_ICON_SIZE)
83	      else
84	        @frames_count = 1
85	        self.src_rect = Rect.new(0, 0, self.bitmap.width, self.bitmap.height)
86	      end
87	      @current_frame = 0
88	    else
89	      self.bitmap = nil
90	    end
91	    changeOrigin
92	  end
93	
94	  def update_frame
95	    @current_frame = (@frames_count * (System.uptime % ANIMATION_DURATION) / ANIMATION_DURATION).floor
96	  end
97	
98	  def update
99	    @updating = true
100	    super
101	    if @animbitmap
102	      @animbitmap.update
103	      self.bitmap = @animbitmap.bitmap
104	      if @frames_count > 1
105	        update_frame
106	        self.src_rect.x = @current_frame * ANIM_ICON_SIZE
107	      end
108	    end
109	    @updating = false
110	  end
111	end

116	class HeldItemIconSprite < Sprite
117	  def initialize(x, y, pokemon, viewport = nil)
118	    super(viewport)
119	    self.x = x
120	    self.y = y
121	    @pokemon = pokemon
122	    @item = nil
123	    self.item = @pokemon.item_id
124	  end
125	
126	  def dispose
127	    @animbitmap&.dispose
128	    super
129	  end
130	
131	  def pokemon=(value)
132	    @pokemon = value
133	    self.item = @pokemon.item_id
134	  end
135	
136	  def item=(value)
137	    return if @item == value
138	    @item = value
139	    @animbitmap&.dispose
140	    @animbitmap = nil
141	    if @item
142	      @animbitmap = AnimatedBitmap.new(GameData::Item.held_icon_filename(@item))
143	      self.bitmap = @animbitmap.bitmap
144	    else
145	      self.bitmap = nil
146	    end
147	  end
148	
149	  def update
150	    super
151	    self.item = @pokemon.item_id
152	    if @animbitmap
153	      @animbitmap.update
154	      self.bitmap = @animbitmap.bitmap
155	    end
156	  end
