2	# The Bag object, which actually contains all the items.
3	#===============================================================================
4	class PokemonBag
5	  attr_reader   :pockets
6	  attr_accessor :last_viewed_pocket
7	  attr_accessor :last_pocket_selections
8	  attr_reader   :registered_items
9	  attr_reader   :ready_menu_selection
10	
11	  def self.pocket_names
12	    return Settings.bag_pocket_names
13	  end
14	
15	  def self.pocket_count
16	    return self.pocket_names.length
17	  end
18	
19	  def initialize
20	    @pockets = []
21	    (0..PokemonBag.pocket_count).each { |i| @pockets[i] = [] }
22	    reset_last_selections
23	    @registered_items = []
24	    @ready_menu_selection = [0, 0, 1]   # Used by the Ready Menu to remember cursor positions
25	  end
26	
27	  def reset_last_selections
28	    @last_viewed_pocket = 1
29	    @last_pocket_selections ||= []
30	    (0..PokemonBag.pocket_count).each { |i| @last_pocket_selections[i] = 0 }
31	  end
32	
33	  def clear
34	    @pockets.each { |pocket| pocket.clear }
35	    (PokemonBag.pocket_count + 1).times { |i| @last_pocket_selections[i] = 0 }
36	  end
37	
38	  #-----------------------------------------------------------------------------
39	
40	  # Gets the index of the current selected item in the pocket
41	  def last_viewed_index(pocket)
42	    if pocket <= 0 || pocket > PokemonBag.pocket_count
43	      raise ArgumentError.new(_INTL("Invalid pocket: {1}", pocket.inspect))
44	    end
45	    return [@last_pocket_selections[pocket], @pockets[pocket].length].min || 0
46	  end
47	
48	  # Sets the index of the current selected item in the pocket
49	  def set_last_viewed_index(pocket, value)
50	    if pocket <= 0 || pocket > PokemonBag.pocket_count
51	      raise ArgumentError.new(_INTL("Invalid pocket: {1}", pocket.inspect))
52	    end
53	    @last_pocket_selections[pocket] = value if value <= @pockets[pocket].length
54	  end
55	
56	  #-----------------------------------------------------------------------------
57	
58	  def quantity(item)
59	    item_data = GameData::Item.try_get(item)
60	    return 0 if !item_data
61	    pocket = item_data.pocket
62	    return ItemStorageHelper.quantity(@pockets[pocket], item_data.id)
63	  end
64	
65	  def has?(item, qty = 1)
66	    return quantity(item) >= qty
67	  end
68	  alias can_remove? has?
69	
70	  def can_add?(item, qty = 1)
71	    item_data = GameData::Item.try_get(item)
72	    return false if !item_data
73	    pocket = item_data.pocket
74	    max_size = max_pocket_size(pocket)
75	    max_size = @pockets[pocket].length + 1 if max_size < 0   # Infinite size
76	    return ItemStorageHelper.can_add?(
77	      @pockets[pocket], max_size, Settings::BAG_MAX_PER_SLOT, item_data.id, qty
78	    )
79	  end
80	
81	  def add(item, qty = 1)
82	    item_data = GameData::Item.try_get(item)
83	    return false if !item_data
84	    pocket = item_data.pocket
85	    max_size = max_pocket_size(pocket)
86	    max_size = @pockets[pocket].length + 1 if max_size < 0   # Infinite size
87	    ret = ItemStorageHelper.add(@pockets[pocket],
88	                                max_size, Settings::BAG_MAX_PER_SLOT, item_data.id, qty)
89	    if ret && Settings::BAG_POCKET_AUTO_SORT[pocket - 1]
90	      @pockets[pocket].sort! { |a, b| GameData::Item.keys.index(a[0]) <=> GameData::Item.keys.index(b[0]) }
91	    end
92	    return ret
93	  end
94	
95	  # Adds qty number of item. Doesn't add anything if it can't add all of them.
96	  def add_all(item, qty = 1)
97	    return false if !can_add?(item, qty)
98	    return add(item, qty)
99	  end
100	
101	  # Deletes as many of item as possible (up to qty), and returns whether it
102	  # managed to delete qty of them.
103	  def remove(item, qty = 1)
104	    item_data = GameData::Item.try_get(item)
105	    return false if !item_data
106	    pocket = item_data.pocket
107	    return ItemStorageHelper.remove(@pockets[pocket], item_data.id, qty)
108	  end
109	
110	  # Deletes qty number of item. Doesn't delete anything if there are less than
111	  # qty of the item in the Bag.
112	  def remove_all(item, qty = 1)
113	    return false if !can_remove?(item, qty)
114	    return remove(item, qty)
115	  end
116	
117	  # This only works if the old and new items are in the same pocket. Used for
118	  # switching on/off certain Key Items. Replaces all old_item in its pocket with
119	  # new_item.
120	  def replace_item(old_item, new_item)
121	    old_item_data = GameData::Item.try_get(old_item)
122	    new_item_data = GameData::Item.try_get(new_item)
123	    return false if !old_item_data || !new_item_data
124	    pocket = old_item_data.pocket
125	    old_id = old_item_data.id
126	    new_id = new_item_data.id
127	    ret = false
128	    @pockets[pocket].each do |item|
129	      next if !item || item[0] != old_id
130	      item[0] = new_id
131	      ret = true
132	    end
133	    return ret
134	  end
135	
136	  #-----------------------------------------------------------------------------
137	
138	  # Returns whether item has been registered for quick access in the Ready Menu.
139	  def registered?(item)
140	    item_data = GameData::Item.try_get(item)
141	    return false if !item_data
142	    return @registered_items.include?(item_data.id)
143	  end
144	
145	  # Registers the item in the Ready Menu.
146	  def register(item)
147	    item_data = GameData::Item.try_get(item)
148	    return if !item_data
149	    @registered_items.push(item_data.id) if !@registered_items.include?(item_data.id)
150	  end
151	
152	  # Unregisters the item from the Ready Menu.
153	  def unregister(item)
154	    item_data = GameData::Item.try_get(item)
155	    @registered_items.delete(item_data.id) if item_data
156	  end
157	
158	  #-----------------------------------------------------------------------------
159	
160	  private
161	
162	  def max_pocket_size(pocket)
163	    return Settings::BAG_MAX_POCKET_SIZE[pocket - 1] || -1
164	  end
165	end

168	# The PC item storage object, which actually contains all the items
169	#===============================================================================
170	class PCItemStorage
171	  attr_reader :items
172	
173	  MAX_SIZE     = 999   # Number of different slots in storage
174	  MAX_PER_SLOT = 999   # Max. number of items per slot
175	
176	  def initialize
177	    @items = []
178	    # Start storage with initial items (e.g. a Potion)
179	    GameData::Metadata.get.start_item_storage.each do |item|
180	      add(item) if GameData::Item.exists?(item)
181	    end
182	  end
183	
184	  def [](i)
185	    return @items[i]
186	  end
187	
188	  def length
189	    return @items.length
190	  end
191	
192	  def empty?
193	    return @items.length == 0
194	  end
195	
196	  def clear
197	    @items.clear
198	  end
199	
200	  # Unused
201	  def get_item(index)
202	    return (index < 0 || index >= @items.length) ? nil : @items[index][0]
203	  end
204	
205	  # Number of the item in the given index
206	  # Unused
207	  def get_item_count(index)
208	    return (index < 0 || index >= @items.length) ? 0 : @items[index][1]
209	  end
210	
211	  def quantity(item)
212	    item = GameData::Item.get(item).id
213	    return ItemStorageHelper.quantity(@items, item)
214	  end
215	
216	  def can_add?(item, qty = 1)
217	    item = GameData::Item.get(item).id
218	    return ItemStorageHelper.can_add?(@items, MAX_SIZE, MAX_PER_SLOT, item, qty)
219	  end
220	
221	  def add(item, qty = 1)
222	    item = GameData::Item.get(item).id
223	    return ItemStorageHelper.add(@items, MAX_SIZE, MAX_PER_SLOT, item, qty)
224	  end
225	
226	  def remove(item, qty = 1)
227	    item = GameData::Item.get(item).id
228	    return ItemStorageHelper.remove(@items, item, qty)
229	  end

232	#===============================================================================
233	# Implements methods that act on arrays of items.  Each element in an item
234	# array is itself an array of [itemID, itemCount].
235	# Used by the Bag, PC item storage, and Triple Triad.
236	#===============================================================================
237	module ItemStorageHelper
238	  # Returns the quantity of item in items
239	  def self.quantity(items, item)
240	    ret = 0
241	    items.each { |i| ret += i[1] if i && i[0] == item }
242	    return ret
243	  end
244	
245	  def self.can_add?(items, max_slots, max_per_slot, item, qty)
246	    raise "Invalid value for qty: #{qty}" if qty < 0
247	    return true if qty == 0
248	    max_slots.times do |i|
249	      item_slot = items[i]
250	      if !item_slot
251	        qty -= [qty, max_per_slot].min
252	        return true if qty == 0
253	      elsif item_slot[0] == item && item_slot[1] < max_per_slot
254	        new_amt = item_slot[1]
255	        new_amt = [new_amt + qty, max_per_slot].min
256	        qty -= (new_amt - item_slot[1])
257	        return true if qty == 0
258	      end
259	    end
260	    return false
261	  end
262	
263	  def self.add(items, max_slots, max_per_slot, item, qty)
264	    raise "Invalid value for qty: #{qty}" if qty < 0
265	    return true if qty == 0
266	    max_slots.times do |i|
267	      item_slot = items[i]
268	      if !item_slot
269	        items[i] = [item, [qty, max_per_slot].min]
270	        qty -= items[i][1]
271	        return true if qty == 0
272	      elsif item_slot[0] == item && item_slot[1] < max_per_slot
273	        new_amt = item_slot[1]
274	        new_amt = [new_amt + qty, max_per_slot].min
275	        qty -= (new_amt - item_slot[1])
276	        item_slot[1] = new_amt
277	        return true if qty == 0
278	      end
279	    end
280	    return false
281	  end
282	
283	  # Deletes an item (items array, max. size per slot, item, no. of items to delete)
284	  def self.remove(items, item, qty)
285	    raise "Invalid value for qty: #{qty}" if qty < 0
286	    return true if qty == 0
287	    ret = false
288	    items.each_with_index do |item_slot, i|
289	      next if !item_slot || item_slot[0] != item
290	      amount = [qty, item_slot[1]].min
291	      item_slot[1] -= amount
292	      qty -= amount
293	      items[i] = nil if item_slot[1] == 0
294	      next if qty > 0
295	      ret = true
296	      break
297	    end
298	    items.compact!
299	    return ret
300	  end
