9	  def pbRunPictures(pictures, sprites)
10	    loop do
11	      pictures.each { |pic| pic.update }
12	      sprites.each_with_index do |sprite, i|
13	        if sprite.is_a?(IconSprite)
14	          setPictureIconSprite(sprite, pictures[i])
15	        else
16	          setPictureSprite(sprite, pictures[i])
17	        end
18	      end
19	      Graphics.update
20	      Input.update
21	      running = false
22	      pictures.each { |pic| running = true if pic.running? }
23	      break if !running
24	    end
25	  end

27	  def pbStartScreen(pokemon, pokemon2, trader1, trader2)
28	    @sprites = {}
29	    @viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
30	    @viewport.z = 99999
31	    @pokemon  = pokemon
32	    @pokemon2 = pokemon2
33	    @trader1  = trader1
34	    @trader2  = trader2
35	    addBackgroundOrColoredPlane(@sprites, "background", "trade_bg",
36	                                Color.new(248, 248, 248), @viewport)
37	    @sprites["rsprite1"] = PokemonSprite.new(@viewport)
38	    @sprites["rsprite1"].setPokemonBitmap(@pokemon, false)
39	    @sprites["rsprite1"].setOffset(PictureOrigin::BOTTOM)
40	    @sprites["rsprite1"].x = Graphics.width / 2
41	    @sprites["rsprite1"].y = 264
42	    @sprites["rsprite1"].z = 10
43	    @pokemon.species_data.apply_metrics_to_sprite(@sprites["rsprite1"], 1)
44	    @sprites["rsprite2"] = PokemonSprite.new(@viewport)
45	    @sprites["rsprite2"].setPokemonBitmap(@pokemon2, false)
46	    @sprites["rsprite2"].setOffset(PictureOrigin::BOTTOM)
47	    @sprites["rsprite2"].x = Graphics.width / 2
48	    @sprites["rsprite2"].y = 264
49	    @sprites["rsprite2"].z = 10
50	    @pokemon2.species_data.apply_metrics_to_sprite(@sprites["rsprite2"], 1)
51	    @sprites["rsprite2"].visible = false
52	    @sprites["msgwindow"] = pbCreateMessageWindow(@viewport)
53	    pbFadeInAndShow(@sprites)
54	  end

56	  def pbScene1
57	    spriteBall = IconSprite.new(0, 0, @viewport)
58	    pictureBall = PictureEx.new(0)
59	    picturePoke = PictureEx.new(0)
60	    ballimage = sprintf("Graphics/Battle animations/ball_%s", @pokemon.poke_ball)
61	    ballopenimage = sprintf("Graphics/Battle animations/ball_%s_open", @pokemon.poke_ball)
62	    # Starting position of ball
63	    pictureBall.setXY(0, Graphics.width / 2, 48)
64	    pictureBall.setName(0, ballimage)
65	    pictureBall.setSrcSize(0, 32, 64)
66	    pictureBall.setOrigin(0, PictureOrigin::CENTER)
67	    pictureBall.setVisible(0, true)
68	    # Starting position of sprite
69	    picturePoke.setXY(0, @sprites["rsprite1"].x, @sprites["rsprite1"].y)
70	    picturePoke.setOrigin(0, PictureOrigin::BOTTOM)
71	    picturePoke.setVisible(0, true)
72	    # Change Pokémon color
73	    picturePoke.moveColor(2, 5, Color.new(248, 176, 140))
74	    # Recall
75	    delay = picturePoke.totalDuration
76	    picturePoke.setSE(delay, "Battle recall")
77	    pictureBall.setName(delay, ballopenimage)
78	    pictureBall.setSrcSize(delay, 32, 64)
79	    # Move sprite to ball
80	    picturePoke.moveZoom(delay, 8, 0)
81	    picturePoke.moveXY(delay, 8, Graphics.width / 2, 48)
82	    picturePoke.setSE(delay + 5, "Battle jump to ball")
83	    picturePoke.setVisible(delay + 8, false)
84	    delay = picturePoke.totalDuration + 1
85	    pictureBall.setName(delay, ballimage)
86	    pictureBall.setSrcSize(delay, 32, 64)
87	    # Make Poké Ball go off the top of the screen
88	    delay = picturePoke.totalDuration + 10
89	    pictureBall.moveXY(delay, 6, Graphics.width / 2, -32)
90	    # Play animation
91	    pbRunPictures(
92	      [picturePoke, pictureBall],
93	      [@sprites["rsprite1"], spriteBall]
94	    )
95	    spriteBall.dispose
96	  end

98	  def pbScene2
99	    spriteBall = IconSprite.new(0, 0, @viewport)
100	    pictureBall = PictureEx.new(0)
101	    picturePoke = PictureEx.new(0)
102	    ballimage = sprintf("Graphics/Battle animations/ball_%s", @pokemon2.poke_ball)
103	    ballopenimage = sprintf("Graphics/Battle animations/ball_%s_open", @pokemon2.poke_ball)
104	    # Starting position of ball
105	    pictureBall.setXY(0, Graphics.width / 2, -32)
106	    pictureBall.setName(0, ballimage)
107	    pictureBall.setSrcSize(0, 32, 64)
108	    pictureBall.setOrigin(0, PictureOrigin::CENTER)
109	    pictureBall.setVisible(0, true)
110	    # Starting position of sprite
111	    picturePoke.setOrigin(0, PictureOrigin::BOTTOM)
112	    picturePoke.setZoom(0, 0)
113	    picturePoke.setColor(0, Color.new(248, 176, 240))
114	    picturePoke.setVisible(0, false)
115	    # Dropping ball
116	    y = Graphics.height - 96 - 16 - 16   # end point of Poké Ball
117	    delay = picturePoke.totalDuration + 2
118	    4.times do |i|
119	      t = [4, 4, 3, 2][i]   # Time taken to rise or fall for each bounce
120	      d = [1, 2, 4, 8][i]   # Fraction of the starting height each bounce rises to
121	      delay -= t if i == 0
122	      if i > 0
123	        pictureBall.setZoomXY(delay, 100 + (5 * (5 - i)), 100 - (5 * (5 - i)))   # Squish
124	        pictureBall.moveZoom(delay, 2, 100)                      # Unsquish
125	        pictureBall.moveXY(delay, t, Graphics.width / 2, y - (100 / d))
126	      end
127	      pictureBall.moveXY(delay + t, t, Graphics.width / 2, y)
128	      pictureBall.setSE(delay + (2 * t), "Battle ball drop")
129	      delay = pictureBall.totalDuration
130	    end
131	    picturePoke.setXY(delay, Graphics.width / 2, y)
132	    # Open Poké Ball
133	    delay = pictureBall.totalDuration + 15
134	    pictureBall.setSE(delay, "Battle recall")
135	    pictureBall.setName(delay, ballopenimage)
136	    pictureBall.setSrcSize(delay, 32, 64)
137	    pictureBall.setVisible(delay + 5, false)
138	    # Pokémon appears and enlarges
139	    picturePoke.setVisible(delay, true)
140	    picturePoke.moveZoom(delay, 8, 100)
141	    picturePoke.moveXY(delay, 8, Graphics.width / 2, @sprites["rsprite2"].y)
142	    # Return Pokémon's color to normal and play cry
143	    delay = picturePoke.totalDuration
144	    picturePoke.moveColor(delay, 5, Color.new(248, 176, 240, 0))
145	    cry = GameData::Species.cry_filename_from_pokemon(@pokemon2)
146	    picturePoke.setSE(delay, cry) if cry
147	    cry_length = (GameData::Species.cry_length(@pokemon2) * 20).ceil
148	    picturePoke.setVisible(delay + cry_length + 4, true)   # Time for the cry to play
149	    # Play animation
150	    pbRunPictures(
151	      [picturePoke, pictureBall],
152	      [@sprites["rsprite2"], spriteBall]
153	    )
154	    spriteBall.dispose
155	  end

157	  def pbEndScreen(need_fade_out = true)
158	    pbDisposeMessageWindow(@sprites["msgwindow"]) if @sprites["msgwindow"]
159	    pbFadeOutAndHide(@sprites) if need_fade_out
160	    pbDisposeSpriteHash(@sprites)
161	    @viewport.dispose
162	    newspecies = @pokemon2.check_evolution_on_trade(@pokemon)
163	    if newspecies
164	      evo = PokemonEvolutionScene.new
165	      evo.pbStartScreen(@pokemon2, newspecies)
166	      evo.pbEvolution(false)
167	      evo.pbEndScreen
168	    end
169	  end

171	  def pbTrade
172	    was_owned = $player.owned?(@pokemon2.species)
173	    $player.pokedex.register(@pokemon2)
174	    $player.pokedex.set_owned(@pokemon2.species)
175	    pbBGMStop
176	    @pokemon.play_cry
177	    speciesname1 = GameData::Species.get(@pokemon.species).name
178	    speciesname2 = GameData::Species.get(@pokemon2.species).name
179	    pbMessageDisplay(@sprites["msgwindow"],
180	                     _ISPRINTF("{1:s}\nID: {2:05d}   OT: {3:s}",
181	                               @pokemon.name, @pokemon.owner.public_id, @pokemon.owner.name) + "\\wtnp[0]") { pbUpdate }
182	    pbMessageWaitForInput(@sprites["msgwindow"], 50, true) { pbUpdate }
183	    pbPlayDecisionSE
184	    pbBGMPlay("Evolution")
185	    pbScene1
186	    pbMessageDisplay(@sprites["msgwindow"],
187	                     _INTL("For {1}'s {2},\n{3} sends {4}.", @trader1, speciesname1, @trader2, speciesname2) + "\1") { pbUpdate }
188	    pbMessageDisplay(@sprites["msgwindow"],
189	                     _INTL("{1} bids farewell to {2}.", @trader2, speciesname2)) { pbUpdate }
190	    pbScene2
191	    pbBGMStop
192	    pbMEPlay("Battle capture success")
193	    pbMessageDisplay(@sprites["msgwindow"],
194	                     _ISPRINTF("{1:s}\nID: {2:05d}   OT: {3:s}",
195	                               @pokemon2.name, @pokemon2.owner.public_id, @pokemon2.owner.name) + "\1") { pbUpdate }
196	    pbMessageDisplay(@sprites["msgwindow"],
197	                     _INTL("Take good care of {1}.", speciesname2)) { pbUpdate }
198	    # Show Pokédex entry for new species if it hasn't been owned before
199	    if Settings::SHOW_NEW_SPECIES_POKEDEX_ENTRY_MORE_OFTEN && !was_owned &&
200	       $player.has_pokedex && $player.pokedex.species_in_unlocked_dex?(@pokemon2.species)
201	      pbMessageDisplay(@sprites["msgwindow"],
202	                       _INTL("{1}'s data was added to the Pokédex.", speciesname2)) { pbUpdate }
203	      $player.pokedex.register_last_seen(@pokemon2)
204	      pbFadeOutIn do
205	        scene = PokemonPokedexInfo_Scene.new
206	        screen = PokemonPokedexInfoScreen.new(scene)
207	        screen.pbDexEntry(@pokemon2.species)
208	        pbEndScreen(false)
209	      end
210	    end
211	  end

217	def pbStartTrade(pokemonIndex, newpoke, nickname, trainerName, trainerGender = 0)
218	  $stats.trade_count += 1
219	  myPokemon = $player.party[pokemonIndex]
220	  yourPokemon = nil
221	  resetmoves = true
222	  if newpoke.is_a?(Pokemon)
223	    newpoke.owner = Pokemon::Owner.new_foreign(trainerName, trainerGender)
224	    yourPokemon = newpoke
225	    resetmoves = false
226	  else
227	    species_data = GameData::Species.try_get(newpoke)
228	    raise _INTL("Species {1} does not exist.", newpoke) if !species_data
229	    yourPokemon = Pokemon.new(species_data.id, myPokemon.level)
230	    yourPokemon.owner = Pokemon::Owner.new_foreign(trainerName, trainerGender)
231	  end
232	  yourPokemon.name          = nickname
233	  yourPokemon.obtain_method = 2   # traded
234	  yourPokemon.reset_moves if resetmoves
235	  yourPokemon.record_first_moves
236	  pbFadeOutInWithMusic do
237	    evo = PokemonTrade_Scene.new
238	    evo.pbStartScreen(myPokemon, yourPokemon, $player.name, trainerName)
239	    evo.pbTrade
240	    evo.pbEndScreen
241	  end
242	  $player.party[pokemonIndex] = yourPokemon
243	end
