308	    # Decides which Dex lists are able to be viewed (i.e. they are unlocked and
309	    # have at least 1 seen species in them), and saves all accessible Dex region
310	    # numbers into {#accessible_dexes}. National Dex comes after all regional
311	    # Dexes.
312	    # If the Dex list shown depends on the player's location, this just decides
313	    # if a species in the current region has been seen - doesn't look at other
314	    # regions.
315	    def refresh_accessible_dexes
316	      @accessible_dexes = []
317	      if Settings::USE_CURRENT_REGION_DEX
318	        region = pbGetCurrentRegion
319	        region = -1 if region >= dexes_count - 1
320	        @accessible_dexes[0] = region if self.seen_any?(region)
321	        return
322	      end
323	      if dexes_count == 1   # Only National Dex is defined
324	        @accessible_dexes.push(-1) if self.unlocked?(0) && self.seen_any?
325	      else   # Regional Dexes + National Dex
326	        dexes_count.times do |i|
327	          dex_list_to_check = (i == dexes_count - 1) ? -1 : i
328	          if self.unlocked?(i) && self.seen_any?(dex_list_to_check)
329	            @accessible_dexes.push(dex_list_to_check)
330	          end
331	        end
332	      end
333	    end

335	    def species_in_unlocked_dex?(species)
336	      return true if @unlocked_dexes.last
337	      (@unlocked_dexes.length - 1).times do |i|
338	        next if !self.unlocked?(i)
339	        return true if pbGetRegionalNumber(i, species) > 0
340	      end
341	      return false
