708	# Only called when in the party screen and having chosen an item to be used on
709	# the selected Pokémon
710	def pbUseItemOnPokemon(item, pkmn, scene)
711	  itm = GameData::Item.get(item)
712	  # TM or HM
713	  if itm.is_machine?
714	    machine = itm.move
715	    return false if !machine
716	    movename = GameData::Move.get(machine).name
717	    if pkmn.shadowPokemon?
718	      pbMessage(_INTL("Shadow Pokémon can't be taught any moves.")) { scene.pbUpdate }
719	    elsif !pkmn.compatible_with_move?(machine)
720	      pbMessage(_INTL("{1} can't learn {2}.", pkmn.name, movename)) { scene.pbUpdate }
721	    else
722	      pbMessage("\\se[PC access]" + _INTL("You booted up the {1}.", itm.portion_name) + "\1") { scene.pbUpdate }
723	      if pbConfirmMessage(_INTL("Do you want to teach {1} to {2}?", movename, pkmn.name)) { scene.pbUpdate }
724	        if pbLearnMove(pkmn, machine, false, true) { scene.pbUpdate }
725	          $bag.remove(item) if itm.consumed_after_use?
726	          return true
727	        end
728	      end
729	    end
730	    return false
731	  end
732	  # Other item
733	  qty = 1
734	  max_at_once = ItemHandlers.triggerUseOnPokemonMaximum(item, pkmn)
735	  max_at_once = [max_at_once, $bag.quantity(item)].min
736	  if max_at_once > 1
737	    qty = scene.scene.pbChooseNumber(
738	      _INTL("How many {1} do you want to use?", itm.portion_name_plural), max_at_once
739	    )
740	    scene.scene.pbSetHelpText("") if scene.is_a?(PokemonPartyScreen)
741	  end
742	  return false if qty <= 0
743	  ret = ItemHandlers.triggerUseOnPokemon(item, qty, pkmn, scene)
744	  scene.pbClearAnnotations
745	  scene.pbHardRefresh
746	  if ret && itm.consumed_after_use?
747	    $bag.remove(item, qty)
748	    if !$bag.has?(item)
749	      pbMessage(_INTL("You used your last {1}.", itm.portion_name)) { scene.pbUpdate }
750	    end
751	  end
752	  return ret
753	end

781	def pbGiveItemToPokemon(item, pkmn, scene, pkmnid = 0)
782	  newitemname = GameData::Item.get(item).portion_name
783	  if pkmn.egg?
784	    scene.pbDisplay(_INTL("Eggs can't hold items."))
785	    return false
786	  elsif pkmn.mail
787	    scene.pbDisplay(_INTL("{1}'s mail must be removed before giving it an item.", pkmn.name))
788	    return false if !pbTakeItemFromPokemon(pkmn, scene)
789	  end
790	  if pkmn.hasItem?
791	    olditemname = pkmn.item.portion_name
792	    if newitemname.starts_with_vowel?
793	      scene.pbDisplay(_INTL("{1} is already holding an {2}.", pkmn.name, olditemname) + "\1")
794	    else
795	      scene.pbDisplay(_INTL("{1} is already holding a {2}.", pkmn.name, olditemname) + "\1")
796	    end
797	    if scene.pbConfirm(_INTL("Would you like to switch the two items?"))
798	      $bag.remove(item)
799	      if !$bag.add(pkmn.item)
800	        raise _INTL("Couldn't re-store deleted item in Bag somehow") if !$bag.add(item)
801	        scene.pbDisplay(_INTL("The Bag is full. The Pokémon's item could not be removed."))
802	      elsif GameData::Item.get(item).is_mail?
803	        if pbWriteMail(item, pkmn, pkmnid, scene)
804	          pkmn.item = item
805	          scene.pbDisplay(_INTL("Took the {1} from {2} and gave it the {3}.", olditemname, pkmn.name, newitemname))
806	          return true
807	        elsif !$bag.add(item)
808	          raise _INTL("Couldn't re-store deleted item in Bag somehow")
809	        end
810	      else
811	        pkmn.item = item
812	        scene.pbDisplay(_INTL("Took the {1} from {2} and gave it the {3}.", olditemname, pkmn.name, newitemname))
813	        return true
814	      end
815	    end
816	  elsif !GameData::Item.get(item).is_mail? || pbWriteMail(item, pkmn, pkmnid, scene)
817	    $bag.remove(item)
818	    pkmn.item = item
819	    scene.pbDisplay(_INTL("{1} is now holding the {2}.", pkmn.name, newitemname))
820	    return true
821	  end
822	  return false
823	end

825	def pbTakeItemFromPokemon(pkmn, scene)
826	  ret = false
827	  if !pkmn.hasItem?
828	    scene.pbDisplay(_INTL("{1} isn't holding anything.", pkmn.name))
829	  elsif !$bag.can_add?(pkmn.item)
830	    scene.pbDisplay(_INTL("The Bag is full. The Pokémon's item could not be removed."))
831	  elsif pkmn.mail
832	    if scene.pbConfirm(_INTL("Save the removed mail in your PC?"))
833	      if pbMoveToMailbox(pkmn)
834	        scene.pbDisplay(_INTL("The mail was saved in your PC."))
835	        pkmn.item = nil
836	        ret = true
837	      else
838	        scene.pbDisplay(_INTL("Your PC's Mailbox is full."))
839	      end
840	    elsif scene.pbConfirm(_INTL("If the mail is removed, its message will be lost. OK?"))
841	      $bag.add(pkmn.item)
842	      scene.pbDisplay(_INTL("Received the {1} from {2}.", pkmn.item.portion_name, pkmn.name))
843	      pkmn.item = nil
844	      pkmn.mail = nil
845	      ret = true
846	    end
847	  else
848	    $bag.add(pkmn.item)
849	    scene.pbDisplay(_INTL("Received the {1} from {2}.", pkmn.item.portion_name, pkmn.name))
850	    pkmn.item = nil
851	    ret = true
852	  end
853	  return ret
854	end

892	# Shows a list of items to choose from, with the chosen item's ID being stored
893	# in the given Game Variable. Only items which the player has are listed.
894	def pbChooseItemFromList(message, variable, *args)
895	  commands = []
896	  itemid   = []
897	  args.each do |item|
898	    next if !GameData::Item.exists?(item)
899	    itm = GameData::Item.get(item)
900	    next if !$bag.has?(itm)
901	    commands.push(itm.name)
902	    itemid.push(itm.id)
903	  end
904	  if commands.length == 0
905	    $game_variables[variable] = :NONE
906	    return nil
907	  end
908	  commands.push(_INTL("Cancel"))
909	  itemid.push(nil)
910	  ret = pbMessage(message, commands, -1)
911	  if ret < 0 || ret >= commands.length - 1
912	    $game_variables[variable] = :NONE
913	    return nil
914	  end
915	  $game_variables[variable] = itemid[ret] || :NONE
916	  return itemid[ret]
