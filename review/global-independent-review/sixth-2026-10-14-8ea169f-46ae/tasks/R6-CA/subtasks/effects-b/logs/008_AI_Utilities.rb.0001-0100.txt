5:   def pbAIRandom(x); return rand(x); end

9:   def each_battler
10:     @battlers.each_with_index do |battler, i|
11:       next if !battler || battler.fainted?
12:       yield battler, i
13:     end
14:   end
15: 
16:   def each_foe_battler(side)
17:     @battlers.each_with_index do |battler, i|
18:       next if !battler || battler.fainted?
19:       yield battler, i if i.even? != side.even?
20:     end
21:   end
22: 
23:   def each_same_side_battler(side)
24:     @battlers.each_with_index do |battler, i|
25:       next if !battler || battler.fainted?
26:       yield battler, i if i.even? == side.even?
27:     end
28:   end
29: 
30:   def each_ally(index)
31:     @battlers.each_with_index do |battler, i|
32:       next if !battler || battler.fainted?
33:       yield battler, i if i != index && i.even? == index.even?
34:     end
35:   end

39:   # Assumes that pkmn's ability is not negated by a global effect (e.g.
40:   # Neutralizing Gas).
41:   # pkmn is either a Battle::AI::AIBattler or a Pokemon.
42:   # move is a Battle::Move or a Pokemon::Move.
43:   def pokemon_can_absorb_move?(pkmn, move, move_type)
44:     return false if pkmn.is_a?(Battle::AI::AIBattler) && !pkmn.ability_active?
45:     # Check pkmn's ability
46:     # Anything with a Battle::AbilityEffects::MoveImmunity handler
47:     case pkmn.ability_id
48:     when :BULLETPROOF
49:       move_data = GameData::Move.get(move.id)
50:       return move_data.has_flag?("Bomb")
51:     when :FLASHFIRE
52:       return move_type == :FIRE
53:     when :LIGHTNINGROD, :MOTORDRIVE, :VOLTABSORB
54:       return move_type == :ELECTRIC
55:     when :SAPSIPPER
56:       return move_type == :GRASS
57:     when :SOUNDPROOF
58:       move_data = GameData::Move.get(move.id)
59:       return move_data.has_flag?("Sound")
60:     when :STORMDRAIN, :WATERABSORB, :DRYSKIN
61:       return move_type == :WATER
62:     when :TELEPATHY
63:       # NOTE: The move is being used by a foe of pkmn.
64:       return false
65:     when :WONDERGUARD
66:       types = pkmn.types
67:       types = pkmn.pbTypes(true) if pkmn.is_a?(Battle::AI::AIBattler)
68:       return !Effectiveness.super_effective_type?(move_type, *types)
69:     end
70:     return false
71:   end
72: 
73:   # Used by Toxic Spikes.
74:   def pokemon_can_be_poisoned?(pkmn)
75:     # Check pkmn's immunity to being poisoned
76:     return false if @battle.field.terrain == :Misty
77:     return false if pkmn.hasType?(:POISON)
78:     return false if pkmn.hasType?(:STEEL)
79:     return false if pkmn.hasAbility?(:IMMUNITY)
80:     return false if pkmn.hasAbility?(:PASTELVEIL)
81:     return false if pkmn.hasAbility?(:FLOWERVEIL) && pkmn.hasType?(:GRASS)
82:     return false if pkmn.hasAbility?(:LEAFGUARD) && [:Sun, :HarshSun].include?(@battle.pbWeather)
83:     return false if pkmn.hasAbility?(:COMATOSE) && pkmn.isSpecies?(:KOMALA)
84:     return false if pkmn.hasAbility?(:SHIELDSDOWN) && pkmn.isSpecies?(:MINIOR) && pkmn.form < 7
85:     return true
86:   end
87: 
88:   def pokemon_airborne?(pkmn)
89:     return false if pkmn.hasItem?(:IRONBALL)
90:     return false if @battle.field.effects[PBEffects::Gravity] > 0
91:     return true if pkmn.hasType?(:FLYING)
92:     return true if pkmn.hasAbility?(:LEVITATE)
93:     return true if pkmn.hasItem?(:AIRBALLOON)
94:     return false
95:   end
