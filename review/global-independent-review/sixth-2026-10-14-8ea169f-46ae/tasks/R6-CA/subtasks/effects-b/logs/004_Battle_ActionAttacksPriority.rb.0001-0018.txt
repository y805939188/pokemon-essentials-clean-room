3:   # Choosing a move/target
4:   #=============================================================================
5:   def pbCanChooseMove?(idxBattler, idxMove, showMessages, sleepTalk = false)
6:     battler = @battlers[idxBattler]
7:     move = battler.moves[idxMove]
8:     return false unless move
9:     if move.pp == 0 && move.total_pp > 0 && !sleepTalk
10:       pbDisplayPaused(_INTL("There's no PP left for this move!")) if showMessages
11:       return false
12:     end
13:     if battler.effects[PBEffects::Encore] > 0
14:       idxEncoredMove = battler.pbEncoredMoveIndex
15:       return false if idxEncoredMove >= 0 && idxMove != idxEncoredMove
16:     end
17:     return battler.pbCanChooseMove?(move, true, showMessages, sleepTalk)
18:   end
