68: # OHKO. Accuracy increases by difference between levels of user and target.
69: #===============================================================================
70: class Battle::Move::OHKO < Battle::Move::FixedDamageMove
71:   def pbFailsAgainstTarget?(user, target, show_message)
72:     if target.level > user.level
73:       @battle.pbDisplay(_INTL("{1} is unaffected!", target.pbThis)) if show_message
74:       return true
75:     end
76:     if target.hasActiveAbility?(:STURDY) && !@battle.moldBreaker
77:       if show_message
78:         @battle.pbShowAbilitySplash(target)
79:         if Battle::Scene::USE_ABILITY_SPLASH
80:           @battle.pbDisplay(_INTL("But it failed to affect {1}!", target.pbThis(true)))
81:         else
82:           @battle.pbDisplay(_INTL("But it failed to affect {1} because of its {2}!",
83:                                   target.pbThis(true), target.abilityName))
84:         end
85:         @battle.pbHideAbilitySplash(target)
86:       end
87:       return true
88:     end
89:     return false
90:   end
91: 
92:   def pbAccuracyCheck(user, target)
93:     acc = @accuracy + user.level - target.level
94:     return @battle.pbRandom(100) < acc
95:   end
96: 
97:   def pbFixedDamage(user, target)
98:     return target.totalhp
99:   end
100: 
101:   def pbHitEffectivenessMessages(user, target, numTargets = 1)
102:     super
103:     @battle.pbDisplay(_INTL("It's a one-hit KO!")) if target.fainted?
104:   end
105: end

108: # OHKO. Accuracy increases by difference between levels of user and target.
109: # Lower accuracy when used by a non-Ice-type Pokémon. Doesn't affect Ice-type

118:     return super

136: #===============================================================================
137: # The target's ally loses 1/16 of its max HP. (Flame Burst)
138: #===============================================================================
139: class Battle::Move::DamageTargetAlly < Battle::Move
140:   def pbEffectWhenDealingDamage(user, target)
141:     hitAlly = []
142:     target.allAllies.each do |b|
143:       next if !b.near?(target.index)
144:       next if !b.takesIndirectDamage?
145:       hitAlly.push([b.index, b.hp])
146:       b.pbReduceHP(b.totalhp / 16, false)
147:     end
148:     if hitAlly.length == 2
149:       @battle.pbDisplay(_INTL("The bursting flame hit {1} and {2}!",
150:                               @battle.battlers[hitAlly[0][0]].pbThis(true),
151:                               @battle.battlers[hitAlly[1][0]].pbThis(true)))
152:     elsif hitAlly.length > 0
153:       hitAlly.each do |b|
154:         @battle.pbDisplay(_INTL("The bursting flame hit {1}!",
155:                                 @battle.battlers[b[0]].pbThis(true)))
156:       end
157:     end
158:     hitAlly.each { |b| @battle.battlers[b[0]].pbItemHPHealCheck }
159:   end
160: end

172: # Power increases the less HP the user has. (Flail, Reversal)
173: #===============================================================================
174: class Battle::Move::PowerLowerWithUserHP < Battle::Move
175:   def pbBaseDamage(baseDmg, user, target)
176:     ret = 20
177:     n = 48 * user.hp / user.totalhp
178:     if n < 2
179:       ret = 200
180:     elsif n < 5
181:       ret = 150
182:     elsif n < 10
183:       ret = 100
184:     elsif n < 17
185:       ret = 80
186:     elsif n < 33
187:       ret = 40
188:     end
189:     return ret
190:   end
191: end
