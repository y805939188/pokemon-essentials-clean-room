1300: # NOTE: This allows Hidden Power to be Fairy-type (if you have that type in your
1301: #       game). I don't care that the official games don't work like that.
1302: def pbHiddenPower(pkmn)
1303:   iv = pkmn.iv
1304:   idxType = 0
1305:   power = 60
1306:   types = []
1307:   GameData::Type.each do |t|
1308:     types[t.icon_position] ||= []
1309:     types[t.icon_position].push(t.id) if !t.pseudo_type && ![:NORMAL, :SHADOW].include?(t.id)
1310:   end
1311:   types.flatten!.compact!
1312:   idxType |= (iv[:HP] & 1)
1313:   idxType |= (iv[:ATTACK] & 1) << 1
1314:   idxType |= (iv[:DEFENSE] & 1) << 2
1315:   idxType |= (iv[:SPEED] & 1) << 3
1316:   idxType |= (iv[:SPECIAL_ATTACK] & 1) << 4
1317:   idxType |= (iv[:SPECIAL_DEFENSE] & 1) << 5
1318:   idxType = (types.length - 1) * idxType / 63
1319:   type = types[idxType]
1320:   if Settings::MECHANICS_GENERATION <= 5
1321:     powerMin = 30
1322:     powerMax = 70
1323:     power |= (iv[:HP] & 2) >> 1
1324:     power |= (iv[:ATTACK] & 2)
1325:     power |= (iv[:DEFENSE] & 2) << 1
1326:     power |= (iv[:SPEED] & 2) << 2
1327:     power |= (iv[:SPECIAL_ATTACK] & 2) << 3
1328:     power |= (iv[:SPECIAL_DEFENSE] & 2) << 4
1329:     power = powerMin + ((powerMax - powerMin) * power / 63)
1330:   end
1331:   return [type, power]
1332: end
