3: #===============================================================================
4: Battle::AI::Handlers::GeneralMoveAgainstTargetScore.add(:shiny_target,
5:   proc { |score, move, user, target, ai, battle|
6:     if target.wild? && target.battler.shiny?
7:       old_score = score
8:       score -= 20
9:       PBDebug.log_score_change(score - old_score, "avoid attacking a shiny wild Pokémon")
10:     end
11:     next score
12:   }

30: # If user is frozen, prefer a move that can thaw the user.
31: #===============================================================================
32: Battle::AI::Handlers::GeneralMoveScore.add(:thawing_move_when_frozen,
33:   proc { |score, move, user, ai, battle|
34:     if ai.trainer.medium_skill? && user.status == :FROZEN
35:       old_score = score
36:       if move.move.thawsUser?
37:         score += 20
38:         PBDebug.log_score_change(score - old_score, "move will thaw the user")
39:       elsif user.check_for_move { |m| m.thawsUser? }
40:         score -= 20   # Don't prefer this move if user knows another move that thaws
41:         PBDebug.log_score_change(score - old_score, "user knows another move will thaw it")
42:       end
43:     end
44:     next score
45:   }
46: )

48: #===============================================================================
49: # Prefer using a priority move if the user is slower than the target and...
50: # - the user is at low HP, or
51: # - the target is predicted to be knocked out by the move.
52: #===============================================================================
53: Battle::AI::Handlers::GeneralMoveAgainstTargetScore.add(:priority_move_against_faster_target,
54:   proc { |score, move, user, target, ai, battle|
55:     if ai.trainer.high_skill? && target.faster_than?(user) && move.rough_priority(user) > 0
56:       # User is at risk of being knocked out
57:       if ai.trainer.has_skill_flag?("HPAware") && user.hp < user.totalhp / 3
58:         old_score = score
59:         score += 8
60:         PBDebug.log_score_change(score - old_score, "user at low HP and move has priority over faster target")
61:       end
62:       # Target is predicted to be knocked out by the move
63:       if move.damagingMove? && move.rough_damage >= target.hp
64:         old_score = score
65:         score += 8
66:         PBDebug.log_score_change(score - old_score, "target at low HP and move has priority over faster target")
67:       end
68:       # Any foe knows Quick Guard and can protect against priority moves
69:       old_score = score
70:       ai.each_foe_battler(user.side) do |b, i|
71:         next if !b.has_move_with_function?("ProtectUserSideFromPriorityMoves")
72:         next if Settings::MECHANICS_GENERATION <= 5 && b.effects[PBEffects::ProtectRate] > 1
73:         score -= 5
74:       end
75:       if score != old_score
76:         PBDebug.log_score_change(score - old_score, "a foe knows Quick Guard and may protect against priority moves")
77:       end
78:     end
79:     next score
80:   }
81: )

83: #===============================================================================
84: # Don't prefer a move that can be Magic Coated if the target (or any foe if the
85: # move doesn't have a target) knows Magic Coat/has Magic Bounce.
86: #===============================================================================
87: Battle::AI::Handlers::GeneralMoveAgainstTargetScore.add(:target_can_Magic_Coat_or_Bounce_move,
88:   proc { |score, move, user, target, ai, battle|
89:     if move.statusMove? && move.move.canMagicCoat? && target.opposes?(user) &&
90:        (target.faster_than?(user) || !target.battler.semiInvulnerable?)
91:       old_score = score
92:       if !battle.moldBreaker && target.has_active_ability?(:MAGICBOUNCE)
93:         score = Battle::AI::MOVE_USELESS_SCORE
94:         PBDebug.log_score_change(score - old_score, "useless because target will Magic Bounce it")
95:       elsif target.has_move_with_function?("BounceBackProblemCausingStatusMoves") &&
96:             target.can_attack? && !target.battler.semiInvulnerable?
97:         score -= 7
98:         PBDebug.log_score_change(score - old_score, "target knows Magic Coat and could bounce it")
99:       end
100:     end
101:     next score
102:   }
103: )

128: # Don't prefer a move that can be Snatched if any other battler knows Snatch.
129: #===============================================================================
130: Battle::AI::Handlers::GeneralMoveScore.add(:any_battler_can_Snatch_move,
131:   proc { |score, move, user, ai, battle|
132:     if move.statusMove? && move.move.canSnatch?
133:       ai.each_battler do |b, i|
134:         next if b.index == user.index
135:         next if b.effects[PBEffects::SkyDrop] >= 0
136:         next if !b.has_move_with_function?("StealAndUseBeneficialStatusMove")
137:         old_score = score
138:         score -= 7
139:         PBDebug.log_score_change(score - old_score, "another battler could Snatch it")
140:         break
141:       end
142:     end
143:     next score
144:   }

147: #===============================================================================
148: # Pick a good move for the Choice items.
149: #===============================================================================
150: Battle::AI::Handlers::GeneralMoveScore.add(:good_move_for_choice_item,
151:   proc { |score, move, user, ai, battle|
152:     next score if !ai.trainer.medium_skill?
153:     next score if !user.has_active_item?([:CHOICEBAND, :CHOICESPECS, :CHOICESCARF]) &&
154:                   !user.has_active_ability?(:GORILLATACTICS)
155:     old_score = score
156:     # Really don't prefer status moves (except Trick)
157:     if move.statusMove? && move.function_code != "UserTargetSwapItems"
158:       score -= 25
159:       PBDebug.log_score_change(score - old_score, "don't want to be Choiced into a status move")
160:       next score
161:     end
162:     # Don't prefer moves which are 0x against at least one type
163:     move_type = move.rough_type
164:     GameData::Type.each do |type_data|
165:       score -= 8 if type_data.immunities.include?(move_type)
166:     end
167:     # Don't prefer moves with lower accuracy
168:     if move.accuracy > 0
169:       score -= (0.4 * (100 - move.accuracy)).to_i   # -0 (100%) to -39 (1%)
170:     end
171:     # Don't prefer moves with low PP
172:     score -= 10 if move.move.pp <= 5
173:     PBDebug.log_score_change(score - old_score, "move is less suitable to be Choiced into")
174:     next score
175:   }
176: )

178: #===============================================================================
179: # Prefer damaging moves if the foe is down to their last Pokémon (opportunistic).
180: # Prefer damaging moves if the AI is down to its last Pokémon but the foe has
181: # more (desperate).
182: #===============================================================================
183: Battle::AI::Handlers::GeneralMoveScore.add(:damaging_move_and_either_side_no_reserves,
184:   proc { |score, move, user, ai, battle|
185:     if ai.trainer.medium_skill? && move.damagingMove?
186:       reserves = battle.pbAbleNonActiveCount(user.idxOwnSide)
187:       foes     = battle.pbAbleNonActiveCount(user.idxOpposingSide)
188:       # Don't mess with scores just because a move is damaging; need to play well
189:       next score if ai.trainer.high_skill? && foes > reserves   # AI is outnumbered
190:       # Prefer damaging moves depending on remaining Pokémon
191:       old_score = score
192:       if foes == 0          # Foe is down to their last Pokémon
193:         score += 10         # => Go for the kill
194:         PBDebug.log_score_change(score - old_score, "prefer damaging moves (no foe party Pokémon left)")
195:       elsif reserves == 0   # AI is down to its last Pokémon, foe has reserves
196:         score += 5          # => Go out with a bang
197:         PBDebug.log_score_change(score - old_score, "prefer damaging moves (no ally party Pokémon left)")
198:       end
199:     end
200:     next score
