25: #===============================================================================
26: # Hits in 2 volleys. The second volley targets the original target's ally if it
27: # has one (that can be targeted), or the original target if not. A battler
28: # cannot be targeted if it is is immune to or protected from this move somehow,
29: # or if this move will miss it. (Dragon Darts)
30: # NOTE: This move sometimes shows a different failure message compared to the
31: #       official games. This is because of the order in which failure checks are
32: #       done (all checks for each target in turn, versus all targets for each
33: #       check in turn). This is considered unimportant, and since correcting it
34: #       would involve extensive code rewrites, it is being ignored.
35: #===============================================================================
36: class Battle::Move::HitTwoTimesTargetThenTargetAlly < Battle::Move
37:   def pbNumHits(user, targets); return 1;    end
38:   def pbRepeatHit?;             return true; end
39: 
40:   def pbModifyTargets(targets, user)
41:     return if targets.length != 1
42:     choices = []
43:     targets[0].allAllies.each { |b| user.pbAddTarget(choices, user, b, self) }
44:     return if choices.length == 0
45:     idxChoice = (choices.length > 1) ? @battle.pbRandom(choices.length) : 0
46:     user.pbAddTarget(targets, user, choices[idxChoice], self, !pbTarget(user).can_choose_distant_target?)
47:   end
48: 
49:   def pbShowFailMessages?(targets)
50:     if targets.length > 1
51:       valid_targets = targets.select { |b| !b.fainted? && !b.damageState.unaffected }
52:       return valid_targets.length <= 1
53:     end
54:     return super
55:   end
56: 
57:   def pbDesignateTargetsForHit(targets, hitNum)
58:     valid_targets = []
59:     targets.each { |b| valid_targets.push(b) if !b.damageState.unaffected }
60:     return [valid_targets[1]] if valid_targets[1] && hitNum == 1
61:     return [valid_targets[0]]
62:   end
63: end

149: #===============================================================================
150: # Hits X times, where X is the number of non-user unfainted status-free Pokémon
151: # in the user's party (not including partner trainers). Fails if X is 0.
152: # Base power of each hit depends on the base Attack stat for the species of that
153: # hit's participant. (Beat Up)
154: #===============================================================================
155: class Battle::Move::HitOncePerUserTeamMember < Battle::Move
156:   def multiHitMove?; return true; end
157: 
158:   def pbMoveFailed?(user, targets)
159:     @beatUpList = []
160:     @battle.eachInTeamFromBattlerIndex(user.index) do |pkmn, i|
161:       next if !pkmn.able? || pkmn.status != :NONE
162:       @beatUpList.push(i)
163:     end
164:     if @beatUpList.length == 0
165:       @battle.pbDisplay(_INTL("But it failed!"))
166:       return true
167:     end
168:     return false
169:   end
170: 
171:   def pbNumHits(user, targets)
172:     return @beatUpList.length
173:   end
174: 
175:   def pbBaseDamage(baseDmg, user, target)
176:     i = @beatUpList.shift   # First element in array, and removes it from array
177:     atk = @battle.pbParty(user.index)[i].baseStats[:ATTACK]
178:     return 5 + (atk / 10)
179:   end
180: end
