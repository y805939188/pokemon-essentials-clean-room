214:   # (called by thing.method) that analyse the data of a particular thing which
215:   # the instance represents.

217:   module InstanceMethods
218:     # @param other [Symbol, self.class, String, Integer]
219:     # @return [Boolean] whether other represents the same thing as this thing
220:     def ==(other)
221:       return false if other.nil?
222:       case other
223:       when Symbol
224:         return @id == other
225:       when self.class
226:         return @id == other.id
227:       when String
228:         return @id == other.to_sym
229:       when Integer
230:         return @id_number == other
231:       end
232:       return false
233:     end
