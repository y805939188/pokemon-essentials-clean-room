543:   #-----------------------------------------------------------------------------

545:   # Return values:
546:   #   0: Regular additional effect chance or isn't an additional effect
547:   #   -999: Additional effect will be negated
548:   #   Other: Amount to add to a move's score
549:   def get_score_change_for_additional_effect(user, target = nil)
550:     # Doesn't have an additional effect
551:     return 0 if @move.addlEffect == 0
552:     # Additional effect will be negated
553:     return -999 if user.has_active_ability?(:SHEERFORCE)
554:     return -999 if target && user.index != target.index &&
555:                    target.has_active_ability?(:SHIELDDUST) && !@ai.battle.moldBreaker
556:     # Prefer if the additional effect will have an increased chance of working
557:     return 5 if @move.addlEffect < 100 &&
558:                 (Settings::MECHANICS_GENERATION >= 6 || function_code != "EffectDependsOnEnvironment") &&
559:                 (user.has_active_ability?(:SERENEGRACE) || user.pbOwnSide.effects[PBEffects::Rainbow] > 0)
560:     # No change to score
561:     return 0
562:   end
