102: Battle::AI::Handlers::MoveEffectAgainstTargetScore.add("FailsIfUserDamagedThisTurn",
103:   proc { |score, move, user, target, ai, battle|
104:     # Check whether user is faster than its foe(s) and could use this move
105:     user_faster_count = 0
106:     foe_faster_count = 0
107:     ai.each_foe_battler(user.side) do |b, i|
108:       if user.faster_than?(b)
109:         user_faster_count += 1
110:       else
111:         foe_faster_count += 1
112:       end
113:     end
114:     next Battle::AI::MOVE_USELESS_SCORE if user_faster_count == 0
115:     score += 15 if foe_faster_count == 0
116:     # Effects that make the target unlikely to act before the user
117:     if ai.trainer.high_skill?
118:       if !target.can_attack?
119:         score += 15
120:       elsif target.effects[PBEffects::Confusion] > 1 ||
121:             target.effects[PBEffects::Attract] == user.index
122:         score += 10
123:       elsif target.battler.paralyzed?
124:         score += 5
125:       end
126:     end
127:     # Don't risk using this move if target is weak
128:     if ai.trainer.has_skill_flag?("HPAware")
129:       score -= 10 if target.hp <= target.totalhp / 2
130:       score -= 10 if target.hp <= target.totalhp / 4
131:     end
132:     next score
133:   }
134: )
