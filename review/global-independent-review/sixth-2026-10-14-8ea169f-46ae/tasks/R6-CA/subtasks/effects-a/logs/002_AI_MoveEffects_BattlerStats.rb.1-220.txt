195:   proc { |score, move, user, ai, battle|
196:     score = ai.get_score_for_target_stat_raise(score, user, move.move.statUp)
197:     if ai.trainer.medium_skill?
198:       current_weight = user.battler.pbWeight
199:       if current_weight > 1
200:         score -= 5 if user.has_move_with_function?("PowerHigherWithUserHeavierThanTarget")
201:         ai.each_foe_battler(user.side) do |b, i|
202:           score -= 5 if b.has_move_with_function?("PowerHigherWithUserHeavierThanTarget")
203:           score += 5 if b.has_move_with_function?("PowerHigherWithTargetWeight")
204:           # User will become susceptible to Sky Drop
205:           if b.has_move_with_function?("TwoTurnAttackInvulnerableInSkyTargetCannotAct") &&
206:              Settings::MECHANICS_GENERATION >= 6
207:             score -= 10 if current_weight >= 2000 && current_weight < 3000
208:           end
209:         end
210:       end
211:     end
212:     next score
213:   }
214: )
