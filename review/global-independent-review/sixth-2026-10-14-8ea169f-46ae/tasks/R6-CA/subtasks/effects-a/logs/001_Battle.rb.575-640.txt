582:   def pbRemoveFromParty(idxBattler, idxParty)
583:     party = pbParty(idxBattler)
584:     # Erase the Pokémon from the party
585:     party[idxParty] = nil
586:     # Rearrange the display order of the team to place the erased Pokémon last
587:     # in it (to avoid gaps)
588:     partyOrders = pbPartyOrder(idxBattler)
589:     partyStarts = pbPartyStarts(idxBattler)
590:     idxTrainer = pbGetOwnerIndexFromPartyIndex(idxBattler, idxParty)
591:     idxPartyStart = partyStarts[idxTrainer]
592:     idxPartyEnd   = (idxTrainer < partyStarts.length - 1) ? partyStarts[idxTrainer + 1] : party.length
593:     origPartyPos = partyOrders[idxParty]   # Position of erased Pokémon initially
594:     partyOrders[idxParty] = idxPartyEnd   # Put erased Pokémon last in the team
595:     party.each_with_index do |_pkmn, i|
596:       next if i < idxPartyStart || i >= idxPartyEnd   # Only check the team
597:       next if partyOrders[i] < origPartyPos   # Appeared before erased Pokémon
598:       partyOrders[i] -= 1   # Appeared after erased Pokémon; bump it up by 1
599:     end
600:   end

602:   def pbSwapBattlers(idxA, idxB)
603:     return false if !@battlers[idxA] || !@battlers[idxB]
604:     # Can't swap if battlers aren't owned by the same trainer
605:     return false if opposes?(idxA, idxB)
606:     return false if pbGetOwnerIndexFromBattlerIndex(idxA) != pbGetOwnerIndexFromBattlerIndex(idxB)
607:     @battlers[idxA],       @battlers[idxB]       = @battlers[idxB],       @battlers[idxA]
608:     @battlers[idxA].index, @battlers[idxB].index = @battlers[idxB].index, @battlers[idxA].index
609:     @choices[idxA],        @choices[idxB]        = @choices[idxB],        @choices[idxA]
610:     @scene.pbSwapBattlerSprites(idxA, idxB)
611:     # Swap the target of any battlers' effects that point at either of the
612:     # swapped battlers, to ensure they still point at the correct target
613:     # NOTE: LeechSeed is not swapped, because drained HP goes to whichever
614:     #       Pokémon is in the position that Leech Seed was used from.
615:     # NOTE: PerishSongUser doesn't need to change, as it's only used to
616:     #       determine which side the Perish Song user was on, and a battler
617:     #       can't change sides.
618:     effectsToSwap = [PBEffects::Attract,
619:                      PBEffects::BideTarget,
620:                      PBEffects::CounterTarget,
621:                      PBEffects::JawLock,
622:                      PBEffects::LockOnPos,
623:                      PBEffects::MeanLook,
624:                      PBEffects::MirrorCoatTarget,
625:                      PBEffects::Octolock,
626:                      PBEffects::SkyDrop,
627:                      PBEffects::TrappingUser]
628:     allBattlers.each do |b|
629:       effectsToSwap.each do |i|
630:         next if b.effects[i] != idxA && b.effects[i] != idxB
631:         b.effects[i] = (b.effects[i] == idxA) ? idxB : idxA
632:       end
633:     end
634:     return true
635:   end
