4: class Battle::AI::AIBattler
5:   attr_reader :index, :side, :party_index
6:   attr_reader :battler
7: 
8:   def initialize(ai, index)
9:     @ai = ai
10:     @index = index
11:     @side = (@ai.battle.opposes?(@index)) ? 1 : 0
12:     refresh_battler
13:   end
14: 
15:   def refresh_battler
16:     old_party_index = @party_index
17:     @battler = @ai.battle.battlers[@index]
18:     @party_index = battler.pokemonIndex
19:   end
20: 
21:   def pokemon;     return battler.pokemon;     end
22:   def level;       return battler.level;       end
23:   def hp;          return battler.hp;          end
24:   def totalhp;     return battler.totalhp;     end
25:   def fainted?;    return battler.fainted?;    end
26:   def status;      return battler.status;      end
27:   def statusCount; return battler.statusCount; end
28:   def gender;      return battler.gender;      end
29:   def turnCount;   return battler.turnCount;   end
30:   def effects;     return battler.effects;     end
31:   def stages;      return battler.stages;      end
32:   def statStageAtMax?(stat); return battler.statStageAtMax?(stat); end
33:   def statStageAtMin?(stat); return battler.statStageAtMin?(stat); end
34:   def moves;       return battler.moves;       end
35: 
36:   def wild?
37:     return @ai.battle.wildBattle? && opposes?
38:   end
39: 
40:   def name
41:     return sprintf("%s (%d)", battler.name, @index)
42:   end
43: 
44:   def opposes?(other = nil)
45:     return @side == 1 if other.nil?
46:     return other.side != @side
47:   end
48: 
49:   def idxOwnSide;      return battler.idxOwnSide;      end
50:   def pbOwnSide;       return battler.pbOwnSide;       end
