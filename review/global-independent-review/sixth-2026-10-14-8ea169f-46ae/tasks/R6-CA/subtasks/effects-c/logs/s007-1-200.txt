5:   proc { |score, move, user, target, ai, battle|
6:     next score if user.wild? || user.item
7:     next score if !target.item || target.battler.unlosableItem?(target.item)
8:     next score if user.battler.unlosableItem?(target.item)
9:     next score if target.effects[PBEffects::Substitute] > 0
10:     next score if target.has_active_ability?(:STICKYHOLD) && !battle.moldBreaker
11:     # User can steal the target's item; score it
12:     user_item_preference = user.wants_item?(target.item_id)
13:     user_no_item_preference = user.wants_item?(:NONE)
14:     user_diff = user_item_preference - user_no_item_preference
15:     user_diff = 0 if !user.item_active?
16:     target_item_preference = target.wants_item?(target.item_id)
17:     target_no_item_preference = target.wants_item?(:NONE)
18:     target_diff = target_no_item_preference - target_item_preference
19:     target_diff = 0 if !target.item_active?
20:     score += user_diff * 4
21:     score -= target_diff * 4
22:     next score
23:   }

69:     user_diff = user_new_item_preference - user_old_item_preference
70:     user_diff = 0 if !user.item_active?
71:     target_new_item_preference = target.wants_item?(user.item_id)

76:     score -= target_diff * 4
