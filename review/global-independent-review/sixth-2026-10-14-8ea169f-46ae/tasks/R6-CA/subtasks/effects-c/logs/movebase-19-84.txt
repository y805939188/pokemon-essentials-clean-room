54:   # This is the code actually used to generate a Battle::Move object. The
55:   # object generated is a subclass of this one which depends on the move's
56:   # function code.
57:   def self.from_pokemon_move(battle, move)
58:     validate move => Pokemon::Move
59:     code = move.function_code || "None"
60:     if code[/^\d/]   # Begins with a digit
61:       class_name = sprintf("Battle::Move::Effect%s", code)
62:     else
63:       class_name = sprintf("Battle::Move::%s", code)
64:     end
65:     if Object.const_defined?(class_name)
66:       return Object.const_get(class_name).new(battle, move)
67:     end
68:     return Battle::Move::Unimplemented.new(battle, move)
69:   end
