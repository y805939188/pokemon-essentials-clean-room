4: Battle::AI::Handlers::MoveEffectScore.add("RedirectAllMovesToUser",
5:   proc { |score, move, user, ai, battle|
6:     # Useless if there is no ally to redirect attacks from
7:     next Battle::AI::MOVE_USELESS_SCORE if user.battler.allAllies.length == 0
8:     # Prefer if ally is at low HP and user is at high HP
9:     if ai.trainer.has_skill_flag?("HPAware") && user.hp > user.totalhp * 2 / 3
10:       ai.each_ally(user.index) do |b, i|
11:         score += 10 if b.hp <= b.totalhp / 3
12:       end
13:     end
14:     next score
15:   }
16: )

118:                   !(move.rough_type == :GHOST && user.has_active_ability?([:LIBERO, :PROTEAN]))
119:     # Don't prefer if user will faint because of using this move

128:       # Prefer if the target can't switch out to remove its curse
129:       score += 10 if !battle.pbCanChooseNonActive?(target.index)
130:     end
131:     if ai.trainer.high_skill?

133:       if user.check_for_move { |m| m.is_a?(Battle::Move::ProtectMove) }
134:         score += 5
135:       end
136:     end

144: Battle::AI::Handlers::MoveEffectAgainstTargetScore.add("EffectDependsOnEnvironment",
145:   proc { |score, move, user, target, ai, battle|
146:     # Determine this move's effect
147:     move.move.pbOnStartUse(user.battler, [target.battler])
148:     function_code = nil
149:     case move.move.secretPower
150:     when 2
151:       function_code = "SleepTarget"
152:     when 10
153:       function_code = "BurnTarget"
154:     when 0, 1
155:       function_code = "ParalyzeTarget"
156:     when 9
157:       function_code = "FreezeTarget"
158:     when 7, 11, 13
159:       function_code = "FlinchTarget"
160:     else
161:       stat_lowered = nil
162:       case move.move.secretPower
163:       when 5
164:         function_code = :ATTACK
165:       when 14
166:         function_code = :DEFENSE
167:       when 3
168:         function_code = :SPECIAL_ATTACK
169:       when 4, 6, 12
170:         function_code = :SPEED
171:       when 8
172:         function_code = :ACCURACY
173:       end
174:       next ai.get_score_for_target_stat_drop(score, target, [stat_lowered, 1]) if stat_lowered
175:     end
176:     if function_code
177:       next Battle::AI::Handlers.apply_move_effect_against_target_score(function_code,
178:          score, move, user, target, ai, battle)
179:     end
180:     next score
181:   }
182: )
