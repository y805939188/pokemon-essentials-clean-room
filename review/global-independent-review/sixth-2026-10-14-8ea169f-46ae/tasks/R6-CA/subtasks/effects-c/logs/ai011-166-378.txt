166:           multipliers[:final_damage_multiplier] *= 2
167:         end
168:       else
169:         Battle::AbilityEffects.triggerDamageCalcFromUser(
170:           user.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
171:         )
172:       end
173:     end
174:     if !@ai.battle.moldBreaker
175:       user_battler.allAllies.each do |b|
176:         next if !b.abilityActive?
177:         Battle::AbilityEffects.triggerDamageCalcFromAlly(
178:           b.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
179:         )
180:       end
181:       if target.ability_active?
182:         case target.ability_id
183:         when :FILTER, :SOLIDROCK
184:           if Effectiveness.super_effective_type?(calc_type, *target.pbTypes(true))
185:             multipliers[:final_damage_multiplier] *= 0.75
186:           end
187:         else
188:           Battle::AbilityEffects.triggerDamageCalcFromTarget(
189:             target.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
190:           )
191:         end
192:       end
193:     end
194:     if target.ability_active?
195:       Battle::AbilityEffects.triggerDamageCalcFromTargetNonIgnorable(
196:         target.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
197:       )
198:     end
199:     if !@ai.battle.moldBreaker
200:       target_battler.allAllies.each do |b|
201:         next if !b.abilityActive?
202:         Battle::AbilityEffects.triggerDamageCalcFromTargetAlly(
203:           b.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
204:         )
205:       end
206:     end
207:     # Item effects that alter damage
208:     if user.item_active?
209:       case user.item_id
210:       when :EXPERTBELT
211:         if Effectiveness.super_effective_type?(calc_type, *target.pbTypes(true))
212:           multipliers[:final_damage_multiplier] *= 1.2
213:         end
214:       when :LIFEORB
215:         multipliers[:final_damage_multiplier] *= 1.3
216:       else
217:         Battle::ItemEffects.triggerDamageCalcFromUser(
218:           user.item, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
219:         )
220:         user.effects[PBEffects::GemConsumed] = nil   # Untrigger consuming of Gems
221:       end
222:     end
223:     if target.item_active? && target.item && !target.item.is_berry?
224:       Battle::ItemEffects.triggerDamageCalcFromTarget(
225:         target.item, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
226:       )
227:     end
228:     # Parental Bond
229:     if user.has_active_ability?(:PARENTALBOND)
230:       multipliers[:power_multiplier] *= (Settings::MECHANICS_GENERATION >= 7) ? 1.25 : 1.5
231:     end
232:     # Me First - n/a because can't predict the move Me First will use
233:     # Helping Hand - n/a
234:     # Charge
235:     if @ai.trainer.medium_skill? &&
236:        user.effects[PBEffects::Charge] > 0 && calc_type == :ELECTRIC
237:       multipliers[:power_multiplier] *= 2
238:     end
239:     # Mud Sport and Water Sport
240:     if @ai.trainer.medium_skill?
241:       case calc_type
242:       when :ELECTRIC
243:         if @ai.battle.allBattlers.any? { |b| b.effects[PBEffects::MudSport] }
244:           multipliers[:power_multiplier] /= 3
245:         end
246:         if @ai.battle.field.effects[PBEffects::MudSportField] > 0
247:           multipliers[:power_multiplier] /= 3
248:         end
249:       when :FIRE
250:         if @ai.battle.allBattlers.any? { |b| b.effects[PBEffects::WaterSport] }
251:           multipliers[:power_multiplier] /= 3
252:         end
253:         if @ai.battle.field.effects[PBEffects::WaterSportField] > 0
254:           multipliers[:power_multiplier] /= 3
255:         end
256:       end
257:     end
258:     # Terrain moves
259:     if @ai.trainer.medium_skill?
260:       terrain_multiplier = (Settings::MECHANICS_GENERATION >= 8) ? 1.3 : 1.5
261:       case @ai.battle.field.terrain
262:       when :Electric
263:         multipliers[:power_multiplier] *= terrain_multiplier if calc_type == :ELECTRIC && user_battler.affectedByTerrain?
264:       when :Grassy
265:         multipliers[:power_multiplier] *= terrain_multiplier if calc_type == :GRASS && user_battler.affectedByTerrain?
266:       when :Psychic
267:         multipliers[:power_multiplier] *= terrain_multiplier if calc_type == :PSYCHIC && user_battler.affectedByTerrain?
268:       when :Misty
269:         multipliers[:power_multiplier] /= 2 if calc_type == :DRAGON && target_battler.affectedByTerrain?
270:       end
271:     end
272:     # Badge multipliers
273:     if @ai.trainer.high_skill? && @ai.battle.internalBattle && target_battler.pbOwnedByPlayer?
274:       # Don't need to check the Atk/Sp Atk-boosting badges because the AI
275:       # won't control the player's Pokémon.
276:       if physicalMove?(calc_type) && @ai.battle.pbPlayer.badge_count >= Settings::NUM_BADGES_BOOST_DEFENSE
277:         multipliers[:defense_multiplier] *= 1.1
278:       elsif specialMove?(calc_type) && @ai.battle.pbPlayer.badge_count >= Settings::NUM_BADGES_BOOST_SPDEF
279:         multipliers[:defense_multiplier] *= 1.1
280:       end
281:     end
282:     # Multi-targeting attacks
283:     if @ai.trainer.high_skill? && targets_multiple_battlers?
284:       multipliers[:final_damage_multiplier] *= 0.75
285:     end
286:     # Weather
287:     if @ai.trainer.medium_skill?
288:       case user_battler.effectiveWeather
289:       when :Sun, :HarshSun
290:         case calc_type
291:         when :FIRE
292:           multipliers[:final_damage_multiplier] *= 1.5
293:         when :WATER
294:           multipliers[:final_damage_multiplier] /= 2
295:         end
296:       when :Rain, :HeavyRain
297:         case calc_type
298:         when :FIRE
299:           multipliers[:final_damage_multiplier] /= 2
300:         when :WATER
301:           multipliers[:final_damage_multiplier] *= 1.5
302:         end
303:       when :Sandstorm
304:         if target.has_type?(:ROCK) && specialMove?(calc_type) &&
305:            function_code != "UseTargetDefenseInsteadOfTargetSpDef"   # Psyshock
306:           multipliers[:defense_multiplier] *= 1.5
307:         end
308:       end
309:     end
310:     # Critical hits
311:     if is_critical
312:       if Settings::NEW_CRITICAL_HIT_RATE_MECHANICS
313:         multipliers[:final_damage_multiplier] *= 1.5
314:       else
315:         multipliers[:final_damage_multiplier] *= 2
316:       end
317:     end
318:     # Random variance - n/a
319:     # STAB
320:     if calc_type && user.has_type?(calc_type)
321:       if user.has_active_ability?(:ADAPTABILITY)
322:         multipliers[:final_damage_multiplier] *= 2
323:       else
324:         multipliers[:final_damage_multiplier] *= 1.5
325:       end
326:     end
327:     # Type effectiveness
328:     typemod = target.effectiveness_of_type_against_battler(calc_type, user, @move)
329:     multipliers[:final_damage_multiplier] *= typemod
330:     # Burn
331:     if @ai.trainer.high_skill? && user.status == :BURN && physicalMove?(calc_type) &&
332:        @move.damageReducedByBurn? && !user.has_active_ability?(:GUTS)
333:       multipliers[:final_damage_multiplier] /= 2
334:     end
335:     # Aurora Veil, Reflect, Light Screen
336:     if @ai.trainer.medium_skill? && !@move.ignoresReflect? && !is_critical &&
337:        !user.has_active_ability?(:INFILTRATOR)
338:       if target.pbOwnSide.effects[PBEffects::AuroraVeil] > 0
339:         if @ai.battle.pbSideBattlerCount(target_battler) > 1
340:           multipliers[:final_damage_multiplier] *= 2 / 3.0
341:         else
342:           multipliers[:final_damage_multiplier] /= 2
343:         end
344:       elsif target.pbOwnSide.effects[PBEffects::Reflect] > 0 && physicalMove?(calc_type)
345:         if @ai.battle.pbSideBattlerCount(target_battler) > 1
346:           multipliers[:final_damage_multiplier] *= 2 / 3.0
347:         else
348:           multipliers[:final_damage_multiplier] /= 2
349:         end
350:       elsif target.pbOwnSide.effects[PBEffects::LightScreen] > 0 && specialMove?(calc_type)
351:         if @ai.battle.pbSideBattlerCount(target_battler) > 1
352:           multipliers[:final_damage_multiplier] *= 2 / 3.0
353:         else
354:           multipliers[:final_damage_multiplier] /= 2
355:         end
356:       end
357:     end
358:     # Minimize
359:     if @ai.trainer.medium_skill? && target.effects[PBEffects::Minimize] && @move.tramplesMinimize?
360:       multipliers[:final_damage_multiplier] *= 2
361:     end
362:     # NOTE: No need to check pbBaseDamageMultiplier, as it's already accounted
363:     #       for in an AI's MoveBasePower handler or can't be checked now anyway.
364:     # NOTE: No need to check pbModifyDamage, as it's already accounted for in an
365:     #       AI's MoveBasePower handler.
366:     ##### Main damage calculation #####
367:     base_dmg = [(base_dmg * multipliers[:power_multiplier]).round, 1].max
368:     atk      = [(atk      * multipliers[:attack_multiplier]).round, 1].max
369:     defense  = [(defense  * multipliers[:defense_multiplier]).round, 1].max
370:     damage   = ((((2.0 * user.level / 5) + 2).floor * base_dmg * atk / defense).floor / 50).floor + 2
371:     damage   = [(damage * multipliers[:final_damage_multiplier]).round, 1].max
372:     ret = damage.floor
373:     ret = target.hp - 1 if @move.nonLethal?(user_battler, target_battler) && ret >= target.hp
374:     return ret
375:   end
