2: # This module stores events that can happen during the game. A procedure can
3: # subscribe to an event by adding itself to the event. It will then be called
4: # whenever the event occurs. Existing events are:
5: #-------------------------------------------------------------------------------
6: #   :on_game_map_setup - When a Game_Map is set up. Typically changes map data.
7: #   :on_new_spriteset_map - When a Spriteset_Map is created. Adds more things to
8: #     show in the overworld.
9: #   :on_frame_update - Once per frame. Various frame/time counters.
10: #   :on_leave_map - When leaving a map. End weather/expired effects.
11: #   :on_enter_map - Upon entering a new map. Set up new effects, end expired
12: #     effects.
13: #   :on_map_or_spriteset_change - Upon entering a new map or when spriteset was
14: #     made. Show things on-screen.
15: #-------------------------------------------------------------------------------
16: #   :on_player_change_direction - When the player turns in a different direction.
17: #   :on_leave_tile - When any event or the player starts to move from a tile.
18: #   :on_step_taken - When any event or the player finishes a step.
19: #   :on_player_step_taken - When the player finishes a step/ends surfing, except
20: #     as part of a move route. Step-based counters.
21: #   :on_player_step_taken_can_transfer - When the player finishes a step/ends
22: #     surfing, except as part of a move route. Step-based effects that can
23: #     transfer the player elsewhere.
24: #   :on_player_interact - When the player presses the Use button in the
25: #     overworld.
26: #-------------------------------------------------------------------------------
27: #   :on_trainer_load - When an NPCTrainer is generated (to battle against or as
28: #     a registered partner). Various modifications to that trainer and their
29: #     Pokémon.
30: #   :on_wild_species_chosen - When a species/level have been chosen for a wild
31: #     encounter. Changes the species/level (e.g. roamer, Poké Radar chain).
32: #   :on_wild_pokemon_created - When a Pokemon object has been created for a wild
33: #     encounter. Various modifications to that Pokémon.
34: #   :on_calling_wild_battle - When a wild battle is called. Prevents that wild
35: #     battle and instead starts a different kind of battle (e.g. Safari Zone).
36: #   :on_start_battle - Just before a battle starts. Memorize/reset information
37: #     about party Pokémon, which is used after battle for evolution checks.
38: #   :on_end_battle - Just after a battle ends. Evolution checks, Pickup/Honey
39: #     Gather, blacking out.
40: #   :on_wild_battle_end - After a wild battle. Updates Poké Radar chain info.
41: #===============================================================================
42: module EventHandlers
43:   @@events = {}
44: 
45:   # Add a named callback for the given event.
46:   def self.add(event, key, proc)
47:     @@events[event] = NamedEvent.new if !@@events.has_key?(event)
48:     @@events[event].add(key, proc)
49:   end
50: 
51:   # Remove a named callback from the given event.
52:   def self.remove(event, key)
53:     @@events[event]&.remove(key)
54:   end
55: 
56:   # Clear all callbacks for the given event.
57:   def self.clear(key)
58:     @@events[key]&.clear
59:   end
60: 
61:   # Trigger all callbacks from an Event if it has been defined.
62:   def self.trigger(event, *args)
63:     return @@events[event]&.trigger(*args)
64:   end
65: end

67: #===============================================================================
68: # This module stores the contents of various menus. Each command in a menu is a
69: # hash of data (containing its name, relative order, code to run when chosen,
70: # etc.).
71: # Menus that use this module are:
72: #-------------------------------------------------------------------------------
73: # Pause menu
74: # Party screen main interact menu
75: # Pokégear main menu
76: # Options screen
77: # PC main menu
78: # Various debug menus (main, Pokémon, battle, battle Pokémon)
79: #===============================================================================
80: module MenuHandlers
81:   @@handlers = {}
82: 
83:   def self.add(menu, option, hash)
84:     @@handlers[menu] = HandlerHash.new if !@@handlers.has_key?(menu)
85:     @@handlers[menu].add(option, hash)
86:   end
87: 
88:   def self.remove(menu, option)
89:     @@handlers[menu]&.remove(option)
90:   end
91: 
92:   def self.clear(menu)
93:     @@handlers[menu]&.clear
94:   end
95: 
96:   def self.each(menu)
97:     return if !@@handlers.has_key?(menu)
98:     @@handlers[menu].each { |option, hash| yield option, hash }
99:   end
100: 
101:   def self.each_available(menu, *args)
102:     return if !@@handlers.has_key?(menu)
103:     options = @@handlers[menu]
104:     keys = options.keys
105:     sorted_keys = keys.sort_by { |option| options[option]["order"] || keys.index(option) }
106:     sorted_keys.each do |option|
107:       hash = options[option]
108:       next if hash["condition"] && !hash["condition"].call(*args)
109:       if hash["name"].is_a?(Proc)
110:         name = hash["name"].call
111:       else
112:         name = _INTL(hash["name"])
113:       end
114:       yield option, hash, name
115:     end
116:   end
117: 
118:   def self.call(menu, option, function, *args)
119:     option_hash = @@handlers[menu][option]
120:     return nil if !option_hash || !option_hash[function]
121:     return option_hash[function].call(*args)
122:   end
123: end
