229:   def pbOnStartUse(user, targets)
230:     # NOTE: This is Gen 7's list plus some of Gen 6 plus a bit of my own.
231:     @secretPower = 0   # Body Slam, paralysis
232:     case @battle.field.terrain
233:     when :Electric
234:       @secretPower = 1   # Thunder Shock, paralysis
235:     when :Grassy
236:       @secretPower = 2   # Vine Whip, sleep
237:     when :Misty
238:       @secretPower = 3   # Fairy Wind, lower Sp. Atk by 1
239:     when :Psychic
240:       @secretPower = 4   # Confusion, lower Speed by 1
241:     else
242:       case @battle.environment
243:       when :Grass, :TallGrass, :Forest, :ForestGrass
244:         @secretPower = 2    # (Same as Grassy Terrain)
245:       when :MovingWater, :StillWater, :Underwater
246:         @secretPower = 5    # Water Pulse, lower Attack by 1
247:       when :Puddle
248:         @secretPower = 6    # Mud Shot, lower Speed by 1
249:       when :Cave
250:         @secretPower = 7    # Rock Throw, flinch
251:       when :Rock, :Sand
252:         @secretPower = 8    # Mud-Slap, lower Acc by 1
253:       when :Snow, :Ice
254:         @secretPower = 9    # Ice Shard, freeze
255:       when :Volcano
256:         @secretPower = 10   # Incinerate, burn
257:       when :Graveyard
258:         @secretPower = 11   # Shadow Sneak, flinch
259:       when :Sky
260:         @secretPower = 12   # Gust, lower Speed by 1
261:       when :Space
262:         @secretPower = 13   # Swift, flinch
263:       when :UltraSpace
264:         @secretPower = 14   # Psywave, lower Defense by 1
265:       end
266:     end
267:   end

269:   # NOTE: This intentionally doesn't use def pbAdditionalEffect, because that
270:   #       method is called per hit and this move's additional effect only occurs
271:   #       once per use, after all the hits have happened (two hits are possible
272:   #       via Parental Bond).
273:   def pbEffectAfterAllHits(user, target)
274:     return if target.fainted?
275:     return if target.damageState.unaffected || target.damageState.substitute
276:     return if user.hasActiveAbility?(:SHEERFORCE)
277:     chance = pbAdditionalEffectChance(user, target)
278:     return if @battle.pbRandom(100) >= chance
279:     case @secretPower
280:     when 2
281:       target.pbSleep if target.pbCanSleep?(user, false, self)
282:     when 10
283:       target.pbBurn(user) if target.pbCanBurn?(user, false, self)
284:     when 0, 1
285:       target.pbParalyze(user) if target.pbCanParalyze?(user, false, self)
286:     when 9
287:       target.pbFreeze if target.pbCanFreeze?(user, false, self)
288:     when 5
289:       if target.pbCanLowerStatStage?(:ATTACK, user, self)
290:         target.pbLowerStatStage(:ATTACK, 1, user)
291:       end
292:     when 14
293:       if target.pbCanLowerStatStage?(:DEFENSE, user, self)
294:         target.pbLowerStatStage(:DEFENSE, 1, user)
295:       end
296:     when 3
297:       if target.pbCanLowerStatStage?(:SPECIAL_ATTACK, user, self)
298:         target.pbLowerStatStage(:SPECIAL_ATTACK, 1, user)
299:       end
300:     when 4, 6, 12
301:       if target.pbCanLowerStatStage?(:SPEED, user, self)
302:         target.pbLowerStatStage(:SPEED, 1, user)
303:       end
304:     when 8
305:       if target.pbCanLowerStatStage?(:ACCURACY, user, self)
306:         target.pbLowerStatStage(:ACCURACY, 1, user)
307:       end
308:     when 7, 11, 13
309:       target.pbFlinch(user)
310:     end
311:   end
