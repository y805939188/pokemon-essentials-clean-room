386:          target.battler.form > 0 && !target.effects[PBEffects::Transform]
387:         old_score = score
388:         score += 8
389:         PBDebug.log_score_change(score - old_score, "can trigger the target's ability")
390:       end
391:       if target.item_active?
392:         if Battle::ItemEffects::OnBeingHit[target.item] ||
393:            (Battle::ItemEffects::AfterMoveUseFromTarget[target.item] &&
394:            (!user.has_active_ability?(:SHEERFORCE) || move.move.addlEffect == 0))
395:           old_score = score
396:           score += 8
397:           PBDebug.log_score_change(score - old_score, "can trigger the target's item")
398:         end
399:       end
400:     end
401:     next score
402:   }

474: #===============================================================================
475: # Don't prefer damaging moves if the target is Biding, unless the move will deal
476: # enough damage to KO the target before it retaliates (assuming the move is used
477: # repeatedly until the target retaliates). Doesn't do a score change if the user
478: # will be immune to Bide's damage.
479: #===============================================================================
480: Battle::AI::Handlers::GeneralMoveAgainstTargetScore.add(:damaging_a_biding_target,
481:   proc { |score, move, user, target, ai, battle|
482:     if ai.trainer.medium_skill? && target.effects[PBEffects::Bide] > 0 && move.damagingMove?
483:       eff = user.effectiveness_of_type_against_battler(:NORMAL, target)   # Bide is Normal type
484:       if !Effectiveness.ineffective?(eff)
485:         # Worth damaging the target if it can be knocked out before Bide ends
486:         if ai.trainer.has_skill_flag?("HPAware")
487:           dmg = move.rough_damage
488:           eor_dmg = target.rough_end_of_round_damage
489:           hits_possible = target.effects[PBEffects::Bide] - 1
490:           eor_dmg *= hits_possible
491:           hits_possible += 1 if user.faster_than?(target)
492:           next score if (dmg * hits_possible) + eor_dmg > target.hp * 1.1
493:         end
494:         old_score = score
495:         score -= 20
496:         PBDebug.log_score_change(score - old_score, "don't want to damage the Biding target")
497:       end
498:     end
499:     next score
500:   }
501: )
