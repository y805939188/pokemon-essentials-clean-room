150: # Hits X times, where X is the number of non-user unfainted status-free Pokémon
151: # in the user's party (not including partner trainers). Fails if X is 0.
152: # Base power of each hit depends on the base Attack stat for the species of that
153: # hit's participant. (Beat Up)
154: #===============================================================================
155: class Battle::Move::HitOncePerUserTeamMember < Battle::Move
156:   def multiHitMove?; return true; end
157: 
158:   def pbMoveFailed?(user, targets)
159:     @beatUpList = []
160:     @battle.eachInTeamFromBattlerIndex(user.index) do |pkmn, i|
161:       next if !pkmn.able? || pkmn.status != :NONE
162:       @beatUpList.push(i)
163:     end
164:     if @beatUpList.length == 0
165:       @battle.pbDisplay(_INTL("But it failed!"))
166:       return true
167:     end
168:     return false
169:   end
170: 
171:   def pbNumHits(user, targets)
172:     return @beatUpList.length
173:   end
174: 
175:   def pbBaseDamage(baseDmg, user, target)
176:     i = @beatUpList.shift   # First element in array, and removes it from array
177:     atk = @battle.pbParty(user.index)[i].baseStats[:ATTACK]
178:     return 5 + (atk / 10)
179:   end
180: end
