285:     return false if self.status != :NONE
286:     if affectedByTerrain? && [:Electric, :Misty].include?(@battle.field.terrain)
287:       return false
288:     end
289:     if !hasActiveAbility?(:SOUNDPROOF) && @battle.allBattlers.any? { |b| b.effects[PBEffects::Uproar] > 0 }
290:       return false
291:     end
292:     if Battle::AbilityEffects.triggerStatusImmunityNonIgnorable(self.ability, self, :SLEEP)
293:       return false
294:     end
295:     # NOTE: Bulbapedia claims that Flower Veil shouldn't prevent sleep due to
296:     #       drowsiness, but I disagree because that makes no sense. Also, the
297:     #       comparable Sweet Veil does prevent sleep due to drowsiness.
298:     if abilityActive? && Battle::AbilityEffects.triggerStatusImmunity(self.ability, self, :SLEEP)
299:       return false
300:     end
301:     allAllies.each do |b|
302:       next if !b.abilityActive?
303:       next if !Battle::AbilityEffects.triggerStatusImmunityFromAlly(b.ability, self, :SLEEP)
304:       return false
305:     end
306:     # NOTE: Bulbapedia claims that Safeguard shouldn't prevent sleep due to
307:     #       drowsiness. I disagree with this too. Compare with the other sided
308:     #       effects Misty/Electric Terrain, which do prevent it.
309:     return false if pbOwnSide.effects[PBEffects::Safeguard] > 0
310:     return true
311:   end
