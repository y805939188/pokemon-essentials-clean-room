67:   def pbHasMegaRing?(idxBattler)
68:     if pbOwnedByPlayer?(idxBattler)
69:       @mega_rings.each { |item| return true if $bag.has?(item) }
70:     else
71:       trainer_items = pbGetOwnerItems(idxBattler)
72:       return false if !trainer_items
73:       @mega_rings.each { |item| return true if trainer_items.include?(item) }
74:     end
75:     return false
76:   end

79:     if !@mega_rings.empty?
80:       if pbOwnedByPlayer?(idxBattler)
81:         @mega_rings.each { |item| return GameData::Item.get(item).name if $bag.has?(item) }
82:       else
83:         trainer_items = pbGetOwnerItems(idxBattler)
84:         if trainer_items
85:           @mega_rings.each { |item| return GameData::Item.get(item).name if trainer_items.include?(item) }
86:         end
87:       end
88:     end
89:     return _INTL("Mega Ring")

92:   def pbCanMegaEvolve?(idxBattler)
93:     return false if $game_switches[Settings::NO_MEGA_EVOLUTION]
94:     return false if !@battlers[idxBattler].hasMega?
95:     return false if @battlers[idxBattler].wild?
96:     return true if $DEBUG && Input.press?(Input::CTRL)
97:     return false if @battlers[idxBattler].effects[PBEffects::SkyDrop] >= 0
98:     return false if !pbHasMegaRing?(idxBattler)
99:     side  = @battlers[idxBattler].idxOwnSide
100:     owner = pbGetOwnerIndexFromBattlerIndex(idxBattler)
101:     return @megaEvolution[side][owner] == -1
102:   end
