64:   def pbFaint(showMessage = true)
65:     if !fainted?
66:       PBDebug.log("!!!***Can't faint with HP greater than 0")
67:       return
68:     end
69:     return if @fainted   # Has already fainted properly
70:     @battle.pbDisplayBrief(_INTL("{1} fainted!", pbThis)) if showMessage
71:     PBDebug.log("[Pokémon fainted] #{pbThis} (#{@index})") if !showMessage
72:     @battle.scene.pbFaintBattler(self)
73:     @battle.pbSetDefeated(self) if opposes?
74:     pbInitEffects(false)
75:     # Reset status
76:     self.status      = :NONE
77:     self.statusCount = 0
78:     # Lose happiness
79:     if @pokemon && @battle.internalBattle
80:       badLoss = @battle.allOtherSideBattlers(@index).any? { |b| b.level >= self.level + 30 }
81:       @pokemon.changeHappiness((badLoss) ? "faintbad" : "faint")
82:     end
83:     # Reset form
84:     @battle.peer.pbOnLeavingBattle(@battle, @pokemon, @battle.usedInBattle[idxOwnSide][@index / 2])
85:     @pokemon.makeUnmega if mega?
86:     @pokemon.makeUnprimal if primal?
87:     # Do other things
88:     @battle.pbClearChoice(@index)   # Reset choice
89:     pbOwnSide.effects[PBEffects::LastRoundFainted] = @battle.turnCount
90:     if $game_temp.party_direct_damage_taken &&
91:        $game_temp.party_direct_damage_taken[@pokemonIndex] &&
92:        pbOwnedByPlayer?
93:       $game_temp.party_direct_damage_taken[@pokemonIndex] = 0
94:     end
95:     # Check other battlers' abilities that trigger upon a battler fainting
96:     pbAbilitiesOnFainting
97:     # Check for end of primordial weather
98:     @battle.pbEndPrimordialWeather
99:   end
