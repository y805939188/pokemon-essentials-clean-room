2:   #=============================================================================
3:   # Choosing to use an item
4:   #=============================================================================
5:   def pbCanUseItemOnPokemon?(item, pkmn, battler, scene, showMessages = true)
6:     if !pkmn || pkmn.egg?
7:       scene.pbDisplay(_INTL("It won't have any effect.")) if showMessages
8:       return false
9:     end
10:     # Embargo
11:     if battler && battler.effects[PBEffects::Embargo] > 0
12:       if showMessages
13:         scene.pbDisplay(_INTL("Embargo's effect prevents the item's use on {1}!",
14:                               battler.pbThis(true)))
15:       end
16:       return false
17:     end
18:     # Hyper Mode and non-Scents
19:     if pkmn.hyper_mode && !GameData::Item.get(item)&.is_scent?
20:       scene.pbDisplay(_INTL("It won't have any effect.")) if showMessages
21:       return false
22:     end
23:     return true
24:   end

26:   # NOTE: Using a Poké Ball consumes all your actions for the round. The method
27:   #       below is one half of making this happen; the other half is in the
28:   #       ItemHandlers::CanUseInBattle for Poké Balls.
29:   def pbItemUsesAllActions?(item)
30:     return true if GameData::Item.get(item).is_poke_ball?
31:     return false
32:   end
33: 
34:   def pbRegisterItem(idxBattler, item, idxTarget = nil, idxMove = nil)
35:     # Register for use of item on a Pokémon in the party
36:     @choices[idxBattler][0] = :UseItem
37:     @choices[idxBattler][1] = item        # ID of item to be used
38:     @choices[idxBattler][2] = idxTarget   # Party index of Pokémon to use item on
39:     @choices[idxBattler][3] = idxMove     # Index of move to recharge (Ethers)
40:     # Delete the item from the Bag. If it turns out it will have no effect, it
41:     # will be re-added to the Bag later.
42:     pbConsumeItemInBag(item, idxBattler)
43:     return true
44:   end
45: 
46:   #=============================================================================
47:   # Using an item
48:   #=============================================================================
49:   def pbConsumeItemInBag(item, idxBattler)
50:     return if !item
51:     return if !GameData::Item.get(item).consumed_after_use?
52:     if pbOwnedByPlayer?(idxBattler)
53:       if !$bag.remove(item)
54:         raise _INTL("Tried to consume item that wasn't in the Bag somehow.")
55:       end
56:     else
57:       items = pbGetOwnerItems(idxBattler)
58:       items.delete_at(items.index(item)) if items
59:     end
60:   end
61: 
62:   def pbReturnUnusedItemToBag(item, idxBattler)
63:     return if !item
64:     return if !GameData::Item.get(item).consumed_after_use?
65:     if pbOwnedByPlayer?(idxBattler)
66:       if $bag&.can_add?(item)
67:         $bag.add(item)
68:       else
69:         raise _INTL("Couldn't return unused item to Bag somehow.")
70:       end
71:     else
72:       items = pbGetOwnerItems(idxBattler)
73:       items&.push(item)
74:     end
75:   end
76: 
77:   def pbUseItemMessage(item, trainerName)
78:     itemName = GameData::Item.get(item).portion_name
79:     if itemName.starts_with_vowel?
80:       pbDisplayBrief(_INTL("{1} used an {2}.", trainerName, itemName))
81:     else
82:       pbDisplayBrief(_INTL("{1} used a {2}.", trainerName, itemName))
83:     end
84:   end
