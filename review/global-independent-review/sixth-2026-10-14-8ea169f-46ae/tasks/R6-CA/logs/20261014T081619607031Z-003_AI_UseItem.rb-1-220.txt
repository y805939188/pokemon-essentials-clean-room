83:   # Decide whether the opponent should use an item on the Pokémon.
84:   def pbChooseToUseItem
85:     item = nil
86:     idxTarget = nil   # Party index (battle_use type 1/2/3) or battler index
87:     idxMove = nil
88:     item, idxTarget, idxMove = choose_item_to_use
89:     return false if !item
90:     # Register use of item
91:     @battle.pbRegisterItem(@user.index, item, idxTarget, idxMove)
92:     PBDebug.log_ai("#{@user.name} will use item #{GameData::Item.get(item).name}")
93:     return true
94:   end
95: 
96:   # Return values are:
97:   #   item ID
98:   #   target index (party index for items with a battle use of 1/2/3, battler
99:   #     index otherwise)
100:   #   move index (for items usable on moves only)
101:   def choose_item_to_use
102:     return nil if !@battle.internalBattle
103:     items = @battle.pbGetOwnerItems(@user.index)
104:     return nil if !items || items.length == 0
105:     # Find all items usable on the Pokémon choosing this action
106:     pkmn = @user.battler.pokemon
107:     usable_items = {}
108:     items.each do |item|
109:       usage = get_usability_of_item_on_pkmn(item, @user.party_index, @user.side)
110:       usage.each_pair do |key, vals|
111:         usable_items[key] ||= []
112:         usable_items[key] += vals
113:       end
114:     end
115:     # Prioritise using a HP restoration item
116:     if usable_items[:hp_heal] && (pkmn.hp <= pkmn.totalhp / 4 ||
117:        (pkmn.hp <= pkmn.totalhp / 2 && pbAIRandom(100) < 30))
118:       usable_items[:hp_heal].sort! { |a, b| (a[2] == b[2]) ? a[3] <=> b[3] : a[2] <=> b[2] }
119:       usable_items[:hp_heal].each do |item|
120:         return item[0], item[1] if item[3] >= (pkmn.totalhp - pkmn.hp) * 0.75
121:       end
122:       return usable_items[:hp_heal].last[0], usable_items[:hp_heal].last[1]
123:     end
124:     # Next prioritise using a status-curing item
125:     if usable_items[:status_cure] &&
126:        ([:SLEEP, :FROZEN].include?(pkmn.status) || pbAIRandom(100) < 40)
127:       usable_items[:status_cure].sort! { |a, b| a[2] <=> b[2] }
128:       return usable_items[:status_cure].first[0], usable_items[:status_cure].first[1]
129:     end
130:     # Next try using an item that raises all stats (Max Mushrooms)
131:     if usable_items[:all_stats_raise] && pbAIRandom(100) < 30
132:       return usable_items[:stat_raise].first[0], usable_items[:stat_raise].first[1]
133:     end
134:     # Next try using an X item
135:     if usable_items[:stat_raise] && pbAIRandom(100) < 30
136:       usable_items[:stat_raise].sort! { |a, b| (a[2] == b[2]) ? a[3] <=> b[3] : a[2] <=> b[2] }
137:       return usable_items[:stat_raise].last[0], usable_items[:stat_raise].last[1]
138:     end
139:     # Find items usable on other Pokémon in the user's team
140:     # NOTE: Currently only checks Revives.
141:     usable_items = {}
142:     @battle.eachInTeamFromBattlerIndex(@user.index) do |team_pkmn, i|
143:       next if !team_pkmn.fainted?   # Remove this line to check unfainted Pokémon too
144:       items.each do |item|
145:         usage = get_usability_of_item_on_pkmn(item, i, @user.side)
146:         usage.each_pair do |key, vals|
147:           usable_items[key] ||= []
148:           usable_items[key] += vals
149:         end
150:       end
151:     end
152:     # Try using a Revive (prefer Max Revive-type items over Revive)
153:     if usable_items[:revive] &&
154:        (@battle.pbAbleNonActiveCount(@user.index) == 0 || pbAIRandom(100) < 40)
155:       usable_items[:revive].sort! { |a, b| (a[2] == b[2]) ? a[1] <=> b[1] : a[2] <=> b[2] }
156:       return usable_items[:revive].last[0], usable_items[:revive].last[1]
157:     end
158:     return nil
159:   end

161:   def get_usability_of_item_on_pkmn(item, party_index, side)
162:     pkmn = @battle.pbParty(side)[party_index]
163:     battler = @battle.pbFindBattler(party_index, side)
164:     ret = {}
165:     return ret if !@battle.pbCanUseItemOnPokemon?(item, pkmn, battler, @battle.scene, false)
166:     return ret if !ItemHandlers.triggerCanUseInBattle(item, pkmn, battler, nil,
167:                                                       false, self, @battle.scene, false)
168:     want_to_cure_status = (pkmn.status != :NONE)
169:     if battler
170:       if want_to_cure_status
171:         want_to_cure_status = @battlers[battler.index].wants_status_problem?(pkmn.status)
172:         want_to_cure_status = false if pkmn.status == :SLEEP && pkmn.statusCount <= 2
173:       end
174:       want_to_cure_status ||= (battler.effects[PBEffects::Confusion] > 1)
175:     end
176:     if HP_HEAL_ITEMS.include?(item)
177:       if pkmn.hp < pkmn.totalhp
178:         heal_amount = HP_HEAL_ITEMS[item]
179:         heal_amount = pkmn.totalhp / 4 if item == :SITURUSBERRY
180:         ret[:hp_heal] ||= []
181:         ret[:hp_heal].push([item, party_index, 5, heal_amount])
182:       end
183:     elsif FULL_RESTORE_ITEMS.include?(item)
184:       prefer_full_restore = (pkmn.hp <= pkmn.totalhp * 2 / 3 && want_to_cure_status)
185:       if pkmn.hp < pkmn.totalhp
186:         ret[:hp_heal] ||= []
187:         ret[:hp_heal].push([item, party_index, (prefer_full_restore) ? 3 : 7, 999])
188:       end
189:       if want_to_cure_status
190:         ret[:status_cure] ||= []
191:         ret[:status_cure].push([item, party_index, (prefer_full_restore) ? 3 : 9])
192:       end
193:     elsif ONE_STATUS_CURE_ITEMS.include?(item)
194:       if want_to_cure_status
195:         ret[:status_cure] ||= []
196:         ret[:status_cure].push([item, party_index, 5])
197:       end
198:     elsif ALL_STATUS_CURE_ITEMS.include?(item)
199:       if want_to_cure_status
200:         ret[:status_cure] ||= []
201:         ret[:status_cure].push([item, party_index, 7])
202:       end
203:     elsif ONE_STAT_RAISE_ITEMS.include?(item)
204:       stat_data = ONE_STAT_RAISE_ITEMS[item]
205:       if battler && stat_raise_worthwhile?(@battlers[battler.index], stat_data[0])
206:         ret[:stat_raise] ||= []
207:         ret[:stat_raise].push([item, party_index, battler.stages[stat_data[0]], stat_data[1]])
208:       end
209:     elsif ALL_STATS_RAISE_ITEMS.include?(item)
210:       if battler
211:         ret[:all_stats_raise] ||= []
212:         ret[:all_stats_raise].push([item, party_index])
213:       end
214:     elsif REVIVE_ITEMS.include?(item)
215:       ret[:revive] ||= []
216:       ret[:revive].push([item, party_index, REVIVE_ITEMS[item]])
217:     end
218:     return ret
219:   end
