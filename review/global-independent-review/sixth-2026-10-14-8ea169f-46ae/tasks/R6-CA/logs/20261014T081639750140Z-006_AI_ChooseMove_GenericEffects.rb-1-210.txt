4: class Battle::AI
5:   # Main method for calculating the score for moves that raise a battler's
6:   # stat(s).
7:   # By default, assumes that a stat raise is a good thing. However, this score
8:   # is inverted (by desire_mult) if the target opposes the user. If the move
9:   # could target a foe but is targeting an ally, the score is also inverted, but
10:   # only because it is inverted again in def pbGetMoveScoreAgainstTarget.
11:   def get_score_for_target_stat_raise(score, target, stat_changes, whole_effect = true,
12:                                       fixed_change = false, ignore_contrary = false)
13:     whole_effect = false if @move.damagingMove?
14:     # Decide whether the target raising its stat(s) is a good thing
15:     desire_mult = 1
16:     if target.opposes?(@user) ||
17:        (@move.pbTarget(@user.battler).targets_foe && target.index != @user.index)
18:       desire_mult = -1
19:     end
20:     # If target has Contrary, use different calculations to score the stat change
21:     if !ignore_contrary && !fixed_change && !@battle.moldBreaker && target.has_active_ability?(:CONTRARY)
22:       if desire_mult > 0 && whole_effect
23:         PBDebug.log_score_change(MOVE_USELESS_SCORE - score, "don't prefer raising target's stats (it has Contrary)")
24:         return MOVE_USELESS_SCORE
25:       end
26:       return get_score_for_target_stat_drop(score, target, stat_changes, whole_effect, fixed_change, true)
27:     end
28:     # Don't make score changes if the move is a damaging move and its additional
29:     # effect (the stat raise(s)) will be negated
30:     add_effect = @move.get_score_change_for_additional_effect(@user, target)
31:     return score if add_effect == -999   # Additional effect will be negated
32:     # Don't make score changes if target will faint from EOR damage
33:     if target.rough_end_of_round_damage >= target.hp
34:       ret = (whole_effect) ? MOVE_USELESS_SCORE : score
35:       PBDebug.log("     ignore stat change (target predicted to faint this round)")
36:       return ret
37:     end
38:     # Don't make score changes if foes have Unaware and target can't make use of
39:     # extra stat stages
40:     if !target.has_move_with_function?("PowerHigherWithUserPositiveStatStages")
41:       foe_is_aware = false
42:       each_foe_battler(target.side) do |b, i|
43:         foe_is_aware = true if !b.has_active_ability?(:UNAWARE)
44:       end
45:       if !foe_is_aware
46:         ret = (whole_effect) ? MOVE_USELESS_SCORE : score
47:         PBDebug.log("     ignore stat change (target's foes have Unaware)")
48:         return ret
49:       end
50:     end
51:     # Figure out which stat raises can happen
52:     real_stat_changes = []
53:     stat_changes.each_with_index do |stat, idx|
54:       next if idx.odd?
55:       if !stat_raise_worthwhile?(target, stat, fixed_change)
56:         if target.index == @user.index
57:           PBDebug.log("     raising the user's #{GameData::Stat.get(stat).name} isn't worthwhile")
58:         else
59:           PBDebug.log("     raising the target's #{GameData::Stat.get(stat).name} isn't worthwhile")
60:         end
61:         next
62:       end
63:       # Calculate amount that stat will be raised by
64:       increment = stat_changes[idx + 1]
65:       increment *= 2 if !fixed_change && !@battle.moldBreaker && target.has_active_ability?(:SIMPLE)
66:       increment = [increment, Battle::Battler::STAT_STAGE_MAXIMUM - target.stages[stat]].min   # The actual stages gained
67:       # Count this as a valid stat raise
68:       real_stat_changes.push([stat, increment]) if increment > 0
69:     end
70:     # Discard move if it can't raise any stats
71:     if real_stat_changes.length == 0
72:       return (whole_effect) ? MOVE_USELESS_SCORE : score
73:     end
74:     # Make score change based on the additional effect chance
75:     score += add_effect
76:     # Make score changes based on the general concept of raising stats at all
77:     score = get_target_stat_raise_score_generic(score, target, real_stat_changes, desire_mult)
78:     # Make score changes based on the specific changes to each stat that will be
79:     # raised
80:     real_stat_changes.each do |change|
81:       old_score = score
82:       score = get_target_stat_raise_score_one(score, target, change[0], change[1], desire_mult)
83:       if target.index == @user.index
84:         PBDebug.log_score_change(score - old_score, "raising the user's #{GameData::Stat.get(change[0]).name} by #{change[1]}")
85:       else
86:         PBDebug.log_score_change(score - old_score, "raising the target's #{GameData::Stat.get(change[0]).name} by #{change[1]}")
87:       end
88:     end
89:     return score
90:   end

94:   # Returns whether the target raising the given stat will have any impact.
95:   def stat_raise_worthwhile?(target, stat, fixed_change = false)
96:     if !fixed_change
97:       return false if !target.battler.pbCanRaiseStatStage?(stat, @user.battler, @move.move)
98:     end
99:     # Check if target won't benefit from the stat being raised
100:     return true if target.has_move_with_function?("SwitchOutUserPassOnEffects",
101:                                                   "PowerHigherWithUserPositiveStatStages")
102:     case stat
103:     when :ATTACK
104:       return false if !target.check_for_move { |m| m.physicalMove?(m.type) &&
105:                                                    m.function_code != "UseUserDefenseInsteadOfUserAttack" &&
106:                                                    m.function_code != "UseTargetAttackInsteadOfUserAttack" }
107:     when :DEFENSE
108:       each_foe_battler(target.side) do |b, i|
109:         return true if b.check_for_move { |m| m.physicalMove?(m.type) ||
110:                                               m.function_code == "UseTargetDefenseInsteadOfTargetSpDef" }
111:       end
112:       return false
113:     when :SPECIAL_ATTACK
114:       return false if !target.check_for_move { |m| m.specialMove?(m.type) }
115:     when :SPECIAL_DEFENSE
116:       each_foe_battler(target.side) do |b, i|
117:         return true if b.check_for_move { |m| m.specialMove?(m.type) &&
118:                                               m.function_code != "UseTargetDefenseInsteadOfTargetSpDef" }
119:       end
120:       return false
121:     when :SPEED
122:       moves_that_prefer_high_speed = [
123:         "PowerHigherWithUserFasterThanTarget",
124:         "PowerHigherWithUserPositiveStatStages"
125:       ]
126:       if !target.has_move_with_function?(*moves_that_prefer_high_speed)
127:         meaningful = false
128:         target_speed = target.rough_stat(:SPEED)
129:         each_foe_battler(target.side) do |b, i|
130:           b_speed = b.rough_stat(:SPEED)
131:           meaningful = true if target_speed < b_speed && target_speed * 2.5 > b_speed
132:           break if meaningful
133:         end
134:         return false if !meaningful
135:       end
136:     when :ACCURACY
137:       min_accuracy = 100
138:       target.battler.moves.each do |m|
139:         next if m.accuracy == 0 || m.is_a?(Battle::Move::OHKO)
140:         min_accuracy = m.accuracy if m.accuracy < min_accuracy
141:       end
142:       if min_accuracy >= 90 && target.stages[:ACCURACY] >= 0
143:         meaningful = false
144:         each_foe_battler(target.side) do |b, i|
145:           meaningful = true if b.stages[:EVASION] > 0
146:           break if meaningful
147:         end
148:         return false if !meaningful
149:       end
150:     when :EVASION
151:     end
152:     return true
153:   end

157:   # Make score changes based on the general concept of raising stats at all.
158:   def get_target_stat_raise_score_generic(score, target, stat_changes, desire_mult = 1)
159:     total_increment = stat_changes.sum { |change| change[1] }
160:     # Prefer if move is a status move and it's the user's first/second turn
161:     if @user.turnCount < 2 && @move.statusMove?
162:       score += total_increment * desire_mult * 5
163:     end
164:     if @trainer.has_skill_flag?("HPAware")
165:       # Prefer if user is at high HP, don't prefer if user is at low HP
166:       if target.index != @user.index
167:         score += total_increment * desire_mult * ((100 * @user.hp / @user.totalhp) - 50) / 8   # +6 to -6 per stage
168:       end
169:       # Prefer if target is at high HP, don't prefer if target is at low HP
170:       score += total_increment * desire_mult * ((100 * target.hp / target.totalhp) - 50) / 8   # +6 to -6 per stage
171:     end
172:     # NOTE: There are no abilities that trigger upon stat raise, but this is
173:     #       where they would be accounted for if they existed.
174:     return score
175:   end

177:   # Make score changes based on the raising of a specific stat.
178:   def get_target_stat_raise_score_one(score, target, stat, increment, desire_mult = 1)
179:     # Figure out how much the stat will actually change by
180:     max_stage = Battle::Battler::STAT_STAGE_MAXIMUM
181:     stage_mul = Battle::Battler::STAT_STAGE_MULTIPLIERS
182:     stage_div = Battle::Battler::STAT_STAGE_DIVISORS
183:     if [:ACCURACY, :EVASION].include?(stat)
184:       stage_mul = Battle::Battler::ACC_EVA_STAGE_MULTIPLIERS
185:       stage_div = Battle::Battler::ACC_EVA_STAGE_DIVISORS
186:     end
187:     old_stage = target.stages[stat]
188:     new_stage = old_stage + increment
189:     inc_mult = (stage_mul[new_stage + max_stage].to_f * stage_div[old_stage + max_stage]) / (stage_div[new_stage + max_stage] * stage_mul[old_stage + max_stage])
190:     inc_mult -= 1
191:     inc_mult *= desire_mult
192:     # Stat-based score changes
193:     case stat
194:     when :ATTACK
195:       # Modify score depending on current stat stage
196:       # More strongly prefer if the target has no special moves
197:       if old_stage >= 2 && increment == 1
198:         score -= 10 * ((target.opposes?(@user)) ? 1 : desire_mult)
199:       else
200:         has_special_moves = target.check_for_move { |m| m.specialMove?(m.type) }
201:         inc = (has_special_moves) ? 8 : 12
202:         score += inc * inc_mult
203:       end
204:     when :DEFENSE
205:       # Modify score depending on current stat stage
206:       if old_stage >= 2 && increment == 1
207:         score -= 10 * ((target.opposes?(@user)) ? 1 : desire_mult)
208:       else
209:         score += 10 * inc_mult
210:       end
