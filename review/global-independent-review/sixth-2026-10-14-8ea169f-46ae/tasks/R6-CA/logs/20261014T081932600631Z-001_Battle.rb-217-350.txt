217:   # Given a battler index, returns the index within @player/@opponent of the
218:   # trainer that controls that battler index.
219:   # NOTE: You shouldn't ever have more trainers on a side than there are battler
220:   #       positions on that side. This method doesn't account for if you do.
221:   def pbGetOwnerIndexFromBattlerIndex(idxBattler)
222:     trainer = (opposes?(idxBattler)) ? @opponent : @player
223:     return 0 if !trainer
224:     case trainer.length
225:     when 2
226:       n = pbSideSize(idxBattler % 2)
227:       return [0, 0, 1][idxBattler / 2] if n == 3
228:       return idxBattler / 2   # Same as [0,1][idxBattler/2], i.e. 2 battler slots
229:     when 3
230:       return idxBattler / 2
231:     end
232:     return 0
233:   end
234: 
235:   def pbGetOwnerFromBattlerIndex(idxBattler)
236:     idxTrainer = pbGetOwnerIndexFromBattlerIndex(idxBattler)
237:     trainer = (opposes?(idxBattler)) ? @opponent : @player
238:     return (trainer.nil?) ? nil : trainer[idxTrainer]
239:   end

241:   def pbGetOwnerIndexFromPartyIndex(idxBattler, idxParty)
242:     ret = -1
243:     pbPartyStarts(idxBattler).each_with_index do |start, i|
244:       break if start > idxParty
245:       ret = i
246:     end
247:     return ret
248:   end

250:   # Only used for the purpose of an error message when one trainer tries to
251:   # switch another trainer's Pokémon.
252:   def pbGetOwnerFromPartyIndex(idxBattler, idxParty)
253:     idxTrainer = pbGetOwnerIndexFromPartyIndex(idxBattler, idxParty)
254:     trainer = (opposes?(idxBattler)) ? @opponent : @player
255:     return (trainer.nil?) ? nil : trainer[idxTrainer]

258:   def pbGetOwnerName(idxBattler)
259:     idxTrainer = pbGetOwnerIndexFromBattlerIndex(idxBattler)
260:     return @opponent[idxTrainer].full_name if opposes?(idxBattler)   # Opponent
261:     return @player[idxTrainer].full_name if idxTrainer > 0   # Ally trainer
262:     return @player[idxTrainer].name   # Player
263:   end

265:   def pbGetOwnerItems(idxBattler)
266:     if opposes?(idxBattler)
267:       return [] if !@items
268:       return @items[pbGetOwnerIndexFromBattlerIndex(idxBattler)]
269:     end
270:     return [] if !@ally_items
271:     return @ally_items[pbGetOwnerIndexFromBattlerIndex(idxBattler)]

274:   # Returns whether the battler in position idxBattler is owned by the same
275:   # trainer that owns the Pokémon in party slot idxParty. This assumes that
276:   # both the battler position and the party slot are from the same side.
277:   def pbIsOwner?(idxBattler, idxParty)
278:     idxTrainer1 = pbGetOwnerIndexFromBattlerIndex(idxBattler)
279:     idxTrainer2 = pbGetOwnerIndexFromPartyIndex(idxBattler, idxParty)
280:     return idxTrainer1 == idxTrainer2
281:   end

288:   # Returns the number of Pokémon positions controlled by the given trainerIndex
289:   # on the given side of battle.
290:   def pbNumPositions(side, idxTrainer)
291:     ret = 0
292:     pbSideSize(side).times do |i|
293:       t = pbGetOwnerIndexFromBattlerIndex((i * 2) + side)
294:       next if t != idxTrainer
295:       ret += 1
296:     end
297:     return ret
298:   end

321:   def pbPlayerDisplayParty(idxBattler = 0)

336:   def pbAbleNonActiveCount(idxBattler = 0)
337:     party = pbParty(idxBattler)
338:     inBattleIndices = allSameSideBattlers(idxBattler).map { |b| b.pokemonIndex }
339:     count = 0
340:     party.each_with_index do |pkmn, idxParty|
341:       next if !pkmn || !pkmn.able?
342:       next if inBattleIndices.include?(idxParty)
343:       count += 1
344:     end
345:     return count
