695:   def get_score_for_terrain(terrain, move_user, starting = false)
696:     ret = 0
697:     ret += 4 if starting && terrain != :None && move_user.has_active_item?(:TERRAINEXTENDER)
698:     # Inherent effects of terrain
699:     each_battler do |b, i|
700:       next if !b.battler.affectedByTerrain?
701:       case terrain
702:       when :Electric
703:         # Immunity to sleep
704:         if b.status == :NONE
705:           ret += (b.opposes?(move_user)) ? -8 : 8
706:         end
707:         if b.effects[PBEffects::Yawn] > 0
708:           ret += (b.opposes?(move_user)) ? -10 : 10
709:         end
710:         # Check for Electric moves
711:         if b.has_damaging_move_of_type?(:ELECTRIC)
712:           ret += (b.opposes?(move_user)) ? -10 : 10
713:         end
714:       when :Grassy
715:         # End of round healing
716:         ret += (b.opposes?(move_user)) ? -8 : 8
717:         # Check for Grass moves
718:         if b.has_damaging_move_of_type?(:GRASS)
719:           ret += (b.opposes?(move_user)) ? -10 : 10
720:         end
721:       when :Misty
722:         # Immunity to status problems/confusion
723:         if b.status == :NONE || b.effects[PBEffects::Confusion] == 0
724:           ret += (b.opposes?(move_user)) ? -8 : 8
725:         end
726:         # Check for Dragon moves
727:         if b.has_damaging_move_of_type?(:DRAGON)
728:           ret += (b.opposes?(move_user)) ? 10 : -10
729:         end
730:       when :Psychic
731:         # Check for priority moves
732:         if b.check_for_move { |m| m.priority > 0 && m.pbTarget(b.battler)&.can_target_one_foe? }
733:           ret += (b.opposes?(move_user)) ? 10 : -10
734:         end
735:         # Check for Psychic moves
736:         if b.has_damaging_move_of_type?(:PSYCHIC)
737:           ret += (b.opposes?(move_user)) ? -10 : 10
738:         end
739:       end
740:     end
741:     # Held items relating to terrain
742:     seed = {
743:       :Electric => :ELECTRICSEED,
744:       :Grassy   => :GRASSYSEED,
745:       :Misty    => :MISTYSEED,
746:       :Psychic  => :PSYCHICSEED
747:     }[terrain]
748:     each_battler do |b, i|
749:       if seed && b.has_active_item?(seed)
750:         ret += (b.opposes?(move_user)) ? -8 : 8
751:       end
752:     end
753:     # Check for abilities/moves affected by the terrain
754:     if @trainer.medium_skill?
755:       abils = {
756:         :Electric => :SURGESURFER,
757:         :Grassy   => :GRASSPELT
758:       }[terrain]
759:       good_moves = {
760:         :Electric => ["DoublePowerInElectricTerrain"],
761:         :Grassy   => ["HealTargetDependingOnGrassyTerrain",
762:                       "HigherPriorityInGrassyTerrain"],
763:         :Misty    => ["UserFaintsPowersUpInMistyTerrainExplosive"],
764:         :Psychic  => ["HitsAllFoesAndPowersUpInPsychicTerrain"]
765:       }[terrain]
766:       bad_moves = {
767:         :Grassy => ["DoublePowerIfTargetUnderground",
768:                     "LowerTargetSpeed1WeakerInGrassyTerrain",
769:                     "RandomPowerDoublePowerIfTargetUnderground"]
770:       }[terrain]
771:       each_battler do |b, i|
772:         next if !b.battler.affectedByTerrain?
773:         # Abilities
774:         if b.has_active_ability?(:MIMICRY)
775:           ret += (b.opposes?(move_user)) ? -5 : 5
776:         end
777:         if abils && b.has_active_ability?(abils)
778:           ret += (b.opposes?(move_user)) ? -8 : 8
779:         end
780:         # Moves
781:         if b.has_move_with_function?("EffectDependsOnEnvironment",
782:                                      "SetUserTypesBasedOnEnvironment",
783:                                      "TypeAndPowerDependOnTerrain",
784:                                      "UseMoveDependingOnEnvironment")
785:           ret += (b.opposes?(move_user)) ? -5 : 5
786:         end
787:         if good_moves && b.has_move_with_function?(*good_moves)
788:           ret += (b.opposes?(move_user)) ? -5 : 5
789:         end
790:         if bad_moves && b.has_move_with_function?(*bad_moves)
791:           ret += (b.opposes?(move_user)) ? 5 : -5
792:         end
793:       end
794:     end
795:     return ret
796:   end
