47:   def pbStartBattle
48:     @properties = {}
49:     @properties["internalBattle"]  = @internalBattle
50:     @properties["player"]          = pbGetTrainerInfo(@player)
51:     @properties["opponent"]        = pbGetTrainerInfo(@opponent)
52:     @properties["party1"]          = Marshal.dump(@party1)
53:     @properties["party2"]          = Marshal.dump(@party2)
54:     @properties["party1starts"]    = Marshal.dump(@party1starts)
55:     @properties["party2starts"]    = Marshal.dump(@party2starts)
56:     @properties["weather"]         = @field.weather
57:     @properties["weatherDuration"] = @field.weatherDuration
58:     @properties["canRun"]          = @canRun
59:     @properties["switchStyle"]     = @switchStyle
60:     @properties["showAnims"]       = @showAnims
61:     @properties["items"]           = Marshal.dump(@items)
62:     @properties["backdrop"]        = @backdrop
63:     @properties["backdropBase"]    = @backdropBase
64:     @properties["time"]            = @time
65:     @properties["environment"]     = @environment
66:     @properties["rules"]           = Marshal.dump(@rules)

80:   def pbRegisterMove(idxBattler, idxMove, showMessages = true)

87:

89:     super

94:     ret = super
95:     @rounds[@roundindex][idxBattler] = [Commands::RUN, @decision]

99:   def pbAutoChooseMove(idxBattler, showMessages = true)

103:   end

106:     if super
107:       @rounds[@roundindex][idxBattler] = [Commands::POKEMON, idxParty]
108:       return true
109:     end
110:     return false
111:   end
112:

114:     if super
115:       @rounds[@roundindex][idxBattler] = [Commands::BAG, item, idxTarget, idxMove]
116:       return true
117:     end
118:     return false
119:   end

122:     @roundindex += 1
123:     @rounds[@roundindex] = [[], [], [], []]

125:   end
