16:     old_party_index = @party_index
17:     @battler = @ai.battle.battlers[@index]
18:     @party_index = battler.pokemonIndex
19:   end

56:   # Returns how much damage this battler will take at the end of this round.
57:   def rough_end_of_round_damage
58:     ret = 0
59:     # Weather
60:     weather = battler.effectiveWeather
61:     if @ai.battle.field.weatherDuration == 1
62:       weather = @ai.battle.field.defaultWeather
63:       weather = :None if @ai.battle.allBattlers.any? { |b| b.hasActiveAbility?([:CLOUDNINE, :AIRLOCK]) }
64:       weather = :None if [:Sun, :Rain, :HarshSun, :HeavyRain].include?(weather) && has_active_item?(:UTILITYUMBRELLA)
65:     end
66:     case weather
67:     when :Sandstorm
68:       ret += [self.totalhp / 16, 1].max if battler.takesSandstormDamage?
69:     when :Hail
70:       ret += [self.totalhp / 16, 1].max if battler.takesHailDamage?
71:     when :ShadowSky
72:       ret += [self.totalhp / 16, 1].max if battler.takesShadowSkyDamage?
73:     end
74:     case ability_id
75:     when :DRYSKIN
76:       ret += [self.totalhp / 8, 1].max if [:Sun, :HarshSun].include?(weather) && battler.takesIndirectDamage?
77:       ret -= [self.totalhp / 8, 1].max if [:Rain, :HeavyRain].include?(weather) && battler.canHeal?
78:     when :ICEBODY
79:       ret -= [self.totalhp / 16, 1].max if weather == :Hail && battler.canHeal?
80:     when :RAINDISH
81:       ret -= [self.totalhp / 16, 1].max if [:Rain, :HeavyRain].include?(weather) && battler.canHeal?
82:     when :SOLARPOWER
83:       ret += [self.totalhp / 8, 1].max if [:Sun, :HarshSun].include?(weather) && battler.takesIndirectDamage?
84:     end
85:     # Future Sight/Doom Desire
86:     # NOTE: Not worth estimating the damage from this.
87:     # Wish
88:     if @ai.battle.positions[@index].effects[PBEffects::Wish] == 1 && battler.canHeal?
89:       ret -= @ai.battle.positions[@index].effects[PBEffects::WishAmount]
90:     end
91:     # Sea of Fire
92:     if @ai.battle.sides[@side].effects[PBEffects::SeaOfFire] > 1 &&
93:        battler.takesIndirectDamage? && !has_type?(:FIRE)
94:       ret += [self.totalhp / 8, 1].max
95:     end
96:     # Grassy Terrain (healing)
97:     if @ai.battle.field.terrain == :Grassy && battler.affectedByTerrain? && battler.canHeal?
98:       ret -= [self.totalhp / 16, 1].max
99:     end
100:     # Leftovers/Black Sludge
101:     if has_active_item?(:BLACKSLUDGE)
102:       if has_type?(:POISON)
103:         ret -= [self.totalhp / 16, 1].max if battler.canHeal?
104:       else
105:         ret += [self.totalhp / 8, 1].max if battler.takesIndirectDamage?
106:       end
107:     elsif has_active_item?(:LEFTOVERS)
108:       ret -= [self.totalhp / 16, 1].max if battler.canHeal?
109:     end
110:     # Aqua Ring
111:     if self.effects[PBEffects::AquaRing] && battler.canHeal?
112:       amt = self.totalhp / 16
113:       amt = (amt * 1.3).floor if has_active_item?(:BIGROOT)
114:       ret -= [amt, 1].max
115:     end
116:     # Ingrain
117:     if self.effects[PBEffects::Ingrain] && battler.canHeal?
118:       amt = self.totalhp / 16
119:       amt = (amt * 1.3).floor if has_active_item?(:BIGROOT)
120:       ret -= [amt, 1].max
121:     end
122:     # Leech Seed
123:     if self.effects[PBEffects::LeechSeed] >= 0
124:       if battler.takesIndirectDamage?
125:         ret += [self.totalhp / 8, 1].max if battler.takesIndirectDamage?
126:       end
127:     else
128:       @ai.each_battler do |b, i|
129:         next if i == @index || b.effects[PBEffects::LeechSeed] != @index
130:         amt = [[b.totalhp / 8, b.hp].min, 1].max
131:         amt = (amt * 1.3).floor if has_active_item?(:BIGROOT)
132:         ret -= [amt, 1].max
133:       end
134:     end
135:     # Hyper Mode (Shadow Pokémon)
136:     if battler.inHyperMode?
137:       ret += [self.totalhp / 24, 1].max
138:     end
139:     # Poison/burn/Nightmare
140:     if self.status == :POISON
141:       if has_active_ability?(:POISONHEAL)
142:         ret -= [self.totalhp / 8, 1].max if battler.canHeal?
143:       elsif battler.takesIndirectDamage?
144:         mult = 2
145:         mult = [self.effects[PBEffects::Toxic] + 1, 16].min if self.statusCount > 0   # Toxic
146:         ret += [mult * self.totalhp / 16, 1].max
147:       end
148:     elsif self.status == :BURN
149:       if battler.takesIndirectDamage?
150:         amt = (Settings::MECHANICS_GENERATION >= 7) ? self.totalhp / 16 : self.totalhp / 8
151:         amt = (amt / 2.0).round if has_active_ability?(:HEATPROOF)
152:         ret += [amt, 1].max
153:       end
154:     elsif battler.asleep? && self.statusCount > 1 && self.effects[PBEffects::Nightmare]
155:       ret += [self.totalhp / 4, 1].max if battler.takesIndirectDamage?
156:     end
157:     # Curse
158:     if self.effects[PBEffects::Curse]
159:       ret += [self.totalhp / 4, 1].max if battler.takesIndirectDamage?
160:     end
161:     # Trapping damage
162:     if self.effects[PBEffects::Trapping] > 1 && battler.takesIndirectDamage?
163:       amt = (Settings::MECHANICS_GENERATION >= 6) ? self.totalhp / 8 : self.totalhp / 16
164:       if @ai.battlers[self.effects[PBEffects::TrappingUser]].has_active_item?(:BINDINGBAND)
165:         amt = (Settings::MECHANICS_GENERATION >= 6) ? self.totalhp / 6 : self.totalhp / 8
166:       end
167:       ret += [amt, 1].max
168:     end
169:     # Perish Song
170:     return 999_999 if self.effects[PBEffects::PerishSong] == 1
171:     # Bad Dreams
172:     if battler.asleep? && self.statusCount > 1 && battler.takesIndirectDamage?
173:       @ai.each_battler do |b, i|
174:         next if i == @index || !b.battler.near?(battler) || !b.has_active_ability?(:BADDREAMS)
175:         ret += [self.totalhp / 8, 1].max
176:       end
177:     end
178:     # Sticky Barb
179:     if has_active_item?(:STICKYBARB) && battler.takesIndirectDamage?
180:       ret += [self.totalhp / 8, 1].max
181:     end
182:     return ret
183:   end

187:   def base_stat(stat)
188:     ret = 0
189:     case stat
190:     when :ATTACK          then ret = battler.attack
191:     when :DEFENSE         then ret = battler.defense
192:     when :SPECIAL_ATTACK  then ret = battler.spatk
193:     when :SPECIAL_DEFENSE then ret = battler.spdef
194:     when :SPEED           then ret = battler.speed
195:     end
196:     return ret
197:   end

200:     return battler.pbSpeed if stat == :SPEED && @ai.trainer.high_skill?
201:     stage_mul = Battle::Battler::STAT_STAGE_MULTIPLIERS
202:     stage_div = Battle::Battler::STAT_STAGE_DIVISORS
203:     if [:ACCURACY, :EVASION].include?(stat)
204:       stage_mul = Battle::Battler::ACC_EVA_STAGE_MULTIPLIERS
205:       stage_div = Battle::Battler::ACC_EVA_STAGE_DIVISORS
