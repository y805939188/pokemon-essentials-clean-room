4: class Battle::AI::AIMove
5:   attr_reader :move
6: 
7:   def initialize(ai)
8:     @ai = ai
9:   end
10: 
11:   def set_up(move)
12:     @move = move
13:     @move.calcType = rough_type
14:     @ai.battle.moldBreaker ||= ["IgnoreTargetAbility",
15:                                 "CategoryDependsOnHigherDamageIgnoreTargetAbility"].include?(function_code)
16:   end
17: 
18:   #-----------------------------------------------------------------------------
19: 
20:   def id;                            return @move.id;                      end
21:   def name;                          return @move.name;                    end
22:   def physicalMove?(thisType = nil); return @move.physicalMove?(thisType); end
23:   def specialMove?(thisType = nil);  return @move.specialMove?(thisType);  end
24:   def damagingMove?;                 return @move.damagingMove?;           end
25:   def statusMove?;                   return @move.statusMove?;             end
26:   def function_code;                 return @move.function_code;           end
27: 
28:   #-----------------------------------------------------------------------------
29: 
30:   def type; return @move.type; end
31: 
32:   def rough_type
33:     return @move.pbCalcType(@ai.user.battler) if @ai.trainer.medium_skill?
34:     return @move.type
35:   end
36: 
37:   #-----------------------------------------------------------------------------
38: 
39:   def pbTarget(user)
40:     return @move.pbTarget((user.is_a?(Battle::AI::AIBattler)) ? user.battler : user)
41:   end
42: 
43:   # Returns whether this move targets multiple battlers.
44:   def targets_multiple_battlers?
45:     user_battler = @ai.user.battler
46:     target_data = pbTarget(user_battler)
47:     return false if target_data.num_targets <= 1
48:     num_targets = 0
49:     case target_data.id
50:     when :AllAllies
51:       @ai.battle.allSameSideBattlers(user_battler).each { |b| num_targets += 1 if b.index != user_battler.index }
52:     when :UserAndAllies
53:       @ai.battle.allSameSideBattlers(user_battler).each { |_b| num_targets += 1 }
54:     when :AllNearFoes
55:       @ai.battle.allOtherSideBattlers(user_battler).each { |b| num_targets += 1 if b.near?(user_battler) }
56:     when :AllFoes
57:       @ai.battle.allOtherSideBattlers(user_battler).each { |_b| num_targets += 1 }
58:     when :AllNearOthers
59:       @ai.battle.allBattlers.each { |b| num_targets += 1 if b.near?(user_battler) }
60:     when :AllBattlers
61:       @ai.battle.allBattlers.each { |_b| num_targets += 1 }
62:     end
63:     return num_targets > 1
64:   end
65: 
66:   #-----------------------------------------------------------------------------
67: 
68:   def rough_priority(user)
69:     ret = @move.pbPriority(user.battler)
70:     if user.ability_active?
71:       ret = Battle::AbilityEffects.triggerPriorityChange(user.ability, user.battler, @move, ret)
72:       user.battler.effects[PBEffects::Prankster] = false   # Untrigger this
73:     end
74:     return ret
75:   end
76: 
77:   #-----------------------------------------------------------------------------
78: 
79:   # Returns this move's base power, taking into account various effects that
80:   # modify it.
81:   def base_power
82:     ret = @move.power
83:     ret = 60 if ret == 1
84:     return ret if !@ai.trainer.medium_skill?
85:     return Battle::AI::Handlers.get_base_power(function_code,
86:        ret, self, @ai.user, @ai.target, @ai, @ai.battle)
87:   end
88: 
89:   # Full damage calculation.
90:   def rough_damage
91:     base_dmg = base_power
92:     return base_dmg if @move.is_a?(Battle::Move::FixedDamageMove)
93:     max_stage = Battle::Battler::STAT_STAGE_MAXIMUM
94:     stage_mul = Battle::Battler::STAT_STAGE_MULTIPLIERS
95:     stage_div = Battle::Battler::STAT_STAGE_DIVISORS
96:     # Get the user and target of this move
97:     user = @ai.user
98:     user_battler = user.battler
99:     target = @ai.target
100:     target_battler = target.battler
101:     # Get the move's type
102:     calc_type = rough_type
103:     # Decide whether the move has 50% chance of higher of being a critical hit
104:     crit_stage = rough_critical_hit_stage
105:     is_critical = crit_stage >= Battle::Move::CRITICAL_HIT_RATIOS.length ||
106:                   Battle::Move::CRITICAL_HIT_RATIOS[crit_stage] <= 2
107:     ##### Calculate user's attack stat #####
108:     if ["CategoryDependsOnHigherDamagePoisonTarget",
109:         "CategoryDependsOnHigherDamageIgnoreTargetAbility"].include?(function_code)
110:       @move.pbOnStartUse(user.battler, [target.battler])   # Calculate category
111:     end
112:     atk, atk_stage = @move.pbGetAttackStats(user.battler, target.battler)
113:     if !target.has_active_ability?(:UNAWARE) || @ai.battle.moldBreaker
114:       atk_stage = max_stage if is_critical && atk_stage < max_stage
115:       atk = (atk.to_f * stage_mul[atk_stage] / stage_div[atk_stage]).floor
116:     end
117:     ##### Calculate target's defense stat #####
118:     defense, def_stage = @move.pbGetDefenseStats(user.battler, target.battler)
119:     if !user.has_active_ability?(:UNAWARE) || @ai.battle.moldBreaker
120:       def_stage = max_stage if is_critical && def_stage > max_stage
121:       defense = (defense.to_f * stage_mul[def_stage] / stage_div[def_stage]).floor
122:     end
123:     ##### Calculate all multiplier effects #####
124:     multipliers = {
125:       :power_multiplier        => 1.0,
126:       :attack_multiplier       => 1.0,
127:       :defense_multiplier      => 1.0,
128:       :final_damage_multiplier => 1.0
129:     }
130:     # Global abilities
131:     if @ai.trainer.medium_skill? &&
132:        ((@ai.battle.pbCheckGlobalAbility(:DARKAURA) && calc_type == :DARK) ||
133:         (@ai.battle.pbCheckGlobalAbility(:FAIRYAURA) && calc_type == :FAIRY))
134:       if @ai.battle.pbCheckGlobalAbility(:AURABREAK)
135:         multipliers[:power_multiplier] *= 3 / 4.0
136:       else
137:         multipliers[:power_multiplier] *= 4 / 3.0
138:       end
139:     end
140:     # Ability effects that alter damage
141:     if user.ability_active?
142:       case user.ability_id
143:       when :AERILATE, :GALVANIZE, :PIXILATE, :REFRIGERATE
144:         multipliers[:power_multiplier] *= 1.2 if type == :NORMAL   # NOTE: Not calc_type.
145:       when :ANALYTIC
146:         if rough_priority(user) <= 0
147:           user_faster = false
148:           @ai.each_battler do |b, i|
149:             user_faster = (i != user.index && user.faster_than?(b))
150:             break if user_faster
151:           end
152:           multipliers[:power_multiplier] *= 1.3 if !user_faster
153:         end
154:       when :NEUROFORCE
155:         if Effectiveness.super_effective_type?(calc_type, *target.pbTypes(true))
156:           multipliers[:final_damage_multiplier] *= 1.25
157:         end
158:       when :NORMALIZE
159:         multipliers[:power_multiplier] *= 1.2 if Settings::MECHANICS_GENERATION >= 7
160:       when :SNIPER
161:         multipliers[:final_damage_multiplier] *= 1.5 if is_critical
162:       when :STAKEOUT
163:         # NOTE: Can't predict whether the target will switch out this round.
164:       when :TINTEDLENS
165:         if Effectiveness.resistant_type?(calc_type, *target.pbTypes(true))
166:           multipliers[:final_damage_multiplier] *= 2
167:         end
168:       else
169:         Battle::AbilityEffects.triggerDamageCalcFromUser(
170:           user.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
171:         )
172:       end
173:     end
174:     if !@ai.battle.moldBreaker
175:       user_battler.allAllies.each do |b|
176:         next if !b.abilityActive?
177:         Battle::AbilityEffects.triggerDamageCalcFromAlly(
178:           b.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
179:         )
180:       end
181:       if target.ability_active?
182:         case target.ability_id
183:         when :FILTER, :SOLIDROCK
184:           if Effectiveness.super_effective_type?(calc_type, *target.pbTypes(true))
185:             multipliers[:final_damage_multiplier] *= 0.75
186:           end
187:         else
188:           Battle::AbilityEffects.triggerDamageCalcFromTarget(
189:             target.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
190:           )
191:         end
192:       end
193:     end
194:     if target.ability_active?
195:       Battle::AbilityEffects.triggerDamageCalcFromTargetNonIgnorable(
196:         target.ability, user_battler, target_battler, @move, multipliers, base_dmg, calc_type
197:       )
198:     end
199:     if !@ai.battle.moldBreaker
200:       target_battler.allAllies.each do |b|
201:         next if !b.abilityActive?
202:         Battle::AbilityEffects.triggerDamageCalcFromTargetAlly(
