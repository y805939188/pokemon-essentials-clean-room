1: module PBDebug
2:   @@log = []
3: 
4:   def self.logonerr
5:     begin
6:       yield
7:     rescue
8:       PBDebug.log("")
9:       PBDebug.log("**Exception: #{$!.message}")
10:       backtrace = ""
11:       $!.backtrace.each { |line| backtrace += line + "\r\n" }
12:       PBDebug.log(backtrace)
13:       PBDebug.log("")
14:       pbPrintException($!)   # if $INTERNAL
15:       PBDebug.flush
16:     end
17:   end
18: 
19:   def self.flush
20:     if $DEBUG && $INTERNAL && @@log.length > 0
21:       File.open("Data/debuglog.txt", "a+b") { |f| f.write(@@log.join) }
22:     end
23:     @@log.clear
24:   end
25: 
26:   def self.log(msg)
27:     if $DEBUG && $INTERNAL
28:       echoln msg.gsub("%", "%%")
29:       @@log.push(msg + "\r\n")
30:       PBDebug.flush   # if @@log.length > 1024
31:     end
32:   end
33: 
34:   def self.log_header(msg)
35:     if $DEBUG && $INTERNAL
36:       echoln Console.markup_style(msg.gsub("%", "%%"), text: :light_purple)
37:       @@log.push(msg + "\r\n")
38:       PBDebug.flush   # if @@log.length > 1024
39:     end
40:   end
41: 
42:   def self.log_message(msg)
43:     if $DEBUG && $INTERNAL
44:       msg = "\"" + msg + "\""
45:       echoln Console.markup_style(msg.gsub("%", "%%"), text: :dark_gray)
46:       @@log.push(msg + "\r\n")
47:       PBDebug.flush   # if @@log.length > 1024
48:     end
49:   end
50: 
51:   def self.log_ai(msg)
52:     if $DEBUG && $INTERNAL
53:       msg = "[AI] " + msg
54:       echoln msg.gsub("%", "%%")
55:       @@log.push(msg + "\r\n")
56:       PBDebug.flush   # if @@log.length > 1024
57:     end
58:   end
59: 
60:   def self.log_score_change(amt, msg)
61:     return if amt == 0
62:     if $DEBUG && $INTERNAL
63:       sign = (amt > 0) ? "+" : "-"
64:       amt_text = sprintf("%3d", amt.abs)
65:       msg = "     #{sign}#{amt_text}: #{msg}"
66:       echoln msg.gsub("%", "%%")
67:       @@log.push(msg + "\r\n")
68:       PBDebug.flush   # if @@log.length > 1024
69:     end
70:   end
71: 
72:   def self.dump(msg)
73:     if $DEBUG && $INTERNAL
74:       File.open("Data/dumplog.txt", "a+b") { |f| f.write("#{msg}\r\n") }
75:     end
76:   end
77: end
