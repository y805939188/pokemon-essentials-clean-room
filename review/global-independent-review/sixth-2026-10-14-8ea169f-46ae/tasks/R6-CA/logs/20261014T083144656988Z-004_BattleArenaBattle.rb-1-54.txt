4: class Battle::SuccessState
5:   attr_accessor :typeMod
6:   attr_accessor :useState    # 0 - not used, 1 - failed, 2 - succeeded
7:   attr_accessor :protected
8:   attr_accessor :skill
9: 
10:   def initialize; clear; end
11: 
12:   def clear(full = true)
13:     @typeMod   = Effectiveness::NORMAL_EFFECTIVE_MULTIPLIER
14:     @useState  = 0
15:     @protected = false
16:     @skill     = 0 if full
17:   end
18: 
19:   def updateSkill
20:     case @useState
21:     when 1
22:       @skill = -2 if !@protected
23:     when 2
24:       if Effectiveness.super_effective?(@typeMod)
25:         @skill = 2
26:       elsif Effectiveness.normal?(@typeMod)
27:         @skill = 1
28:       elsif Effectiveness.not_very_effective?(@typeMod)
29:         @skill = -1
30:       else   # Ineffective
31:         @skill = -2
32:       end
33:     end
34:     clear(false)
35:   end
36: end
