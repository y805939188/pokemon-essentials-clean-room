182:   # Sets the number of battler slots on each side of the field independently.
183:   # For "1v2" names, the first number is for the player's side and the second
184:   # number is for the opposing side.
185:   def setBattleMode(mode)
186:     @sideSizes =
187:       case mode
188:       when "triple", "3v3" then [3, 3]
189:       when "3v2"           then [3, 2]
190:       when "3v1"           then [3, 1]
191:       when "2v3"           then [2, 3]
192:       when "double", "2v2" then [2, 2]
193:       when "2v1"           then [2, 1]
194:       when "1v3"           then [1, 3]
195:       when "1v2"           then [1, 2]
196:       else                      [1, 1]   # Single, 1v1 (default)
197:       end
198:   end
