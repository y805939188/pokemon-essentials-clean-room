1133: def pbCopyBitmap(dstbm, srcbm, x, y, opacity = 255)
1134:   rc = Rect.new(0, 0, srcbm.width, srcbm.height)
1135:   dstbm.blt(x, y, srcbm, rc, opacity)
1136: end
1137: 
1138: def pbDrawImagePositions(bitmap, textpos)
1139:   textpos.each do |i|
1140:     srcbitmap = AnimatedBitmap.new(pbBitmapName(i[0]))
1141:     x = i[1]
1142:     y = i[2]
1143:     srcx = i[3] || 0
1144:     srcy = i[4] || 0
1145:     width = (i[5] && i[5] >= 0) ? i[5] : srcbitmap.width
1146:     height = (i[6] && i[6] >= 0) ? i[6] : srcbitmap.height
1147:     srcrect = Rect.new(srcx, srcy, width, height)
1148:     bitmap.blt(x, y, srcbitmap.bitmap, srcrect)
1149:     srcbitmap.dispose
1150:   end
1151: end
