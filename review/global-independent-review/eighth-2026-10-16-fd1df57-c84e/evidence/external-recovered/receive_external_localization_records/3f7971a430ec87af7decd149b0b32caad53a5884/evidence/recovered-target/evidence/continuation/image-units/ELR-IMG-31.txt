71:   def pbStartSceneBrief(species)
72:     @viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
73:     @viewport.z = 99999
74:     dexnum = 0
75:     dexnumshift = false
76:     if $player.pokedex.unlocked?(-1)   # National Dex is unlocked
77:       species_data = GameData::Species.try_get(species)
78:       if species_data
79:         nationalDexList = [:NONE]
80:         GameData::Species.each_species { |s| nationalDexList.push(s.species) }
81:         dexnum = nationalDexList.index(species_data.species) || 0
82:         dexnumshift = true if dexnum > 0 && Settings::DEXES_WITH_OFFSETS.include?(-1)
83:       end
84:     else
85:       ($player.pokedex.dexes_count - 1).times do |i|   # Regional Dexes
86:         next if !$player.pokedex.unlocked?(i)
87:         num = pbGetRegionalNumber(i, species)
88:         next if num <= 0
89:         dexnum = num
90:         dexnumshift = true if Settings::DEXES_WITH_OFFSETS.include?(i)
91:         break
92:       end
93:     end
94:     @dexlist = [{
95:       :species => species,
96:       :name    => "",
97:       :height  => 0,
98:       :weight  => 0,
99:       :number  => dexnum,
100:       :shift   => dexnumshift
101:     }]
102:     @index = 0
103:     @page = 1
104:     @brief = true
105:     @typebitmap = AnimatedBitmap.new(_INTL("Graphics/UI/Pokedex/icon_types"))
106:     @sprites = {}
107:     @sprites["background"] = IconSprite.new(0, 0, @viewport)
108:     @sprites["infosprite"] = PokemonSprite.new(@viewport)
109:     @sprites["infosprite"].setOffset(PictureOrigin::CENTER)
110:     @sprites["infosprite"].x = 104
111:     @sprites["infosprite"].y = 136
112:     @sprites["overlay"] = BitmapSprite.new(Graphics.width, Graphics.height, @viewport)
113:     pbSetSystemFont(@sprites["overlay"].bitmap)
114:     pbUpdateDummyPokemon
115:     drawPage(@page)
116:     pbFadeInAndShow(@sprites) { pbUpdate }
117:   end
