43:   def initializeDataBoxGraphic(sideSize)
44:     onPlayerSide = @battler.index.even?
45:     # Get the data box graphic and set whether the HP numbers/Exp bar are shown
46:     if sideSize == 1   # One Pokémon on side, use the regular dara box BG
47:       bgFilename = [_INTL("Graphics/UI/Battle/databox_normal"),
48:                     _INTL("Graphics/UI/Battle/databox_normal_foe")][@battler.index % 2]
49:       if onPlayerSide
50:         @show_hp_numbers = true
51:         @show_exp_bar    = true
52:       end
53:     else   # Multiple Pokémon on side, use the thin dara box BG
54:       bgFilename = [_INTL("Graphics/UI/Battle/databox_thin"),
55:                     _INTL("Graphics/UI/Battle/databox_thin_foe")][@battler.index % 2]
56:     end
57:     @databoxBitmap&.dispose
58:     @databoxBitmap = AnimatedBitmap.new(bgFilename)
59:     # Determine the co-ordinates of the data box and the left edge padding width
60:     if onPlayerSide
61:       @spriteX = Graphics.width - 244
62:       @spriteY = Graphics.height - 192
63:       @spriteBaseX = 34
64:     else
65:       @spriteX = -16
66:       @spriteY = 36
67:       @spriteBaseX = 16
68:     end
69:     case sideSize
70:     when 2
71:       @spriteX += [-12,  12,  0,  0][@battler.index]
72:       @spriteY += [-20, -34, 34, 20][@battler.index]
73:     when 3
74:       @spriteX += [-12,  12, -6,  6,  0,  0][@battler.index]
75:       @spriteY += [-42, -46,  4,  0, 50, 46][@battler.index]
76:     end
77:   end
