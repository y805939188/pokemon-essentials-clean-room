64:   def drawItem(index, _count, rect)
65:     textpos = []
66:     rect = Rect.new(rect.x + 16, rect.y + 16, rect.width - 16, rect.height)
67:     thispocket = @bag.pockets[@pocket]
68:     if index == self.itemCount - 1
69:       textpos.push([_INTL("CLOSE BAG"), rect.x, rect.y + 2, :left, self.baseColor, self.shadowColor])
70:     else
71:       item = (@filterlist) ? thispocket[@filterlist[@pocket][index]][0] : thispocket[index][0]
72:       baseColor   = self.baseColor
73:       shadowColor = self.shadowColor
74:       if @sorting && index == self.index
75:         baseColor   = Color.new(224, 0, 0)
76:         shadowColor = Color.new(248, 144, 144)
77:       end
78:       textpos.push(
79:         [@adapter.getDisplayName(item), rect.x, rect.y + 2, :left, baseColor, shadowColor]
80:       )
81:       item_data = GameData::Item.get(item)
82:       showing_register_icon = false
83:       if item_data.is_important?
84:         if @bag.registered?(item)
85:           pbDrawImagePositions(
86:             self.contents,
87:             [[_INTL("Graphics/UI/Bag/icon_register"), rect.x + rect.width - 72, rect.y + 8, 0, 0, -1, 24]]
88:           )
89:           showing_register_icon = true
90:         elsif pbCanRegisterItem?(item)
91:           pbDrawImagePositions(
92:             self.contents,
93:             [[_INTL("Graphics/UI/Bag/icon_register"), rect.x + rect.width - 72, rect.y + 8, 0, 24, -1, 24]]
94:           )
95:           showing_register_icon = true
96:         end
97:       end
98:       if item_data.show_quantity? && !showing_register_icon
99:         qty = (@filterlist) ? thispocket[@filterlist[@pocket][index]][1] : thispocket[index][1]
100:         qtytext = _ISPRINTF("x{1: 3d}", qty)
101:         xQty    = rect.x + rect.width - self.contents.text_size(qtytext).width - 16
102:         textpos.push([qtytext, xQty, rect.y + 2, :left, baseColor, shadowColor])
103:       end
104:     end
105:     pbDrawTextPositions(self.contents, textpos)
106:   end
