154     def self.get_species_form(species, form)
155       return nil if !species || !form
156       validate species => [Symbol, self, String]
157       validate form => Integer
158       species = species.species if species.is_a?(self)
159       species = species.to_sym if species.is_a?(String)
160       trial = sprintf("%s_%d", species, form).to_sym
161       species_form = (DATA[trial].nil?) ? species : trial
162       return (DATA.has_key?(species_form)) ? DATA[species_form] : nil
163     end
