55:   def set_up_screen(number)
56:     @label_screens.each_with_index do |screen, i|
57:       @labels[i].moveOpacity((screen == number) ? 10 : 0, 10, (screen == number) ? 255 : 0)
58:     end
59:     @key_screens.each_with_index do |screen, i|
60:       @keys[i].moveOpacity((screen == number) ? 10 : 0, 10, (screen == number) ? 255 : 0)
61:     end
62:     pictureWait   # Update event scene with the changes
63:   end
64: 
65:   def pbOnScreenEnd(scene, *args)
66:     last_screen = [@label_screens.max, @key_screens.max].max
67:     if @current_screen >= last_screen
68:       # End scene
69:       $game_temp.background_bitmap = Graphics.snap_to_bitmap
70:       Graphics.freeze
71:       @viewport.color = Color.black   # Ensure screen is black
72:       Graphics.transition(8, "fadetoblack")
73:       $game_temp.background_bitmap.dispose
74:       scene.dispose
75:     else
76:       # Next screen
77:       @current_screen += 1
78:       onCTrigger.clear
79:       set_up_screen(@current_screen)
80:       onCTrigger.set(method(:pbOnScreenEnd))
81:     end
82:   end
