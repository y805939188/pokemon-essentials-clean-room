220:   def initialize(viewport, z)
221:     super(viewport)
222:     self.x = 0
223:     self.y = Graphics.height - 96
224:     @battler   = nil
225:     @shiftMode = 0
226:     # NOTE: @mode is for the display of the Mega Evolution button.
227:     #       0=don't show, 1=show unpressed, 2=show pressed
228:     if USE_GRAPHICS
229:       # Create bitmaps
230:       @buttonBitmap  = AnimatedBitmap.new(_INTL("Graphics/UI/Battle/cursor_fight"))
231:       @typeBitmap    = AnimatedBitmap.new(_INTL("Graphics/UI/types"))
232:       @megaEvoBitmap = AnimatedBitmap.new(_INTL("Graphics/UI/Battle/cursor_mega"))
233:       @shiftBitmap   = AnimatedBitmap.new(_INTL("Graphics/UI/Battle/cursor_shift"))
234:       # Create background graphic
235:       background = IconSprite.new(0, Graphics.height - 96, viewport)
236:       background.setBitmap("Graphics/UI/Battle/overlay_fight")
237:       addSprite("background", background)
238:       # Create move buttons
239:       @buttons = Array.new(Pokemon::MAX_MOVES) do |i|
240:         button = Sprite.new(viewport)
241:         button.bitmap = @buttonBitmap.bitmap
242:         button.x = self.x + 4
243:         button.x += (i.even? ? 0 : (@buttonBitmap.width / 2) - 4)
244:         button.y = self.y + 6
245:         button.y += (((i / 2) == 0) ? 0 : BUTTON_HEIGHT - 4)
246:         button.src_rect.width  = @buttonBitmap.width / 2
247:         button.src_rect.height = BUTTON_HEIGHT
248:         addSprite("button_#{i}", button)
249:         next button
250:       end
251:       # Create overlay for buttons (shows move names)
252:       @overlay = BitmapSprite.new(Graphics.width, Graphics.height - self.y, viewport)
253:       @overlay.x = self.x
254:       @overlay.y = self.y
255:       pbSetNarrowFont(@overlay.bitmap)
256:       addSprite("overlay", @overlay)
257:       # Create overlay for selected move's info (shows move's PP)
258:       @infoOverlay = BitmapSprite.new(Graphics.width, Graphics.height - self.y, viewport)
259:       @infoOverlay.x = self.x
260:       @infoOverlay.y = self.y
261:       pbSetNarrowFont(@infoOverlay.bitmap)
262:       addSprite("infoOverlay", @infoOverlay)
263:       # Create type icon
264:       @typeIcon = Sprite.new(viewport)
265:       @typeIcon.bitmap = @typeBitmap.bitmap
266:       @typeIcon.x      = self.x + 416
267:       @typeIcon.y      = self.y + 20
268:       @typeIcon.src_rect.height = TYPE_ICON_HEIGHT
269:       addSprite("typeIcon", @typeIcon)
270:       # Create Mega Evolution button
271:       @megaButton = Sprite.new(viewport)
272:       @megaButton.bitmap = @megaEvoBitmap.bitmap
273:       @megaButton.x      = self.x + 120
274:       @megaButton.y      = self.y - (@megaEvoBitmap.height / 2)
275:       @megaButton.src_rect.height = @megaEvoBitmap.height / 2
276:       addSprite("megaButton", @megaButton)
277:       # Create Shift button
278:       @shiftButton = Sprite.new(viewport)
279:       @shiftButton.bitmap = @shiftBitmap.bitmap
280:       @shiftButton.x      = self.x + 4
281:       @shiftButton.y      = self.y - @shiftBitmap.height
282:       addSprite("shiftButton", @shiftButton)
283:     else
284:       # Create message box (shows type and PP of selected move)
285:       @msgBox = Window_AdvancedTextPokemon.newWithSize(
286:         "", self.x + 320, self.y, Graphics.width - 320, Graphics.height - self.y, viewport
287:       )
288:       @msgBox.baseColor   = TEXT_BASE_COLOR
289:       @msgBox.shadowColor = TEXT_SHADOW_COLOR
290:       pbSetNarrowFont(@msgBox.contents)
291:       addSprite("msgBox", @msgBox)
292:       # Create command window (shows moves)
293:       @cmdWindow = Window_CommandPokemon.newWithSize(
294:         [], self.x, self.y, 320, Graphics.height - self.y, viewport
295:       )
296:       @cmdWindow.columns       = 2
297:       @cmdWindow.columnSpacing = 4
298:       @cmdWindow.ignore_input  = true
299:       pbSetNarrowFont(@cmdWindow.contents)
300:       addSprite("cmdWindow", @cmdWindow)
