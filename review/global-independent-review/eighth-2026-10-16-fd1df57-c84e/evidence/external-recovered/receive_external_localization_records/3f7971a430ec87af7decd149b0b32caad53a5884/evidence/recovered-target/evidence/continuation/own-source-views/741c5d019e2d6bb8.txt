19:   def pbStartScene(pokemon, moves)
20:     @pokemon = pokemon
21:     @moves = moves
22:     moveCommands = []
23:     moves.each { |m| moveCommands.push(GameData::Move.get(m).name) }
24:     # Create sprite hash
25:     @viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
26:     @viewport.z = 99999
27:     @sprites = {}
28:     addBackgroundPlane(@sprites, "bg", "Move Reminder/bg", @viewport)
29:     @sprites["pokeicon"] = PokemonIconSprite.new(@pokemon, @viewport)
30:     @sprites["pokeicon"].setOffset(PictureOrigin::CENTER)
31:     @sprites["pokeicon"].x = 320
32:     @sprites["pokeicon"].y = 84
33:     @sprites["background"] = IconSprite.new(0, 0, @viewport)
34:     @sprites["background"].setBitmap("Graphics/UI/Move Reminder/cursor")
35:     @sprites["background"].y = 78
36:     @sprites["background"].src_rect = Rect.new(0, 72, 258, 72)
37:     @sprites["overlay"] = BitmapSprite.new(Graphics.width, Graphics.height, @viewport)
38:     pbSetSystemFont(@sprites["overlay"].bitmap)
39:     @sprites["commands"] = Window_CommandPokemon.new(moveCommands, 32)
40:     @sprites["commands"].height = 32 * (VISIBLEMOVES + 1)
41:     @sprites["commands"].visible = false
42:     @sprites["msgwindow"] = Window_AdvancedTextPokemon.new("")
43:     @sprites["msgwindow"].visible = false
44:     @sprites["msgwindow"].viewport = @viewport
45:     @typebitmap = AnimatedBitmap.new(_INTL("Graphics/UI/types"))
46:     pbDrawMoveList
47:     pbDeactivateWindows(@sprites)
48:     # Fade in all sprites
49:     pbFadeInAndShow(@sprites) { pbUpdate }
50:   end
52:   def pbDrawMoveList
53:     overlay = @sprites["overlay"].bitmap
54:     overlay.clear
55:     @pokemon.types.each_with_index do |type, i|
56:       type_number = GameData::Type.get(type).icon_position
57:       type_rect = Rect.new(0, type_number * 28, 64, 28)
58:       type_x = (@pokemon.types.length == 1) ? 400 : 366 + (70 * i)
59:       overlay.blt(type_x, 70, @typebitmap.bitmap, type_rect)
60:     end
61:     textpos = [
62:       [_INTL("Teach which move?"), 16, 14, :left, Color.new(88, 88, 80), Color.new(168, 184, 184)]
63:     ]
64:     imagepos = []
65:     yPos = 88
66:     VISIBLEMOVES.times do |i|
67:       moveobject = @moves[@sprites["commands"].top_item + i]
68:       if moveobject
69:         moveData = GameData::Move.get(moveobject)
70:         type_number = GameData::Type.get(moveData.display_type(@pokemon)).icon_position
71:         imagepos.push([_INTL("Graphics/UI/types"), 12, yPos - 4, 0, type_number * 28, 64, 28])
72:         textpos.push([moveData.name, 80, yPos, :left, Color.new(248, 248, 248), Color.black])
73:         textpos.push([_INTL("PP"), 112, yPos + 32, :left, Color.new(64, 64, 64), Color.new(176, 176, 176)])
74:         if moveData.total_pp > 0
75:           textpos.push([moveData.total_pp.to_s + "/" + moveData.total_pp.to_s, 230, yPos + 32, :right,
76:                         Color.new(64, 64, 64), Color.new(176, 176, 176)])
77:         else
78:           textpos.push(["--", 230, yPos + 32, :right, Color.new(64, 64, 64), Color.new(176, 176, 176)])
79:         end
80:       end
81:       yPos += 64
82:     end
83:     imagepos.push(["Graphics/UI/Move Reminder/cursor",
84:                    0, 78 + ((@sprites["commands"].index - @sprites["commands"].top_item) * 64),
85:                    0, 0, 258, 72])
86:     selMoveData = GameData::Move.get(@moves[@sprites["commands"].index])
87:     power = selMoveData.display_damage(@pokemon)
88:     category = selMoveData.display_category(@pokemon)
89:     accuracy = selMoveData.display_accuracy(@pokemon)
90:     textpos.push([_INTL("CATEGORY"), 272, 120, :left, Color.new(248, 248, 248), Color.black])
91:     textpos.push([_INTL("POWER"), 272, 152, :left, Color.new(248, 248, 248), Color.black])
92:     textpos.push([power <= 1 ? power == 1 ? "???" : "---" : power.to_s, 468, 152, :center,
93:                   Color.new(64, 64, 64), Color.new(176, 176, 176)])
94:     textpos.push([_INTL("ACCURACY"), 272, 184, :left, Color.new(248, 248, 248), Color.black])
95:     textpos.push([accuracy == 0 ? "---" : "#{accuracy}%", 468, 184, :center,
96:                   Color.new(64, 64, 64), Color.new(176, 176, 176)])
97:     pbDrawTextPositions(overlay, textpos)
98:     imagepos.push(["Graphics/UI/category", 436, 116, 0, category * 28, 64, 28])
99:     if @sprites["commands"].index < @moves.length - 1
100:       imagepos.push(["Graphics/UI/Move Reminder/buttons", 48, 350, 0, 0, 76, 32])
101:     end
102:     if @sprites["commands"].index > 0
103:       imagepos.push(["Graphics/UI/Move Reminder/buttons", 134, 350, 76, 0, 76, 32])
104:     end
105:     pbDrawImagePositions(overlay, imagepos)
106:     drawTextEx(overlay, 272, 216, 230, 5, selMoveData.description,
107:                Color.new(64, 64, 64), Color.new(176, 176, 176))
108:   end
