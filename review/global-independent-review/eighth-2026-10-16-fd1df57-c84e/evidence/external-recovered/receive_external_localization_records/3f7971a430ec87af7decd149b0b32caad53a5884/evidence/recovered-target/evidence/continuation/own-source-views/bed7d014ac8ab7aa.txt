97:   def addBitmap(x, y, bitmap)
98:     # _bitmap_ can be a Bitmap or an AnimatedBitmap
99:     # (update method isn't called if it's animated)
100:     # EventScene doesn't take ownership of the passed-in bitmap
101:     num = @pictures.length
102:     picture = PictureEx.new(num)
103:     picture.setXY(0, x, y)
104:     picture.setVisible(0, true)
105:     @pictures[num] = picture
106:     @picturesprites[num] = PictureSprite.new(@viewport, picture)
107:     @picturesprites[num].setCustomBitmap(bitmap)
108:     return picture
109:   end
