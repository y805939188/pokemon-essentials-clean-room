def pbGet(id)
  return 0 if !id || !$game_variables
  return $game_variables[id]
end

# Sets the value of a variable.
def pbSet(id, value)
  return if !id || id < 0
  $game_variables[id] = value if $game_variables
  $game_map.need_refresh = true if $game_map
end
