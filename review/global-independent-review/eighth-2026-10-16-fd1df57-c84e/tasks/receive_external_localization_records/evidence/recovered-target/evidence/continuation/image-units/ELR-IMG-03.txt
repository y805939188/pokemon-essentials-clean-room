147:   def pbPayout
148:     @replay = false
149:     payout = 0
150:     bonus = 0
151:     wonRow = []
152:     # Get reel pictures
153:     reel1 = @sprites["reel1"].showing
154:     reel2 = @sprites["reel2"].showing
155:     reel3 = @sprites["reel3"].showing
156:     combinations = [[reel1[1], reel2[1], reel3[1]],   # Centre row
157:                     [reel1[0], reel2[0], reel3[0]],   # Top row
158:                     [reel1[2], reel2[2], reel3[2]],   # Bottom row
159:                     [reel1[0], reel2[1], reel3[2]],   # Diagonal top left -> bottom right
160:                     [reel1[2], reel2[1], reel3[0]]]   # Diagonal bottom left -> top right
161:     combinations.length.times do |i|
162:       break if i >= 1 && @wager <= 1 # One coin = centre row only
163:       break if i >= 3 && @wager <= 2 # Two coins = three rows only
164:       wonRow[i] = true
165:       case combinations[i]
166:       when [1, 1, 1]   # Three Magnemites
167:         payout += 8
168:       when [2, 2, 2]   # Three Shellders
169:         payout += 8
170:       when [3, 3, 3]   # Three Pikachus
171:         payout += 15
172:       when [4, 4, 4]   # Three Psyducks
173:         payout += 15
174:       when [5, 5, 6], [5, 6, 5], [6, 5, 5], [6, 6, 5], [6, 5, 6], [5, 6, 6]   # 777 multi-colored
175:         payout += 90
176:         bonus = 1 if bonus < 1
177:       when [5, 5, 5], [6, 6, 6]   # Red 777, blue 777
178:         payout += 300
179:         bonus = 2 if bonus < 2
180:       when [7, 7, 7]   # Three replays
181:         @replay = true
182:       else
183:         if combinations[i][0] == 0   # Left cherry
184:           if combinations[i][1] == 0   # Centre cherry as well
185:             payout += 4
186:           else
187:             payout += 2
188:           end
189:         else
190:           wonRow[i] = false
191:         end
192:       end
193:     end
194:     @sprites["payout"].score = payout
195:     if payout > 0 || @replay
196:       if bonus > 0
197:         pbMEPlay("Slots big win")
198:       else
199:         pbMEPlay("Slots win")
200:       end
201:       # Show winning animation
202:       timer_start = System.uptime
203:       loop do
204:         frame = ((System.uptime - timer_start) / 0.125).to_i
205:         @sprites["window2"].bitmap&.clear
206:         @sprites["window1"].setBitmap(_INTL("Graphics/UI/Slot Machine/win"))
207:         @sprites["window1"].src_rect.set(152 * (frame % 4), 0, 152, 208)
208:         if bonus > 0
209:           @sprites["window2"].setBitmap(_INTL("Graphics/UI/Slot Machine/bonus"))
210:           @sprites["window2"].src_rect.set(152 * (bonus - 1), 0, 152, 208)
211:         end
212:         @sprites["light1"].visible = true
213:         @sprites["light1"].src_rect.set(0, 26 * (frame % 4), 96, 26)
214:         @sprites["light2"].visible = true
215:         @sprites["light2"].src_rect.set(0, 26 * (frame % 4), 96, 26)
216:         (1..5).each do |i|
217:           if wonRow[i - 1]
218:             @sprites["row#{i}"].visible = frame.even?
219:           else
220:             @sprites["row#{i}"].visible = false
221:           end
222:         end
223:         Graphics.update
224:         Input.update
225:         update
226:         break if System.uptime - timer_start >= 3.0
227:       end
228:       @sprites["light1"].visible = false
229:       @sprites["light2"].visible = false
230:       @sprites["window1"].src_rect.set(0, 0, 152, 208)
231:       # Pay out
232:       timer_start = System.uptime
233:       last_paid_tick = -1
234:       loop do
235:         break if @sprites["payout"].score <= 0
236:         Graphics.update
237:         Input.update
238:         update
239:         this_tick = ((System.uptime - timer_start) * 20).to_i   # Pay out 1 coin every 1/20 seconds
240:         if this_tick != last_paid_tick
241:           @sprites["payout"].score -= 1
242:           @sprites["credit"].score += 1
243:           this_tick = last_paid_tick
244:         end
245:         if Input.trigger?(Input::USE) || @sprites["credit"].score == Settings::MAX_COINS
246:           @sprites["credit"].score += @sprites["payout"].score
247:           @sprites["payout"].score = 0
248:         end
249:       end
250:       # Wait
251:       timer_start = System.uptime
252:       loop do
253:         Graphics.update
254:         Input.update
255:         update
256:         break if System.uptime - timer_start >= 0.5
257:       end
258:     else
259:       # Show losing animation
260:       timer_start = System.uptime
261:       loop do
262:         frame = ((System.uptime - timer_start) / 0.25).to_i
263:         @sprites["window2"].bitmap&.clear
264:         @sprites["window1"].setBitmap(_INTL("Graphics/UI/Slot Machine/lose"))
265:         @sprites["window1"].src_rect.set(152 * (frame % 2), 0, 152, 208)
266:         Graphics.update
267:         Input.update
268:         update
269:         break if System.uptime - timer_start >= 2.0
270:       end
271:     end
272:     @wager = 0
273:   end
