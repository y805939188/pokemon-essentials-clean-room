354:   def drawPageArea
355:     @sprites["background"].setBitmap(_INTL("Graphics/UI/Pokedex/bg_area"))
356:     overlay = @sprites["overlay"].bitmap
357:     base   = Color.new(88, 88, 80)
358:     shadow = Color.new(168, 184, 184)
359:     @sprites["areahighlight"].bitmap.clear
360:     # Get all points to be shown as places where @species can be encountered
361:     points = pbGetEncounterPoints
362:     # Draw coloured squares on each point of the Town Map with a nest
363:     pointcolor   = Color.new(0, 248, 248)
364:     pointcolorhl = Color.new(192, 248, 248)
365:     town_map_width = 1 + PokemonRegionMap_Scene::RIGHT - PokemonRegionMap_Scene::LEFT
366:     sqwidth = PokemonRegionMap_Scene::SQUARE_WIDTH
367:     sqheight = PokemonRegionMap_Scene::SQUARE_HEIGHT
368:     points.length.times do |j|
369:       next if !points[j]
370:       x = (j % town_map_width) * sqwidth
371:       x += (Graphics.width - @sprites["areamap"].bitmap.width) / 2
372:       y = (j / town_map_width) * sqheight
373:       y += (Graphics.height + 32 - @sprites["areamap"].bitmap.height) / 2
374:       @sprites["areahighlight"].bitmap.fill_rect(x, y, sqwidth, sqheight, pointcolor)
375:       if j - town_map_width < 0 || !points[j - town_map_width]
376:         @sprites["areahighlight"].bitmap.fill_rect(x, y - 2, sqwidth, 2, pointcolorhl)
377:       end
378:       if j + town_map_width >= points.length || !points[j + town_map_width]
379:         @sprites["areahighlight"].bitmap.fill_rect(x, y + sqheight, sqwidth, 2, pointcolorhl)
380:       end
381:       if j % town_map_width == 0 || !points[j - 1]
382:         @sprites["areahighlight"].bitmap.fill_rect(x - 2, y, 2, sqheight, pointcolorhl)
383:       end
384:       if (j + 1) % town_map_width == 0 || !points[j + 1]
385:         @sprites["areahighlight"].bitmap.fill_rect(x + sqwidth, y, 2, sqheight, pointcolorhl)
386:       end
387:     end
388:     # Set the text
389:     textpos = []
390:     if points.length == 0
391:       pbDrawImagePositions(
392:         overlay,
393:         [["Graphics/UI/Pokedex/overlay_areanone", 108, 188]]
394:       )
395:       textpos.push([_INTL("Area unknown"), Graphics.width / 2, (Graphics.height / 2) + 6, :center, base, shadow])
396:     end
397:     textpos.push([@mapdata.name, 414, 50, :center, base, shadow])
398:     textpos.push([_INTL("{1}'s area", GameData::Species.get(@species).name),
399:                   Graphics.width / 2, 358, :center, base, shadow])
400:     pbDrawTextPositions(overlay, textpos)
401:   end
