45:     # Get the data box graphic and set whether the HP numbers/Exp bar are shown
46:     if sideSize == 1   # One Pokémon on side, use the regular dara box BG

54:       bgFilename = [_INTL("Graphics/UI/Battle/databox_thin"),
55:                     _INTL("Graphics/UI/Battle/databox_thin_foe")][@battler.index % 2]
56:     end

58:     @databoxBitmap = AnimatedBitmap.new(bgFilename)

60:     if onPlayerSide

62:       @spriteY = Graphics.height - 192
63:       @spriteBaseX = 34
64:     else
65:       @spriteX = -16
66:       @spriteY = 36
67:       @spriteBaseX = 16
68:     end
69:     case sideSize
70:     when 2
71:       @spriteX += [-12,  12,  0,  0][@battler.index]
72:       @spriteY += [-20, -34, 34, 20][@battler.index]
73:     when 3
