151     hash[:weaknesses].uniq! if hash[:weaknesses].is_a?(Array)
152     hash[:resistances].uniq! if hash[:resistances].is_a?(Array)
153     hash[:immunities].uniq! if hash[:immunities].is_a?(Array)

156   def validate_all_compiled_types
157     type_names = []
158     GameData::Type.each do |type|
159       # Ensure all weaknesses/resistances/immunities are valid types
160       type.weaknesses.each do |other_type|
161         next if GameData::Type.exists?(other_type)
162         raise _INTL("'{1}' is not a defined type (type {2}, Weaknesses).", other_type.to_s, type.id)
163       end
164       type.resistances.each do |other_type|
165         next if GameData::Type.exists?(other_type)
166         raise _INTL("'{1}' is not a defined type (type {2}, Resistances).", other_type.to_s, type.id)
167       end
168       type.immunities.each do |other_type|
169         next if GameData::Type.exists?(other_type)
170         raise _INTL("'{1}' is not a defined type (type {2}, Immunities).", other_type.to_s, type.id)
171       end
172       # Get type names for translating
173       type_names.push(type.real_name)
174     end
175     MessageTypes.setMessagesAsHash(MessageTypes::TYPE_NAMES, type_names)
176   end

211   def validate_compiled_move(hash)
212     if (hash[:category] || 2) == 2 && (hash[:power] || 0) != 0
213       raise _INTL("Move {1} is defined as a Status move with a non-zero base damage.", hash[:real_name]) + "\n" + FileLineData.linereport
214     elsif (hash[:category] || 2) != 2 && (hash[:power] || 0) == 0
215       print _INTL("Warning: Move {1} is defined as Physical or Special but has a base damage of 0. Changing it to a Status move.", hash[:real_name]) + "\n" + FileLineData.linereport
216       hash[:category] = 2
217     end
218   end
