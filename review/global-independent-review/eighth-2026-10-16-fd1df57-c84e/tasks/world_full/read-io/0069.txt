    8 # Make all wild Pokémon shiny while a certain Switch is ON (see Settings).
    9 EventHandlers.add(:on_wild_pokemon_created, :make_shiny_switch,
   10   proc { |pkmn|
   11     pkmn.shiny = true if $game_switches[Settings::SHINY_WILD_POKEMON_SWITCH]
   12   }
   13 )
   14 
   15 # In the Safari Zone and Bug-Catching Contests, wild Pokémon reroll their IVs up
   16 # to 4 times if they don't have a perfect IV.
   17 EventHandlers.add(:on_wild_pokemon_created, :reroll_ivs_in_safari_and_bug_contest,
   18   proc { |pkmn|
   19     next if !pbInSafari? && !pbInBugContest?
   20     rerolled = false
   21     4.times do
   22       break if pkmn.iv.any? { |_stat, val| val == Pokemon::IV_STAT_LIMIT }
   23       rerolled = true
   24       GameData::Stat.each_main do |s|
   25         pkmn.iv[s.id] = rand(Pokemon::IV_STAT_LIMIT + 1)
   26       end
   27     end
   28     pkmn.calc_stats if rerolled
   29   }
   30 )
   31 
   32 # In Gen 6 and later, Legendary/Mythical/Ultra Beast Pokémon are guaranteed to
   33 # have at least 3 perfect IVs.
   34 EventHandlers.add(:on_wild_pokemon_created, :some_perfect_ivs_for_legendaries,
   35   proc { |pkmn|
   36     next if !Settings::LEGENDARIES_HAVE_SOME_PERFECT_IVS
   37     data = pkmn.species_data
   38     next if !data.has_flag?("Legendary") && !data.has_flag?("Mythical") && !data.has_flag?("UltraBeast")
   39     stats = []
   40     GameData::Stat.each_main { |s| stats.push(s.id) }
   41     perfect_stats = stats.sample(3)
   42     perfect_stats.each { |s| pkmn.iv[s] = Pokemon::IV_STAT_LIMIT }
   43     pkmn.calc_stats
   44   }
   45 )

   47 # Used in the random dungeon map. Makes the levels of all wild Pokémon in that
   48 # map depend on the levels of Pokémon in the player's party.
   49 # This is a simple method, and can/should be modified to account for evolutions
   50 # and other such details.  Of course, you don't HAVE to use this code.
   51 EventHandlers.add(:on_wild_pokemon_created, :level_depends_on_party,
   52   proc { |pkmn|
   53     next if !$game_map.metadata&.has_flag?("ScaleWildEncounterLevels")
   54     new_level = pbBalancedLevel($player.party) - 4 + rand(5)   # For variety
   55     new_level = new_level.clamp(1, GameData::GrowthRate.max_level)
   56     pkmn.level = new_level
   57     pkmn.calc_stats
   58     pkmn.reset_moves
   59   }
   60 )
