  261     else
  262       battle.defaultWeather = battleRules["defaultWeather"]
  263     end
  264     # Environment
  265     if battleRules["environment"].nil?
  266       battle.environment = pbGetEnvironment
  267     else
  268       battle.environment = battleRules["environment"]
  269     end
  270     # Backdrop graphic filename
  271     if !battleRules["backdrop"].nil?
  272       backdrop = battleRules["backdrop"]
  273     elsif $PokemonGlobal.nextBattleBack
  274       backdrop = $PokemonGlobal.nextBattleBack
  275     elsif $PokemonGlobal.surfing
  276       backdrop = "water"   # This applies wherever you are, including in caves
  277     elsif $game_map.metadata
  278       back = $game_map.metadata.battle_background
  279       backdrop = back if back && back != ""
  280     end
  281     backdrop = "indoor1" if !backdrop
  282     battle.backdrop = backdrop
  283     # Choose a name for bases depending on environment
  284     if battleRules["base"].nil?
  285       environment_data = GameData::Environment.try_get(battle.environment)
  286       base = environment_data.battle_base if environment_data
  287     else
  288       base = battleRules["base"]
  289     end
  290     battle.backdropBase = base if base
  291     # Time of day
  292     if $game_map.metadata&.battle_environment == :Cave
  293       battle.time = 2   # This makes Dusk Balls work properly in caves
  294     elsif Settings::TIME_SHADING
  295       timeNow = pbGetTimeNow
  296       if PBDayNight.isNight?(timeNow)
  297         battle.time = 2
  298       elsif PBDayNight.isEvening?(timeNow)
  299         battle.time = 1
  300       else
  301         battle.time = 0
  302       end
  303     end
  304   end
  305 
  306   def after_battle(outcome, can_lose)
  307     $player.party.each do |pkmn|
  308       pkmn.statusCount = 0 if pkmn.status == :POISON   # Bad poison becomes regular
  309       pkmn.makeUnmega
  310       pkmn.makeUnprimal
  311     end
  312     if $PokemonGlobal.partner
  313       $player.heal_party
  314       $PokemonGlobal.partner[3].each do |pkmn|
  315         pkmn.heal
  316         pkmn.makeUnmega
  317         pkmn.makeUnprimal
  318       end
  319     end
  320     if [2, 5].include?(outcome) && can_lose   # if loss or draw
  321       $player.party.each { |pkmn| pkmn.heal }
  322       timer_start = System.uptime
  323       until System.uptime - timer_start >= 0.25
  324         Graphics.update
  325       end
  326     end
  327     EventHandlers.trigger(:on_end_battle, outcome, can_lose)
  328     $game_player.straighten
  329   end

  331   # Save the result of the battle in a Game Variable (1 by default)
  332   #    0 - Undecided or aborted
  333   #    1 - Player won
  334   #    2 - Player lost
  335   #    3 - Player or wild Pokémon ran from battle, or player forfeited the match
  336   #    4 - Wild Pokémon was caught
  337   #    5 - Draw
  338   def set_outcome(outcome, outcome_variable = 1, trainer_battle = false)
  339     case outcome
  340     when 1, 4   # Won, caught
  341       $stats.wild_battles_won += 1 if !trainer_battle
  342       $stats.trainer_battles_won += 1 if trainer_battle
  343     when 2, 3, 5   # Lost, fled, draw
  344       $stats.wild_battles_lost += 1 if !trainer_battle
  345       $stats.trainer_battles_lost += 1 if trainer_battle
  346     end
  347     pbSet(outcome_variable, outcome)
  348   end
  349 end

  353 #===============================================================================
  354 class WildBattle
  355   # Used when walking in tall grass, hence the additional code.
  356   def self.start(*args, can_override: false)
  357     foe_party = WildBattle.generate_foes(*args)
  358     # Potentially call a different WildBattle.start-type method instead (for
  359     # roaming Pokémon, Safari battles, Bug Contest battles)
  360     if foe_party.length == 1 && can_override
  361       handled = [nil]
  362       EventHandlers.trigger(:on_calling_wild_battle, foe_party[0], handled)
  363       return handled[0] if !handled[0].nil?
  364     end
  365     # Perform the battle
  366     outcome = WildBattle.start_core(*foe_party)
  367     # Used by the Poké Radar to update/break the chain
  368     if foe_party.length == 1 && can_override
  369       EventHandlers.trigger(:on_wild_battle_end, foe_party[0].species, foe_party[0].level, outcome)
  370     end
  371     # Return false if the player lost or drew the battle, and true if any other result
  372     return outcome != 2 && outcome != 5
  373   end
  374 
  375   def self.start_core(*args)
  376     outcome_variable = $game_temp.battle_rules["outcomeVar"] || 1
  377     can_lose         = $game_temp.battle_rules["canLose"] || false
  378     # Skip battle if the player has no able Pokémon, or if holding Ctrl in Debug mode
  379     if BattleCreationHelperMethods.skip_battle?
  380       return BattleCreationHelperMethods.skip_battle(outcome_variable)
  381     end
  382     # Record information about party Pokémon to be used at the end of battle
  383     # (e.g. comparing levels for an evolution check)
  384     EventHandlers.trigger(:on_start_battle)
  385     # Generate array of foes
  386     foe_party = WildBattle.generate_foes(*args)
  387     # Generate information for the player and partner trainer(s)
  388     player_trainers, ally_items, player_party, player_party_starts = BattleCreationHelperMethods.set_up_player_trainers(foe_party)
  389     # Create the battle scene (the visual side of it)
  390     scene = BattleCreationHelperMethods.create_battle_scene
  391     # Create the battle class (the mechanics side of it)
  392     battle = Battle.new(scene, player_party, foe_party, player_trainers, nil)
  393     battle.party1starts = player_party_starts
  394     battle.ally_items   = ally_items
  395     # Set various other properties in the battle class
  396     setBattleRule("#{foe_party.length}v#{foe_party.length}") if $game_temp.battle_rules["size"].nil?
  397     BattleCreationHelperMethods.prepare_battle(battle)
  398     $game_temp.clear_battle_rules
  399     # Perform the battle itself
  400     outcome = 0
  401     pbBattleAnimation(pbGetWildBattleBGM(foe_party), (foe_party.length == 1) ? 0 : 2, foe_party) do
  402       pbSceneStandby { outcome = battle.pbStartBattle }
  403       BattleCreationHelperMethods.after_battle(outcome, can_lose)
  404     end
  405     Input.update
  406     # Save the result of the battle in a Game Variable (1 by default)
  407     BattleCreationHelperMethods.set_outcome(outcome, outcome_variable)
  408     return outcome
  409   end
  410 
  411   def self.generate_foes(*args)
  412     ret = []
  413     species_id = nil
  414     args.each do |arg|
  415       case arg
  416       when Pokemon
  417         raise _INTL("Species {1} was given but not a level.", species_id) if species_id
  418         ret.push(arg)
  419       when Array
  420         raise _INTL("Species {1} was given but not a level.", species_id) if species_id
  421         species = GameData::Species.get(arg[0]).id
  422         pkmn = pbGenerateWildPokemon(species, arg[1])
  423         ret.push(pkmn)
  424       else
  425         if species_id   # Expecting level
  426           if !arg.is_a?(Integer) || !(1..GameData::GrowthRate.max_level).include?(arg)
  427             raise _INTL("Expected a level (1..{1}) but {2} is not a number or not a valid level.", GameData::GrowthRate.max_level, arg)
  428           end
  429           ret.push(pbGenerateWildPokemon(species_id, arg))
  430           species_id = nil
  431         else   # Expecting species ID
  432           if !GameData::Species.exists?(arg)
  433             raise _INTL("Species {1} does not exist.", arg)
  434           end
  435           species_id = arg
  436         end
  437       end
  438     end
  439     raise _INTL("Species {1} was given but not a level.", species_id) if species_id
  440     return ret
  441   end
  442 end

  446 #===============================================================================
  447 class TrainerBattle
  448   # Used by most trainer events, which can be positioned in such a way that
  449   # multiple trainer events spot the player at once. The extra code in this
  450   # method deals with that case and can cause a double trainer battle instead.
  451   def self.start(*args)
  452     # If there is another NPC trainer who spotted the player at the same time,
  453     # and it is possible to have a double battle (the player has 2+ able Pokémon
  454     # or has a partner trainer), then record this first NPC trainer into
  455     # $game_temp.waiting_trainer and end this method. That second NPC event will
  456     # then trigger and cause the battle to happen against this first trainer and
  457     # themselves.
  458     if !$game_temp.waiting_trainer && pbMapInterpreterRunning? && pbCanDoubleBattle?
  459       thisEvent = pbMapInterpreter.get_self
  460       # Find all other triggered trainer events
  461       triggeredEvents = $game_player.pbTriggeredTrainerEvents([2], false, true)
  462       otherEvent = []
  463       triggeredEvents.each do |i|
  464         next if i.id == thisEvent.id
  465         next if $game_self_switches[[$game_map.map_id, i.id, "A"]]
  466         otherEvent.push(i)
  467       end
  468       # If there is exactly 1 other triggered trainer event, this trainer can be
  469       # stored up to battle with that one
  470       if otherEvent.length == 1
  471         trainers, _items, _end_speeches, _party, _party_starts = TrainerBattle.generate_foes(*args)
  472         # If this is just 1 trainer with 6 or fewer Pokémon, it can be stored up
  473         # to battle alongside the other trainer
  474         if trainers.length == 1 && trainers[0].party.length <= Settings::MAX_PARTY_SIZE
  475           $game_temp.waiting_trainer = [trainers[0], thisEvent.id]
  476           return false
  477         end
  478       end
  479     end
  480     # Perform the battle
  481     if $game_temp.waiting_trainer
  482       new_args = args + [$game_temp.waiting_trainer[0]]
  483       outcome = TrainerBattle.start_core(*new_args)
  484       pbMapInterpreter.pbSetSelfSwitch($game_temp.waiting_trainer[1], "A", true) if outcome == 1
  485       $game_temp.waiting_trainer = nil
  486     else
  487       outcome = TrainerBattle.start_core(*args)
  488     end
  489     # Return true if the player won the battle, and false if any other result
  490     return outcome == 1
  491   end

  494     outcome_variable = $game_temp.battle_rules["outcomeVar"] || 1
  495     can_lose         = $game_temp.battle_rules["canLose"] || false
  496     # Skip battle if the player has no able Pokémon, or if holding Ctrl in Debug mode
  497     if BattleCreationHelperMethods.skip_battle?
  498       return BattleCreationHelperMethods.skip_battle(outcome_variable, true)
  499     end
  500     # Record information about party Pokémon to be used at the end of battle (e.g.
  501     # comparing levels for an evolution check)
  502     EventHandlers.trigger(:on_start_battle)
  503     # Generate information for the foes
  504     foe_trainers, foe_items, foe_party, foe_party_starts = TrainerBattle.generate_foes(*args)
  505     # Generate information for the player and partner trainer(s)
  506     player_trainers, ally_items, player_party, player_party_starts = BattleCreationHelperMethods.set_up_player_trainers(foe_party)
  507     # Create the battle scene (the visual side of it)
  508     scene = BattleCreationHelperMethods.create_battle_scene
  509     # Create the battle class (the mechanics side of it)
  510     battle = Battle.new(scene, player_party, foe_party, player_trainers, foe_trainers)
