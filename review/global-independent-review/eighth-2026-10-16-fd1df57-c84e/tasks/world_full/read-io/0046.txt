   30   def add_battle_rule(rule, var = nil)
   31     rules = self.battle_rules
   32     case rule.to_s.downcase
   33     when "single", "1v1", "1v2", "2v1", "1v3", "3v1",
   34          "double", "2v2", "2v3", "3v2", "triple", "3v3"
   35       rules["size"] = rule.to_s.downcase
   36     when "canlose"                then rules["canLose"]             = true
   37     when "cannotlose"             then rules["canLose"]             = false
   38     when "canrun"                 then rules["canRun"]              = true
   39     when "cannotrun"              then rules["canRun"]              = false
   40     when "roamerflees"            then rules["roamerFlees"]         = true
   41     when "canswitch"              then rules["canSwitch"]           = true
   42     when "cannotswitch"           then rules["canSwitch"]           = false
   43     when "noexp"                  then rules["expGain"]             = false
   44     when "nomoney"                then rules["moneyGain"]           = false
   45     when "disablepokeballs"       then rules["disablePokeBalls"]    = true
   46     when "forcecatchintoparty"    then rules["forceCatchIntoParty"] = true
   47     when "switchstyle"            then rules["switchStyle"]         = true
   48     when "setstyle"               then rules["switchStyle"]         = false
   49     when "anims"                  then rules["battleAnims"]         = true
   50     when "noanims"                then rules["battleAnims"]         = false
   51     when "terrain"
   52       rules["defaultTerrain"] = GameData::BattleTerrain.try_get(var)&.id
   53     when "weather"
   54       rules["defaultWeather"] = GameData::BattleWeather.try_get(var)&.id
   55     when "environment", "environ"
   56       rules["environment"] = GameData::Environment.try_get(var)&.id
   57     when "backdrop", "battleback" then rules["backdrop"]            = var
   58     when "base"                   then rules["base"]                = var
   59     when "outcome", "outcomevar"  then rules["outcomeVar"]          = var
   60     when "nopartner"              then rules["noPartner"]           = true
   61     else
   62       raise _INTL("Battle rule \"{1}\" does not exist.", rule)
   63     end
   64   end
   65 end

   70 def setBattleRule(*args)
   71   r = nil
   72   args.each do |arg|
   73     if r
   74       $game_temp.add_battle_rule(r, arg)
   75       r = nil
   76     else
   77       case arg.downcase
   78       when "terrain", "weather", "environment", "environ", "backdrop",
   79            "battleback", "base", "outcome", "outcomevar"
   80         r = arg
   81         next
   82       end
   83       $game_temp.add_battle_rule(arg)
   84     end
   85   end
   86   raise _INTL("Argument {1} expected a variable after it but didn't have one.", r) if r
   87 end

   89 # Used to determine the environment in battle, and also the form of Burmy/
   90 # Wormadam.
   91 def pbGetEnvironment
   92   ret = :None
   93   map_env = $game_map.metadata&.battle_environment
   94   ret = map_env if map_env
   95   if $game_temp.encounter_type &&
   96      GameData::EncounterType.get($game_temp.encounter_type).type == :fishing
   97     terrainTag = $game_player.pbFacingTerrainTag
   98   else
   99     terrainTag = $game_player.terrain_tag
  100   end
  101   tile_environment = terrainTag.battle_environment
  102   if ret == :Forest && [:Grass, :TallGrass].include?(tile_environment)
  103     ret = :ForestGrass
  104   elsif tile_environment
  105     ret = tile_environment
  106   end
  107   return ret
  108 end

  125 def pbCanDoubleBattle?
  126   return true if $player.able_pokemon_count >= 2
  127   return $PokemonGlobal.partner && $player.able_pokemon_count >= 1
  128 end

  130 def pbCanTripleBattle?
  131   return true if $player.able_pokemon_count >= 3
  132   return $PokemonGlobal.partner && $player.able_pokemon_count >= 2

  141   # Skip battle if the player has no able Pokémon, or if holding Ctrl in Debug mode
  142   def skip_battle?
  143     return true if $player.able_pokemon_count == 0
  144     return true if $DEBUG && Input.press?(Input::CTRL)
  145     return false
  146   end

  148   def skip_battle(outcome_variable, trainer_battle = false)
  149     pbMessage(_INTL("SKIPPING BATTLE...")) if !trainer_battle && $player.pokemon_count > 0
  150     pbMessage(_INTL("SKIPPING BATTLE...")) if trainer_battle && $DEBUG
  151     pbMessage(_INTL("AFTER WINNING...")) if trainer_battle && $player.able_pokemon_count > 0
  152     $game_temp.clear_battle_rules
  153     if $game_temp.memorized_bgm && $game_system.is_a?(Game_System)
  154       $game_system.bgm_pause
  155       $game_system.bgm_position = $game_temp.memorized_bgm_position
  156       $game_system.bgm_resume($game_temp.memorized_bgm)
  157     end
  158     $game_temp.memorized_bgm            = nil
  159     $game_temp.memorized_bgm_position   = 0
  160     $PokemonGlobal.nextBattleBGM        = nil
  161     $PokemonGlobal.nextBattleVictoryBGM = nil
  162     $PokemonGlobal.nextBattleCaptureME  = nil
  163     $PokemonGlobal.nextBattleBack       = nil
  164     $PokemonEncounters.reset_step_count
  165     outcome = 1   # Win
  166     outcome = 0 if trainer_battle && $player.able_pokemon_count == 0   # Undecided
  167     pbSet(outcome_variable, outcome)
  168     return outcome
  169   end

  171   def partner_can_participate?(foe_party)
  172     return false if !$PokemonGlobal.partner || $game_temp.battle_rules["noPartner"]
  173     return true if foe_party.length > 1
  174     if $game_temp.battle_rules["size"]
  175       return false if $game_temp.battle_rules["size"] == "single" ||
  176                       $game_temp.battle_rules["size"][/^1v/i]   # "1v1", "1v2", "1v3", etc.
  177       return true
  178     end
  179     return false
  180   end

  182   # Generate information for the player and partner trainer(s)
  183   def set_up_player_trainers(foe_party)
  184     trainer_array = [$player]
  185     ally_items    = []
  186     pokemon_array = $player.party
  187     party_starts  = [0]
  188     if partner_can_participate?(foe_party)
  189       ally = NPCTrainer.new($PokemonGlobal.partner[1], $PokemonGlobal.partner[0])
  190       ally.id    = $PokemonGlobal.partner[2]
  191       ally.party = $PokemonGlobal.partner[3]
  192       data = GameData::Trainer.try_get($PokemonGlobal.partner[0], $PokemonGlobal.partner[1], $PokemonGlobal.partner[2])
  193       ally_items[1] = data&.items.clone || []
  194       trainer_array.push(ally)
  195       pokemon_array = []
  196       $player.party.each { |pkmn| pokemon_array.push(pkmn) }
  197       party_starts.push(pokemon_array.length)
  198       ally.party.each { |pkmn| pokemon_array.push(pkmn) }
  199       setBattleRule("double") if $game_temp.battle_rules["size"].nil?
  200     end
  201     return trainer_array, ally_items, pokemon_array, party_starts
  202   end

  208   # Sets up various battle parameters and applies special rules.
  209   def prepare_battle(battle)
  210     battleRules = $game_temp.battle_rules
  211     # The size of the battle, i.e. how many Pokémon on each side (default: "single")
  212     battle.setBattleMode(battleRules["size"]) if !battleRules["size"].nil?
  213     # Whether the game won't black out even if the player loses (default: false)
  214     battle.canLose = battleRules["canLose"] if !battleRules["canLose"].nil?
  215     # Whether the player can choose to run from the battle (default: true)
  216     battle.canRun = battleRules["canRun"] if !battleRules["canRun"].nil?
  217     # Whether the player can manually choose to switch out Pokémon (default: true)
  218     battle.canSwitch = battleRules["canSwitch"] if !battleRules["canSwitch"].nil?
  219     # Whether wild Pokémon always try to run from battle (default: nil)
  220     battle.rules["alwaysflee"] = battleRules["roamerFlees"]
  221     # Whether Pokémon gain Exp/EVs from defeating/catching a Pokémon (default: true)
  222     battle.expGain = battleRules["expGain"] if !battleRules["expGain"].nil?
  223     # Whether the player gains/loses money at the end of the battle (default: true)
  224     battle.moneyGain = battleRules["moneyGain"] if !battleRules["moneyGain"].nil?
  225     # Whether Poké Balls cannot be thrown at all
  226     battle.disablePokeBalls = battleRules["disablePokeBalls"] if !battleRules["disablePokeBalls"].nil?
  227     # Whether the player is asked what to do with a new Pokémon when their party is full
  228     battle.sendToBoxes = $PokemonSystem.sendtoboxes if Settings::NEW_CAPTURE_CAN_REPLACE_PARTY_MEMBER
  229     battle.sendToBoxes = 2 if battleRules["forceCatchIntoParty"]
  230     # Whether the player is able to switch when an opponent's Pokémon faints
  231     battle.switchStyle = ($PokemonSystem.battlestyle == 0)
  232     battle.switchStyle = battleRules["switchStyle"] if !battleRules["switchStyle"].nil?
  233     # Whether battle animations are shown
  234     battle.showAnims = ($PokemonSystem.battlescene == 0)
  235     battle.showAnims = battleRules["battleAnims"] if !battleRules["battleAnims"].nil?
  236     # Terrain
  237     if battleRules["defaultTerrain"].nil?
  238       if Settings::OVERWORLD_WEATHER_SETS_BATTLE_TERRAIN
  239         case $game_screen.weather_type
  240         when :Storm
  241           battle.defaultTerrain = :Electric
  242         when :Fog
  243           battle.defaultTerrain = :Misty
  244         end
  245       end
  246     else
  247       battle.defaultTerrain = battleRules["defaultTerrain"]
  248     end
  249     # Weather
  250     if battleRules["defaultWeather"].nil?
  251       case GameData::Weather.get($game_screen.weather_type).category
  252       when :Rain, :Storm
  253         battle.defaultWeather = :Rain
  254       when :Hail
  255         battle.defaultWeather = :Hail
  256       when :Sandstorm
  257         battle.defaultWeather = :Sandstorm
  258       when :Sun
  259         battle.defaultWeather = :Sun
  260       end
