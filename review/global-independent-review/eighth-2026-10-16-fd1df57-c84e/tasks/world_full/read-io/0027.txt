  511       route.list.push(RPG::MoveCommand.new(commands[i],
  512                                            [commands[i + 1], commands[i + 2],
  513                                             commands[i + 3], commands[i + 4]]))
  514       i += 4
  515     else
  516       route.list.push(RPG::MoveCommand.new(commands[i]))
  517     end
  518     i += 1
  519   end
  520   route.list.push(RPG::MoveCommand.new(PBMoveRoute::THROUGH_OFF))
  521   route.list.push(RPG::MoveCommand.new(0))
  522   event&.force_move_route(route)
  523   return route
  524 end

  553 def pbTurnTowardEvent(event, otherEvent)
  554   sx = 0
  555   sy = 0
  556   if $map_factory
  557     relativePos = $map_factory.getThisAndOtherEventRelativePos(otherEvent, event)
  558     sx = relativePos[0]
  559     sy = relativePos[1]
  560   else
  561     sx = event.x - otherEvent.x
  562     sy = event.y - otherEvent.y
  563   end
  564   sx += (event.width - otherEvent.width) / 2.0
  565   sy -= (event.height - otherEvent.height) / 2.0
  566   return if sx == 0 && sy == 0
  567   if sx.abs > sy.abs
  568     (sx > 0) ? event.turn_left : event.turn_right
  569   else
  570     (sy > 0) ? event.turn_up : event.turn_down
  571   end
  572 end

  574 def pbMoveTowardPlayer(event)
  575   maxsize = [$game_map.width, $game_map.height].max
  576   return if !pbEventCanReachPlayer?(event, $game_player, maxsize)
  577   loop do
  578     x = event.x
  579     y = event.y
  580     event.move_toward_player
  581     break if event.x == x && event.y == y
  582     while event.moving?
  583       Graphics.update
  584       Input.update
  585       pbUpdateSceneMap
  586     end
  587   end
  588   $PokemonMap&.addMovedEvent(event.id)
  589 end

  602 def pbSetEscapePoint
  603   $PokemonGlobal.escapePoint = [] if !$PokemonGlobal.escapePoint
  604   xco = $game_player.x
  605   yco = $game_player.y
  606   case $game_player.direction
  607   when 2   # Down
  608     yco -= 1
  609     dir = 8
  610   when 4   # Left
  611     xco += 1
  612     dir = 6
  613   when 6   # Right
  614     xco -= 1
  615     dir = 4
  616   when 8   # Up
  617     yco += 1
  618     dir = 2
  619   end
  620   $PokemonGlobal.escapePoint = [$game_map.map_id, xco, yco, dir]
  621 end

  656 def pbItemBall(item, quantity = 1)
  657   item = GameData::Item.get(item)
  658   return false if !item || quantity < 1
  659   itemname = (quantity > 1) ? item.portion_name_plural : item.portion_name
  660   pocket = item.pocket
  661   move = item.move
  662   if $bag.add(item, quantity)   # If item can be picked up
  663     meName = (item.is_key_item?) ? "Key item get" : "Item get"
  664     if item == :DNASPLICERS
  665       pbMessage("\\me[#{meName}]" + _INTL("You found \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
  666     elsif item.is_machine?   # TM or HM
  667       if quantity > 1
  668         pbMessage("\\me[Machine get]" + _INTL("You found {1} \\c[1]{2} {3}\\c[0]!",
  669                                               quantity, itemname, GameData::Move.get(move).name) + "\\wtnp[70]")
  670       else
  671         pbMessage("\\me[Machine get]" + _INTL("You found \\c[1]{1} {2}\\c[0]!",
  672                                               itemname, GameData::Move.get(move).name) + "\\wtnp[70]")
  673       end
  674     elsif quantity > 1
  675       pbMessage("\\me[#{meName}]" + _INTL("You found {1} \\c[1]{2}\\c[0]!", quantity, itemname) + "\\wtnp[40]")
  676     elsif itemname.starts_with_vowel?
  677       pbMessage("\\me[#{meName}]" + _INTL("You found an \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
  678     else
  679       pbMessage("\\me[#{meName}]" + _INTL("You found a \\c[1]{1}\\c[0]!", itemname) + "\\wtnp[40]")
  680     end
  681     pbMessage(_INTL("You put the {1} in\\nyour Bag's <icon=bagPocket{2}>\\c[1]{3}\\c[0] pocket.",
  682                     itemname, pocket, PokemonBag.pocket_names[pocket - 1]))
  683     return true
  684   end
  685   # Can't add the item
  686   if item.is_machine?   # TM or HM
  687     if quantity > 1
  688       pbMessage(_INTL("You found {1} \\c[1]{2} {3}\\c[0]!", quantity, itemname, GameData::Move.get(move).name))
  689     else
  690       pbMessage(_INTL("You found \\c[1]{1} {2}\\c[0]!", itemname, GameData::Move.get(move).name))
  691     end
  692   elsif quantity > 1
  693     pbMessage(_INTL("You found {1} \\c[1]{2}\\c[0]!", quantity, itemname))
  694   elsif itemname.starts_with_vowel?
  695     pbMessage(_INTL("You found an \\c[1]{1}\\c[0]!", itemname))
  696   else
  697     pbMessage(_INTL("You found a \\c[1]{1}\\c[0]!", itemname))
  698   end
  699   pbMessage(_INTL("But your Bag is full..."))
  700   return false
  701 end
