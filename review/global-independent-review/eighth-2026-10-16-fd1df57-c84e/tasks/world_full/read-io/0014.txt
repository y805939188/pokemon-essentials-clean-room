  682 #===============================================================================
  683 # Surf
  684 #===============================================================================
  685 def pbSurf
  686   return false if !$game_player.can_ride_vehicle_with_follower?
  687   move = :SURF
  688   movefinder = $player.get_pokemon_with_move(move)
  689   if !pbCheckHiddenMoveBadge(Settings::BADGE_FOR_SURF, false) || (!$DEBUG && !movefinder)
  690     return false
  691   end
  692   if pbConfirmMessage(_INTL("The water is a deep blue color... Would you like to use Surf on it?"))
  693     speciesname = (movefinder) ? movefinder.name : $player.name
  694     pbMessage(_INTL("{1} used {2}!", speciesname, GameData::Move.get(move).name))
  695     pbCancelVehicles
  696     pbHiddenMoveAnimation(movefinder)
  697     surfbgm = GameData::Metadata.get.surf_BGM
  698     pbCueBGM(surfbgm, 0.5) if surfbgm
  699     pbStartSurfing
  700     return true
  701   end
  702   return false
  703 end
  704 
  705 def pbStartSurfing
  706   pbCancelVehicles
  707   $PokemonEncounters.reset_step_count
  708   $PokemonGlobal.surfing = true
  709   $stats.surf_count += 1
  710   pbUpdateVehicle
  711   $game_temp.surf_base_coords = $map_factory.getFacingCoords($game_player.x, $game_player.y, $game_player.direction)
  712   $game_player.jumpForward
  713 end
  714 
  715 def pbEndSurf(_xOffset, _yOffset)
  716   return false if !$PokemonGlobal.surfing
  717   return false if $game_player.pbFacingTerrainTag.can_surf
  718   base_coords = [$game_player.x, $game_player.y]
  719   if $game_player.jumpForward
  720     $game_temp.surf_base_coords = base_coords
  721     $game_temp.ending_surf = true
  722     return true
  723   end
  724   return false
  725 end

  788 #===============================================================================
  789 # Sweet Scent
  790 #===============================================================================
  791 def pbSweetScent
  792   if $game_screen.weather_type != :None
  793     pbMessage(_INTL("The sweet scent faded for some reason..."))
  794     return
  795   end
  796   viewport = Viewport.new(0, 0, Graphics.width, Graphics.height)
  797   viewport.z = 99999
  798   viewport.color.red   = 255
  799   viewport.color.green = 32
  800   viewport.color.blue  = 32
  801   viewport.color.alpha -= 10
  802   pbSEPlay("Sweet Scent")
  803   start_alpha = viewport.color.alpha
  804   duration = 2.0
  805   fade_time = 0.4
  806   pbWait(duration) do |delta_t|
  807     if delta_t < duration / 2
  808       viewport.color.alpha = lerp(start_alpha, start_alpha + 128, fade_time, delta_t)
  809     else
  810       viewport.color.alpha = lerp(start_alpha + 128, start_alpha, fade_time, delta_t - duration + fade_time)
  811     end
  812   end
  813   viewport.dispose
  814   pbSEStop(0.5)
  815   enctype = $PokemonEncounters.encounter_type
  816   if !enctype || !$PokemonEncounters.encounter_possible_here? ||
  817      !pbEncounter(enctype, false)
  818     pbMessage(_INTL("There appears to be nothing here..."))
  819   end
  820 end

  887 #===============================================================================
  888 # Waterfall
  889 #===============================================================================
  890 # Starts the ascending of a waterfall.
  891 def pbAscendWaterfall
  892   return if $game_player.direction != 8   # Can't ascend if not facing up
  893   terrain = $game_player.pbFacingTerrainTag
  894   return if !terrain.waterfall && !terrain.waterfall_crest
  895   $stats.waterfall_count += 1
  896   $PokemonGlobal.ascending_waterfall = true
  897   $game_player.through = true
  898 end
  899 
  900 # Triggers after finishing each step while ascending/descending a waterfall.
  901 def pbTraverseWaterfall
  902   if $game_player.direction == 2   # Facing down; descending
  903     terrain = $game_player.pbTerrainTag
  904     if ($DEBUG && Input.press?(Input::CTRL)) ||
  905        (!terrain.waterfall && !terrain.waterfall_crest)
  906       $PokemonGlobal.descending_waterfall = false
  907       $game_player.through = false
  908       return
  909     end
  910     $stats.waterfalls_descended += 1 if !$PokemonGlobal.descending_waterfall
  911     $PokemonGlobal.descending_waterfall = true
  912     $game_player.through = true
  913   elsif $PokemonGlobal.ascending_waterfall
  914     terrain = $game_player.pbTerrainTag
  915     if ($DEBUG && Input.press?(Input::CTRL)) ||
  916        (!terrain.waterfall && !terrain.waterfall_crest)
  917       $PokemonGlobal.ascending_waterfall = false
  918       $game_player.through = false
  919       return
  920     end
  921     $PokemonGlobal.ascending_waterfall = true
  922     $game_player.through = true
  923   end
  924 end
  925 
  926 def pbWaterfall
  927   move = :WATERFALL
  928   movefinder = $player.get_pokemon_with_move(move)
  929   if !pbCheckHiddenMoveBadge(Settings::BADGE_FOR_WATERFALL, false) || (!$DEBUG && !movefinder)
  930     pbMessage(_INTL("A wall of water is crashing down with a mighty roar."))
  931     return false
  932   end
  933   if pbConfirmMessage(_INTL("It's a large waterfall. Would you like to use Waterfall?"))
  934     speciesname = (movefinder) ? movefinder.name : $player.name
  935     pbMessage(_INTL("{1} used {2}!", speciesname, GameData::Move.get(move).name))
  936     pbHiddenMoveAnimation(movefinder)
  937     pbAscendWaterfall
  938     return true
  939   end
  940   return false
  941 end
