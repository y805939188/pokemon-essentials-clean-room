    1 #===============================================================================
    2 # ** Game_Variables
    3 #-------------------------------------------------------------------------------
    4 #  This class handles variables. It's a wrapper for the built-in class "Array."
    5 #  Refer to "$game_variables" for the instance of this class.
    6 #===============================================================================
    7 class Game_Variables
    8   def initialize
    9     @data = []
   10   end
   11 
   12   # Get Variable
   13   #     variable_id : variable ID
   14   def [](variable_id)
   15     return @data[variable_id] if variable_id <= 5000 && !@data[variable_id].nil?
   16     return 0
   17   end
   18 
   19   # Set Variable
   20   #     variable_id : variable ID
   21   #     value       : the variable's value
   22   def []=(variable_id, value)
   23     @data[variable_id] = value if variable_id <= 5000
   24   end
   25 end
