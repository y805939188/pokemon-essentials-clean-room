   10     #   :full          - every node in the map
   11     #   :no_corners    - every node except for one in each corner
   12     #   :ring          - every node around the edge of the map
   13     #   :antiring      - every node except one that touches an edge of the map
   14     #   :plus          - every node in a plus (+) shape
   15     #   :diagonal_up   - every node in a line from bottom left to top right (/)
   16     #   :diagonal_down - every node in a line from top left to bottom right (\)
   17     #   :cross         - every node in a cross (x) shape
   18     #   :quadrants     - every node except the middles of each edge (i.e. each corner bulges out)

   55     # @param area [Symbol, String, self]
   56     # @param version [Integer]
   57     # @return [self]
   58     def self.try_get(area, version = 0)
   59       validate area => [Symbol, self, String]
   60       validate version => Integer
   61       area = area.id if area.is_a?(self)
   62       area = area.to_sym if area.is_a?(String)
   63       trial = sprintf("%s_%d", area, version).to_sym
   64       area_version = (DATA[trial].nil?) ? area : trial
   65       return (DATA.has_key?(area_version)) ? DATA[area_version] : self.new({})
   66     end

   68     def initialize(hash)
   69       @id                             = hash[:id]
   70       @area                           = hash[:area]
   71       @version                        = hash[:version]                 || 0
   72       @cell_count_x                   = (hash[:dungeon_size]) ? hash[:dungeon_size][0] : 5
   73       @cell_count_y                   = (hash[:dungeon_size]) ? hash[:dungeon_size][1] : 5
   74       @cell_width                     = (hash[:cell_size]) ? hash[:cell_size][0] : 10
   75       @cell_height                    = (hash[:cell_size]) ? hash[:cell_size][1] : 10
   76       @room_min_width                 = (hash[:min_room_size]) ? hash[:min_room_size][0] : 5
   77       @room_min_height                = (hash[:min_room_size]) ? hash[:min_room_size][1] : 5
   78       @room_max_width                 = (hash[:max_room_size]) ? hash[:max_room_size][0] : @cell_width - 1
   79       @room_max_height                = (hash[:max_room_size]) ? hash[:max_room_size][1] : @cell_height - 1
   80       @corridor_width                 = hash[:corridor_width]          || 2
   81       @random_corridor_shift          = hash[:random_corridor_shift]
   82       @node_layout                    = hash[:node_layout]             || :full
   83       @room_layout                    = hash[:room_layout]             || :full
   84       @room_chance                    = hash[:room_chance]             || 70
   85       @extra_connections_count        = hash[:extra_connections_count] || 2
   86       @floor_patch_radius             = (hash[:floor_patches]) ? hash[:floor_patches][0] : 3
   87       @floor_patch_chance             = (hash[:floor_patches]) ? hash[:floor_patches][1] : 75
   88       @floor_patch_smooth_rate        = (hash[:floor_patches]) ? hash[:floor_patches][2] : 25
   89       @floor_decoration_density       = (hash[:floor_decorations]) ? hash[:floor_decorations][0] : 50
   90       @floor_decoration_large_density = (hash[:floor_decorations]) ? hash[:floor_decorations][1] : 200
   91       @void_decoration_density        = (hash[:void_decorations]) ? hash[:void_decorations][0] : 50
   92       @void_decoration_large_density  = (hash[:void_decorations]) ? hash[:void_decorations][1] : 200
   93       @rng_seed                       = hash[:rng_seed]
   94       @flags                          = hash[:flags]                   || []
   95       @pbs_file_suffix                = hash[:pbs_file_suffix]         || ""

   98     def has_flag?(flag)
   99       return @flags.any? { |f| f.downcase == flag.downcase }
  100     end

  102     def rand_cell_center
  103       x = (@cell_width / 2) + rand(-2..2)
  104       y = (@cell_height / 2) + rand(-2..2)
  105       return x, y
  106     end
  107 
  108     def rand_room_size
  109       width = @room_min_width
  110       if @room_max_width > @room_min_width
  111         width = rand(@room_min_width..@room_max_width)
  112       end
  113       height = @room_min_height
  114       if @room_max_height > @room_min_height
  115         height = rand(@room_min_height..@room_max_height)
  116       end
  117       return width, height
  118     end

  120     alias __orig__get_property_for_PBS get_property_for_PBS unless method_defined?(:__orig__get_property_for_PBS)
  121     def get_property_for_PBS(key)
  122       case key
  123       when "SectionName"      then return [@area, (@version > 0) ? @version : nil]
  124       when "DungeonSize"      then return [@cell_count_x, @cell_count_y]
  125       when "CellSize"         then return [@cell_width, @cell_height]
  126       when "MinRoomSize"      then return [@room_min_width, @room_min_height]
  127       when "MaxRoomSize"      then return [@room_max_width, @room_max_height]
  128       when "FloorPatches"     then return [@floor_patch_radius, @floor_patch_chance, @floor_patch_smooth_rate]
  129       when "FloorDecorations" then return [@floor_decoration_density, @floor_decoration_large_density]
  130       when "VoidDecorations"  then return [@void_decoration_density, @void_decoration_large_density]
  131       end
  132       return __orig__get_property_for_PBS(key)
  133     end
