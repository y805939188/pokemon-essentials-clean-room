  321                 "st", "sw", "th", "tr", "tw", "vl", "zh"]
  322         name += (rand(3) > 0) ? set1.sample : set2.sample
  323       when "E"   # ending consonant

  329                 "ns", "th", "zh"]
  330         name += (rand(3) > 0) ? set1.sample : set2.sample
  331       when "f"   # consonant and vowel
  332         set = ["iz", "us", "or"]
  333         name += set.sample

  339         name += set.sample

  358   end
  359   name = name[0, maxLength]
  360   case upper

  386 # Returns the Regional Pokédex number of the given species in the given Regional
  387 # Dex. The parameter "region" is zero-based. For example, if two regions are
  388 # defined, they would each be specified as 0 and 1.
  389 def pbGetRegionalNumber(region, species)
  390   dex_list = pbLoadRegionalDexes[region]
  391   return 0 if !dex_list || dex_list.length == 0
  392   species_data = GameData::Species.try_get(species)
  393   return 0 if !species_data
  394   dex_list.each_with_index do |s, index|
  395     return index + 1 if s == species_data.species
  396   end
  397   return 0
  398 end

  411 def pbGetRegionalDexLength(region_dex)

  429 def pbMoveTutorAnnotations(move, movelist = nil)
  430   ret = []
  431   $player.party.each_with_index do |pkmn, i|
  432     if pkmn.egg?
  433       ret[i] = _INTL("NOT ABLE")
  434     elsif pkmn.hasMove?(move)
  435       ret[i] = _INTL("LEARNED")
  436     else
  437       species = pkmn.species
  438       if movelist&.any? { |j| j == species }
  439         # Checked data from movelist given in parameter
  440         ret[i] = _INTL("ABLE")
  441       elsif pkmn.compatible_with_move?(move)
  442         # Checked data from Pokémon's tutor moves in pokemon.txt
  443         ret[i] = _INTL("ABLE")
  444       else
  445         ret[i] = _INTL("NOT ABLE")
  446       end
  447     end
  448   end
  449   return ret
  450 end

  452 def pbMoveTutorChoose(move, movelist = nil, bymachine = false, oneusemachine = false)
  453   ret = false
  454   move = GameData::Move.get(move).id
  455   if movelist.is_a?(Array)
  456     movelist.map! { |m| GameData::Move.get(m).id }
  457   end
  458   pbFadeOutIn do
  459     movename = GameData::Move.get(move).name
  460     annot = pbMoveTutorAnnotations(move, movelist)
  461     scene = PokemonParty_Scene.new
  462     screen = PokemonPartyScreen.new(scene, $player.party)
  463     screen.pbStartScene(_INTL("Teach which Pokémon?"), false, annot)
  464     loop do
  465       chosen = screen.pbChoosePokemon
  466       break if chosen < 0
  467       pokemon = $player.party[chosen]
  468       if pokemon.egg?
  469         pbMessage(_INTL("Eggs can't be taught any moves.")) { screen.pbUpdate }
  470       elsif pokemon.shadowPokemon?
  471         pbMessage(_INTL("Shadow Pokémon can't be taught any moves.")) { screen.pbUpdate }
  472       elsif movelist && movelist.none? { |j| j == pokemon.species }
  473         pbMessage(_INTL("{1} can't learn {2}.", pokemon.name, movename)) { screen.pbUpdate }
  474       elsif !pokemon.compatible_with_move?(move)
  475         pbMessage(_INTL("{1} can't learn {2}.", pokemon.name, movename)) { screen.pbUpdate }
  476       elsif pbLearnMove(pokemon, move, false, bymachine) { screen.pbUpdate }
  477         $stats.moves_taught_by_item += 1 if bymachine
  478         $stats.moves_taught_by_tutor += 1 if !bymachine
  479         pokemon.add_first_move(move) if oneusemachine
  480         ret = true
  481         break
  482       end
  483     end
  484     screen.pbEndScene
  485   end
  486   return ret   # Returns whether the move was learned by a Pokemon
  487 end

  541 def pbHideVisibleObjects
  542   visibleObjects = []
  543   ObjectSpace.each_object(Sprite) do |o|
  544     if !o.disposed? && o.visible
  545       visibleObjects.push(o)
  546       o.visible = false
  547     end
  548   end
  549   ObjectSpace.each_object(Viewport) do |o|
  550     if !pbDisposed?(o) && o.visible
  551       visibleObjects.push(o)
  552       o.visible = false
  553     end
  554   end
  555   ObjectSpace.each_object(Plane) do |o|
  556     if !o.disposed? && o.visible
  557       visibleObjects.push(o)
  558       o.visible = false
  559     end
  560   end
  561   ObjectSpace.each_object(Tilemap) do |o|
  562     if !o.disposed? && o.visible
  563       visibleObjects.push(o)
  564       o.visible = false
  565     end
  566   end
  567   ObjectSpace.each_object(Window) do |o|
  568     if !o.disposed? && o.visible
  569       visibleObjects.push(o)
  570       o.visible = false
  571     end
  572   end
  573   return visibleObjects
  574 end

  583 def pbLoadRpgxpScene(scene)
  584   return if !$scene.is_a?(Scene_Map)
  585   oldscene = $scene
  586   $scene = scene
  587   Graphics.freeze
  588   oldscene.dispose
  589   visibleObjects = pbHideVisibleObjects
  590   Graphics.transition
  591   Graphics.freeze
  592   while $scene && !$scene.is_a?(Scene_Map)
  593     $scene.main
  594   end
  595   Graphics.transition
  596   Graphics.freeze
  597   $scene = oldscene
  598   $scene.createSpritesets
  599   pbShowObjects(visibleObjects)
  600   Graphics.transition
  601 end
