    1 module GameData
    2   class TerrainTag
    3     attr_reader :id
    4     attr_reader :id_number
    5     attr_reader :real_name
    6     attr_reader :can_surf
    7     attr_reader :waterfall   # The main part only, not the crest
    8     attr_reader :waterfall_crest
    9     attr_reader :can_fish
   10     attr_reader :can_dive
   11     attr_reader :deep_bush
   12     attr_reader :shows_grass_rustle
   13     attr_reader :land_wild_encounters
   14     attr_reader :double_wild_encounters
   15     attr_reader :battle_environment
   16     attr_reader :ledge
   17     attr_reader :ice
   18     attr_reader :bridge
   19     attr_reader :shows_reflections
   20     attr_reader :must_walk
   21     attr_reader :must_walk_or_run
   22     attr_reader :ignore_passability
   23 
   24     DATA = {}
   25 
   26     extend ClassMethods
   27     include InstanceMethods
   28 
   29     # @param other [Symbol, self, String, Integer]
   30     # @return [self]
   31     def self.try_get(other)
   32       return self.get(:None) if other.nil?
   33       validate other => [Symbol, self, String, Integer]
   34       return other if other.is_a?(self)
   35       other = other.to_sym if other.is_a?(String)
   36       return (self::DATA.has_key?(other)) ? self::DATA[other] : self.get(:None)
   37     end
   38 
   39     def self.load; end
   40     def self.save; end
   41 
   42     def initialize(hash)
   43       @id                     = hash[:id]
   44       @id_number              = hash[:id_number]
   45       @real_name              = hash[:id].to_s                || "Unnamed"
   46       @can_surf               = hash[:can_surf]               || false
   47       @waterfall              = hash[:waterfall]              || false
   48       @waterfall_crest        = hash[:waterfall_crest]        || false
   49       @can_fish               = hash[:can_fish]               || false
   50       @can_dive               = hash[:can_dive]               || false
   51       @deep_bush              = hash[:deep_bush]              || false
   52       @shows_grass_rustle     = hash[:shows_grass_rustle]     || false
   53       @land_wild_encounters   = hash[:land_wild_encounters]   || false
   54       @double_wild_encounters = hash[:double_wild_encounters] || false
   55       @battle_environment     = hash[:battle_environment]
   56       @ledge                  = hash[:ledge]                  || false
   57       @ice                    = hash[:ice]                    || false
   58       @bridge                 = hash[:bridge]                 || false
   59       @shows_reflections      = hash[:shows_reflections]      || false
   60       @must_walk              = hash[:must_walk]              || false
   61       @must_walk_or_run       = hash[:must_walk_or_run]       || false
   62       @ignore_passability     = hash[:ignore_passability]     || false
   63     end
   64 
   65     alias name real_name
   66 
   67     def can_surf_freely
   68       return @can_surf && !@waterfall && !@waterfall_crest
   69     end
   70   end
   71 end
