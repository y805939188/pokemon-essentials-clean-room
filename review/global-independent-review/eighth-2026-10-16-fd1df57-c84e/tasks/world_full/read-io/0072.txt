   40   def self.tableNeighbors(data, x, y, layer = nil)
   41     return 0 if x < 0 || x >= data.xsize
   42     return 0 if y < 0 || y >= data.ysize
   43     if layer.nil?
   44       t = data[x, y]
   45     else
   46       t = data[x, y, layer]
   47     end
   48     xp1 = [x + 1, data.xsize - 1].min
   49     yp1 = [y + 1, data.ysize - 1].min
   50     xm1 = [x - 1, 0].max
   51     ym1 = [y - 1, 0].max
   52     i = 0
   53     if layer.nil?
   54       i |= 0x01 if data[  x, ym1] == t   # N
   55       i |= 0x02 if data[xp1, ym1] == t   # NE
   56       i |= 0x04 if data[xp1,   y] == t   # E
   57       i |= 0x08 if data[xp1, yp1] == t   # SE
   58       i |= 0x10 if data[  x, yp1] == t   # S
   59       i |= 0x20 if data[xm1, yp1] == t   # SW
   60       i |= 0x40 if data[xm1,   y] == t   # W
   61       i |= 0x80 if data[xm1, ym1] == t   # NW
   62     else
   63       i |= 0x01 if data[  x, ym1, layer] == t   # N
   64       i |= 0x02 if data[xp1, ym1, layer] == t   # NE
   65       i |= 0x04 if data[xp1,   y, layer] == t   # E
   66       i |= 0x08 if data[xp1, yp1, layer] == t   # SE
   67       i |= 0x10 if data[  x, yp1, layer] == t   # S
   68       i |= 0x20 if data[xm1, yp1, layer] == t   # SW
   69       i |= 0x40 if data[xm1,   y, layer] == t   # W
   70       i |= 0x80 if data[xm1, ym1, layer] == t   # NW
   71     end
   72     return i
   73   end

   75   def self.fromTileset(tileset)
   76     bmtileset = pbGetTileset(tileset.tileset_name)
   77     bmautotiles = []
   78     7.times do |i|
   79       bmautotiles.push(pbGetAutotile(tileset.autotile_names[i]))
   80     end
   81     return self.new(bmtileset, bmautotiles)
   82   end

   84   def initialize(tileset, autotiles)
   85     if tileset.mega?
   86       @tileset = TilemapRenderer::TilesetWrapper.wrapTileset(tileset)
   87       tileset.dispose
   88       @shouldWrap = true
   89     else
   90       @tileset = tileset
   91       @shouldWrap = false
   92     end
   93     @autotiles = autotiles
   94   end

   96   def dispose
   97     @tileset&.dispose
   98     @tileset = nil
   99     @autotiles.each_with_index do |autotile, i|
  100       autotile.dispose
  101       @autotiles[i] = nil
  102     end
  103   end

  105   def bltSmallAutotile(bitmap, x, y, cxTile, cyTile, id, frame)
  106     return if id >= 384 || frame < 0 || !@autotiles
  107     autotile = @autotiles[(id / 48) - 1]
  108     return if !autotile || autotile.disposed?
  109     cxTile = [cxTile / 2, 1].max
  110     cyTile = [cyTile / 2, 1].max
  111     if autotile.height == 32
  112       anim = frame * 32
  113       src_rect = Rect.new(anim, 0, 32, 32)
  114       bitmap.stretch_blt(Rect.new(x, y, cxTile * 2, cyTile * 2), autotile, src_rect)
  115     else
  116       anim = frame * 96
  117       id %= 48
  118       tiles = AUTOTILE_PATTERNS[id >> 3][id & 7]
  119       src = Rect.new(0, 0, 0, 0)
  120       4.times do |i|
  121         tile_position = tiles[i] - 1
  122         src.set(((tile_position % 6) * 16) + anim, (tile_position / 6) * 16, 16, 16)
  123         bitmap.stretch_blt(Rect.new((i % 2 * cxTile) + x, (i / 2 * cyTile) + y, cxTile, cyTile),
  124                            autotile, src)
  125       end
  126     end
  127   end

  129   def bltSmallRegularTile(bitmap, x, y, cxTile, cyTile, id)
  130     return if id < 384 || !@tileset || @tileset.disposed?
  131     rect = Rect.new(((id - 384) % 8) * 32, ((id - 384) / 8) * 32, 32, 32)
  132     rect = TilemapRenderer::TilesetWrapper.getWrappedRect(rect) if @shouldWrap
  133     bitmap.stretch_blt(Rect.new(x, y, cxTile, cyTile), @tileset, rect)
  134   end
  135 
  136   def bltSmallTile(bitmap, x, y, cxTile, cyTile, id, frame = 0)
  137     if id >= 384
  138       bltSmallRegularTile(bitmap, x, y, cxTile, cyTile, id)
  139     elsif id > 0
  140       bltSmallAutotile(bitmap, x, y, cxTile, cyTile, id, frame)
  141     end
  142   end
  143 
  144   def bltAutotile(bitmap, x, y, id, frame)
  145     bltSmallAutotile(bitmap, x, y, 32, 32, id, frame)
  146   end
  147 
  148   def bltRegularTile(bitmap, x, y, id)
  149     bltSmallRegularTile(bitmap, x, y, 32, 32, id)
  150   end
  151 
  152   def bltTile(bitmap, x, y, id, frame = 0)
  153     if id >= 384
  154       bltRegularTile(bitmap, x, y, id)
  155     elsif id > 0
  156       bltAutotile(bitmap, x, y, id, frame)
  157     end
  158   end

  164 def createMinimap(mapid)
  165   map = load_data(sprintf("Data/Map%03d.rxdata", mapid)) rescue nil
  166   return Bitmap.new(32, 32) if !map
  167   bitmap = Bitmap.new(map.width * 4, map.height * 4)
  168   black = Color.black
  169   tilesets = $data_tilesets
  170   tileset = tilesets[map.tileset_id]
  171   return bitmap if !tileset
  172   helper = TileDrawingHelper.fromTileset(tileset)
  173   map.height.times do |y|
  174     map.width.times do |x|
  175       3.times do |z|
  176         id = map.data[x, y, z]
  177         id = 0 if !id
  178         helper.bltSmallTile(bitmap, x * 4, y * 4, 4, 4, id)
  179       end
  180     end
  181   end
  182   bitmap.fill_rect(0, 0, bitmap.width, 1, black)
  183   bitmap.fill_rect(0, bitmap.height - 1, bitmap.width, 1, black)
  184   bitmap.fill_rect(0, 0, 1, bitmap.height, black)
  185   bitmap.fill_rect(bitmap.width - 1, 0, 1, bitmap.height, black)
  186   return bitmap
  187 end

  189 def bltMinimapAutotile(dstBitmap, x, y, srcBitmap, id)
  190   return if id >= 48 || !srcBitmap || srcBitmap.disposed?
  191   anim = 0
  192   cxTile = 3
  193   cyTile = 3
  194   tiles = TileDrawingHelper::AUTOTILE_PATTERNS[id >> 3][id & 7]
  195   src = Rect.new(0, 0, 0, 0)
  196   4.times do |i|
  197     tile_position = tiles[i] - 1
  198     src.set((tile_position % 6 * cxTile) + anim,
  199             tile_position / 6 * cyTile, cxTile, cyTile)
  200     dstBitmap.blt((i % 2 * cxTile) + x, (i / 2 * cyTile) + y, srcBitmap, src)
  201   end
  202 end

  204 def passable?(passages, tile_id)
  205   return false if tile_id.nil?
  206   passage = passages[tile_id]
  207   return (passage && passage < 15)
  208 end

  211 def getPassabilityMinimap(mapid)
  212   map = load_data(sprintf("Data/Map%03d.rxdata", mapid))
  213   tileset = $data_tilesets[map.tileset_id]
  214   minimap = AnimatedBitmap.new("Graphics/UI/minimap_tiles")
  215   ret = Bitmap.new(map.width * 6, map.height * 6)
  216   passtable = Table.new(map.width, map.height)
  217   passages = tileset.passages
  218   map.width.times do |i|
  219     map.height.times do |j|
  220       pass = true
  221       [2, 1, 0].each do |z|
  222         if !passable?(passages, map.data[i, j, z])
  223           pass = false
  224           break
  225         end
  226       end
  227       passtable[i, j] = pass ? 1 : 0
  228     end
  229   end
  230   neighbors = TileDrawingHelper::NEIGHBORS_TO_AUTOTILE_INDEX
  231   map.width.times do |i|
  232     map.height.times do |j|
  233       next if passtable[i, j] != 0
  234       nb = TileDrawingHelper.tableNeighbors(passtable, i, j)
  235       tile = neighbors[nb]
  236       bltMinimapAutotile(ret, i * 6, j * 6, minimap.bitmap, tile)
  237     end
  238   end
  239   minimap.dispose
  240   return ret
