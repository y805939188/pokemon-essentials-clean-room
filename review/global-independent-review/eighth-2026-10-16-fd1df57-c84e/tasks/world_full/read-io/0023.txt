    1 #===============================================================================
    2 # ** Game_SelfSwitches
    3 #-------------------------------------------------------------------------------
    4 #  This class handles self switches. It's a wrapper for the built-in class
    5 #  "Hash." Refer to "$game_self_switches" for the instance of this class.
    6 #===============================================================================
    7 class Game_SelfSwitches
    8   def initialize
    9     @data = {}
   10   end
   11 
   12   # Get Self Switch
   13   #     key : key
   14   def [](key)
   15     return @data[key] == true
   16   end
   17 
   18   # Set Self Switch
   19   #     key   : key
   20   #     value : ON (true) / OFF (false)
   21   def []=(key, value)
   22     @data[key] = value
   23   end
   24 end
