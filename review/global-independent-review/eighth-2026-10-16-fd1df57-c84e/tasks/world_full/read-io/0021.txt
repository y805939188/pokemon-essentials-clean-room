    1 #===============================================================================
    2 # ** Game_Switches
    3 #-------------------------------------------------------------------------------
    4 #  This class handles switches. It's a wrapper for the built-in class "Array."
    5 #  Refer to "$game_switches" for the instance of this class.
    6 #===============================================================================
    7 class Game_Switches
    8   def initialize
    9     @data = []
   10   end
   11 
   12   # Get Switch
   13   #     switch_id : switch ID
   14   def [](switch_id)
   15     return @data[switch_id] if switch_id <= 5000 && @data[switch_id]
   16     return false
   17   end
   18 
   19   # Set Switch
   20   #     switch_id : switch ID
   21   #     value     : ON (true) / OFF (false)
   22   def []=(switch_id, value)
   23     @data[switch_id] = value if switch_id <= 5000
   24   end
   25 end
