    1 #===============================================================================
    2 # ** Game_Temp
    3 #-------------------------------------------------------------------------------
    4 #  This class handles temporary data that is not included with save data.
    5 #  Refer to "$game_temp" for the instance of this class.
    6 #===============================================================================
    7 class Game_Temp
    8   # Flags requesting something to happen
    9   attr_accessor :menu_calling             # menu calling flag
   10   attr_accessor :ready_menu_calling       # ready menu calling flag
   11   attr_accessor :debug_calling            # debug calling flag
   12   attr_accessor :interact_calling         # EventHandlers.trigger(:on_player_interact) flag
   13   attr_accessor :battle_abort             # battle flag: interrupt (unused)
   14   attr_accessor :title_screen_calling     # return to title screen flag
   15   attr_accessor :common_event_id          # common event ID to start
   16   # Flags indicating something is happening
   17   attr_accessor :in_menu                  # menu is open
   18   attr_accessor :in_storage               # in-Pokémon storage flag
   19   attr_accessor :in_battle                # in-battle flag
   20   attr_accessor :message_window_showing   # message window showing
   21   attr_accessor :ending_surf              # jumping off surf base flag
   22   attr_accessor :surf_base_coords         # [x, y] while jumping on/off, or nil
   23   attr_accessor :in_mini_update           # performing mini update flag
   24   # Battle
   25   attr_accessor :battleback_name          # battleback file name
   26   attr_accessor :force_single_battle      # force next battle to be 1v1 flag
   27   attr_accessor :waiting_trainer          # [trainer, event ID] or nil
   28   attr_accessor :last_battle_record       # record of actions in last recorded battle
   29   # Player transfers
   30   attr_accessor :player_transferring      # player place movement flag
   31   attr_accessor :player_new_map_id        # player destination: map ID
   32   attr_accessor :player_new_x             # player destination: x-coordinate
   33   attr_accessor :player_new_y             # player destination: y-coordinate
   34   attr_accessor :player_new_direction     # player destination: direction
   35   attr_accessor :fly_destination          # [map ID, x, y] or nil
   36   # Transitions
   37   attr_accessor :transition_processing    # transition processing flag
   38   attr_accessor :transition_name          # transition file name
   39   attr_accessor :background_bitmap
   40   attr_accessor :fadestate                # for sprite hashes
   41   # Other
   42   attr_accessor :begun_new_game           # new game flag (true fron new game until saving)
   43   attr_accessor :menu_beep                # menu: play sound effect flag
   44   attr_accessor :menu_last_choice         # pause menu: index of last selection
   45   attr_accessor :memorized_bgm            # set when trainer intro BGM is played
   46   attr_accessor :memorized_bgm_position   # set when trainer intro BGM is played
   47   attr_accessor :darkness_sprite          # DarknessSprite or nil
   48   attr_accessor :mart_prices
   49 
   50   #-----------------------------------------------------------------------------
   51   # * Object Initialization
   52   #-----------------------------------------------------------------------------
   53   def initialize
   54     # Flags requesting something to happen
   55     @menu_calling           = false
   56     @ready_menu_calling     = false
   57     @debug_calling          = false
   58     @interact_calling       = false
   59     @battle_abort           = false
   60     @title_screen_calling   = false
   61     @common_event_id        = 0
   62     # Flags indicating something is happening
   63     @in_menu                = false
   64     @in_storage             = false
   65     @in_battle              = false
   66     @message_window_showing = false
   67     @ending_surf            = false
   68     @in_mini_update         = false
   69     # Battle
   70     @battleback_name        = ""
   71     @force_single_battle    = false
   72     # Player transfers
   73     @player_transferring    = false
   74     @player_new_map_id      = 0
   75     @player_new_x           = 0
   76     @player_new_y           = 0
   77     @player_new_direction   = 0
   78     # Transitions
   79     @transition_processing  = false
   80     @transition_name        = ""
   81     @fadestate              = 0
   82     # Other
   83     @begun_new_game         = false
   84     @menu_beep              = false
   85     @memorized_bgm          = nil
   86     @memorized_bgm_position = 0
   87     @menu_last_choice       = 0
   88     @mart_prices            = {}
   89   end
   90 
   91   def clear_mart_prices
   92     @mart_prices = {}
   93   end
   94 end
