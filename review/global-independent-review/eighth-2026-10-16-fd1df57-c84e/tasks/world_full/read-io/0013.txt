  515 #===============================================================================
  516 # Headbutt
  517 #===============================================================================
  518 def pbHeadbuttEffect(event = nil)
  519   pbSEPlay("Headbutt")
  520   pbWait(1.0)
  521   event = $game_player.pbFacingEvent(true) if !event
  522   a = (event.x + (event.x / 24).floor + 1) * (event.y + (event.y / 24).floor + 1)
  523   a = (a * 2 / 5) % 10   # Even 2x as likely as odd, 0 is 1.5x as likely as odd
  524   b = $player.public_ID % 10   # Practically equal odds of each value
  525   chance = 1                 # ~50%
  526   if a == b                    # 10%
  527     chance = 8
  528   elsif a > b && (a - b).abs < 5   # ~30.3%
  529     chance = 5
  530   elsif a < b && (a - b).abs > 5   # ~9.7%
  531     chance = 5
  532   end
  533   if rand(10) >= chance
  534     pbMessage(_INTL("Nope. Nothing..."))
  535   else
  536     enctype = (chance == 1) ? :HeadbuttLow : :HeadbuttHigh
  537     if pbEncounter(enctype)
  538       $stats.headbutt_battles += 1
  539     else
  540       pbMessage(_INTL("Nope. Nothing..."))
  541     end
  542   end
  543 end

  545 def pbHeadbutt(event = nil)
  546   move = :HEADBUTT
  547   movefinder = $player.get_pokemon_with_move(move)
  548   if !$DEBUG && !movefinder
  549     pbMessage(_INTL("A Pokémon could be in this tree. Maybe a Pokémon could shake it."))
  550     return false
  551   end
  552   if pbConfirmMessage(_INTL("A Pokémon could be in this tree. Would you like to use Headbutt?"))
  553     $stats.headbutt_count += 1
  554     speciesname = (movefinder) ? movefinder.name : $player.name
  555     pbMessage(_INTL("{1} used {2}!", speciesname, GameData::Move.get(move).name))
  556     pbHiddenMoveAnimation(movefinder)
  557     pbHeadbuttEffect(event)
  558     return true
  559   end
  560   return false
  561 end
