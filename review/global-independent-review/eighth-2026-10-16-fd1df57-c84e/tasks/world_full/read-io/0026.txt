  320 # NOTE: Assumes the event is 1x1 tile in size. Only returns one tile.
  321 def pbFacingTile(direction = nil, event = nil)
  322   return $map_factory.getFacingTile(direction, event) if $map_factory
  323   return pbFacingTileRegular(direction, event)
  324 end

  326 # NOTE: Assumes the event is 1x1 tile in size. Only returns one tile.
  327 def pbFacingTileRegular(direction = nil, event = nil)
  328   event = $game_player if !event
  329   return [0, 0, 0] if !event
  330   x = event.x
  331   y = event.y
  332   direction = event.direction if !direction
  333   x_offset = [0, -1, 0, 1, -1, 0, 1, -1, 0, 1][direction]
  334   y_offset = [0, 1, 1, 1, 0, 0, 0, -1, -1, -1][direction]
  335   return [$game_map.map_id, x + x_offset, y + y_offset]
  336 end

  338 # Returns whether event is in line with the player, is facing the player and is
  339 # within distance tiles of the player.
  340 def pbEventFacesPlayer?(event, player, distance)
  341   return false if !event || !player || distance <= 0
  342   x_min = x_max = y_min = y_max = -1
  343   case event.direction
  344   when 2   # Down
  345     x_min = event.x
  346     x_max = event.x + event.width - 1
  347     y_min = event.y + 1
  348     y_max = event.y + distance
  349   when 4   # Left
  350     x_min = event.x - distance
  351     x_max = event.x - 1
  352     y_min = event.y - event.height + 1
  353     y_max = event.y
  354   when 6   # Right
  355     x_min = event.x + event.width
  356     x_max = event.x + event.width - 1 + distance
  357     y_min = event.y - event.height + 1
  358     y_max = event.y
  359   when 8   # Up
  360     x_min = event.x
  361     x_max = event.x + event.width - 1
  362     y_min = event.y - event.height + 1 - distance
  363     y_max = event.y - event.height
  364   else
  365     return false
  366   end
  367   return player.x >= x_min && player.x <= x_max &&
  368          player.y >= y_min && player.y <= y_max
  369 end
  370 
  371 # Returns whether event is able to walk up to the player.
  372 def pbEventCanReachPlayer?(event, player, distance)
  373   return false if !pbEventFacesPlayer?(event, player, distance)
  374   delta_x = (event.direction == 6) ? 1 : (event.direction == 4) ? -1 : 0
  375   delta_y = (event.direction == 2) ? 1 : (event.direction == 8) ? -1 : 0
  376   case event.direction
  377   when 2   # Down
  378     real_distance = player.y - event.y - 1
  379   when 4   # Left
  380     real_distance = event.x - player.x - 1
  381   when 6   # Right
  382     real_distance = player.x - event.x - event.width
  383   when 8   # Up
  384     real_distance = event.y - event.height - player.y
  385   end
  386   if real_distance > 0
  387     real_distance.times do |i|
  388       return false if !event.can_move_from_coordinate?(event.x + (i * delta_x), event.y + (i * delta_y), event.direction)
  389     end
  390   end
  391   return true
  392 end
  393 
  394 # Returns whether the two events are standing next to each other and facing each
  395 # other.
  396 def pbFacingEachOther(event1, event2)
  397   return pbEventFacesPlayer?(event1, event2, 1) && pbEventFacesPlayer?(event2, event1, 1)
  398 end

  403 def pbCueBGM(bgm, seconds, volume = nil, pitch = nil)
  404   return if !bgm
  405   bgm = pbResolveAudioFile(bgm, volume, pitch)
  406   playingBGM = $game_system.playing_bgm
  407   if !playingBGM || playingBGM.name != bgm.name || playingBGM.pitch != bgm.pitch
  408     pbBGMFade(seconds)
  409     $game_temp.cue_bgm = bgm
  410     if !$game_temp.cue_bgm_delay
  411       $game_temp.cue_bgm_delay = seconds * 0.6
  412       $game_temp.cue_bgm_timer_start = System.uptime
  413     end
  414   elsif playingBGM
  415     pbBGMPlay(bgm)
  416   end
  417 end

  489 def pbMoveRoute(event, commands, waitComplete = false)
  490   route = RPG::MoveRoute.new
  491   route.repeat    = false
  492   route.skippable = true
  493   route.list.clear
  494   route.list.push(RPG::MoveCommand.new(PBMoveRoute::THROUGH_ON))
  495   i = 0
  496   while i < commands.length
  497     case commands[i]
  498     when PBMoveRoute::WAIT, PBMoveRoute::SWITCH_ON, PBMoveRoute::SWITCH_OFF,
  499        PBMoveRoute::CHANGE_SPEED, PBMoveRoute::CHANGE_FREQUENCY, PBMoveRoute::OPACITY,
  500        PBMoveRoute::BLENDING, PBMoveRoute::PLAY_SE, PBMoveRoute::SCRIPT
  501       route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1]]))
  502       i += 1
  503     when PBMoveRoute::SCRIPT_ASYNC
  504       route.list.push(RPG::MoveCommand.new(PBMoveRoute::SCRIPT, [commands[i + 1]]))
  505       route.list.push(RPG::MoveCommand.new(PBMoveRoute::WAIT, [0]))
  506       i += 1
  507     when PBMoveRoute::JUMP
  508       route.list.push(RPG::MoveCommand.new(commands[i], [commands[i + 1], commands[i + 2]]))
  509       i += 2
  510     when PBMoveRoute::GRAPHIC
