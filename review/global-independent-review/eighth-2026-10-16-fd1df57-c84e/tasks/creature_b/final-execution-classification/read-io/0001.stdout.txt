00030 def certificate(rows, item_clause=True):
00031     edges = collections.defaultdict(dict)
00032     for row in rows:
00033         right = row['item'] if row['item'] and item_clause else '@free:' + row['species']
00034         edges[row['species']].setdefault(right, row['index'])
00035     held = {}
00036     def augment(species, visited):
00037         for item, row_index in edges[species].items():
00038             if item in visited:
00039                 continue
00040             visited.add(item)
00041             if item not in held or augment(held[item][0], visited):
00042                 held[item] = (species, row_index)
00043                 return True
00044         return False
00045     for species in edges:
00046         augment(species, set())
00047     return dict(maximum_matching=len(held), witness_rows=[v[1] for v in held.values()])
