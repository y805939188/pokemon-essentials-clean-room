00090         type_data = GameData::Type.get(move_type)
00091         ret = sprintf("Graphics/Items/%s_%s", prefix, type_data.id)
00092         return ret if pbResolveBitmap(ret)
00093         if !item_data.is_TM?
00094           ret = sprintf("Graphics/Items/machine_%s", type_data.id)
