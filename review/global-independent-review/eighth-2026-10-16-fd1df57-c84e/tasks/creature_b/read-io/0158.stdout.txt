00502       i += 1

00506       i += 1
00507     when PBMoveRoute::JUMP

00512                                            [commands[i + 1], commands[i + 2],
00513                                             commands[i + 3], commands[i + 4]]))
00514       i += 4
00515     else
00516       route.list.push(RPG::MoveCommand.new(commands[i]))
