00544     # NOTE: This code assumes each side has a maximum of 3 battlers on it, and
00545     #       is not generalised to larger side sizes.
00546     if !singleBattle?
00547       swaps = []   # Each element is an array of two battler indices to swap
00548       2.times do |side|
00549         next if pbSideSize(side) == 1   # Only battlers on sides of size 2+ need to move
00550         # Check if any battler on this side is near any battler on the other side

00552         allSameSideBattlers(side).each do |battler|
00553           anyNear = allOtherSideBattlers(battler).any? { |other| nearBattlers?(other.index, battler.index) }
00554           break if anyNear
00555         end
00556         break if anyNear
00557         # No battlers on this side are near any battlers on the other side; try
00558         # to move them
00559         # NOTE: If we get to here (assuming both sides are of size 3 or less),
00560         #       there is definitely only 1 able battler on this side, so we
00561         #       don't need to worry about multiple battlers trying to move into
00562         #       the same position. If you add support for a side of size 4+,
00563         #       this code will need revising to account for that, as well as to
00564         #       add more complex code to ensure battlers will end up near each
00565         #       other.
00566         allSameSideBattlers(side).each do |battler|
00567           # Get the position to move to
00568           pos = -1
00569           case pbSideSize(side)
00570           when 2 then pos = [2, 3, 0, 1][battler.index]   # The unoccupied position
00571           when 3 then pos = (side == 0) ? 2 : 3    # The centre position
00572           end
00573           next if pos < 0
00574           # Can't move if the same trainer doesn't control both positions
00575           idxOwner = pbGetOwnerIndexFromBattlerIndex(battler.index)
00576           next if pbGetOwnerIndexFromBattlerIndex(pos) != idxOwner
00577           swaps.push([battler.index, pos])
00578         end
00579       end
00580       # Move battlers around
00581       swaps.each do |pair|
00582         next if pbSideSize(pair[0]) == 2 && swaps.length > 1

00586           pbDisplay(_INTL("{1} moved across!", @battlers[pair[1]].pbThis))
00587         when 3

00589         end
