00196   # Calculate the mean of all levels
00197   sum = 0
00198   party.each { |p| sum += p.level }
00199   return 1 if sum == 0
00200   mLevel = GameData::GrowthRate.max_level
00201   average = sum.to_f / party.length
00202   # Calculate the standard deviation
00203   varianceTimesN = 0
00204   party.each do |pkmn|
00205     deviation = pkmn.level - average
00206     varianceTimesN += deviation * deviation
00207   end
00208   # NOTE: This is the "population" standard deviation calculation, since no
00209   # sample is being taken.
00210   stdev = Math.sqrt(varianceTimesN / party.length)
00211   mean = 0
00212   weights = []
00213   # Skew weights according to standard deviation
00214   party.each do |pkmn|
00215     weight = pkmn.level.to_f / sum
00216     if weight < 0.5
00217       weight -= (stdev / mLevel.to_f)
00218       weight = 0.001 if weight <= 0.001
00219     else
00220       weight += (stdev / mLevel.to_f)
00221       weight = 0.999 if weight >= 0.999
00222     end
00223     weights.push(weight)
00224   end
00225   weightSum = 0
00226   weights.each { |w| weightSum += w }
00227   # Calculate the weighted mean, assigning each weight to each level's
00228   # contribution to the sum
00229   party.each_with_index { |pkmn, i| mean += pkmn.level * weights[i] }
00230   mean /= weightSum
00231   mean = mean.round
00232   mean = 1 if mean < 1
00233   # Add 2 to the mean to challenge the player
00234   mean += 2
