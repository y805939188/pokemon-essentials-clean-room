313:   def pbMain
314:     loop do
315:       Graphics.update
316:       Input.update
317:       update
318:       @sprites["window1"].bitmap&.clear
319:       @sprites["window2"].bitmap&.clear
320:       if @sprites["credit"].score == Settings::MAX_COINS
321:         pbMessage(_INTL("You've got {1} Coins.", Settings::MAX_COINS.to_s_formatted))
322:         break
323:       elsif $player.coins == 0
324:         pbMessage(_INTL("You've run out of Coins.\nGame over!"))
325:         break
326:       elsif @gameRunning   # Reels are spinning
327:         @sprites["window1"].setBitmap(_INTL("Graphics/UI/Slot Machine/stop"))
328:         timer_start = System.uptime
329:         loop do
330:           frame = ((System.uptime - timer_start) / 0.25).to_i
331:           @sprites["window1"].src_rect.set(152 * (frame % 4), 0, 152, 208)
332:           Graphics.update
333:           Input.update
334:           update
335:           if Input.trigger?(Input::USE)
336:             pbSEPlay("Slots stop")
337:             if @sprites["reel1"].spinning?
338:               @sprites["reel1"].stopSpinning(@replay)
339:               @sprites["button1"].visible = true
340:             elsif @sprites["reel2"].spinning?
341:               @sprites["reel2"].stopSpinning(@replay)
342:               @sprites["button2"].visible = true
343:             elsif @sprites["reel3"].spinning?
344:               @sprites["reel3"].stopSpinning(@replay)
345:               @sprites["button3"].visible = true
346:             end
347:           end
348:           if !@sprites["reel3"].spinning?
349:             @gameEnd = true
350:             @gameRunning = false
351:           end
352:           break if !@gameRunning
353:         end
354:       elsif @gameEnd   # Reels have been stopped
355:         pbPayout
356:         # Reset graphics
357:         @sprites["button1"].visible = false
358:         @sprites["button2"].visible = false
359:         @sprites["button3"].visible = false
360:         (1..5).each do |i|
361:           @sprites["row#{i}"].visible = false
362:         end
363:         @gameEnd = false
364:       else   # Awaiting coins for the next spin
365:         @sprites["window1"].setBitmap(_INTL("Graphics/UI/Slot Machine/insert"))
366:         timer_start = System.uptime
367:         loop do
368:           frame = ((System.uptime - timer_start) / 0.4).to_i
369:           @sprites["window1"].src_rect.set(152 * (frame % 2), 0, 152, 208)
370:           if @wager > 0
371:             @sprites["window2"].setBitmap(_INTL("Graphics/UI/Slot Machine/press"))
372:             @sprites["window2"].src_rect.set(152 * (frame % 2), 0, 152, 208)
373:           end
374:           Graphics.update
375:           Input.update
376:           update
377:           if Input.trigger?(Input::DOWN) && @wager < 3 && @sprites["credit"].score > 0
378:             pbSEPlay("Slots coin")
379:             @wager += 1
380:             @sprites["credit"].score -= 1
381:             if @wager >= 3
382:               @sprites["row5"].visible = true
383:               @sprites["row4"].visible = true
384:             elsif @wager >= 2
385:               @sprites["row3"].visible = true
386:               @sprites["row2"].visible = true
387:             elsif @wager >= 1
388:               @sprites["row1"].visible = true
389:             end
390:           elsif @wager >= 3 || (@wager > 0 && @sprites["credit"].score == 0) ||
391:                 (Input.trigger?(Input::USE) && @wager > 0) || @replay
392:             if @replay
393:               @wager = 3
394:               (1..5).each { |i| @sprites["row#{i}"].visible = true }
395:             end
396:             @sprites["reel1"].startSpinning
397:             @sprites["reel2"].startSpinning
398:             @sprites["reel3"].startSpinning
399:             @gameRunning = true
400:           elsif Input.trigger?(Input::BACK) && @wager == 0
401:             break
402:           end
403:           break if @gameRunning
404:         end
405:         break if !@gameRunning
406:       end
407:     end
408:     old_coins = $player.coins
409:     $player.coins = @sprites["credit"].score
410:     if $player.coins > old_coins
411:       $stats.coins_won += $player.coins - old_coins
412:     elsif $player.coins < old_coins
413:       $stats.coins_lost += old_coins - $player.coins
414:     end
415:   end
