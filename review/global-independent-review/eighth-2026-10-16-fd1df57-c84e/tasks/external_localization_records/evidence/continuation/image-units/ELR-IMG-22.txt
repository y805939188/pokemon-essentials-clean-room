303:   def drawPage(page)
304:     if @pokemon.egg?
305:       drawPageOneEgg
306:       return
307:     end
308:     @sprites["pokemon"].setPokemonBitmap(@pokemon)
309:     @sprites["pokeicon"].pokemon = @pokemon
310:     @sprites["itemicon"].item = @pokemon.item_id
311:     overlay = @sprites["overlay"].bitmap
312:     overlay.clear
313:     base   = Color.new(248, 248, 248)
314:     shadow = Color.new(104, 104, 104)
315:     # Set background image
316:     @sprites["background"].setBitmap("Graphics/UI/Summary/bg_#{page}")
317:     imagepos = []
318:     # Show the Poké Ball containing the Pokémon
319:     ballimage = sprintf("Graphics/UI/Summary/icon_ball_%s", @pokemon.poke_ball)
320:     imagepos.push([ballimage, 14, 60])
321:     # Show status/fainted/Pokérus infected icon
322:     status = -1
323:     if @pokemon.fainted?
324:       status = GameData::Status.count - 1
325:     elsif @pokemon.status != :NONE
326:       status = GameData::Status.get(@pokemon.status).icon_position
327:     elsif @pokemon.pokerusStage == 1
328:       status = GameData::Status.count
329:     end
330:     if status >= 0
331:       imagepos.push([_INTL("Graphics/UI/statuses"), 124, 100, 0, 16 * status, 44, 16])
332:     end
333:     # Show Pokérus cured icon
334:     if @pokemon.pokerusStage == 2
335:       imagepos.push(["Graphics/UI/Summary/icon_pokerus", 176, 100])
336:     end
337:     # Show shininess star
338:     imagepos.push(["Graphics/UI/shiny", 2, 134]) if @pokemon.shiny?
339:     # Draw all images
340:     pbDrawImagePositions(overlay, imagepos)
341:     # Write various bits of text
342:     pagename = [_INTL("INFO"),
343:                 _INTL("TRAINER MEMO"),
344:                 _INTL("SKILLS"),
345:                 _INTL("MOVES"),
346:                 _INTL("RIBBONS")][page - 1]
347:     textpos = [
348:       [pagename, 26, 22, :left, base, shadow],
349:       [@pokemon.name, 46, 68, :left, base, shadow],
350:       [@pokemon.level.to_s, 46, 98, :left, Color.new(64, 64, 64), Color.new(176, 176, 176)],
351:       [_INTL("Item"), 66, 324, :left, base, shadow]
352:     ]
353:     # Write the held item's name
354:     if @pokemon.hasItem?
355:       textpos.push([@pokemon.item.name, 16, 358, :left, Color.new(64, 64, 64), Color.new(176, 176, 176)])
356:     else
357:       textpos.push([_INTL("None"), 16, 358, :left, Color.new(192, 200, 208), Color.new(208, 216, 224)])
358:     end
359:     # Write the gender symbol
360:     if @pokemon.male?
361:       textpos.push([_INTL("♂"), 178, 68, :left, Color.new(24, 112, 216), Color.new(136, 168, 208)])
362:     elsif @pokemon.female?
363:       textpos.push([_INTL("♀"), 178, 68, :left, Color.new(248, 56, 32), Color.new(224, 152, 144)])
364:     end
365:     # Draw all text
366:     pbDrawTextPositions(overlay, textpos)
367:     # Draw the Pokémon's markings
368:     drawMarkings(overlay, 84, 292)
369:     # Draw page-specific information
370:     case page
371:     when 1 then drawPageOne
372:     when 2 then drawPageTwo
373:     when 3 then drawPageThree
374:     when 4 then drawPageFour
375:     when 5 then drawPageFive
376:     end
377:   end
