4 class Dir
5   # Reads all files in a directory
6   def self.get(dir, filters = "*", full = true)
7     files = []
8     filters = [filters] if !filters.is_a?(Array)
9     self.chdir(dir) do
10       filters.each do |filter|
11         self.glob(filter) { |f| files.push(full ? (dir + "/" + f) : f) }
12       end
13     end
14     return files.sort
15   end
16 
17   # Generates entire file/folder tree from a certain directory
18   def self.all(dir, filters = "*", full = true)
19     # sets variables for starting
20     files = []
21     subfolders = []
22     self.get(dir, filters, full).each do |file|
23       # engages in recursion to read the entire file tree
24       if self.safe?(file)   # Is a directory
25         subfolders += self.all(file, filters, full)
26       else   # Is a file
27         files += [file]
28       end
29     end
30     # returns all found files
31     return files + subfolders
32   end

34   # Checks for existing directory
35   def self.safe?(dir)
36     return FileTest.directory?(dir)
37   end

39   # Creates all the required directories for filename path
40   def self.create(path)
41     path.gsub!("\\", "/")   # Windows compatibility
42     # get path tree
43     dirs = path.split("/")
44     full = ""
45     dirs.each do |dir|
46       full += dir + "/"
47       # creates directories
48       self.mkdir(full) if !self.safe?(full)
49     end
50   end

52   # Generates entire folder tree from a certain directory
53   def self.all_dirs(dir)
54     # sets variables for starting
55     dirs = []
56     self.get(dir, "*", true).each do |file|
57       # engages in recursion to read the entire folder tree
58       dirs += self.all_dirs(file) if self.safe?(file)
59     end
60     # returns all found directories
61     return dirs.length > 0 ? (dirs + [dir]) : [dir]
62   end

64   # Deletes all the files in a directory and all the sub directories (allows for non-empty dirs)
