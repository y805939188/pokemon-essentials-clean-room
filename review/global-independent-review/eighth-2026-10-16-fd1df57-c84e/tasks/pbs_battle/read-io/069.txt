00560   # Gives an item to this Pokémon to hold.
00561   # @param value [Symbol, String, GameData::Item, nil] ID of the item to give
00562   #   to this Pokémon
00563   def item=(value)
00564     return if value && !GameData::Item.exists?(value)
00565     @item = (value) ? GameData::Item.get(value).id : value
00566   end

00568   # Returns whether this Pokémon is holding an item. If an item id is passed,
00569   # returns whether the Pokémon is holding that item.
00570   # @param check_item [Symbol, String, GameData::Item, nil] item ID to check
00571   # @return [Boolean] whether the Pokémon is holding the specified item or
00572   #   an item at all
00573   def hasItem?(check_item = nil)
00574     return !@item.nil? if check_item.nil?
00575     held_item = self.item
00576     return held_item && held_item == check_item
00577   end

00579   # @return [Array<Array<Symbol>>] the items this species can be found holding in the wild
00580   def wildHoldItems
00581     sp_data = species_data
00582     return [sp_data.wild_item_common, sp_data.wild_item_uncommon, sp_data.wild_item_rare]
00583   end

00585   # @return [Mail, nil] mail held by this Pokémon (nil if there is none)
00586   def mail
00587     @mail = nil if @mail && (!@mail.item || !hasItem?(@mail.item))
