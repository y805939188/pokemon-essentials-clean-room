00043   # Returns whether this move targets multiple battlers.
00044   def targets_multiple_battlers?
00045     user_battler = @ai.user.battler
00046     target_data = pbTarget(user_battler)
00047     return false if target_data.num_targets <= 1
00048     num_targets = 0
00049     case target_data.id
00050     when :AllAllies
00051       @ai.battle.allSameSideBattlers(user_battler).each { |b| num_targets += 1 if b.index != user_battler.index }
00052     when :UserAndAllies
00053       @ai.battle.allSameSideBattlers(user_battler).each { |_b| num_targets += 1 }
00054     when :AllNearFoes
00055       @ai.battle.allOtherSideBattlers(user_battler).each { |b| num_targets += 1 if b.near?(user_battler) }
00056     when :AllFoes
00057       @ai.battle.allOtherSideBattlers(user_battler).each { |_b| num_targets += 1 }
00058     when :AllNearOthers
00059       @ai.battle.allBattlers.each { |b| num_targets += 1 if b.near?(user_battler) }
00060     when :AllBattlers
00061       @ai.battle.allBattlers.each { |_b| num_targets += 1 }
00062     end
00063     return num_targets > 1
00064   end

00384   # Full accuracy calculation.
00385   def rough_accuracy
00386     # Determine user and target
00387     user = @ai.user
00388     user_battler = user.battler
00389     target = @ai.target
00390     target_battler = target.battler
00391     # OHKO move accuracy
00392     if @move.is_a?(Battle::Move::OHKO)
00393       ret = self.accuracy + user.level - target.level
00394       ret -= 10 if function_code == "OHKOIce" && !user.has_type?(:ICE)
00395       return [ret, 0].max
00396     end
00397     # "Always hit" effects and "always hit" accuracy
00398     if @ai.trainer.medium_skill?
00399       return 100 if target.effects[PBEffects::Telekinesis] > 0
00400       return 100 if target.effects[PBEffects::Minimize] && @move.tramplesMinimize? &&
00401                     Settings::MECHANICS_GENERATION >= 6
00402     end
00403     # Get base accuracy
00404     baseAcc = self.accuracy
00405     return 100 if baseAcc == 0
00406     # Get the move's type
00407     type = rough_type
00408     # Calculate all modifier effects
00409     modifiers = {}
00410     modifiers[:base_accuracy]  = baseAcc
00411     modifiers[:accuracy_stage] = user.stages[:ACCURACY]
00412     modifiers[:evasion_stage]  = target.stages[:EVASION]
00413     modifiers[:accuracy_multiplier] = 1.0
00414     modifiers[:evasion_multiplier]  = 1.0
00415     apply_rough_accuracy_modifiers(user, target, type, modifiers)
00416     # Check if move certainly misses/can't miss
00417     return 0 if modifiers[:base_accuracy] < 0
00418     return 100 if modifiers[:base_accuracy] == 0
00419     # Calculation
00420     max_stage = Battle::Battler::STAT_STAGE_MAXIMUM
00421     accStage = [[modifiers[:accuracy_stage], -max_stage].max, max_stage].min + max_stage
00422     evaStage = [[modifiers[:evasion_stage], -max_stage].max, max_stage].min + max_stage
00423     stageMul = Battle::Battler::ACC_EVA_STAGE_MULTIPLIERS
00424     stageDiv = Battle::Battler::ACC_EVA_STAGE_DIVISORS
00425     accuracy = 100.0 * stageMul[accStage] / stageDiv[accStage]
00426     evasion  = 100.0 * stageMul[evaStage] / stageDiv[evaStage]
00427     accuracy = (accuracy * modifiers[:accuracy_multiplier]).round
00428     evasion  = (evasion  * modifiers[:evasion_multiplier]).round
00429     evasion = 1 if evasion < 1
00430     return modifiers[:base_accuracy] * accuracy / evasion
00431   end

00504     target_battler = target.battler
00505     return -1 if target_battler.pbOwnSide.effects[PBEffects::LuckyChant] > 0
00506     crit_stage = 0
00507     # Ability effects that alter critical hit rate
00508     if user.ability_active?
00509       crit_stage = Battle::AbilityEffects.triggerCriticalCalcFromUser(user_battler.ability,
00510          user_battler, target_battler, crit_stage)
00511       return -1 if crit_stage < 0
00512     end
00513     if !@ai.battle.moldBreaker && target.ability_active?
00514       crit_stage = Battle::AbilityEffects.triggerCriticalCalcFromTarget(target_battler.ability,
00515          user_battler, target_battler, crit_stage)
00516       return -1 if crit_stage < 0
00517     end
00518     # Item effects that alter critical hit rate
00519     if user.item_active?
00520       crit_stage = Battle::ItemEffects.triggerCriticalCalcFromUser(user_battler.item,
00521          user_battler, target_battler, crit_stage)
00522       return -1 if crit_stage < 0
00523     end
00524     if target.item_active?
00525       crit_stage = Battle::ItemEffects.triggerCriticalCalcFromTarget(user_battler.item,
00526          user_battler, target_battler, crit_stage)
00527       return -1 if crit_stage < 0
00528     end
00529     # Other effects
00530     case @move.pbCritialOverride(user_battler, target_battler)
00531     when 1  then return 99
00532     when -1 then return -1
00533     end
00534     return 99 if crit_stage > 50   # Merciless
00535     return 99 if user_battler.effects[PBEffects::LaserFocus] > 0
00536     crit_stage += 1 if @move.highCriticalRate?
00537     crit_stage += user_battler.effects[PBEffects::FocusEnergy]
00538     crit_stage += 1 if user_battler.inHyperMode? && @move.type == :SHADOW
00539     crit_stage = [crit_stage, Battle::Move::CRITICAL_HIT_RATIOS.length - 1].min
00540     return crit_stage
00541   end

00555                    target.has_active_ability?(:SHIELDDUST) && !@ai.battle.moldBreaker
00556     # Prefer if the additional effect will have an increased chance of working

00559                 (user.has_active_ability?(:SERENEGRACE) || user.pbOwnSide.effects[PBEffects::Rainbow] > 0)
