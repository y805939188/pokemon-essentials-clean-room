00690   def get_csv_record(rec, schema)
00691     ret = []
00692     repeat = false
00693     start = 0
00694     schema_length = schema[1].length
00695     case schema[1][0, 1]   # First character in schema
00696     when "*"
00697       repeat = true
00698       start = 1
00699     when "^"
00700       start = 1
00701       schema_length -= 1
00702     end
00703     subarrays = repeat && schema[1].length - start > 1   # Whether ret is an array of arrays
00704     # Split the string on commas into an array of values to apply the schema to
00705     values = split_csv_line(rec)
00706     # Apply the schema to each value in the line
00707     idx = -1   # Index of value to look at in values
00708     loop do
00709       record = []
00710       (start...schema[1].length).each do |i|
00711         idx += 1
00712         sche = schema[1][i, 1]
00713         if sche[/[A-Z]/]   # Upper case = optional
00714           if nil_or_empty?(values[idx])
00715             record.push(nil)
00716             next
00717           end
00718         end
00719         if sche.downcase == "q"   # Unformatted text
00720           record.push(rec)
00721           idx = values.length
00722           break
00723         else
00724           record.push(cast_csv_value(values[idx], sche, schema[2 + i - start]))
00725         end
00726       end
00727       if !record.empty?
00728         if subarrays
00729           ret.push(record)
00730         else
00731           ret.concat(record)
00732         end
00733       end
00734       break if !repeat || idx >= values.length - 1
00735     end
00736     return (!repeat && schema_length == 1) ? ret[0] : ret

00742   def pbWriteCsvRecord(record, file, schema)
00743     rec = (record.is_a?(Array)) ? record.flatten : [record]
00744     start = (["*", "^"].include?(schema[1][0, 1])) ? 1 : 0
00745     index = -1
00746     loop do
00747       (start...schema[1].length).each do |i|
00748         index += 1
00749         value = rec[index]
00750         if schema[1][i, 1][/[A-Z]/]   # Optional
00751           # Check the rest of the values for non-nil things
00752           later_value_found = false
00753           (index...rec.length).each do |j|
00754             later_value_found = true if !rec[j].nil?
00755             break if later_value_found
00756           end
00757           if !later_value_found
00758             start = -1
00759             break
00760           end
00761         end
00762         file.write(",") if index > 0
00763         next if value.nil?
00764         case schema[1][i, 1]
00765         when "e", "E"   # Enumerable
00766           enumer = schema[2 + i - start]
00767           case enumer
00768           when Array
00769             file.write((value.is_a?(Integer) && !enumer[value].nil?) ? enumer[value] : value)
00770           when Symbol, String
00771             if GameData.const_defined?(enumer.to_sym)
00772               mod = GameData.const_get(enumer.to_sym)
00773               file.write(mod.get(value).id.to_s)
00774             else
00775               mod = Object.const_get(enumer.to_sym)
00776               file.write(getConstantName(mod, value))
00777             end
00778           when Module
00779             file.write(getConstantName(enumer, value))
00780           when Hash
00781             if value.is_a?(String)
00782               file.write(value)
00783             else
00784               enumer.each_key do |key|
00785                 next if enumer[key] != value
00786                 file.write(key)
00787                 break
00788               end
00789             end
00790           end
00791         when "y", "Y"   # Enumerable or integer
00792           enumer = schema[2 + i - start]
00793           case enumer
00794           when Array
00795             file.write((value.is_a?(Integer) && !enumer[value].nil?) ? enumer[value] : value)
00796           when Symbol, String
00797             if !Kernel.const_defined?(enumer.to_sym) && GameData.const_defined?(enumer.to_sym)
00798               mod = GameData.const_get(enumer.to_sym)
00799               if mod.exists?(value)
00800                 file.write(mod.get(value).id.to_s)
00801               else
00802                 file.write(value.to_s)
00803               end
00804             else
00805               mod = Object.const_get(enumer.to_sym)
00806               file.write(getConstantNameOrValue(mod, value))
00807             end
00808           when Module
00809             file.write(getConstantNameOrValue(enumer, value))
00810           when Hash
00811             if value.is_a?(String)
00812               file.write(value)
00813             else
00814               has_enum = false
00815               enumer.each_key do |key|
00816                 next if enumer[key] != value
00817                 file.write(key)
00818                 has_enum = true
00819                 break
00820               end
00821               file.write(value) if !has_enum
00822             end
00823           end
00824         else
00825           if value.is_a?(String)
00826             file.write((schema[1][i, 1].downcase == "q") ? value : csvQuote(value))
00827           elsif value.is_a?(Symbol)
00828             file.write(csvQuote(value.to_s))
00829           elsif value == true
00830             file.write("true")
00831           elsif value == false
00832             file.write("false")
00833           else
00834             file.write(value.inspect)
00835           end
00836         end
00837       end
00838       break if start > 0 && index >= rec.length - 1
00839       break if start <= 0
00840     end
00841     return record
