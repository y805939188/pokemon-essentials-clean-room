00073 module Effectiveness
00074   INEFFECTIVE                   = 0
00075   NOT_VERY_EFFECTIVE            = 1
00076   NORMAL_EFFECTIVE              = 2
00077   SUPER_EFFECTIVE               = 4
00078   INEFFECTIVE_MULTIPLIER        = INEFFECTIVE.to_f / NORMAL_EFFECTIVE
00079   NOT_VERY_EFFECTIVE_MULTIPLIER = NOT_VERY_EFFECTIVE.to_f / NORMAL_EFFECTIVE
00080   NORMAL_EFFECTIVE_MULTIPLIER   = 1.0

00085   def ineffective?(value)

00135     ret = NORMAL_EFFECTIVE_MULTIPLIER
