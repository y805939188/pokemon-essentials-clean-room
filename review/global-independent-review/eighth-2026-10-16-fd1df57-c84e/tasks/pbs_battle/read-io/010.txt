00040   def self.linereport
00041     if @section

00043         return _INTL("File {1}, section {2}\n{3}", @file, @section, @value) + "\n\n"
00044       else
00045         return _INTL("File {1}, section {2}, key {3}\n{4}", @file, @section, @key, @value) + "\n\n"
00046       end
00047     else
00048       return _INTL("File {1}, line {2}\n{3}", @file, @lineno, @linedata) + "\n\n"

00059   def findIndex(a)
00060     index = -1
00061     count = 0
00062     a.each do |i|
00063       if yield i
00064         index = count
00065         break
00066       end
00067       count += 1
00068     end

00072   def prepline(line)
00073     line.sub!(/\s*\#.*$/, "")
00074     line.sub!(/^\s+/, "")
00075     line.sub!(/\s+$/, "")
00076     return line
00077   end

00079   def csvQuote(str, always = false)
00080     return "" if nil_or_empty?(str)
00081     if always || str[/[,\"]/]   # || str[/^\s/] || str[/\s$/] || str[/^#/]
00082       str = str.gsub(/\"/, "\\\"")
00083       str = "\"#{str}\""
00084     end

00095   def pbEachFileSectionEx(f, schema = nil)
00096     lineno      = 1
00097     havesection = false
00098     sectionname = nil
00099     lastsection = {}
00100     f.each_line do |line|
00101       if lineno == 1 && line[0].ord == 0xEF && line[1].ord == 0xBB && line[2].ord == 0xBF
00102         line = line[3, line.length - 3]
00103       end
00104       line.force_encoding(Encoding::UTF_8)
00105       if !line[/^\#/] && !line[/^\s*$/]
00106         line = prepline(line)
00107         if line[/^\s*\[\s*(.*)\s*\]\s*$/]   # Of the format: [something]
00108           yield lastsection, sectionname if havesection
00109           sectionname = $~[1]
00110           havesection = true
00111           lastsection = {}
00112         else
00113           if sectionname.nil?
00114             FileLineData.setLine(line, lineno)
00115             raise _INTL("Expected a section at the beginning of the file.\nThis error may also occur if the file was not saved in UTF-8.") + "\n" + FileLineData.linereport
00116           end
00117           if !line[/^\s*(\w+)\s*=\s*(.*)$/]
00118             FileLineData.setSection(sectionname, nil, line)
00119             raise _INTL("Bad line syntax (expected syntax like XXX=YYY).") + "\n" + FileLineData.linereport
00120           end
00121           r1 = $~[1]
00122           r2 = $~[2]
00123           if schema && schema[r1] && schema[r1][1][0] == "^"
00124             lastsection[r1] ||= []
00125             lastsection[r1].push(r2.gsub(/\s+$/, ""))
00126           else
00127             lastsection[r1] = r2.gsub(/\s+$/, "")
00128           end
00129         end
00130       end
00131       lineno += 1
00132       Graphics.update if lineno % 1000 == 0
00133     end
00134     yield lastsection, sectionname if havesection
00135   end
