00004   def compile_PBS_file_generic(game_data, *paths)
00005     if game_data.const_defined?(:OPTIONAL) && game_data::OPTIONAL
00006       return if paths.none? { |p| FileTest.exist?(p) }
00007     end
00008     game_data::DATA.clear
00009     schema = game_data.schema
00010     # Read from PBS file(s)
00011     paths.each do |path|
00012       compile_pbs_file_message_start(path)
00013       base_filename = game_data::PBS_BASE_FILENAME
00014       base_filename = base_filename[0] if base_filename.is_a?(Array)   # For Species
00015       file_suffix = File.basename(path, ".txt")[base_filename.length + 1, path.length] || ""
00016       File.open(path, "rb") do |f|
00017         FileLineData.file = path   # For error reporting
00018         # Read a whole section's lines at once, then run through this code.
00019         # contents is a hash containing all the XXX=YYY lines in that section, where
00020         # the keys are the XXX and the values are the YYY (as unprocessed strings).
00021         idx = 0
00022         pbEachFileSection(f, schema) do |contents, section_name|
00023           echo "." if idx % 100 == 0
00024           Graphics.update if idx % 500 == 0
00025           idx += 1
00026           data_hash = {
00027             :id              => section_name.to_sym,
00028             :pbs_file_suffix => file_suffix
00029           }
00030           # Go through schema hash of compilable data and compile this section
00031           schema.each_key do |key|
00032             FileLineData.setSection(section_name, key, contents[key])   # For error reporting
00033             if key == "SectionName"
00034               data_hash[schema[key][0]] = get_csv_record(section_name, schema[key])
00035               next
00036             end
00037             # Skip empty properties
00038             next if contents[key].nil?
00039             # Compile value for key
00040             if schema[key][1][0] == "^"
00041               contents[key].each do |val|
00042                 value = get_csv_record(val, schema[key])
00043                 value = nil if value.is_a?(Array) && value.empty?
00044                 data_hash[schema[key][0]] ||= []
00045                 data_hash[schema[key][0]].push(value)
00046               end
00047               data_hash[schema[key][0]].compact!
00048             else
00049               value = get_csv_record(contents[key], schema[key])
00050               value = nil if value.is_a?(Array) && value.empty?
00051               data_hash[schema[key][0]] = value
00052             end
00053           end
00054           # Validate and modify the compiled data
00055           yield false, data_hash if block_given?
00056           if game_data.exists?(data_hash[:id])
00057             raise _INTL("Section name '{1}' is used twice.", data_hash[:id]) + "\n" + FileLineData.linereport
00058           end
00059           # Add section's data to records
00060           game_data.register(data_hash)
00061         end
00062       end
00063       process_pbs_file_message_end
00064     end
00065     yield true, nil if block_given?
00066     # Save all data
00067     game_data.save
00068   end

00104   def compile_connections(*paths)
00105     hashenum = {
00106       "N" => "N", "North" => "N",
00107       "E" => "E", "East"  => "E",
00108       "S" => "S", "South" => "S",
00109       "W" => "W", "West"  => "W"
00110     }
00111     schema = [nil, "iyiiyi", nil, hashenum, nil, nil, hashenum]
00112     records = []
00113     paths.each do |path|
00114       compile_pbs_file_message_start(path)
00115       pbCompilerEachPreppedLine(path) do |line, lineno|
00116         FileLineData.setLine(line, lineno)
00117         record = get_csv_record(line, schema)
00118         if !pbRgssExists?(sprintf("Data/Map%03d.rxdata", record[0]))
00119           print _INTL("Warning: Map {1}, as mentioned in the map connection data, was not found.", record[0]) + "\n" + FileLineData.linereport
00120         elsif !pbRgssExists?(sprintf("Data/Map%03d.rxdata", record[3]))
00121           print _INTL("Warning: Map {1}, as mentioned in the map connection data, was not found.", record[3]) + "\n" + FileLineData.linereport
00122         end
00123         case record[1]
00124         when "N"
00125           raise _INTL("North side of first map must connect with south side of second map.") + "\n" + FileLineData.linereport if record[4] != "S"
00126         when "S"
00127           raise _INTL("South side of first map must connect with north side of second map.") + "\n" + FileLineData.linereport if record[4] != "N"
00128         when "E"
00129           raise _INTL("East side of first map must connect with west side of second map.") + "\n" + FileLineData.linereport if record[4] != "W"
00130         when "W"
00131           raise _INTL("West side of first map must connect with east side of second map.") + "\n" + FileLineData.linereport if record[4] != "E"
00132         end
00133         records.push(record)
00134       end
00135       process_pbs_file_message_end
00136     end
00137     save_data(records, "Data/map_connections.dat")
00138   end

00211   def validate_compiled_move(hash)
00212     if (hash[:category] || 2) == 2 && (hash[:power] || 0) != 0
00213       raise _INTL("Move {1} is defined as a Status move with a non-zero base damage.", hash[:real_name]) + "\n" + FileLineData.linereport
00214     elsif (hash[:category] || 2) != 2 && (hash[:power] || 0) == 0
00215       print _INTL("Warning: Move {1} is defined as Physical or Special but has a base damage of 0. Changing it to a Status move.", hash[:real_name]) + "\n" + FileLineData.linereport
00216       hash[:category] = 2
00217     end
00218   end
