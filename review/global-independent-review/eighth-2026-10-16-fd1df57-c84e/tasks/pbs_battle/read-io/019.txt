00075   # Loads and validates the map. Called when loading a saved game.
00076   def self.load_map
00077     $game_map = $map_factory.map
00078     magic_number_matches = ($game_system.magic_number == $data_system.magic_number)
00079     if !magic_number_matches || $PokemonGlobal.safesave
00080       pbMapInterpreter.setup(nil, 0) if pbMapInterpreterRunning?
00081       begin
00082         $map_factory.setup($game_map.map_id)
00083       rescue Errno::ENOENT
00084         if $DEBUG
00085           pbMessage(_INTL("Map {1} was not found.", $game_map.map_id))
00086           map = pbWarpToMap
00087           exit unless map
00088           $map_factory.setup(map[0])
00089           $game_player.moveto(map[1], map[2])
00090         else
00091           raise _INTL("The map was not found. The game cannot continue.")
00092         end
00093       end
00094       $game_player.center($game_player.x, $game_player.y)
00095     else
00096       $map_factory.setMapChanged($game_map.map_id)
00097     end
00098     if $game_map.events.nil?
00099       raise _INTL("The map is corrupt. The game cannot continue.")
00100     end
00101     $PokemonEncounters = PokemonEncounters.new
00102     $PokemonEncounters.setup($game_map.map_id)
00103     pbUpdateVehicle
00104   end

00106   # Saves the game. Returns whether the operation was successful.
00107   # @param save_file [String] the save file path
00108   # @param safe [Boolean] whether $PokemonGlobal.safesave should be set to true
00109   # @return [Boolean] whether the operation was successful
00110   # @raise [SaveData::InvalidValueError] if an invalid value is being saved
00111   def self.save(save_file = SaveData::FILE_PATH, safe: false)
00112     validate save_file => String, safe => [TrueClass, FalseClass]
00113     $PokemonGlobal.safesave = safe
00114     $game_system.save_count += 1
00115     $game_system.magic_number = $data_system.magic_number
00116     $stats.set_time_last_saved
00117     begin
00118       SaveData.save_to_file(save_file)
00119       Graphics.frame_reset
00120     rescue IOError, SystemCallError
00121       $game_system.save_count -= 1
00122       return false
00123     end
00124     return true
