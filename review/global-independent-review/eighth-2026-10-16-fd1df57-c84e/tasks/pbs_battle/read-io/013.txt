00924   def edit_and_rewrite_pbs_file_text(filename)
00925     return if !block_given?
00926     lines = []
00927     File.open(filename, "rb") do |f|
00928       f.each_line { |line| lines.push(line) }
00929     end
00930     changed = false
00931     lines.each { |line| changed = true if yield line }
00932     if changed
00933       Console.markup_style("Changes made to file #{filename}.", text: :yellow)
00934       File.open(filename, "wb") do |f|
00935         lines.each { |line| f.write(line) }
00936       end
00937     end
00938   end

00967   def get_all_pbs_files_to_compile
00968     # Get the GameData classes and their respective base PBS filenames
00969     ret = GameData.get_all_pbs_base_filenames
00970     ret.merge!({
00971       :BattleFacility => "battle_facility_lists",
00972       :Connection     => "map_connections",
00973       :RegionalDex    => "regional_dexes"
00974     })
00975     ret.each { |key, val| ret[key] = [val] }   # [base_filename, ["PBS/file.txt", etc.]]
00976     # Look through all PBS files and match them to a GameData class based on
00977     # their base filenames
00978     text_files_keys = ret.keys.sort! { |a, b| ret[b][0].length <=> ret[a][0].length }
00979     Dir.chdir("PBS/") do
00980       Dir.glob("*.txt") do |f|
00981         base_name = File.basename(f, ".txt")
00982         text_files_keys.each do |key|
00983           next if base_name != ret[key][0] && !f.start_with?(ret[key][0] + "_")
00984           ret[key][1] ||= []
00985           ret[key][1].push("PBS/" + f)
00986           break
00987         end
00988       end
00989     end
00990     return ret
00991   end

01039   def main
01040     return if !$DEBUG
01041     begin
01042       mustCompile = false
01043       # If no PBS file, create one and fill it, then recompile
01044       if !FileTest.directory?("PBS")
01045         Dir.mkdir("PBS") rescue nil
01046         GameData.load_all
01047         write_all
01048         mustCompile = true
01049       end
01050       # Get all data files and PBS files to be checked for their last modified times
01051       data_files = GameData.get_all_data_filenames
01052       data_files += [   # Extra .dat files for data that isn't a GameData class
01053         ["map_connections.dat", true],
01054         ["regional_dexes.dat", true],
01055         ["trainer_lists.dat", true]
01056       ]
01057       text_files = get_all_pbs_files_to_compile
01058       latestDataTime = 0
01059       latestTextTime = 0
01060       # Should recompile if new maps were imported
01061       mustCompile |= import_new_maps
01062       # Check data files for their latest modify time
01063       data_files.each do |filename|   # filename = [string, boolean (whether mandatory)]
01064         if FileTest.exist?("Data/" + filename[0])
01065           begin
01066             File.open("Data/#{filename[0]}") do |file|
01067               latestDataTime = [latestDataTime, file.mtime.to_i].max
01068             end
01069           rescue SystemCallError
01070             mustCompile = true
01071           end
01072         elsif filename[1]
01073           mustCompile = true
01074           break
01075         end
01076       end
01077       # Check PBS files for their latest modify time
01078       text_files.each do |key, value|
01079         next if !value || !value[1].is_a?(Array)
01080         value[1].each do |filepath|
01081           begin
01082             File.open(filepath) { |file| latestTextTime = [latestTextTime, file.mtime.to_i].max }
01083           rescue SystemCallError
01084           end
01085         end
01086       end
01087       # Decide to compile if a PBS file was edited more recently than any .dat files
01088       mustCompile |= (latestTextTime >= latestDataTime)
01089       # Should recompile if holding Ctrl
01090       Input.update
01091       mustCompile = true if $full_compile || Input.press?(Input::CTRL)
01092       # Delete old data files in preparation for recompiling
01093       if mustCompile
01094         data_files.each do |filename|
01095           begin
01096             File.delete("Data/#{filename[0]}") if FileTest.exist?("Data/#{filename[0]}")
01097           rescue SystemCallError
01098           end
01099         end
01100       end
01101       # Recompile all data
01102       compile_all(mustCompile)
01103     rescue Exception
01104       e = $!
01105       raise e if e.class.to_s == "Reset" || e.is_a?(Reset) || e.is_a?(SystemExit)
01106       pbPrintException(e)
01107       data_files.each do |filename|
01108         begin
01109           File.delete("Data/#{filename[0]}") if FileTest.exist?("Data/#{filename[0]}")
01110         rescue SystemCallError
01111         end
01112       end
01113       raise Reset.new if e.is_a?(Hangup)
01114       raise "Unknown exception when compiling."
01115     end
01116   end
